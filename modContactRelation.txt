'=============================================================================
' RelationScan  (epinglable au QAT / ruban)
'=============================================================================
Public Sub RelationScan()
    Dim errNo As Long, errDesc As String, heavyOn As Boolean
    If mBusy Then
        modUi.Inform T("Le rapport de relance est d~ej~a en cours.", "The reconnect report is already running."), UiTitle()
        Exit Sub
    End If
    On Error GoTo ErrH
    LoadSettings

    Dim dFrom As Date
    dFrom = DateAdd("m", -mMonths, Date)

    ' v1.2 (F7-06) : plus de "progression sur la barre d'etat" (Outlook n'en
    ' a pas de pilotable) ; un toast donne l'avancement.
    If MsgBox(T("Chercher tes mails envoy~es (tous les dossiers de mails) des ", _
                "Look for your sent e-mails (all mail folders) in the ") & _
              ScopeLabel() & vbCrLf & _
              T("depuis le ", "since ") & Format$(dFrom, "dd/mm/yyyy") & _
              T(" (fen~xtre ", " (window ") & mMonths & T(" mois), seuil par d~efaut ", " months), default threshold ") & _
              mThreshold & T(" jours ?", " days?") & vbCrLf & vbCrLf & _
              T("Le balayage peut prendre plusieurs minutes : Outlook reste occup~e, un toast donne l'avancement.", _
                "The scan can take several minutes: Outlook stays busy, a toast shows the progress."), _
              vbOKCancel + vbQuestion, UiTitle()) <> vbOK Then Exit Sub

    mBusy = True
    Set mLastSent = CreateObject("Scripting.Dictionary")
    mLastSent.CompareMode = vbTextCompare
    Set mLastByKey = CreateObject("Scripting.Dictionary")
    Set mNameCache = CreateObject("Scripting.Dictionary")
    mMailsSeen = 0
    mContactsSeen = 0
    mFoldersRead = 0
    mRowsRead = 0
    mTableErrors = 0
    mStoreSummary = ""
    mDateCol = "SentOn"
    Set mGroupKeys = New Collection
    modContactCommon.OwnIdentity mOwnAddr, mOwnKeys
    mLastToastMs = modCommon.TickMs()
    ' Les DoEvents du balayage ne laissent pas partir un traitement lourd differe.
    modDefer.BeginHeavy
    heavyOn = True

    ' Passe 1 : mails envoyes, tous les dossiers de mails de chaque boite
    ' retenue (multi-boites), en mode liste.
    Dim ns As Outlook.NameSpace, idx As Long, st As Outlook.Store, mailsBefore As Long
    Dim sts As Outlook.Stores, nSt As Long
    Set ns = Application.Session
    Set sts = ns.Stores
    nSt = sts.Count
    For idx = 1 To nSt
        Set st = sts.Item(idx)
        If Not StoreWanted(st.DisplayName) Then
            mStoreSummary = mStoreSummary & "<li>" & modCommon.EscapeHTML(st.DisplayName) & " : " & _
                            modCommon.EscapeHTML(T("hors p~erim~etre (RelationScanStores)", "out of scope (RelationScanStores)")) & "</li>"
        ElseIf IsPublicStore(st) Then
            mStoreSummary = mStoreSummary & "<li>" & modCommon.EscapeHTML(st.DisplayName) & " : " & _
                            modCommon.EscapeHTML(T("dossiers publics, non balay~es", "public folders, not scanned")) & "</li>"
        Else
            mailsBefore = mMailsSeen
            ScanStoreSent st, dFrom
            mStoreSummary = mStoreSummary & "<li>" & modCommon.EscapeHTML(st.DisplayName) & " : " & _
                            (mMailsSeen - mailsBefore) & _
                            modCommon.EscapeHTML(T(" mails envoy~es trouv~es (tous dossiers)", " sent e-mails found (all folders)")) & "</li>"
        End If
    Next idx

    ' Passe 2 : croisement avec les fiches de tous les dossiers de contacts.
    Dim rows As Collection
    Set rows = New Collection
    For idx = 1 To nSt
        ScanContactBranch sts.Item(idx).GetRootFolder, rows
    Next idx
    modDefer.EndHeavy
    heavyOn = False

    ' Rapport HTML trie par retard decroissant.
    Dim outPath As String
    outPath = WriteReport(rows, dFrom)
    modDebugLog.DInfo MOD_NAME, "RelationScan : " & mFoldersRead & " dossiers, " & mRowsRead & " lignes, " & _
                                mMailsSeen & " mails envoyes, " & mContactsSeen & " fiches, " & rows.Count & _
                                " a relancer, tables refusees=" & mTableErrors & " -> " & outPath
    mBusy = False

    If MsgBox(rows.Count & T(" contact(s) ~a relancer (sur ", " contact(s) to reconnect with (out of ") & _
              mContactsSeen & T(" fiches externes examin~ees).", " external cards examined).") & vbCrLf & _
              T("Rapport : ", "Report: ") & outPath & vbCrLf & vbCrLf & _
              T("Ouvrir le rapport ?", "Open the report?"), vbYesNo + vbInformation, UiTitle()) = vbYes Then
        CreateObject("Shell.Application").ShellExecute outPath
    End If
    Exit Sub
ErrH:
    errNo = Err.Number
    errDesc = Err.Description
    On Error Resume Next
    If heavyOn Then modDefer.EndHeavy
    mBusy = False
    modDebugLog.DError MOD_NAME, "RelationScan err " & errNo & " - " & errDesc
    MsgBox T("Erreur : ", "Error: ") & errNo & " - " & errDesc, vbExclamation, UiTitle()
End Sub

Private Sub LoadSettings()
    modConfig.EnsureLoaded
    mMonths = modConfig.CfgL("RelationScanMonths", 48)
    If mMonths < 1 Then mMonths = 48
    mThreshold = modConfig.CfgL("RelationScanThresholdDays", 180)
    If mThreshold < 1 Then mThreshold = 180
    mExclude = "|" & LCase$(Replace(modConfig.Cfg("InternalDomains", ""), " ", "")) & "|" & _
               LCase$(Replace(modConfig.Cfg("RelationScanExcludeDomains", ""), " ", "")) & "|"
    mStoresWanted = Trim$(modConfig.Cfg("RelationScanStores", ""))
    mSkipFolders = modConfig.Cfg("ContactSkipFolders", modContactCommon.ContactSkipDefault())
End Sub

Private Function UiTitle() As String
    UiTitle = T("Relations ~a relancer", "Relations to reconnect")
End Function

Private Function ScopeLabel() As String
    If Len(mStoresWanted) = 0 Then
        ScopeLabel = T("boites du profil (toutes)", "mailboxes of the profile (all)")
    Else
        ScopeLabel = T("boites correspondant ~a : ", "mailboxes matching: ") & mStoresWanted
    End If
End Function

Private Function StoreWanted(ByVal displayName As String) As Boolean
    If Len(mStoresWanted) = 0 Then
        StoreWanted = True
        Exit Function
    End If
    Dim parts() As String, idx As Long, nameLow As String
    nameLow = LCase$(displayName)
    parts = Split(mStoresWanted, "|")
    For idx = LBound(parts) To UBound(parts)
        If Len(Trim$(parts(idx))) > 0 Then
            If InStr(nameLow, LCase$(Trim$(parts(idx)))) > 0 Then
                StoreWanted = True
                Exit Function
            End If
        End If
    Next idx
End Function

'=============================================================================
' Passe 1 (v1.2, F7-05) : TOUS les dossiers de mails d'une boite (les envois
' ranges par ReplyInPlace / QuickMove compris), chacun lu en mode LISTE.
' Seuls les mails envoyes par toi dans la fenetre sont retenus. Aucun mail
' ouvert, aucun filtre de date ecrit (comparaison en VBA).
'=============================================================================
Private Sub ScanStoreSent(ByVal st As Outlook.Store, ByVal dFrom As Date)
    On Error Resume Next
    WalkSentFolders st.GetRootFolder, SpecialFolderIds(st), dFrom
    Err.Clear
End Sub

Private Sub WalkSentFolders(ByVal fld As Object, ByVal skipIds As String, ByVal dFrom As Date)
    On Error Resume Next
    Dim fid As String, itemType As Long, subs As Object, child As Object
    If fld Is Nothing Then Exit Sub
    fid = SafeStr(fld.EntryID)
    If Len(fid) = 0 Then Exit Sub
    If InStr(skipIds, "|" & fid & "|") > 0 Then Exit Sub
    If modContactCommon.IsHiddenFolder(fld) Then Exit Sub
    itemType = -1
    itemType = fld.DefaultItemType
    If itemType = OL_MAIL_ITEM Then
        ScanSentFolder fld, dFrom
        mFoldersRead = mFoldersRead + 1
        DoEvents
        MaybeProgress
    End If
    ' Sous-dossiers illisibles : sortir (sous On Error Resume Next, un For Each
    ' sur Nothing executerait le corps une fois -> recursion sans fin).
    Set subs = Nothing
    Set subs = fld.Folders
    If subs Is Nothing Then
        Err.Clear
        Exit Sub
    End If
    For Each child In subs
        If Not child Is Nothing Then WalkSentFolders child, skipIds, dFrom
    Next child
    Err.Clear
End Sub

' Elements supprimes, Boite d'envoi, Brouillons, Conflits, Problemes de
' synchronisation (+ echecs locaux / serveur), Courrier indesirable, RSS.
Private Function SpecialFolderIds(ByVal st As Object) As String
    On Error Resume Next
    Dim ids As Variant, k As Long, fld As Object, res As String
    ids = Array(3, 4, 16, 19, 20, 21, 22, 23, 25)
    res = "|"
    For k = LBound(ids) To UBound(ids)
        Set fld = Nothing
        Set fld = st.GetDefaultFolder(CLng(ids(k)))
        If Not fld Is Nothing Then res = res & SafeStr(fld.EntryID) & "|"
        Err.Clear
    Next k
    SpecialFolderIds = res
End Function

Private Function IsPublicStore(ByVal st As Object) As Boolean
    On Error Resume Next
    Dim typ As Long
    typ = -1
    typ = st.ExchangeStoreType
    IsPublicStore = (typ = STORE_PUBLIC)
    Err.Clear
End Function

Private Sub ScanSentFolder(ByVal fld As Object, ByVal dFrom As Date)
    On Error Resume Next
    Dim tbl As Object, rw As Object, v As Variant, n As Long
    Set tbl = Nothing
    Err.Clear
    Set tbl = fld.GetTable()
    If Err.Number <> 0 Or tbl Is Nothing Then
        mTableErrors = mTableErrors + 1
        If mTableErrors <= 3 Then modDebugLog.DError MOD_NAME, "Table refusee (" & Err.Number & ") : " & SafeStr(fld.FolderPath)
        Err.Clear
        Exit Sub
    End If
    tbl.Columns.RemoveAll
    AddSentColumns tbl
    If Err.Number <> 0 And mDateCol = "SentOn" Then
        ' Repli unique : date de reception (egale a l'envoi pour un mail envoye).
        modDebugLog.DError MOD_NAME, "Colonne SentOn refusee (" & Err.Number & ") : repli ReceivedTime"
        Err.Clear
        mDateCol = "ReceivedTime"
        tbl.Columns.RemoveAll
        AddSentColumns tbl
    End If
    If Err.Number <> 0 Then
        mTableErrors = mTableErrors + 1
        If mTableErrors <= 3 Then
            modDebugLog.DError MOD_NAME, "Colonnes refusees (" & Err.Number & " - " & Err.Description & ") : " & _
                SafeStr(fld.FolderPath)
        End If
        Err.Clear
        Exit Sub
    End If
    Do Until tbl.EndOfTable
        Set rw = Nothing
        Set rw = tbl.GetNextRow
        If rw Is Nothing Then Exit Do
        v = Empty
        v = rw.GetValues()
        If IsArray(v) Then ProcessSentRow v, dFrom
        Err.Clear
        n = n + 1
        mRowsRead = mRowsRead + 1
        If (n Mod 250) = 0 Then
            DoEvents
            If (n Mod 2000) = 0 Then MaybeProgress
        End If
    Loop
    Set rw = Nothing
    Set tbl = Nothing
End Sub

' Pas de gestionnaire d'erreur : une colonne refusee remonte a ScanSentFolder.
Private Sub AddSentColumns(ByVal tbl As Object)
    tbl.Columns.Add "MessageClass"
    tbl.Columns.Add mDateCol
    tbl.Columns.Add PR_SENT_REP_SMTP
    tbl.Columns.Add PR_SENT_REP_EMAIL
    tbl.Columns.Add PR_SENDER_SMTP
    tbl.Columns.Add "SenderName"
    tbl.Columns.Add PR_DISPLAY_TO
    tbl.Columns.Add PR_DISPLAY_CC
End Sub

' Une ligne de table : mail (IPM.Note*) envoye par toi dans la fenetre ->
' ses destinataires A / Cc sont notes.
Private Sub ProcessSentRow(ByRef v As Variant, ByVal dFrom As Date)
    Dim b As Long, dt As Date, lst As String, cc As String
    b = LBound(v)
    If Left$(LCase$(VStr(v(b + COL_CLASS))), 8) <> "ipm.note" Then Exit Sub
    dt = VDate(v(b + COL_DATE))
    If dt < dFrom Then Exit Sub                ' hors fenetre (0 = date illisible)
    If Not IsOwnSender(v, b) Then Exit Sub
    mMailsSeen = mMailsSeen + 1
    lst = VStr(v(b + COL_TO))
    cc = VStr(v(b + COL_CC))
    If Len(cc) > 0 Then lst = lst & ";" & cc
    If Len(lst) > 0 Then RecordDisplayRecipients lst, dt
End Sub

' Expediteur = toi : adresse representee ou expediteur parmi tes adresses
' (modSmtp.IsMyAddress : alias compris), adresse Exchange interne parmi les
' tiennes, ou - sans aucune adresse lisible - ton nom.
Private Function IsOwnSender(ByRef v As Variant, ByVal b As Long) As Boolean
    Dim smtp As String, raw As String, snd As String, k As String
    smtp = LCase$(Trim$(VStr(v(b + COL_REPSMTP))))
    raw = LCase$(Trim$(VStr(v(b + COL_REPEMAIL))))
    snd = LCase$(Trim$(VStr(v(b + COL_SNDSMTP))))
    If InStr(smtp, "@") > 0 Then
        If IsMine(smtp) Then
            IsOwnSender = True
            Exit Function
        End If
    End If
    If InStr(raw, "@") > 0 And Left$(raw, 1) <> "/" Then
        If IsMine(raw) Then
            IsOwnSender = True
            Exit Function
        End If
    End If
    If InStr(snd, "@") > 0 Then
        If IsMine(snd) Then
            IsOwnSender = True
            Exit Function
        End If
    End If
    If Left$(raw, 1) = "/" Then
        If InStr(mOwnAddr, "|" & raw & "|") > 0 Then
            IsOwnSender = True
            Exit Function
        End If
    End If
    If InStr(smtp, "@") = 0 And InStr(snd, "@") = 0 And Len(raw) = 0 Then
        k = NameKeyCached(VStr(v(b + COL_SNDNAME)))
        If Len(k) > 0 Then IsOwnSender = (InStr(mOwnKeys, "|" & k & "|") > 0)
    End If
End Function

Private Function IsMine(ByVal adr As String) As Boolean
    IsMine = (InStr(mOwnAddr, "|" & adr & "|") > 0)
    If Not IsMine Then IsMine = modSmtp.IsMyAddress(adr)
End Function

' "Jean Dupont; Dupont, Marie <marie.dupont@x.fr>; 'paul@y.fr'" : adresse
' visible -> notee par adresse (domaines exclus ecartes, nom compris) ;
' toujours notee aussi par cle de nom.
Private Sub RecordDisplayRecipients(ByVal lst As String, ByVal dt As Date)
    Dim parts() As String, k As Long, p As String, adr As String, key As String
    parts = Split(lst, ";")
    For k = LBound(parts) To UBound(parts)
        p = Trim$(parts(k))
        If Len(p) > 2 Then
            adr = ""
            If InStr(p, "@") > 0 Then adr = modContactCommon.MailIn(p)
            If Len(adr) > 0 Then
                If IsExcluded(adr) Then GoTo NextPart     ' interne / exclu : ni adresse ni nom
                NoteDate mLastSent, adr, dt
            End If
            key = NameKeyCached(p)
            If Len(key) > 0 Then NoteDate mLastByKey, key, dt
        End If
NextPart:
    Next k
End Sub

Private Sub NoteDate(ByVal d As Object, ByVal k As String, ByVal dt As Date)
    If d.Exists(k) Then
        If dt > d(k) Then d(k) = dt
    Else
        d.Add k, dt
    End If
End Sub

Private Function NameKeyCached(ByVal s As String) As String
    Dim k As String
    If Len(s) = 0 Then Exit Function
    If mNameCache.Exists(s) Then
        NameKeyCached = mNameCache(s)
        Exit Function
    End If
    k = modContactCommon.NameKeySorted(s)
    If mNameCache.Count > 60000 Then mNameCache.RemoveAll
    mNameCache(s) = k
    NameKeyCached = k
End Function

' Toast d'avancement toutes les 30 s (Outlook n'a pas de barre d'etat).
Private Sub MaybeProgress()
    If modCommon.ElapsedMs(mLastToastMs) < PROGRESS_MS Then Exit Sub
    mLastToastMs = modCommon.TickMs()
    modUi.Toast T("Rapport de relance en cours", "Reconnect report running") & vbLf & _
                mFoldersRead & T(" dossiers, ", " folders, ") & mMailsSeen & T(" mails envoy~es, ", " sent e-mails, ") & _
                mContactsSeen & T(" fiches", " cards"), uiInfo, True
End Sub

Private Function VStr(ByVal x As Variant) As String
    On Error Resume Next
    If IsNull(x) Or IsEmpty(x) Or IsArray(x) Then Exit Function
    VStr = CStr(x)
End Function

Private Function VDate(ByVal x As Variant) As Date
    On Error Resume Next
    Dim d As Date
    If IsDate(x) Then
        d = CDate(x)
        If d > #1/1/4000# Or d < #1/1/1990# Then d = 0     ' "aucune date" d'Outlook (4501) ou aberrante
    End If
    VDate = d
End Function

' Adresse interne ou d'un domaine exclu ?
Private Function IsExcluded(ByVal addr As String) As Boolean
    If InStr(addr, "@") = 0 Then
        IsExcluded = True
        Exit Function
    End If
    Dim dom As String
    dom = LCase$(Mid$(addr, InStr(addr, "@") + 1))
    If Len(mExclude) > 2 Then IsExcluded = (InStr(mExclude, "|" & dom & "|") > 0)
End Function

'=============================================================================
' Passe 2 : dossiers de contacts (recursif, exclusions partagees).
'=============================================================================
Private Sub ScanContactBranch(ByVal fld As Outlook.MAPIFolder, ByVal rows As Collection)
    On Error Resume Next
    Dim isContacts As Boolean
    isContacts = (fld.DefaultItemType = olContactItem)
    If Err.Number <> 0 Then isContacts = False: Err.Clear
    If isContacts Then
        If Not IsSkippedFolder(fld) Then ScanContactFolder fld, rows
    End If
    Dim child As Outlook.MAPIFolder
    For Each child In fld.Folders
        ScanContactBranch child, rows
    Next child
    Err.Clear
End Sub

Private Sub ScanContactFolder(ByVal fld As Outlook.MAPIFolder, ByVal rows As Collection)
    On Error Resume Next
    Dim itm As Object, k As Long, addr As String, bestAddr As String
    Dim lastDate As Date, threshold As Long, daysSince As Long, hasExternal As Boolean
    Dim keyAdded As Boolean, byName As Boolean, nk As String, nk2 As String
    For Each itm In fld.Items
        If TypeName(itm) = "ContactItem" Then
            hasExternal = False
            lastDate = 0
            bestAddr = ""
            byName = False
            For k = 1 To 3
                addr = LCase$(Trim$(SafeStr(CallByName(itm, "Email" & k & "Address", VbGet))))
                If InStr(addr, "@") > 0 Then
                    If Not IsExcluded(addr) Then
                        hasExternal = True
                        If Len(bestAddr) = 0 Then bestAddr = addr
                        If mLastSent.Exists(addr) Then
                            If mLastSent(addr) > lastDate Then
                                lastDate = mLastSent(addr)
                                bestAddr = addr
                            End If
                        End If
                    End If
                End If
            Next k
            If hasExternal Then
                ' v1.2 (F7-05) : destinataire sans adresse visible -> par le nom
                ' (prenom + nom, puis nom complet de la fiche).
                nk = modContactCommon.NameKeySorted(SafeStr(itm.FirstName) & " " & SafeStr(itm.LastName))
                nk2 = modContactCommon.NameKeySorted(SafeStr(itm.FullName))
                If Len(nk) > 0 Then
                    If mLastByKey.Exists(nk) Then
                        If mLastByKey(nk) > lastDate Then
                            lastDate = mLastByKey(nk)
                            byName = True
                        End If
                    End If
                End If
                If Len(nk2) > 0 And nk2 <> nk Then
                    If mLastByKey.Exists(nk2) Then
                        If mLastByKey(nk2) > lastDate Then
                            lastDate = mLastByKey(nk2)
                            byName = True
                        End If
                    End If
                End If
                mContactsSeen = mContactsSeen + 1
                threshold = ContactThreshold(itm)
                If lastDate = 0 Then
                    daysSince = NEVER_DAYS
                Else
                    daysSince = DateDiff("d", lastDate, Date)
                End If
                If daysSince > threshold Then
                    rows.Add MakeRow(itm, fld, bestAddr, lastDate, daysSince, threshold, byName)
                    If Not keyAdded Then
                        mGroupKeys.Add fld.Store.DisplayName & Chr$(7) & fld.FolderPath
                        keyAdded = True
                    End If
                End If
                If (mContactsSeen Mod 100) = 0 Then
                    DoEvents
                    MaybeProgress
                End If
            End If
        End If
    Next itm
    Err.Clear
End Sub

' Cadence propre a la fiche (UserProperty KeepInTouchDays), sinon defaut.
Private Function ContactThreshold(ByVal ct As Object) As Long
    On Error Resume Next
    ContactThreshold = mThreshold
    Dim up As Object
    Set up = ct.UserProperties.Find(UP_KEEPDAYS)
    If Not up Is Nothing Then
        If IsNumeric(up.Value) Then
            If CLng(up.Value) > 0 Then ContactThreshold = CLng(up.Value)
        End If
    End If
    Err.Clear
End Function

' Ligne de rapport, champs separes par Chr$(7) (les noms de dossiers peuvent
' contenir | et des emojis) : boite, dossier, jamais(0/1), jours (largeur
' fixe pour le tri), nom, adresse, date ISO, seuil. v1.2 : byName = date
' trouvee par le nom du destinataire ("(nom)" apres la date).
Private Function MakeRow(ByVal ct As Object, ByVal fld As Outlook.MAPIFolder, _
                         ByVal addr As String, ByVal lastDate As Date, _
                         ByVal daysSince As Long, ByVal threshold As Long, _
                         Optional ByVal byName As Boolean = False) As String
    Dim nm As String, dateTxt As String
    nm = SafeStr(ct.FileAs)
    If Len(nm) = 0 Then nm = SafeStr(ct.FullName)
    If Len(nm) = 0 Then nm = addr
    If lastDate > 0 Then
        dateTxt = Format$(lastDate, "yyyy-mm-dd")
        If byName Then dateTxt = dateTxt & " " & T("(nom)", "(name)")
    Else
        dateTxt = "-"
    End If
    MakeRow = fld.Store.DisplayName & Chr$(7) & fld.FolderPath & Chr$(7) & _
              IIf(daysSince >= NEVER_DAYS, "1", "0") & Chr$(7) & _
              Format$(daysSince, "000000") & Chr$(7) & nm & Chr$(7) & addr & Chr$(7) & _
              dateTxt & Chr$(7) & threshold
End Function

'=============================================================================
' Rapport HTML - UTF-8 (ADODB.Stream), groupe par boite aux lettres puis par
' dossier dans l'ordre de parcours ; contacts tries par retard decroissant,
' les "jamais ecrit sur la fenetre" en dernier (tries par nom).
'=============================================================================
Private Function WriteReport(ByVal rows As Collection, ByVal dFrom As Date) As String
    Dim html As String, outPath As String
    Dim ki As Long, ri As Long, sj As Long
    Dim keyStr As String, keyParts() As String, storeName As String, folderPath As String
    Dim lastStore As String, rowTxt As String, fields() As String
    Dim overArr() As String, overCnt As Long, nevArr() As String, nevCnt As Long
    Dim tmp As String

    html = "<!DOCTYPE html><html><head><meta charset=""utf-8""><title>RelationScan</title>" & _
           "<style>body{font-family:Segoe UI,Arial,sans-serif;font-size:13px;margin:20px}" & _
           "h3{margin:24px 0 6px 0}table{border-collapse:collapse;width:100%}" & _
           "th,td{border:1px solid #ccc;padding:4px 8px;text-align:left}" & _
           "th{background:#f0f0f0}tr.folder td{background:#e8e8f4;font-weight:bold}" & _
           ".never{color:#b00}</style></head><body>" & _
           "<h2>" & modCommon.EscapeHTML(T("Relations ~a relancer", "Relations to reconnect")) & "</h2>" & _
           "<p>" & modCommon.EscapeHTML(T("Approche multi-boites : mails envoy~es (tous les dossiers de mails) et fiches de toutes les boites retenues.", _
                                          "Multi-mailbox approach: sent e-mails (all mail folders) and contact cards of all retained mailboxes.")) & "</p>" & _
           "<p>" & modCommon.EscapeHTML(T("(nom) : dernier envoi reconnu par le nom du destinataire, son adresse n'~etant pas visible dans la liste des destinataires.", _
                                          "(name): last e-mail matched by the recipient's name, the address not being visible in the recipient list.")) & "</p>" & _
           "<p><b>" & modCommon.EscapeHTML(T("Boites balay~ees (mails envoy~es, tous dossiers)", "Mailboxes scanned (sent e-mails, all folders)")) & _
           "</b></p><ul>" & mStoreSummary & "</ul>" & _
           "<p>" & modCommon.EscapeHTML(T("Fen~xtre : depuis le ", "Window: since ")) & Format$(dFrom, "dd/mm/yyyy") & _
           " (" & mMonths & " " & modCommon.EscapeHTML(T("mois", "months")) & ") - " & _
           modCommon.EscapeHTML(T("seuil par d~efaut ", "default threshold ")) & mThreshold & " j - " & _
           mMailsSeen & " mails - " & mContactsSeen & " " & modCommon.EscapeHTML(T("fiches externes", "external cards")) & _
           " - " & rows.Count & " " & modCommon.EscapeHTML(T("~a relancer", "to reconnect")) & "</p>"

    Dim headerRow As String
    headerRow = "<tr><th>" & modCommon.EscapeHTML(T("Contact", "Contact")) & "</th><th>" & _
                modCommon.EscapeHTML(T("Adresse", "Address")) & "</th><th>" & _
                modCommon.EscapeHTML(T("Dernier mail sortant", "Last outgoing e-mail")) & "</th><th>" & _
                modCommon.EscapeHTML(T("Jours", "Days")) & "</th><th>" & _
                modCommon.EscapeHTML(T("Seuil", "Threshold")) & "</th></tr>"

    For ki = 1 To mGroupKeys.Count
        keyStr = mGroupKeys(ki)
        keyParts = Split(keyStr, Chr$(7))
        storeName = keyParts(0)
        folderPath = keyParts(1)

        ' Lignes du groupe : en retard (tri jours decroissant) puis jamais (tri nom).
        overCnt = 0: nevCnt = 0
        ReDim overArr(1 To rows.Count + 1)
        ReDim nevArr(1 To rows.Count + 1)
        For ri = 1 To rows.Count
            rowTxt = rows(ri)
            If Left$(rowTxt, Len(keyStr) + 1) = keyStr & Chr$(7) Then
                fields = Split(rowTxt, Chr$(7))
                ' cle de tri en tete : jours (decroissant) ou nom (croissant)
                If fields(2) = "1" Then
                    nevCnt = nevCnt + 1
                    nevArr(nevCnt) = LCase$(fields(4)) & Chr$(7) & fields(4) & Chr$(7) & fields(5) & Chr$(7) & fields(6) & Chr$(7) & fields(3) & Chr$(7) & fields(7)
                Else
                    overCnt = overCnt + 1
                    overArr(overCnt) = fields(3) & Chr$(7) & fields(4) & Chr$(7) & fields(5) & Chr$(7) & fields(6) & Chr$(7) & fields(3) & Chr$(7) & fields(7)
                End If
            End If
        Next ri
        ' Tri par insertion : overArr DECROISSANT, nevArr CROISSANT.
        For ri = 2 To overCnt
            tmp = overArr(ri)
            sj = ri - 1
            Do While sj >= 1
                If overArr(sj) < tmp Then
                    overArr(sj + 1) = overArr(sj)
                    sj = sj - 1
                Else
                    Exit Do
                End If
            Loop
            overArr(sj + 1) = tmp
        Next ri
        For ri = 2 To nevCnt
            tmp = nevArr(ri)
            sj = ri - 1
            Do While sj >= 1
                If nevArr(sj) > tmp Then
                    nevArr(sj + 1) = nevArr(sj)
                    sj = sj - 1
                Else
                    Exit Do
                End If
            Loop
            nevArr(sj + 1) = tmp
        Next ri

        ' Changement de boite -> nouvelle section + nouveau tableau.
        If storeName <> lastStore Then
            If Len(lastStore) > 0 Then html = html & "</table>"
            html = html & "<h3>" & modCommon.EscapeHTML(storeName) & "</h3><table>" & headerRow
            lastStore = storeName
        End If
        html = html & "<tr class=""folder""><td colspan=""5"">" & _
               modCommon.EscapeHTML(FolderShort(storeName, folderPath)) & "</td></tr>"
        For ri = 1 To overCnt
            html = html & RowHTML(Split(overArr(ri), Chr$(7)), False)
        Next ri
        For ri = 1 To nevCnt
            html = html & RowHTML(Split(nevArr(ri), Chr$(7)), True)
        Next ri
    Next ki
    If Len(lastStore) > 0 Then html = html & "</table>"
    If mGroupKeys.Count = 0 Then _
        html = html & "<p>" & modCommon.EscapeHTML(T("Aucun contact ~a relancer sur la fen~xtre.", "No contact to reconnect with over the window.")) & "</p>"
    html = html & "</body></html>"

    outPath = modConfig.MacroVBADir() & "\RelationScan-" & Format$(Now, "yyyymmdd-hhnnss") & ".html"
    ' Ecriture UTF-8 (Print # ecrirait en ANSI : accents et emojis casses).
    ' v1.2 (F7-14) : ecriture partagee (modContactCommon.WriteUtf8).
    If Not modContactCommon.WriteUtf8(outPath, html) Then _
        Err.Raise vbObjectError + 513, MOD_NAME, T("~Ecriture du rapport impossible", "Cannot write the report")
    WriteReport = outPath
End Function

' Cellules d'une ligne (Variant tableau issu de Split) :
' cle de tri, nom, adresse, date, jours, seuil.
Private Function RowHTML(ByVal cells As Variant, ByVal isNever As Boolean) As String
    Dim daysTxt As String, dateTxt As String
    If isNever Then
        daysTxt = "<span class=""never"">" & modCommon.EscapeHTML(T("jamais (fen~xtre)", "never (window)")) & "</span>"
        dateTxt = "-"
    Else
        daysTxt = CStr(CLng(cells(4)))
        dateTxt = cells(3)
    End If
    RowHTML = "<tr><td>" & modCommon.EscapeHTML(cells(1)) & "</td><td>" & _
              modCommon.EscapeHTML(cells(2)) & "</td><td>" & dateTxt & "</td><td>" & _
              daysTxt & "</td><td>" & cells(5) & "</td></tr>"
End Function

' "\\Boite\Contacts\Reseau" -> "Contacts\Reseau" (prefixe boite retire).
Private Function FolderShort(ByVal storeName As String, ByVal folderPath As String) As String
    FolderShort = folderPath
    If Left$(folderPath, Len("\\" & storeName & "\")) = "\\" & storeName & "\" Then _
        FolderShort = Mid$(folderPath, Len("\\" & storeName & "\") + 1)
End Function

End Function
