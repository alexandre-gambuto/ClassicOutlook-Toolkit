Public Sub RunEmailAnalysis()
    DebugLog String(70, "=")
    DebugLog "=== LANCEMENT EMAIL ANALYTICS " & APP_VERSION & " (OPTIMISE) ==="
    DebugLog "Date/Heure : " & Format(Now, "dd/mm/yyyy hh:mm:ss")
    DebugLog String(70, "=")
    
    On Error GoTo ErrorHandler
    
    ' Reset compteurs
    m_TotalProcessed = 0
    m_TotalIgnored = 0
    m_TotalErrors = 0
    
    ' --- Configuration ---
    Set fConfig = New frmConfig
    fConfig.Show vbModal
    
    Dim wasCancelled As Boolean
    On Error Resume Next
    wasCancelled = fConfig.Cancelled
    If Err.Number <> 0 Then wasCancelled = True: Err.Clear
    On Error GoTo ErrorHandler
    
    If wasCancelled Then
        g_DeltaMode = False
        m_DeltaRows = Empty
        DebugLog "   -> Annule par l'utilisateur"
        Unload fConfig
        Exit Sub
    End If
    
    Dim selectedStores As Collection
    On Error Resume Next
    Set selectedStores = fConfig.SelectedStores
    On Error GoTo ErrorHandler
    
    If selectedStores Is Nothing Then
        MsgBox T("Aucune bo~ite s~electionn~ee.", "No mailbox selected."), vbExclamation, T("Attention", "Warning")
        Unload fConfig
        Exit Sub
    End If
    
    If selectedStores.Count = 0 Then
        MsgBox T("Aucune bo~ite s~electionn~ee.", "No mailbox selected."), vbExclamation, T("Attention", "Warning")
        Unload fConfig
        Exit Sub
    End If
    
    DebugLog "   -> " & selectedStores.Count & " boites selectionnees"
    
    ' V23.1 : forcer g_DebugMode = True temporairement pour voir les logs de diagnostic
    g_DebugMode = True
    
    DebugLog "   Lecture IsTestMode..."
    g_TestMode = fConfig.IsTestMode
    DebugLog "   Lecture UseExchangeResolution..."
    g_UseExchangeResolution = fConfig.UseExchangeResolution
    DebugLog "   Lecture AnalyzeLength..."
    g_AnalyzeLength = fConfig.AnalyzeLength
    DebugLog "   Lecture PeriodStart..."
    g_PeriodStart = fConfig.PeriodStart
    DebugLog "   Lecture PeriodEnd..."
    g_PeriodEnd = fConfig.PeriodEnd
    DebugLog "   Lecture DebugMode (optionnel)..."
    On Error Resume Next
    Dim dbgTmp As Boolean
    dbgTmp = fConfig.DebugMode
    On Error GoTo ErrorHandler
    DebugLog "   Toutes les lectures frmConfig OK - DebugMode final = " & dbgTmp
    g_DebugMode = dbgTmp  ' applique la valeur reelle (ou False si la propriete n'existe pas)
    
    Set dictStoreColors = CreateObject("Scripting.Dictionary")
    Set dictStoreDomains = CreateObject("Scripting.Dictionary")
    Set dictStoreEmailUser = CreateObject("Scripting.Dictionary")
    Set dictStoreKeyword = CreateObject("Scripting.Dictionary")
    Set dictStoreAddressMode = CreateObject("Scripting.Dictionary")
    Set dictStoreDateMin = CreateObject("Scripting.Dictionary")
    
    Dim storeName As Variant
    For Each storeName In selectedStores
        dictStoreColors(CStr(storeName)) = fConfig.GetStoreColor(CStr(storeName))
        dictStoreDomains(CStr(storeName)) = fConfig.GetStoreDomain(CStr(storeName))
        dictStoreEmailUser(CStr(storeName)) = LCase(CStr(fConfig.GetStoreEmailUser(CStr(storeName))))
        dictStoreKeyword(CStr(storeName)) = LCase(CStr(fConfig.GetStoreKeyword(CStr(storeName))))
        dictStoreAddressMode(CStr(storeName)) = fConfig.GetStoreAddressMode(CStr(storeName))
        dictStoreDateMin(CStr(storeName)) = fConfig.GetStoreDateMin(CStr(storeName))
    Next storeName
    
    Unload fConfig

    ' V25 : en delta, la periode est IMPOSEE par l'etat lu, pas par le
    ' formulaire. La justesse des chiffres ne doit pas dependre d'un champ
    ' que l'utilisateur pourrait modifier.
    If g_DeltaMode Then
        g_PeriodStart = g_DeltaStart
        g_PeriodEnd = g_DeltaEnd
        g_TestMode = False
        Heartbeat "DELTA : periode imposee " & Format$(g_PeriodStart, "YYYY-MM-DD") & _
                  " -> " & Format$(g_PeriodEnd, "YYYY-MM-DD")
    End If

    ' V23.4 : journal de bord sur disque (survit a un blocage / kill Excel)
    ' V23.5 : chemin host-agnostique (ThisWorkbook n'existe pas partout),
    '         range a cote de debug.log et config.ini du toolkit.
    On Error Resume Next
    Dim traceDir As String
    traceDir = ResolveDesktopPath() & "\Macro VBA\Outlook"
    If Len(Dir$(traceDir, vbDirectory)) = 0 Then traceDir = ResolveDesktopPath()
    m_TraceFile = traceDir & "\EmailAnalytics_trace.log"
    If Len(Dir$(m_TraceFile)) > 0 Then Kill m_TraceFile
    On Error GoTo ErrorHandler
    Heartbeat "=== DEBUT ANALYSE " & APP_VERSION & _
              " | ExchangeResolution=" & g_UseExchangeResolution & _
              " | AnalyzeLength=" & g_AnalyzeLength & _
              " | TestMode=" & g_TestMode & " ==="
    
    Call InitializeDictionaries
    Call InitializeFolderCaches  ' V22

    ' V25 : recharge les compteurs de l'analyse precedente AVANT le scan.
    ' A partir d'ici, les dictionnaires contiennent l'historique ; le scan
    ' ne fera qu'y ajouter les mois manquants.
    If g_DeltaMode Then Call RehydrateFromState
    
    Set fProgress = New frmProgress
    fProgress.Show vbModeless
    
    m_StartTime = Timer
    m_StartDate = Date
    
    Dim success As Boolean
    success = AnalyzeEmails(selectedStores)
    
    DebugLog "[ETAPE 9] Analyse terminee, success = " & success
    DebugLog "   Traites  : " & Format(m_TotalProcessed, "#,##0")
    DebugLog "   Ignores  : " & Format(m_TotalIgnored, "#,##0")
    DebugLog "   Erreurs  : " & Format(m_TotalErrors, "#,##0")
    
    If success And Not fProgress.Cancelled Then
        fProgress.SetComplete "Generation du rapport Excel..."
        
        Dim excelPath As String
        excelPath = GenerateExcelReport()
        
        Dim elapsedSeconds As Double
        elapsedSeconds = GetElapsedSeconds()
        
        If Len(excelPath) > 0 Then
            fProgress.SetComplete "Rapport genere : " & excelPath
            Dim speed As String
            If elapsedSeconds > 0 Then
                speed = Format(m_TotalProcessed / (elapsedSeconds / 60), "#,##0") & " emails/min"
            Else
                speed = "N/A"
            End If
            
            MsgBox T("Analyse termin~ee !", "Analysis complete!") & vbCrLf & vbCrLf & _
                   T("Emails analys~es : ", "Emails analyzed: ") & Format(m_TotalProcessed, "#,##0") & vbCrLf & _
                   T("Emails ignor~es : ", "Emails ignored: ") & Format(m_TotalIgnored, "#,##0") & vbCrLf & _
                   T("Dur~ee : ", "Duration: ") & Format(elapsedSeconds / 60, "0.0") & " min" & vbCrLf & _
                   T("Vitesse : ", "Speed: ") & speed & vbCrLf & vbCrLf & _
                   T("Fichier : ", "File: ") & excelPath, vbInformation, T("Succ~fs", "Success")
        End If
    ElseIf fProgress.Cancelled Then
        fProgress.SetError "Analyse annulee par l'utilisateur"
    End If
    
    Do While fProgress.Visible
        DoEvents
    Loop
    
    Unload fProgress
    
    Heartbeat "=== FIN ANALYSE OK | traites=" & m_TotalProcessed & _
              " | ignores=" & m_TotalIgnored & " ==="
    
    DebugLog String(70, "=")
    DebugLog "ANALYSE TERMINEE ! (" & APP_VERSION & ")"
    DebugLog "Duree : " & Format(GetElapsedSeconds() / 60, "0.0") & " minutes"
    DebugLog "Emails analyses : " & Format(m_TotalProcessed, "#,##0")
    If GetElapsedSeconds() > 0 Then
        DebugLog "Vitesse : " & Format(m_TotalProcessed / (GetElapsedSeconds() / 60), "#,##0") & " emails/min"
    End If
    DebugLog String(70, "=")

    ' V25 : le mode delta ne doit JAMAIS survivre a un parcours. Sans ce
    ' desarmement, un lancement normal enchaine derriere reutiliserait la
    ' periode delta et l'etat recharge, en silence.
    g_DeltaMode = False
    m_DeltaRows = Empty

    Exit Sub
    
ErrorHandler:
    ' V23.6 : capturer Err AVANT tout appel (Heartbeat contient
    ' On Error Resume Next, qui remet Err a zero)
    Dim errNum As Long, errDesc As String
    errNum = Err.Number
    errDesc = Err.Description
    g_DeltaMode = False
    m_DeltaRows = Empty
    Heartbeat "!!! ERREUR " & errNum & " : " & errDesc
    DebugLog "!!! ERREUR !!! " & errNum & " : " & errDesc
    MsgBox "Error " & errNum & " : " & errDesc, vbCritical, "Error"
    On Error Resume Next
    If Not fProgress Is Nothing Then Unload fProgress
    If Not fConfig Is Nothing Then Unload fConfig
End Sub

' ============================================================================
' V22 : Logging conditionnel
' ============================================================================
Private Sub DebugLog(msg As String)
    If g_DebugMode Then Debug.Print msg
End Sub

' ============================================================================
' V23.4 : Journal de bord sur disque. Ouvre / ecrit / ferme A CHAQUE ligne
' pour que la derniere ligne survive meme si Excel est fige ou tue.
' Volume : ~1 ligne par dossier + 1 ligne / 250 items -> cout negligeable.
' ============================================================================
Private Sub Heartbeat(msg As String)
    On Error Resume Next
    If Len(m_TraceFile) = 0 Then Exit Sub
    Dim f As Integer
    f = FreeFile
    Open m_TraceFile For Append As #f

'   StatMailboxN = fragment_nom ; mot-cle ; #RRGGBB [; email_utilisateur]
' Lecture en un bloc (Input$) + normalisation CRLF/CR/LF : un config.ini

Public Sub ReloadStatMailboxConfig()

Private Sub EnsureStatConfigLoaded()
    If m_StatCfgLoaded Then Exit Sub
    m_StatCfgLoaded = True
    
    Set dictStatPresets = CreateObject("Scripting.Dictionary")
    m_StatPresetCount = 0
    m_StatDefaultSet = False
    m_StatDefaultColor = RGB(136, 136, 136)
    
    On Error Resume Next
    
    Dim cfgPath As String
    cfgPath = ResolveDesktopPath() & "\Macro VBA\Outlook\config.ini"
    If Len(Dir$(cfgPath)) = 0 Then Exit Sub
    
    Dim f As Integer, raw As String
    f = FreeFile
    Open cfgPath For Input As #f
    raw = Input$(LOF(f), f)
    Close #f
    If Len(raw) = 0 Then Exit Sub
    
    raw = Replace(raw, vbCrLf, vbLf)
    raw = Replace(raw, vbCr, vbLf)
    
    Dim arrLines() As String
    arrLines = Split(raw, vbLf)
    
    Dim i As Long, s As String, eq As Long
    Dim keyName As String, keyVal As String
    Dim parts() As String, colorOK As Boolean, colorVal As Long
    
    For i = LBound(arrLines) To UBound(arrLines)
        s = Trim$(arrLines(i))
        If Len(s) > 0 Then
            If Left$(s, 1) <> ";" And Left$(s, 1) <> "[" Then
                eq = InStr(s, "=")
                If eq > 1 Then
                    keyName = LCase$(Trim$(Left$(s, eq - 1)))
                    keyVal = Trim$(Mid$(s, eq + 1))
                    
                    If keyName = "statmailboxdefault" Then
                        colorVal = HexToColor(keyVal, colorOK)
                        If colorOK Then
                            m_StatDefaultColor = colorVal
                            m_StatDefaultSet = True
                        End If
                    ElseIf Left$(keyName, 11) = "statmailbox" And Len(keyName) > 11 Then
                        If IsNumeric(Mid$(keyName, 12)) Then
                            parts = Split(keyVal, ";")
                            If UBound(parts) >= 0 Then
                                Dim fragment As String
                                fragment = Trim$(parts(0))
                                If Len(fragment) > 0 Then
                                    Dim p As Object
                                    Set p = CreateObject("Scripting.Dictionary")
                                    p("Fragment") = fragment
                                    p("Keyword") = ""
                                    p("EmailUser") = ""
                                    p("ColorSet") = False
                                    p("ColorRGB") = 0
                                    p("DateMinSet") = False
                                    If UBound(parts) >= 1 Then p("Keyword") = Trim$(parts(1))
                                    If UBound(parts) >= 2 Then
                                        colorVal = HexToColor(Trim$(parts(2)), colorOK)
                                        If colorOK Then
                                            p("ColorRGB") = colorVal
                                            p("ColorSet") = True
                                        End If
                                    End If
                                    If UBound(parts) >= 3 Then p("EmailUser") = Trim$(parts(3))
                                    If UBound(parts) >= 4 Then
                                        Dim dParsed As Date
                                        If ParseIsoDate(Trim$(parts(4)), dParsed) Then
                                            p("DateMin") = dParsed
                                            p("DateMinSet") = True
                                        End If
                                    End If
                                    
                                    ' Domain derive : email prioritaire, sinon
                                    ' fragment s'il ressemble a un domaine
                                    Dim dom As String
                                    dom = ""
                                    If InStr(CStr(p("EmailUser")), "@") > 0 Then
                                        dom = Mid$(CStr(p("EmailUser")), InStr(CStr(p("EmailUser")), "@") + 1)
                                    ElseIf InStr(fragment, ".") > 0 And InStr(fragment, "@") = 0 And InStr(fragment, " ") = 0 Then
                                        dom = fragment
                                    End If
                                    p("Domain") = LCase$(dom)
                                    
                                    m_StatPresetCount = m_StatPresetCount + 1
                                    dictStatPresets.Add CStr(m_StatPresetCount), p
                                End If
                            End If
                        End If
                    End If
                End If
            End If
        End If
    Next i
End Sub

' V23.9 : parse "YYYY-MM-DD" (ou toute date reconnue par IsDate) -> Date.
' Renvoie False si la chaine n'est pas une date valide.
Private Function ParseIsoDate(ByVal s As String, ByRef outDate As Date) As Boolean
    ParseIsoDate = False
    s = Trim$(s)
    If Len(s) = 0 Then Exit Function
    If Len(s) = 10 Then
        If Mid$(s, 5, 1) = "-" And Mid$(s, 8, 1) = "-" Then
            Dim yy As Long, mo As Long, dd As Long
            yy = Val(Left$(s, 4)): mo = Val(Mid$(s, 6, 2)): dd = Val(Mid$(s, 9, 2))
            If yy >= 1900 And mo >= 1 And mo <= 12 And dd >= 1 And dd <= 31 Then
                On Error Resume Next
                outDate = DateSerial(yy, mo, dd)
                If Err.Number = 0 Then ParseIsoDate = True
                On Error GoTo 0
                Exit Function
            End If
        End If
    End If
    If IsDate(s) Then
        outDate = CDate(s)
        ParseIsoDate = True
    End If
End Function

' "#RRGGBB" ou "RRGGBB" -> Long RGB. ok=False si format invalide.
Private Function HexToColor(ByVal h As String, ByRef ok As Boolean) As Long
    On Error GoTo Bad
    ok = False
    HexToColor = 0
    h = Trim$(h)
    If Left$(h, 1) = "#" Then h = Mid$(h, 2)
    If Len(h) <> 6 Then Exit Function
    Dim r As Long, g As Long, b As Long
    r = CLng("&H" & Mid$(h, 1, 2))
    g = CLng("&H" & Mid$(h, 3, 2))
    b = CLng("&H" & Mid$(h, 5, 2))
    HexToColor = RGB(r, g, b)
    ok = True
    Exit Function
Bad:
    ok = False
End Function

' V22 : CALCUL DU TEMPS ECOULE (gere le passage par minuit)
' ============================================================================
Private Function GetElapsedSeconds() As Double
    Dim currentTimer As Double
    currentTimer = Timer
    
    If Date > m_StartDate Then
        GetElapsedSeconds = (86400 - m_StartTime) + currentTimer + ((Date - m_StartDate - 1) * 86400)
    Else
        GetElapsedSeconds = currentTimer - m_StartTime
    End If
    
    If GetElapsedSeconds < 0 Then GetElapsedSeconds = 0
End Function

' ============================================================================
' ANALYSE DES EMAILS
' ============================================================================
Private Function AnalyzeEmails(selectedStores As Collection) As Boolean
    On Error GoTo ErrorHandler
    
    Dim olNS As Object
    Dim store As Object
    Dim rootFolder As Object
    Dim storeName As Variant
    Dim storeIndex As Long
    Dim totalEmails As Long
    
    Set olNS = Application.GetNamespace("MAPI")
    
    If fProgress Is Nothing Then
        AnalyzeEmails = False
        Exit Function
    End If
    
    fProgress.UpdateProgress 0, 0, "Comptage...", 0, selectedStores.Count
    
    ' Comptage des emails
    totalEmails = 0
    For Each storeName In selectedStores
        DebugLog "    Comptage boite: " & storeName
        Heartbeat "COMPTAGE " & storeName
        For Each store In olNS.Stores
            If store.DisplayName = CStr(storeName) Then
                totalEmails = totalEmails + CountEmailsInStore(store)
                DebugLog "      -> " & totalEmails & " emails cumules"
                Exit For
            End If
        Next store
    Next storeName
    
    DebugLog "   -> Total a traiter: " & Format(totalEmails, "#,##0")
    Heartbeat "Total a traiter: " & totalEmails
    m_TotalToProcess = totalEmails
    m_TotalProcessed = 0
    
    ' Analyse de chaque boite
    storeIndex = 0
    For Each storeName In selectedStores
        storeIndex = storeIndex + 1
        DebugLog "   [" & storeIndex & "/" & selectedStores.Count & "] " & storeName
        Heartbeat "STORE [" & storeIndex & "/" & selectedStores.Count & "] " & storeName
        
        For Each store In olNS.Stores
            If store.DisplayName = CStr(storeName) Then
                Set rootFolder = store.GetRootFolder
                Call ProcessFolder(rootFolder, CStr(storeName), storeIndex, selectedStores.Count)
                Exit For
            End If
        Next store
        
        If fProgress.Cancelled Then Exit For
    Next storeName
    
    AnalyzeEmails = Not fProgress.Cancelled
    Exit Function
    
ErrorHandler:
    DebugLog "  [AnalyzeEmails] ERREUR: " & Err.Number & " - " & Err.Description
    AnalyzeEmails = False
End Function

' ============================================================================
' V22 : IsExcludedFolder OPTIMISE avec cache dictionnaire
' Au lieu de 30+ InStr a chaque appel, on met en cache le resultat
' ============================================================================
Private Function IsExcludedFolder(folderName As String) As Boolean
    ' Cache hit ?
    If dictExcludedFolderCache.Exists(folderName) Then
        IsExcludedFolder = dictExcludedFolderCache(folderName)
        Exit Function
    End If
    
    Dim lowerName As String
    lowerName = LCase(folderName)
    
    Dim result As Boolean
    result = False
    
    ' Une seule passe de verification, regroupee par patterns
    Select Case True
        ' Corbeille / supprimes
        Case InStr(lowerName, "supprim") > 0, _
             InStr(lowerName, "deleted") > 0, _
             InStr(lowerName, "junk") > 0, _
             InStr(lowerName, "spam") > 0, _
             InStr(lowerName, "indesirable") > 0
            result = True
        ' Synchronisation / Conflits
        Case InStr(lowerName, "sync issues") > 0, _
             InStr(lowerName, "probleme") > 0 And InStr(lowerName, "sync") > 0, _
             InStr(lowerName, "personmetadata") > 0, _
             InStr(lowerName, "conflits") > 0, _
             InStr(lowerName, "conflicts") > 0, _
             InStr(lowerName, "local failures") > 0, _
             InStr(lowerName, "server failures") > 0
            result = True
        ' Recuperables / Historique
        Case InStr(lowerName, "recoverable") > 0, _
             InStr(lowerName, "recuperable") > 0, _
             InStr(lowerName, "conversation history") > 0, _
             InStr(lowerName, "historique") > 0
            result = True
        ' RSS et autres
        Case InStr(lowerName, "rss") > 0, _
             InStr(lowerName, "quick step") > 0, _
             InStr(lowerName, "social activity") > 0, _
             InStr(lowerName, "suggested contacts") > 0
            result = True
        ' Calendrier, Contacts, Taches, Notes
        Case InStr(lowerName, "calendrier") > 0, _
             InStr(lowerName, "calendar") > 0, _
             InStr(lowerName, "contacts") > 0, _
             InStr(lowerName, "taches") > 0, _
             InStr(lowerName, "tasks") > 0, _
             InStr(lowerName, "notes") > 0, _
             InStr(lowerName, "journal") > 0
            result = True
    End Select
    
    ' Mise en cache
    dictExcludedFolderCache(folderName) = result
    IsExcludedFolder = result
End Function

' ============================================================================
' V22 : BOUCLE PRINCIPALE ULTRA-OPTIMISEE
' - Boucle par index (plus rapide que For Each sur items Outlook)
' - MessageClass au lieu de TypeName
' - Cache des objets Items
' - Early exit sur exclusions
' ============================================================================
Private Sub ProcessFolder(folder As Object, storeName As String, storeIndex As Long, storeCount As Long)
    On Error Resume Next
    
    If IsExcludedFolder(folder.Name) Then Exit Sub
    If fProgress.Cancelled Then Exit Sub
    
    ' Cache du flag "dossier d'envoi"
    Dim isSentFolder As Boolean
    isSentFolder = IsSentFolderName(folder.Name)
    
    ' V23.4 : trace AVANT l'acces aux Items (un dossier casse peut bloquer ici)
    Heartbeat "DOSSIER " & folder.FolderPath
    
    ' V22 : Cache des Items dans une variable locale
    Dim folderItems As Object
    Set folderItems = folder.Items
    
    Dim itemCount As Long
    itemCount = folderItems.Count
    
    If itemCount > 0 Then
        Heartbeat "  items=" & itemCount & " | deja traites=" & m_TotalProcessed
        Dim item As Object
        Dim counter As Long: counter = 0
        Dim localProcessed As Long: localProcessed = 0
        Dim i As Long
        Dim msgClass As String
        
        ' V22 : Boucle par INDEX (plus rapide et plus robuste)
        For i = 1 To itemCount
            If fProgress.Cancelled Then Exit Sub
            
            ' V23.4 : position courante (index BRUT, y compris items ignores)
            If i Mod PROGRESS_UPDATE_FREQUENCY = 0 Then
                Heartbeat "  idx " & i & " / " & itemCount & " | total=" & m_TotalProcessed
            End If
            
            Set item = Nothing
            Set item = folderItems.Item(i)
            
            If Not item Is Nothing Then
                ' V22 : Test via MessageClass (plus rapide que TypeName)
                msgClass = ""
                msgClass = item.MessageClass
                
                ' IPM.Note = Email standard, IPM.Note.* = variantes d'emails
                If Left$(msgClass, 8) = "IPM.Note" Then
                    counter = counter + 1
                    
                    If (Not g_TestMode) Or (counter Mod TEST_SAMPLING = 0) Then
                        Call ProcessEmail(item, storeName, isSentFolder)
                        m_TotalProcessed = m_TotalProcessed + 1
                        localProcessed = localProcessed + 1
                        
                        ' V22 : Deux frequences separees
                        ' - Progress update plus frequent (250)
                        ' - DoEvents moins frequent (500) car plus couteux
                        If localProcessed Mod PROGRESS_UPDATE_FREQUENCY = 0 Then
                            fProgress.UpdateProgress m_TotalProcessed, m_TotalToProcess, storeName, storeIndex, storeCount
                            
                            Dim elapsed As Double, remaining As Long
                            elapsed = GetElapsedSeconds()
                            If elapsed > 0 And m_TotalProcessed > 0 Then
                                remaining = CLng((elapsed / m_TotalProcessed) * (m_TotalToProcess - m_TotalProcessed))
                                fProgress.SetTimeInfo CLng(elapsed), remaining
                            End If
                        End If
                        
                        If localProcessed Mod DOEVENTS_FREQUENCY = 0 Then
                            DoEvents
                        End If
                    End If
                Else
                    m_TotalIgnored = m_TotalIgnored + 1
                End If
            End If
        Next i
    End If
    
    Set folderItems = Nothing
    
    ' Recursion sur sous-dossiers
    Dim subFolder As Object
    For Each subFolder In folder.Folders
        Call ProcessFolder(subFolder, storeName, storeIndex, storeCount)
    Next subFolder
    
    On Error GoTo 0
End Sub

' ============================================================================
' V22 : IsSentFolderName avec cache
' ============================================================================
Private Function IsSentFolderName(folderName As String) As Boolean
    If dictSentFolderCache.Exists(folderName) Then
        IsSentFolderName = dictSentFolderCache(folderName)
        Exit Function
    End If
    
    Dim lowerName As String
    lowerName = LCase(folderName)
    
    Dim result As Boolean
    result = False
    
    Select Case lowerName
        Case "elements envoyes", "envoyes", "sent items", "sent", "sent mail", _
             "elements envoyes [", "envoyes [", _
             Chr(233) & "l" & Chr(233) & "ments envoy" & Chr(233) & "s"  ' "Elements envoyes"
            result = True
    End Select
    
    ' Fallback : contient "envoy" ou "sent "
    If Not result Then
        If InStr(lowerName, "envoy") > 0 Or InStr(lowerName, "sent") > 0 Then
            ' Verification fine pour eviter faux positifs
            If InStr(lowerName, "envoy") > 0 And Len(lowerName) < 30 Then result = True
            If InStr(lowerName, "sent") > 0 And Len(lowerName) < 20 Then result = True
        End If
    End If
    
    dictSentFolderCache(folderName) = result
    IsSentFolderName = result
End Function

' ============================================================================
' V22 : TRAITEMENT D'UN EMAIL - VERSION OPTIMISEE
' - Cache de TOUTES les proprietes COM en variables locales
' - Pre-calcul des cles temporelles via Year/Month/Day (pas Format)
' - Gestion erreur fine
' ============================================================================
Private Sub ProcessEmail(mail As Object, storeName As String, Optional isSentFolder As Boolean = False)
    On Error GoTo ErrHandler
    
    ' === V22 : Cache des proprietes COM en variables locales ===
    Dim subj As String
    Dim senderName As String, senderAddr As String, senderEmailType As String
    Dim emailDate As Date
    Dim bodyText As String
    
    subj = Trim$(CStr(mail.Subject))
    senderName = CStr(mail.SenderName)
    senderAddr = CStr(mail.SenderEmailAddress)
    senderEmailType = CStr(mail.SenderEmailType)
    emailDate = mail.ReceivedTime
    
    ' === EXCLUSION : Emails vides ou sans sujet ===
    If Len(subj) = 0 Then
        If Len(Trim$(senderName)) = 0 And Len(Trim$(senderAddr)) = 0 Then
            m_TotalIgnored = m_TotalIgnored + 1
            Exit Sub
        End If
    End If
    
    ' === EXCLUSION : Journaux de synchronisation (verification UNIQUE avec LCase cache) ===
    Dim subjLower As String
    subjLower = LCase$(subj)
    If InStr(subjLower, "journal de synchronisation") > 0 Or _
       InStr(subjLower, "sync issues") > 0 Or _
       InStr(subjLower, "synchronization log") > 0 Or _
       InStr(subjLower, "probleme de synchronisation") > 0 Then
        m_TotalIgnored = m_TotalIgnored + 1
        Exit Sub
    End If
    
    ' === FILTRES DATES ===
    If emailDate < g_PeriodStart Or emailDate > g_PeriodEnd Then
        m_TotalIgnored = m_TotalIgnored + 1
        Exit Sub
    End If
    
    Dim yearNum As Long
    yearNum = Year(emailDate)
    If yearNum < 2000 Then
        m_TotalIgnored = m_TotalIgnored + 1
        Exit Sub
    End If
    
    ' Date minimum par boite
    If dictStoreDateMin.Exists(storeName) Then
        Dim dateMin As Date
        dateMin = dictStoreDateMin(storeName)
        If dateMin > DateSerial(1990, 1, 1) Then
            If emailDate < dateMin Then
                m_TotalIgnored = m_TotalIgnored + 1
                Exit Sub
            End If
        End If
    End If
    
    ' === Resolution adresse expediteur ===
    If Len(senderAddr) = 0 Then
        If Not mail.Sender Is Nothing Then
            senderAddr = ResolveEmailAddressFromSender(mail.Sender, senderEmailType, senderAddr)
        End If
    Else
        If senderEmailType = "SMTP" Then
            senderAddr = LCase$(senderAddr)
        ElseIf g_UseExchangeResolution And senderEmailType = "EX" Then
            If Not mail.Sender Is Nothing Then
                senderAddr = ResolveEmailAddressFromSender(mail.Sender, senderEmailType, senderAddr)
            End If
        End If
    End If
    
    ' Detection Sent/Received
    Dim isSent As Boolean
    If isSentFolder Then
        isSent = True
    Else
        isSent = IsSentByUser(senderAddr, senderName, storeName)
    End If
    
    Dim direction As String
    If isSent Then direction = "Sent" Else direction = "Received"
    
    ' === STATS PAR BOITE ===
    IncrementDict dictStoreStats, storeName & "|" & direction
    IncrementDict dictStoreStats, storeName & "|Total"

    ' V24 : A / Cc / Indirect - uniquement sur les mails RECUS (sur un envoi
    ' la question n'a pas de sens : l'utilisateur est l'expediteur).
    If Not isSent Then
        IncrementDict dictToCc, storeName & "|" & Format$(emailDate, "YYYY-MM") & _
                                "|" & RecipientRole(mail, storeName)
    End If
    
    ' === V22 : Pre-calcul OPTIMISE des cles temporelles ===
    ' Format() est tres lent ; Year/Month/Day + concatenation sont 10x plus rapides
    Dim monthNum As Long, dayNum As Long, hourNum As Long, dowNum As Long
    monthNum = Month(emailDate)
    dayNum = Day(emailDate)
    hourNum = Hour(emailDate)
    dowNum = Weekday(emailDate, vbMonday)
    
    Dim yearStr As String, monthStr As String, dayStr As String, hourStr As String
    yearStr = CStr(yearNum)
    monthStr = Format$(monthNum, "00")
    dayStr = Format$(dayNum, "00")
    hourStr = Format$(hourNum, "00")
    
    Dim monthKey As String, dayKey As String
    monthKey = yearStr & "-" & monthStr
    dayKey = monthKey & "-" & dayStr
    
    Dim weekKey As String, yearKey As String, quarterKey As String, semesterKey As String
    yearKey = yearStr
    weekKey = yearStr & "-W" & Format$(DatePart("ww", emailDate, vbMonday), "00")
    quarterKey = yearStr & "-Q" & ((monthNum - 1) \ 3 + 1)
    If monthNum <= 6 Then
        semesterKey = yearStr & "-S1"
    Else
        semesterKey = yearStr & "-S2"
    End If
    
    Dim hourKey As String, dowKey As String
    hourKey = hourStr
    dowKey = CStr(dowNum)
    
    ' === STATS TEMPORELLES ===
    IncrementDict dictMonthly, monthKey & "|" & storeName & "|" & direction
    IncrementDict dictMonthly, monthKey & "|" & storeName & "|Total"
    IncrementDict dictMonthly, monthKey & "|_TOTAL|" & direction
    IncrementDict dictMonthly, monthKey & "|_TOTAL|Total"
    
    IncrementDict dictWeekly, weekKey & "|" & direction
    IncrementDict dictWeekly, weekKey & "|Total"
    
    IncrementDict dictDaily, dayKey & "|" & direction
    IncrementDict dictDaily, dayKey & "|Total"
    
    IncrementDict dictQuarterly, quarterKey & "|" & direction
    IncrementDict dictQuarterly, quarterKey & "|Total"
    
    IncrementDict dictYearly, yearKey & "|" & direction
    IncrementDict dictYearly, yearKey & "|Total"
    
    IncrementDict dictSemesterly, semesterKey & "|" & storeName & "|" & direction
    IncrementDict dictSemesterly, semesterKey & "|" & storeName & "|Total"
    
    IncrementDict dictHourly, hourKey & "|" & direction
    IncrementDict dictHourly, hourKey & "|Total"
    
    IncrementDict dictDayOfWeek, dowKey & "|" & direction
    IncrementDict dictDayOfWeek, dowKey & "|Total"
    
    ' === HEATMAPS ===
    Dim heatKey As String
    heatKey = dowKey & "|" & hourNum
    If isSent Then
        IncrementDict dictHeatmapSent, heatKey
    Else
        IncrementDict dictHeatmapReceived, heatKey
    End If
    IncrementDict dictHeatmapTotal, heatKey
    
    ' ============================================================================
    ' LOGIQUE INTERNE / EXTERNE / INCONNU
    ' ============================================================================
    Dim storeDomain As String
    storeDomain = ""
    If dictStoreDomains.Exists(storeName) Then storeDomain = LCase$(CStr(dictStoreDomains(storeName)))
    
    Dim emailClassification As String
    emailClassification = "Unknown"
    
    Dim displayRecip As String  ' V22 : pour raw data en fin
    Dim displaySender As String
    Dim recipName As String, recipAddr As String
    
    If isSent Then
        ' === EMAIL ENVOYE : Analyser les destinataires ===
        Dim recipients As Object
        Set recipients = mail.Recipients
        
        Dim recipCount As Long: recipCount = 0
        Dim hasExternal As Boolean: hasExternal = False
        Dim hasInternal As Boolean: hasInternal = False
        Dim allUnknown As Boolean: allUnknown = True
        
        Dim recip As Object
        Dim resolvedRecipAddr As String
        Dim recipDomain As String
        
        For Each recip In recipients
            recipCount = recipCount + 1
            
            If g_UseExchangeResolution Then
                resolvedRecipAddr = ForceResolveRecipientSMTP(recip, recipName)
            Else
                recipName = CStr(recip.Name)
                resolvedRecipAddr = GetSimpleRecipientAddress(recip)
            End If
            recipAddr = resolvedRecipAddr
            
            displayRecip = GetBestContactDisplay(recipAddr, recipName)
            
            If Len(displayRecip) > 0 Then
                IncrementDict dictSentTo, displayRecip
            End If
            
            If IsValidSMTPAddress(resolvedRecipAddr) Then
                allUnknown = False
                recipDomain = LCase$(ExtractDomain(resolvedRecipAddr))
                
                If Len(storeDomain) > 0 And Len(recipDomain) > 0 Then
                    If recipDomain = storeDomain Then
                        hasInternal = True
                        If Len(displayRecip) > 0 Then IncrementDict dictSentToInternal, displayRecip
                    Else
                        hasExternal = True
                        If Len(displayRecip) > 0 Then IncrementDict dictSentToExternal, displayRecip
                    End If
                Else
                    hasExternal = True
                    If Len(displayRecip) > 0 Then IncrementDict dictSentToExternal, displayRecip
                End If
            Else
                If Len(displayRecip) > 0 Then IncrementDict dictSentToUnknown, displayRecip
            End If
        Next recip
        
        Set recipients = Nothing
        
        If allUnknown Then
            emailClassification = "Unknown"
        ElseIf hasExternal Then
            emailClassification = "External"
        Else
            emailClassification = "Internal"
        End If
        
        If recipCount > 0 Then
            IncrementDict dictNbRecipients, CStr(recipCount)
        End If
        
        ' === Length message envoye (V20/V22) ===
        If g_AnalyzeLength Then
            bodyText = CStr(mail.Body)
            Dim msgLen As Long
            msgLen = ExtractCurrentMessageLength(bodyText)
            If msgLen > 0 Then
                Dim lenKey As String
                ' V22 : Cle numerique simple au lieu de Format(...) - plus rapide
                lenKey = CStr(dictMsgLength.Count + 1)
                dictMsgLength.Add lenKey, msgLen
                
                If Not dictMsgLengthByStore.Exists(storeName) Then
                    Dim newStoreDict As Object
                    Set newStoreDict = CreateObject("Scripting.Dictionary")
                    Set dictMsgLengthByStore(storeName) = newStoreDict
                End If
                Dim storeDict As Object
                Set storeDict = dictMsgLengthByStore(storeName)
                storeDict.Add CStr(storeDict.Count + 1), msgLen
            End If
        End If
        
    Else
        ' === EMAIL RECU : Analyser l'expediteur ===
        displaySender = GetBestContactDisplay(senderAddr, senderName)
        
        If Len(displaySender) > 0 Then
            IncrementDict dictReceivedFrom, displaySender
        End If
        
        Dim resolvedSenderAddr As String
        If g_UseExchangeResolution Then
            resolvedSenderAddr = ForceResolveSenderSMTP(mail)
        Else
            resolvedSenderAddr = senderAddr
        End If
        
        If IsValidSMTPAddress(resolvedSenderAddr) Then
            Dim senderDomain As String
            senderDomain = LCase$(ExtractDomain(resolvedSenderAddr))
            
            If Len(storeDomain) > 0 And Len(senderDomain) > 0 Then
                If senderDomain = storeDomain Then
                    emailClassification = "Internal"
                    If Len(displaySender) > 0 Then IncrementDict dictReceivedFromInternal, displaySender
                Else
                    emailClassification = "External"
                    If Len(displaySender) > 0 Then IncrementDict dictReceivedFromExternal, displaySender
                End If
            Else
                emailClassification = "External"
                If Len(displaySender) > 0 Then IncrementDict dictReceivedFromExternal, displaySender
            End If
        Else
            emailClassification = "Unknown"
            If Len(displaySender) > 0 Then IncrementDict dictReceivedFromUnknown, displaySender
        End If
    End If
    
    ' Stats Internal/External
    IncrementDict dictInternalExternal, storeName & "|" & emailClassification
    IncrementDict dictInternalExternal, "_TOTAL|" & emailClassification
    
    ' === Raw data (5000 premiers) ===
    If dictRawData.Count < MAX_RAW_DATA Then
        Dim rawKey As String, rawVal As String
        rawKey = CStr(dictRawData.Count + 1)  ' V22 : plus de Format()
        
        Dim dateStr As String
        ' V22 : Concatenation directe au lieu de Format - beaucoup plus rapide
        dateStr = yearStr & "-" & monthStr & "-" & dayStr & " " & hourStr & ":" & Format$(Minute(emailDate), "00")
        
        rawVal = storeName & "|" & dateStr & "|" & direction & "|"
        If isSent Then
            rawVal = rawVal & Left$(GetBestContactDisplay(recipAddr, recipName), 50)
        Else
            rawVal = rawVal & Left$(displaySender, 50)
        End If
        rawVal = rawVal & "|" & Left$(subj, 100)
        dictRawData.Add rawKey, rawVal
    End If
    
    Exit Sub
    
ErrHandler:
    ' V22 : Erreur sur un email = on passe au suivant, on n'interrompt pas
    m_TotalErrors = m_TotalErrors + 1
    If g_DebugMode And (m_TotalErrors <= 10) Then
        Debug.Print "  [ProcessEmail] Erreur " & Err.Number & " : " & Err.Description & " (email ignore)"
    End If
End Sub

Private Function ResolveEmailAddressFromSender(sender As Object, emailType As String, emailAddr As String) As String
    On Error Resume Next
    
    If g_UseExchangeResolution And emailType = "EX" And Len(emailAddr) > 0 Then
        If dictResolveCache Is Nothing Then Set dictResolveCache = CreateObject("Scripting.Dictionary")
        If dictResolveCache.Exists(emailAddr) Then
            ResolveEmailAddressFromSender = dictResolveCache(emailAddr)
            Exit Function
        End If
        ResolveEmailAddressFromSender = ResolveSenderDirect(sender, emailType, emailAddr)
        dictResolveCache(emailAddr) = ResolveEmailAddressFromSender
        Exit Function
    End If
    
    ResolveEmailAddressFromSender = ResolveSenderDirect(sender, emailType, emailAddr)
End Function

Private Function ResolveSenderDirect(sender As Object, emailType As String, emailAddr As String) As String
    On Error Resume Next
    
    If emailType = "SMTP" Then
        ResolveSenderDirect = LCase$(emailAddr)
        Exit Function
    End If
    
    If g_UseExchangeResolution Then
        If emailType = "EX" Then
            Dim exUser As Object
            Set exUser = sender.GetExchangeUser
            If Not exUser Is Nothing Then
                If Len(exUser.PrimarySmtpAddress) > 0 Then
                    ResolveSenderDirect = LCase$(CStr(exUser.PrimarySmtpAddress))
                    Exit Function
                End If
            End If
        End If
    End If
    
    If InStr(emailAddr, "@") > 0 Then
        ResolveSenderDirect = LCase$(emailAddr)
    Else
        ResolveSenderDirect = LCase$(CStr(sender.Name))
    End If
End Function

'============================================================================
' V24 - ROLE DU DESTINATAIRE : A / Cc / Indirect
'
' Sur un mail RECU, dit si l'utilisateur figure en A, en Cc, ou dans aucun
' des deux ("Indirect" : liste de diffusion, boite partagee, Cci).
'
' On lit les CHAINES .To et .CC, pas la collection Recipients : elles sont
' deja resolues et ne coutent rien, alors qu'iterer Recipients rouvre l'item
' et ramene le probleme des adresses au format EX. Contrepartie : la
' reconnaissance repose sur les memes reperes que IsSentByUser - l'email de
' la boite (4e champ de StatMailboxN) et le mot-cle (2e champ). Un panier
' "Indirect" anormalement gros signalerait que ces reperes sont mal regles.
'
' Les trois categories totalisent 100 % : tout mail recu tombe dans une et
' une seule.
'============================================================================
Private Function RecipientRole(mail As Object, storeName As String) As String
    On Error Resume Next
    RecipientRole = "Indirect"

    Dim toStr As String, ccStr As String
    toStr = LCase$(CStr(mail.To))
    ccStr = LCase$(CStr(mail.CC))
    If Len(toStr) = 0 And Len(ccStr) = 0 Then Exit Function

    Dim userEmail As String, motCle As String
    If dictStoreEmailUser.Exists(storeName) Then userEmail = LCase$(CStr(dictStoreEmailUser(storeName)))
    If dictStoreKeyword.Exists(storeName) Then motCle = LCase$(CStr(dictStoreKeyword(storeName)))

    If MatchesIdentity(toStr, userEmail, motCle) Then
        RecipientRole = "To"
    ElseIf MatchesIdentity(ccStr, userEmail, motCle) Then
        RecipientRole = "Cc"
    End If
End Function

Private Function MatchesIdentity(hay As String, userEmail As String, motCle As String) As Boolean
    MatchesIdentity = False
    If Len(hay) = 0 Then Exit Function
    If Len(userEmail) > 0 Then
        If InStr(hay, userEmail) > 0 Then MatchesIdentity = True: Exit Function
    End If
    If Len(motCle) > 0 Then
        If InStr(hay, motCle) > 0 Then MatchesIdentity = True
    End If
End Function

Private Function IsSentByUser(senderAddr As String, senderName As String, storeName As String) As Boolean

    Dim modeAdresse As Boolean

        Dim userEmail As String
        If dictStoreEmailUser.Exists(storeName) Then userEmail = CStr(dictStoreEmailUser(storeName))
        If Len(userEmail) > 0 And Len(senderAddr) > 0 Then

                IsSentByUser = True
            End If
        End If
    End If
End Function

' ============================================================================
' EXTRACTION LONGUEUR MESSAGE - optimise V22
' ============================================================================
Private Function ExtractCurrentMessageLength(bodyText As String) As Long
    On Error Resume Next
    ExtractCurrentMessageLength = 0
    If Len(bodyText) = 0 Then Exit Function
    
    Dim msgText As String
    msgText = bodyText
    
    ' V22 : Test les marqueurs les plus frequents en premier
    Dim minPos As Long: minPos = Len(msgText) + 1
    Dim pos As Long
    
    ' Marqueur 1 : "-----Original Message-----" / "-----Message d'origine-----"
    pos = InStr(1, msgText, "-----", vbTextCompare)
    If pos > 10 And pos < minPos Then
        ' Verifier que c'est bien un marqueur d'en-tete
        Dim checkStr As String
        checkStr = Mid$(msgText, pos, 30)
        If InStr(1, checkStr, "Message", vbTextCompare) > 0 Or InStr(1, checkStr, "Original", vbTextCompare) > 0 Then
            minPos = pos
        End If
    End If
    
    ' Marqueur 2 : "________________________________" (32 underscores)
    pos = InStr(1, msgText, "________________________________", vbTextCompare)
    If pos > 10 And pos < minPos Then minPos = pos
    
    ' Marqueur 3 : "From:" en debut de ligne
    pos = InStr(1, msgText, vbCrLf & "From:", vbTextCompare)
    If pos > 10 And pos < minPos Then minPos = pos
    
    ' Marqueur 4 : "De :" en debut de ligne
    pos = InStr(1, msgText, vbCrLf & "De :", vbTextCompare)
    If pos > 10 And pos < minPos Then minPos = pos
    
    ' Marqueur 5 : "Le " suivi de date (debut de reponse francaise)
    pos = InStr(1, msgText, vbCrLf & "Le ", vbTextCompare)
    If pos > 10 And pos < minPos Then minPos = pos
    
    If minPos <= Len(msgText) Then
        msgText = Left$(msgText, minPos - 1)
    End If
    
    ' Nettoyer
    msgText = Replace(msgText, vbCrLf, " ")
    msgText = Replace(msgText, vbCr, " ")
    msgText = Replace(msgText, vbLf, " ")
    msgText = Trim$(msgText)
    
    ExtractCurrentMessageLength = Len(msgText)
End Function

Private Function IsValidSMTPAddress(addr As String) As Boolean

' ----------------------------------------------------------------------------
' ResolveDesktopPath (V23.2)
' Real Desktop folder, OneDrive Known-Folder-Move aware. USERPROFILE\Desktop
' is WRONG on redirected machines (invisible legacy folder, or missing).
' First existing candidate wins; falls back to the user profile root.
' ----------------------------------------------------------------------------
Private Function ResolveDesktopPath() As String
    On Error Resume Next
    Dim cands As Variant
    Dim i As Long
    Dim p As String
    cands = Array( _
        Environ("OneDriveCommercial") & "\Bureau", _
        Environ("OneDriveCommercial") & "\Desktop", _
        Environ("OneDrive") & "\Bureau", _
        Environ("OneDrive") & "\Desktop", _
        Environ("USERPROFILE") & "\OneDrive\Bureau", _
        Environ("USERPROFILE") & "\OneDrive\Desktop", _
        Environ("USERPROFILE") & "\Bureau", _
        Environ("USERPROFILE") & "\Desktop")
    For i = LBound(cands) To UBound(cands)
        p = CStr(cands(i))
        ' Skip candidates whose environment variable was empty ("\Bureau").
        If Left$(p, 1) <> "\" Then
            If Dir(p, vbDirectory) <> "" Then
                ResolveDesktopPath = p
                Exit Function
            End If
        End If
    Next i
    ResolveDesktopPath = Environ("USERPROFILE")
End Function

' ============================================================================
' UTILITAIRE : Tri decroissant
' ============================================================================
Public Function GetSortedKeysByValueDesc(dict As Object) As Variant
    Dim keys As Variant, vals As Variant
    Dim i As Long, j As Long
    Dim tempK As Variant, tempV As Variant, count As Long
    
    count = dict.Count
    If count = 0 Then
        GetSortedKeysByValueDesc = Array()
        Exit Function
    End If
    
    ReDim keys(0 To count - 1)
    ReDim vals(0 To count - 1)
    
    i = 0
    Dim k As Variant
    For Each k In dict.Keys
        keys(i) = k
        vals(i) = dict(k)
        i = i + 1
    Next k
    
    ' Tri par insertion pour petites listes, tri rapide pour grandes
    If count < 30 Then
        For i = 0 To count - 2
            For j = i + 1 To count - 1
                If vals(i) < vals(j) Then
                    tempK = keys(i): tempV = vals(i)
                    keys(i) = keys(j): vals(i) = vals(j)
                    keys(j) = tempK: vals(j) = tempV
                End If
            Next j
        Next i
    Else
        Call QuickSortByValueDesc(keys, vals, 0, count - 1)
    End If
    
    GetSortedKeysByValueDesc = keys
End Function

' V22 : QuickSort pour tri rapide des grandes listes
Private Sub QuickSortByValueDesc(ByRef keys As Variant, ByRef vals As Variant, _
                                   ByVal low As Long, ByVal high As Long)
    Dim i As Long, j As Long
    Dim pivot As Variant, tempK As Variant, tempV As Variant
    
    If low < high Then
        pivot = vals((low + high) \ 2)
        i = low
        j = high
        
        Do
            Do While vals(i) > pivot: i = i + 1: Loop
            Do While vals(j) < pivot: j = j - 1: Loop
            
            If i <= j Then
                tempK = keys(i): tempV = vals(i)
                keys(i) = keys(j): vals(i) = vals(j)
                keys(j) = tempK: vals(j) = tempV
                i = i + 1
                j = j - 1
            End If
        Loop Until i > j
        
        If low < j Then Call QuickSortByValueDesc(keys, vals, low, j)
        If i < high Then Call QuickSortByValueDesc(keys, vals, i, high)
    End If
End Sub

' V22 : Ecriture des headers en bloc + coloration
Private Sub WriteEvolutionHeaders(ws As Object, row As Long, startCol As Long, firstHeader As String, _
                                    boitesList As Variant, nMailboxes As Long)

' V22 : Ecriture des donnees d'evolution en bloc (tres rapide)
Private Function WriteEvolutionDataBlock(ws As Object, startRow As Long, sortedKeys As Variant, _
                                           boitesList As Variant, dict As Object, suffix As String, _
                                           lastCol As Long) As Long
    Dim nRows As Long
    If IsArray(sortedKeys) Then
        nRows = UBound(sortedKeys) - LBound(sortedKeys) + 1
    Else
        nRows = 0
    End If
    
    If nRows = 0 Then
        WriteEvolutionDataBlock = startRow - 1
        Exit Function
    End If
    
    Dim nMailboxes As Long
    nMailboxes = UBound(boitesList) - LBound(boitesList) + 1
    
    Dim dataArr() As Variant
    ReDim dataArr(1 To nRows, 1 To nMailboxes + 1)
    
    Dim i As Long, j As Long
    For i = 0 To nRows - 1
        dataArr(i + 1, 1) = sortedKeys(i)
        For j = 0 To nMailboxes - 1
            dataArr(i + 1, j + 2) = GetDictValue(dict, CStr(sortedKeys(i)) & "|" & CStr(boitesList(j)) & "|" & suffix)
        Next j
    Next i
    
    ws.Range(ws.Cells(startRow, 1), ws.Cells(startRow + nRows - 1, lastCol)).Value = dataArr
    
    WriteEvolutionDataBlock = startRow + nRows - 1
End Function

' Helper pour creer un graphique d'evolution avec couleurs coherentes
Private Sub CreateEvolutionChart(ws As Object, headerRow As Long, lastRow As Long, lastCol As Long, _
                                  boitesList As Variant, chartLeft As Long, chartTop As Long, chartTitle As String, _
                                  ByVal asLine As Boolean)
    On Error Resume Next
    
    If lastRow <= headerRow + 1 Then Exit Sub
    
    Dim chart As Object
    Set chart = ws.ChartObjects.Add(Left:=chartLeft, Top:=chartTop, Width:=550, Height:=280)
    
    With chart.Chart
        .ChartType = IIf(asLine, 4, 51)  ' V23.8 : 4 = courbe pour le mensuel
        .SetSourceData ws.Range(ws.Cells(headerRow, 1), ws.Cells(lastRow, lastCol))
        .HasTitle = True
        .ChartTitle.Text = chartTitle
        .HasLegend = True
        .Legend.Position = -4107
        
        Dim serIdx As Long, boiteIdx As Long
        boiteIdx = 0
        For serIdx = 1 To .SeriesCollection.Count
            If boiteIdx <= UBound(boitesList) Then
                If asLine Then
                    .SeriesCollection(serIdx).Format.Line.ForeColor.RGB = GetStoreColor(CStr(boitesList(boiteIdx)))
                    .SeriesCollection(serIdx).Format.Line.Weight = 3
                Else
                    .SeriesCollection(serIdx).Format.Fill.ForeColor.RGB = GetStoreColor(CStr(boitesList(boiteIdx)))
                End If
                boiteIdx = boiteIdx + 1
            End If
        Next serIdx
    End With
    
    On Error GoTo 0
End Sub

Private Sub AppendEvolutionGlobalSection(ws As Object, ByVal startRow As Long, _
                                         ByVal chartLeft As Long, ByVal chartTop As Double, _
                                         ByVal chartHeight As Long, ByVal chartSpacing As Long)
    On Error Resume Next

    Dim rowM As Long, rowS As Long, rowY As Long
    Dim hdrM As Long, hdrS As Long, hdrY As Long

    r = startRow
    ws.Cells(r, 1).Value = T("SYNTH~fSE GLOBALE (toutes bo~ites)", "GLOBAL SUMMARY (all mailboxes)")
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 1).Font.Size = 12
    r = r + 2

    arr = GlobalPeriodArray("month")
    hdrM = r + 1
    r = WriteGlobalTable(ws, r, T("Par mois", "By month"), T("Mois", "Month"), arr)
    rowM = r: r = r + 3

    arr = GlobalPeriodArray("quarter")
    r = WriteGlobalTable(ws, r, T("Par trimestre", "By quarter"), T("Trimestre", "Quarter"), arr)
    r = r + 3

    hdrS = r + 1
    r = WriteGlobalTable(ws, r, T("Par semestre", "By half-year"), T("Semestre", "Half-year"), arr)
    rowS = r: r = r + 3

    arr = GlobalPeriodArray("year")
    hdrY = r + 1

    rowY = r

    Dim topPts As Double
    topPts = ws.Cells(startRow, 1).Top
    If topPts < chartTop Then topPts = chartTop

    If rowM > hdrM Then
        Call AddStacked100Chart(ws, chartLeft, topPts, T("Proportion envoy~es/re" & Chr(231) & "us par mois", "Sent/received share by month"), hdrM, rowM)
        topPts = topPts + chartHeight + chartSpacing
    End If
    If rowS > hdrS Then
        Call AddStacked100Chart(ws, chartLeft, topPts, T("Proportion envoy~es/re" & Chr(231) & "us par semestre", "Sent/received share by half-year"), hdrS, rowS)
        topPts = topPts + chartHeight + chartSpacing
    End If
    If rowY > hdrY Then
        Call AddStacked100Chart(ws, chartLeft, topPts, T("Proportion envoy~es/re" & Chr(231) & "us par ann~ee", "Sent/received share by year"), hdrY, rowY)
    End If

    On Error GoTo 0
End Sub

' Ecrit une table globale (titre + entete + donnees). Renvoie la derniere
' ligne de DONNEES ecrite (ou la ligne d'entete si aucune donnee).
Private Function WriteGlobalTable(ws As Object, ByVal startRow As Long, _
                                  ByVal tableTitle As String, ByVal periodHdr As String, _
                                  arr As Variant) As Long
    ws.Cells(startRow, 1).Value = tableTitle
    ws.Cells(startRow, 1).Font.Bold = True

    Dim hdr As Long: hdr = startRow + 1
    ws.Cells(hdr, 1).Value = periodHdr
    ws.Cells(hdr, 2).Value = T("Envoy~es", "Sent")
    ws.Cells(hdr, 3).Value = T("Re" & Chr(231) & "us", "Received")
    ws.Cells(hdr, 4).Value = T("Total", "Total")
    ws.Range(ws.Cells(hdr, 1), ws.Cells(hdr, 4)).Font.Bold = True

    If IsEmpty(arr) Then
        WriteGlobalTable = hdr
        Exit Function
    End If

    Dim n As Long: n = UBound(arr, 1)
    If n < 1 Then
        WriteGlobalTable = hdr
        Exit Function
    End If

    ws.Range(ws.Cells(hdr + 1, 1), ws.Cells(hdr + n, 4)).Value = arr
    ws.Range(ws.Cells(hdr + 1, 2), ws.Cells(hdr + n, 4)).NumberFormat = "#,##0"
    WriteGlobalTable = hdr + n
End Function

' Renvoie un tableau (1..n, 1..4) = libelle, Envoyes, Recus, Total agrege
' sur toutes les boites pour la granularite demandee. Empty si aucune donnee.
Private Function GlobalPeriodArray(ByVal kind As String) As Variant
    Dim srcDict As Object
    Select Case kind
        Case "month": Set srcDict = dictMonthly
        Case "quarter": Set srcDict = dictQuarterly
        Case "semester": Set srcDict = dictSemesterly
        Case "year": Set srcDict = dictYearly
    End Select
    If srcDict Is Nothing Then Exit Function

    Dim setKeys As Object
    Set setKeys = CreateObject("Scripting.Dictionary")
    Dim k As Variant, parts As Variant, pk As String
    For Each k In srcDict.Keys
        parts = Split(CStr(k), "|")
        pk = CStr(parts(0))
        If kind = "month" Then
            If UBound(parts) >= 2 Then
                If CStr(parts(1)) = "_TOTAL" Then
                    If Not setKeys.Exists(pk) Then setKeys.Add pk, True
                End If
            End If
        Else
            If Not setKeys.Exists(pk) Then setKeys.Add pk, True
        End If
    Next k
    If setKeys.Count = 0 Then Exit Function

    Dim sortedKeys As Variant
    sortedKeys = SortDictKeys(setKeys)
    Dim n As Long: n = UBound(sortedKeys) - LBound(sortedKeys) + 1
    Dim outArr() As Variant
    ReDim outArr(1 To n, 1 To 4)

    Dim i As Long, sentV As Long, recvV As Long
    For i = 0 To n - 1
        pk = CStr(sortedKeys(i))
        sentV = 0: recvV = 0
        Select Case kind
            Case "month"
                sentV = GetDictValue(dictMonthly, pk & "|_TOTAL|Sent")
                recvV = GetDictValue(dictMonthly, pk & "|_TOTAL|Received")
            Case "quarter"
                sentV = GetDictValue(dictQuarterly, pk & "|Sent")
                recvV = GetDictValue(dictQuarterly, pk & "|Received")
            Case "year"
                sentV = GetDictValue(dictYearly, pk & "|Sent")
                recvV = GetDictValue(dictYearly, pk & "|Received")
            Case "semester"
                For Each k In dictSemesterly.Keys
                    parts = Split(CStr(k), "|")
                    If UBound(parts) >= 2 Then
                        If CStr(parts(0)) = pk Then
                            If CStr(parts(2)) = "Sent" Then
                                sentV = sentV + CLng(dictSemesterly(k))
                            ElseIf CStr(parts(2)) = "Received" Then
                                recvV = recvV + CLng(dictSemesterly(k))
                            End If
                        End If
                    End If
                Next k
        End Select
        outArr(i + 1, 1) = pk
        outArr(i + 1, 2) = sentV
        outArr(i + 1, 3) = recvV
        outArr(i + 1, 4) = sentV + recvV
    Next i

    GlobalPeriodArray = outArr
End Function

' Fonction de couleur pour Heatmap (Vert -> Jaune -> Rouge)
Private Function GetHeatmapColor(val As Long, maxVal As Long) As Long
    Dim intensity As Double
    intensity = val / maxVal
    
    Dim r As Long, g As Long, b As Long
    
    If intensity <= 0.5 Then
        Dim t1 As Double: t1 = intensity * 2
        r = CLng(0 + t1 * 255)
        g = CLng(200 + t1 * 55)
        b = 0
    Else
        Dim t2 As Double: t2 = (intensity - 0.5) * 2
        r = 255
        g = CLng(255 - t2 * 255)
        b = 0
    End If
    
    GetHeatmapColor = RGB(r, g, b)
End Function

' ============================================================================
' UTILITAIRE : GetTopN - V22 avec QuickSort pour gros dictionnaires
' ============================================================================
Public Function GetTopN(dict As Object, n As Long) As Variant
    Dim result() As Variant
    Dim items() As Variant
    Dim i As Long
    Dim count As Long
    
    count = dict.Count
    If count = 0 Then
        ReDim result(0 To 1, 0 To -1)  ' Tableau vide (UBound=-1)
        GetTopN = result
        Exit Function
    End If
    
    ReDim items(0 To 1, 0 To count - 1)
    i = 0
    Dim k As Variant
    For Each k In dict.Keys
        items(0, i) = k
        items(1, i) = dict(k)
        i = i + 1
    Next k
    
    ' V22 : Tri decroissant - QuickSort pour > 30 elements, sinon bulle
    If count > 30 Then
        Call QuickSortTopN(items, 0, count - 1)
    Else
        Dim j As Long, temp1 As Variant, temp2 As Variant
        For i = 0 To count - 2
            For j = i + 1 To count - 1
                If items(1, i) < items(1, j) Then
                    temp1 = items(0, i): temp2 = items(1, i)
                    items(0, i) = items(0, j): items(1, i) = items(1, j)
                    items(0, j) = temp1: items(1, j) = temp2
                End If
            Next j
        Next i
    End If
    
    Dim resultCount As Long
    resultCount = n
    If resultCount > count Then resultCount = count
    
    ReDim result(0 To 1, 0 To resultCount - 1)
    For i = 0 To resultCount - 1
        result(0, i) = items(0, i)
        result(1, i) = items(1, i)
    Next i
    
    GetTopN = result
End Function

' V22 : QuickSort descendant sur items(0=key, 1=val)
Private Sub QuickSortTopN(ByRef items As Variant, ByVal low As Long, ByVal high As Long)
    Dim i As Long, j As Long
    Dim pivot As Variant, tempK As Variant, tempV As Variant
    
    If low < high Then
        pivot = items(1, (low + high) \ 2)
        i = low
        j = high
        
        Do
            Do While items(1, i) > pivot: i = i + 1: Loop
            Do While items(1, j) < pivot: j = j - 1: Loop
            
            If i <= j Then
                tempK = items(0, i): tempV = items(1, i)
                items(0, i) = items(0, j): items(1, i) = items(1, j)
                items(0, j) = tempK: items(1, j) = tempV
                i = i + 1
                j = j - 1
            End If
        Loop Until i > j
        
        If low < j Then Call QuickSortTopN(items, low, j)
        If i < high Then Call QuickSortTopN(items, i, high)
    End If
End Sub

            Dim storeDict As Object
            Set storeDict = dictMsgLengthByStore(storeName)

                Dim storeLengths() As Long
                Dim storeCount As Long: storeCount = storeDict.Count
                ReDim storeLengths(0 To storeCount - 1)
                
                Dim idxL As Long: idxL = 0
                Dim storeTotal As Double: storeTotal = 0
                Dim storeMin As Long: storeMin = 999999999
                Dim storeMax As Long: storeMax = 0

                For Each sk In storeDict.Keys

                    If storeLengths(idxL) < storeMin Then storeMin = storeLengths(idxL)
                    If storeLengths(idxL) > storeMax Then storeMax = storeLengths(idxL)
                    idxL = idxL + 1
                Next sk
                
                Call SortLongArray(storeLengths)
                
                Dim storeMediane As Long
                If storeCount Mod 2 = 0 Then
                    storeMediane = (storeLengths(storeCount \ 2 - 1) + storeLengths(storeCount \ 2)) \ 2
                Else
                    storeMediane = storeLengths(storeCount \ 2)
                End If
                
                idxStore = idxStore + 1
                storeArr(idxStore, 1) = storeName
                storeArr(idxStore, 2) = storeCount
                storeArr(idxStore, 3) = Round(storeTotal / storeCount, 0)
                storeArr(idxStore, 4) = storeMediane
                storeArr(idxStore, 5) = storeMin
                storeArr(idxStore, 6) = storeMax
            End If
        Next storeName
        
        If idxStore > 0 Then
            ws.Range(ws.Cells(startRowStore + 2, 1), ws.Cells(startRowStore + 1 + idxStore, 6)).Value = storeArr
            ws.Range(ws.Cells(startRowStore + 2, 2), ws.Cells(startRowStore + 1 + idxStore, 6)).NumberFormat = "#,##0"

                    ws.Cells(row, 1).Interior.Color = GetStoreColor(CStr(storeName))

                    row = row + 1
                End If
            Next storeName
        End If
    End If
    
    ws.Columns("A:F").AutoFit
End Sub

' ============================================================================
' UTILITAIRE : Tri tableau de Long (QuickSort)
' ============================================================================
Public Sub SortLongArray(ByRef arr() As Long)
    If UBound(arr) - LBound(arr) < 1 Then Exit Sub
    Call QuickSortLong(arr, LBound(arr), UBound(arr))
End Sub

Private Sub QuickSortLong(ByRef arr() As Long, ByVal low As Long, ByVal high As Long)
    Dim pivot As Long, i As Long, j As Long, temp As Long
    
    If low < high Then
        Dim mid As Long: mid = (low + high) \ 2
        If arr(low) > arr(mid) Then: temp = arr(low): arr(low) = arr(mid): arr(mid) = temp
        If arr(low) > arr(high) Then: temp = arr(low): arr(low) = arr(high): arr(high) = temp
        If arr(mid) > arr(high) Then: temp = arr(mid): arr(mid) = arr(high): arr(high) = temp
        
        pivot = arr(mid)
        i = low
        j = high
        
        Do
            Do While arr(i) < pivot: i = i + 1: Loop
            Do While arr(j) > pivot: j = j - 1: Loop
            
            If i <= j Then
                temp = arr(i): arr(i) = arr(j): arr(j) = temp
                i = i + 1
                j = j - 1
            End If
        Loop Until i > j
        
        If low < j Then Call QuickSortLong(arr, low, j)
        If i < high Then Call QuickSortLong(arr, i, high)
    End If
End Sub

' ============================================================================
' UTILITAIRE : Tri des cles d'un dictionnaire
' ============================================================================
Public Function SortDictKeys(dict As Object) As Variant
    If dict Is Nothing Then
        SortDictKeys = Array()
        Exit Function
    End If
    
    If dict.Count = 0 Then
        SortDictKeys = Array()
        Exit Function
    End If
    
    Dim keys() As String
    ReDim keys(0 To dict.Count - 1)
    
    Dim i As Long: i = 0
    Dim k As Variant
    For Each k In dict.Keys
        keys(i) = CStr(k)
        i = i + 1
    Next k
    
    Call QuickSortStrings(keys, LBound(keys), UBound(keys))
    
    SortDictKeys = keys
End Function

Private Sub QuickSortStrings(ByRef arr() As String, ByVal low As Long, ByVal high As Long)
    Dim pivot As String, i As Long, j As Long, temp As String
    
    If low < high Then
        pivot = arr((low + high) \ 2)
        i = low
        j = high
        
        Do
            Do While arr(i) < pivot: i = i + 1: Loop
            Do While arr(j) > pivot: j = j - 1: Loop
            
            If i <= j Then
                temp = arr(i): arr(i) = arr(j): arr(j) = temp
                i = i + 1
                j = j - 1
            End If
        Loop Until i > j
        
        If low < j Then Call QuickSortStrings(arr, low, j)
        If i < high Then Call QuickSortStrings(arr, i, high)
    End If
End Sub

    ws.Cells(r, 1).Value = "MAILBOX"
    ws.Cells(r, 2).Value = "EMAIL"

            If dictStoreKeyword.Exists(k) Then ws.Cells(r, 3).Value = CStr(dictStoreKeyword(k))
            r = r + 1
            If r > 28 Then Exit For
        Next k
    End If

    ' --- Compteurs ---
    Dim names(1 To 23) As String
    Dim ds(1 To 23) As Object
    names(1) = "dictStoreStats":            Set ds(1) = dictStoreStats
    names(2) = "dictMonthly":               Set ds(2) = dictMonthly

'----------------------------------------------------------------------------
' LastCompleteMonth - dernier mois ENTIEREMENT couvert par l'analyse.
' Le mois courant est toujours exclu : il est incomplet par nature, et le
' figer dans l'etat interdirait de le completer plus tard.
'----------------------------------------------------------------------------
Private Function LastCompleteMonth() As String
    On Error Resume Next
    Dim d As Date
    d = g_PeriodEnd
    If d > Now Then d = Now
    If Year(d) = Year(Now) And Month(d) = Month(Now) Then d = DateAdd("m", -1, d)
    LastCompleteMonth = Format$(d, "YYYY-MM")
End Function

Private Sub CreateSheetToCc(ByRef xlWB As Object)
    On Error GoTo Fin

    Dim ws As Object
    Set ws = xlWB.Worksheets.Add(After:=xlWB.Worksheets(xlWB.Worksheets.count))
    ws.Name = T("A - Cc", "To - Cc")

    ws.Cells(1, 1).Value = T("R~EPARTITION A / Cc / INDIRECT (mails re" & Chr(231) & "us)", _
                             "TO / Cc / INDIRECT BREAKDOWN (received mail)")
    ws.Cells(1, 1).Font.Bold = True

    ws.Cells(2, 1).Value = T("Indirect = ni en A ni en Cc : liste de diffusion, bo~ite partag~ee, Cci.", _

        ws.Cells(4, 1).Value = T("Aucun mail re" & Chr(231) & "u sur la p~eriode.", "No received mail over the period.")

    Set byStore = CreateObject("Scripting.Dictionary")
    Set byMonth = CreateObject("Scripting.Dictionary")
    Set months = CreateObject("Scripting.Dictionary")
    Set stores = CreateObject("Scripting.Dictionary")

    Dim k As Variant, parts() As String, gTo As Long, gCc As Long, gIn As Long
    For Each k In dictToCc.Keys
        parts = Split(CStr(k), "|")
        If UBound(parts) = 2 Then
            IncrementDict byStore, parts(0) & "|" & parts(2), CLng(dictToCc(k))
            IncrementDict byMonth, parts(1) & "|" & parts(2), CLng(dictToCc(k))
            If Not months.Exists(parts(1)) Then months.Add parts(1), 1
            If Not stores.Exists(parts(0)) Then stores.Add parts(0), 1
            Select Case parts(2)
                Case "To":  gTo = gTo + CLng(dictToCc(k))
                Case "Cc":  gCc = gCc + CLng(dictToCc(k))
                Case Else:  gIn = gIn + CLng(dictToCc(k))
            End Select
        End If
    Next k

    r = 4
    ws.Cells(r, 1).Value = T("PAR BO~ITE", "BY MAILBOX")
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 1).Font.Size = 12

    For i = LBound(sortedStores) To UBound(sortedStores)
        sn = CStr(sortedStores(i))
        a = GetDictValue(byStore, sn & "|To")
        b = GetDictValue(byStore, sn & "|Cc")
        c = GetDictValue(byStore, sn & "|Indirect")
        tot = a + b + c
        If tot > 0 Then
            r = r + 1
            ws.Cells(r, 1).Value = sn
            ws.Cells(r, 1).Interior.Color = GetStoreColor(sn)
            ws.Cells(r, 1).Font.Color = RGB(255, 255, 255)
            ws.Cells(r, 2).Value = a
            ws.Cells(r, 3).Value = b
            ws.Cells(r, 4).Value = c
            ws.Cells(r, 5).Value = a / tot
            ws.Cells(r, 6).Value = b / tot

        End If

    ws.Range(ws.Cells(headerRowStores + 1, 5), ws.Cells(r, 7)).NumberFormat = "0.0%"

    r = r + 1

    If gTot > 0 Then ws.Cells(r, 3).Value = gTo / gTot
    r = r + 1

    If gTot > 0 Then ws.Cells(r, 3).Value = gCc / gTot
    r = r + 1
    ws.Cells(r, 1).Value = T("Indirect", "Indirect"): ws.Cells(r, 2).Value = gIn
    If gTot > 0 Then ws.Cells(r, 3).Value = gIn / gTot
    r = r + 1
    ws.Cells(r, 1).Value = T("Total re" & Chr(231) & "u", "Total received")
    ws.Cells(r, 2).Value = gTot
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 2).Font.Bold = True
    ws.Range(ws.Cells(r - 3, 2), ws.Cells(r, 2)).NumberFormat = "#,##0"
    ws.Range(ws.Cells(r - 3, 3), ws.Cells(r - 1, 3)).NumberFormat = "0.0%"

    ' --- Bloc 3 : suivi mensuel des pourcentages ---
    r = r + 2
    Dim monthHeader As Long
    monthHeader = r
    ws.Cells(r, 1).Value = T("~EVOLUTION MENSUELLE (%)", "MONTHLY TREND (%)")
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 1).Font.Size = 12
    r = r + 1
    ws.Cells(r, 1).Value = T("Mois", "Month")
    ws.Cells(r, 2).Value = "% A / To"
    ws.Cells(r, 3).Value = "% Cc"
    ws.Cells(r, 4).Value = T("% Indirect", "% Indirect")
    ws.Cells(r, 5).Value = T("Total re" & Chr(231) & "u", "Total received")

        mk = CStr(sortedMonths(i))

        b = GetDictValue(byMonth, mk & "|Cc")
        c = GetDictValue(byMonth, mk & "|Indirect")

            ws.Cells(r, 2).Value = a / tot
            ws.Cells(r, 3).Value = b / tot
            ws.Cells(r, 4).Value = c / tot
            ws.Cells(r, 5).Value = tot
        End If
    Next i
    ws.Range(ws.Cells(firstDataRow, 2), ws.Cells(r, 4)).NumberFormat = "0.0%"
    ws.Range(ws.Cells(firstDataRow, 5), ws.Cells(r, 5)).NumberFormat = "#,##0"

    ' Courbe des pourcentages
    If r >= firstDataRow + 1 Then
        Dim chartObj As Object
        Set chartObj = ws.ChartObjects.Add(Left:=460, Top:=(monthHeader - 1) * 15, Width:=560, Height:=300)

                .ChartType = 4          ' xlLine

                .HasTitle = True
                .ChartTitle.Text = T("Part des mails re" & Chr(231) & "us en A / Cc / Indirect", _
                                     "Share of received mail in To / Cc / Indirect")
            End With
        End If
    End If

    ws.Columns("A:G").AutoFit
Fin:
    On Error Resume Next
End Sub

' PolishWorkbook (V24) - finition, en un seul endroit.
'
' Placement = 3 (xlFreeFloating) sur TOUS les graphiques : par defaut ils
' suivent les cellules, si bien que toute colonne masquee ou redimensionnee
' apres coup les fait glisser - c'est ce qui superposait le 3e graphique du
' tableau de bord au tableau des boites.
'
' PageSetup n'est volontairement PAS touche : chaque propriete declenche un
' aller-retour avec le pilote d'impression, soit plusieurs secondes sur 14
' feuilles, pour un benefice cosmetique.
'----------------------------------------------------------------------------
Private Sub PolishWorkbook(ByRef xlWB As Object)
    On Error Resume Next

    Dim colors As Object
    Set colors = CreateObject("Scripting.Dictionary")
    colors(T("Tableau de bord", "Dashboard")) = RGB(0, 169, 186)
    colors(T("R~esum~e", "Summary")) = RGB(0, 169, 186)
    colors(T("A - Cc", "To - Cc")) = RGB(0, 169, 186)
    colors(T("Par heure", "By hour")) = RGB(45, 38, 97)
    colors(T("Par jour semaine", "By weekday")) = RGB(45, 38, 97)
    colors(T("Par mois", "By month")) = RGB(45, 38, 97)
    colors(T("Par ann~ee", "By year")) = RGB(45, 38, 97)
    colors(T("Evolution par bo~ite", "Trend by mailbox")) = RGB(45, 38, 97)
    colors(T("Heatmaps", "Heatmaps")) = RGB(45, 38, 97)
    colors(T("Contacts", "Contacts")) = RGB(216, 122, 0)
    colors(T("Contacts int-ext", "Contacts int-ext")) = RGB(216, 122, 0)
    colors(T("Nb destinataires", "Nb recipients")) = RGB(216, 122, 0)
    colors(T("Longueur messages", "Message length")) = RGB(216, 122, 0)
    colors(T("DATA", "DATA")) = RGB(128, 128, 128)

    Dim ws As Object, i As Long
    For Each ws In xlWB.Worksheets

        ' 1. Graphiques flottants
        For i = 1 To ws.ChartObjects.count
            ws.ChartObjects(i).Placement = 3
        Next i

        ' 2. Feuille d'etat : masquee, jamais coloree
        If ws.Name = STATE_SHEET Then
            ws.Visible = 0          ' xlSheetHidden
        Else
            If colors.Exists(ws.Name) Then ws.Tab.Color = colors(ws.Name)
        End If
    Next ws

    ' 3. Filtre automatique sur les feuilles tabulaires
    ApplyFilter xlWB, T("Contacts", "Contacts"), 2
    ApplyFilter xlWB, T("Contacts int-ext", "Contacts int-ext"), 2
    ApplyFilter xlWB, T("DATA", "DATA"), 2

    ' 4. Volets figes - necessite d'activer la feuille, donc en dernier.
    FreezeAt xlWB, T("Contacts", "Contacts"), 3
    FreezeAt xlWB, T("DATA", "DATA"), 3
    FreezeAt xlWB, T("A - Cc", "To - Cc"), 6
End Sub

Private Sub ApplyFilter(ByRef xlWB As Object, ByVal sheetName As String, ByVal headerRow As Long)
    On Error Resume Next
    Dim ws As Object
    Set ws = xlWB.Worksheets(sheetName)
    If ws Is Nothing Then Exit Sub
    If ws.AutoFilterMode Then ws.AutoFilterMode = False
    ws.Rows(headerRow).AutoFilter
End Sub

Private Sub FreezeAt(ByRef xlWB As Object, ByVal sheetName As String, ByVal firstFreeRow As Long)
    On Error Resume Next
    Dim ws As Object
    Set ws = xlWB.Worksheets(sheetName)
    If ws Is Nothing Then Exit Sub
    ws.Activate
    ws.Cells(firstFreeRow, 1).Select
    ws.Application.ActiveWindow.FreezePanes = True
End Sub

'----------------------------------------------------------------------------
' RunEmailAnalysisDelta - POINT D'ENTREE (bouton / Alt+F8).
'
' 1. demande le rapport precedent
' 2. lit sa feuille _ETAT (configuration + dernier mois complet + compteurs)
' 3. calcule les mois manquants
' 4. enchaine sur RunEmailAnalysis, qui pre-remplira la fenetre et
'    n'analysera que ces mois.
'----------------------------------------------------------------------------
Public Sub RunEmailAnalysisDelta()
    On Error GoTo ErrH

    g_DeltaMode = False
    g_DeltaSourcePath = ""

    Dim p As String
    p = AskForPreviousReport()
    If Len(p) = 0 Then Exit Sub

    If Not LoadStateFromWorkbook(p) Then Exit Sub

    ' Bornes : du mois SUIVANT le dernier mois complet, jusqu'au dernier mois
    ' entierement ecoule. Le mois courant est exclu.
    Dim lastDone As Date
    lastDone = MonthKeyToDate(g_DeltaLastMonth)
    If lastDone = 0 Then
        MsgBox T("Le fichier choisi ne porte pas de mois de r~ef~erence exploitable.", _
                 "The selected file carries no usable reference month."), _
               vbExclamation, "Analytics " & APP_VERSION
        Exit Sub
    End If

    g_DeltaStart = DateSerial(Year(lastDone), Month(lastDone) + 1, 1)
    g_DeltaEnd = DateSerial(Year(Now), Month(Now), 1) - 1
    g_DeltaEnd = DateSerial(Year(g_DeltaEnd), Month(g_DeltaEnd), Day(g_DeltaEnd)) + 0.99999

    If g_DeltaStart > g_DeltaEnd Then
        MsgBox T("Rien ~a ajouter : l'analyse pr~ec~edente couvre d~ej~a tous les mois complets.", _
                 "Nothing to add: the previous analysis already covers every complete month.") & _
               vbCrLf & vbCrLf & _
               T("Dernier mois analys~e : ", "Last month analyzed: ") & g_DeltaLastMonth, _
               vbInformation, "Analytics " & APP_VERSION
        Exit Sub
    End If

    g_DeltaMode = True
    g_DeltaSourcePath = p

    DebugLog "DELTA : source=" & p & " | dernier mois=" & g_DeltaLastMonth & _
             " | a analyser " & Format$(g_DeltaStart, "YYYY-MM-DD") & " -> " & _
             Format$(g_DeltaEnd, "YYYY-MM-DD")

    Call RunEmailAnalysis
    Exit Sub

ErrH:
    g_DeltaMode = False
    MsgBox T("Erreur en mode delta : ", "Delta mode error: ") & Err.Number & " - " & Err.Description, _
           vbCritical, "Analytics " & APP_VERSION
End Sub

'----------------------------------------------------------------------------
' AskForPreviousReport - selection du rapport precedent. Demarre dans le
' dossier ou les rapports sont ecrits depuis la V24.
'----------------------------------------------------------------------------
Private Function AskForPreviousReport() As String
    On Error Resume Next
    AskForPreviousReport = ""

    Dim prev As String
    prev = CurDir$
    ChDir ReportFolderPath()

    Dim f As Variant
    f = Application.GetOpenFilename( _
            T("Rapport d'analyse (*.xlsx),*.xlsx", "Analysis report (*.xlsx),*.xlsx"), _
            1, T("Choisir le rapport pr~ec~edent", "Select the previous report"))

    ChDir prev
    If VarType(f) = vbBoolean Then Exit Function
    AskForPreviousReport = CStr(f)
End Function

'----------------------------------------------------------------------------
' LoadStateFromWorkbook - lit la feuille _ETAT du classeur choisi.
'
' Ouverture en LECTURE SEULE et fermeture sans enregistrer : le rapport
' precedent ne doit jamais etre modifie par une analyse delta.
'
' Le bloc de compteurs est lu EN UNE FOIS dans un tableau Variant. Lu cellule
' par cellule, 100 000 lignes prendraient plusieurs minutes en COM.
'----------------------------------------------------------------------------
Private Function LoadStateFromWorkbook(ByVal path As String) As Boolean
    LoadStateFromWorkbook = False
    On Error GoTo ErrH

    Dim wb As Object, ws As Object
    Set wb = Application.Workbooks.Open(path, , True)   ' ReadOnly

    On Error Resume Next
    Set ws = wb.Worksheets(STATE_SHEET)
    On Error GoTo ErrH

    If ws Is Nothing Then
        wb.Close False
        ' v- 2026-08-22 : STATE_SHEET sort de T(). C'est un NOM DE FEUILLE
        ' Excel, donc une donnee du contrat : le faire traverser A() est
        ' faux par principe, et le rendrait faux en fait le jour ou ce nom
        ' contiendrait un "~". Le titre passe par T() (il melait une
        ' etiquette technique et un numero de version).
        MsgBox T("Ce rapport n'a pas de feuille ", "This report has no ") & _
                 STATE_SHEET & _
                 T(" : il a ~et~e produit par une version ant~erieure ~a la V24.", _
                   " sheet: it was produced before V24.") & vbCrLf & vbCrLf & _
                 T("Une derni~ere analyse COMPL~ETE est n~ecessaire pour cr~eer " & _
                   "l'~etat. Les analyses en delta seront possibles ensuite.", _
                   "One last FULL analysis is required to build the state. " & _
                   "Delta runs will work afterwards."), _
               vbExclamation + vbMsgBoxSetForeground, _
               T("Statistiques ", "Analytics ") & APP_VERSION
        Exit Function
    End If

    g_DeltaLastMonth = Trim$(CStr(ws.Cells(16, 2).Value))

    ' Configuration des boites telle qu'utilisee lors de l'analyse d'origine.
    Set m_DeltaConfig = CreateObject("Scripting.Dictionary")
    Dim r As Long
    For r = 20 To 28
        Dim sn As String
        sn = Trim$(CStr(ws.Cells(r, 1).Value))
        If Len(sn) > 0 Then
            m_DeltaConfig(sn) = CStr(ws.Cells(r, 2).Value) & "|" & CStr(ws.Cells(r, 3).Value)
        End If
    Next r

    Dim nRows As Long
    nRows = CLng(ws.Cells(18, 2).Value)
    If nRows > 0 Then
        m_DeltaRows = ws.Range(ws.Cells(31, 1), ws.Cells(30 + nRows, 3)).Value
    Else
        m_DeltaRows = Empty
    End If

    wb.Close False
    Set wb = Nothing

    LoadStateFromWorkbook = True
    Exit Function

ErrH:
    On Error Resume Next
    If Not wb Is Nothing Then wb.Close False
    MsgBox T("Lecture de l'~etat impossible : ", "Cannot read state: ") & Err.Description, _
           vbCritical, "Analytics " & APP_VERSION
End Function

'----------------------------------------------------------------------------
' RehydrateFromState - recharge les compteurs lus dans _ETAT.
' Appelee APRES InitializeDictionaries, AVANT le scan.
'----------------------------------------------------------------------------
Private Sub RehydrateFromState()
    On Error GoTo Fin
    If IsEmpty(m_DeltaRows) Then Exit Sub

    Dim i As Long, dn As String, k As String, v As Long, n As Long
    For i = LBound(m_DeltaRows, 1) To UBound(m_DeltaRows, 1)
        dn = Trim$(CStr(m_DeltaRows(i, 1)))
        If Len(dn) > 0 Then
            k = CStr(m_DeltaRows(i, 2))
            v = CLng(m_DeltaRows(i, 3))
            If SeedDict(dn, k, v) Then n = n + 1
        End If
    Next i

    Heartbeat "DELTA : " & n & " compteurs recharges depuis " & g_DeltaLastMonth
Fin:
    On Error Resume Next
End Sub

'----------------------------------------------------------------------------
' SeedDict - aiguillage nom -> dictionnaire. Table en dur, comme partout
' ailleurs dans ce projet : VBA ne resout pas un nom de variable a partir
' d'une chaine. histMsgLength500 est ignore ici (il ne sert qu'a recalculer
' une mediane, pas au scan).
'----------------------------------------------------------------------------
Private Function SeedDict(ByVal dictName As String, ByVal k As String, ByVal v As Long) As Boolean
    On Error Resume Next
    SeedDict = True
    Select Case dictName
        Case "dictStoreStats":            dictStoreStats(k) = v
        Case "dictMonthly":               dictMonthly(k) = v
        Case "dictWeekly":                dictWeekly(k) = v
        Case "dictDaily":                 dictDaily(k) = v
        Case "dictQuarterly":             dictQuarterly(k) = v
        Case "dictYearly":                dictYearly(k) = v
        Case "dictSemesterly":            dictSemesterly(k) = v
        Case "dictHourly":                dictHourly(k) = v
        Case "dictDayOfWeek":             dictDayOfWeek(k) = v
        Case "dictHeatmapSent":           dictHeatmapSent(k) = v
        Case "dictHeatmapReceived":       dictHeatmapReceived(k) = v
        Case "dictHeatmapTotal":          dictHeatmapTotal(k) = v
        Case "dictSentTo":                dictSentTo(k) = v
        Case "dictReceivedFrom":          dictReceivedFrom(k) = v
        Case "dictSentToInternal":        dictSentToInternal(k) = v
        Case "dictSentToExternal":        dictSentToExternal(k) = v
        Case "dictSentToUnknown":         dictSentToUnknown(k) = v
        Case "dictReceivedFromInternal":  dictReceivedFromInternal(k) = v
        Case "dictReceivedFromExternal":  dictReceivedFromExternal(k) = v
        Case "dictReceivedFromUnknown":   dictReceivedFromUnknown(k) = v
        Case "dictNbRecipients":          dictNbRecipients(k) = v
        Case "dictInternalExternal":      dictInternalExternal(k) = v
        Case "dictToCc":                  dictToCc(k) = v
        Case Else:                        SeedDict = False
    End Select
End Function

'----------------------------------------------------------------------------
' MonthKeyToDate - "YYYY-MM" -> 1er du mois. 0 si illisible.
'----------------------------------------------------------------------------
Private Function MonthKeyToDate(ByVal mk As String) As Date
    On Error Resume Next
    MonthKeyToDate = 0
    If Len(mk) < 7 Then Exit Function
    Dim y As Long, m As Long
    y = CLng(Left$(mk, 4))
    m = CLng(Mid$(mk, 6, 2))
    If y < 2000 Or m < 1 Or m > 12 Then Exit Function
    MonthKeyToDate = DateSerial(y, m, 1)
End Function

    p = Split(CStr(m_DeltaConfig(storeName)), "|")

End Function

Private Sub DetectYearRange(ByRef minYear As Long, ByRef maxYear As Long)
    On Error Resume Next
    minYear = 0
    maxYear = 0
    
    If dictYearly Is Nothing Then Exit Sub
    If dictYearly.Count = 0 Then Exit Sub
    
    Dim k As Variant, parts As Variant, y As Long
    For Each k In dictYearly.Keys
        parts = Split(CStr(k), "|")
        If UBound(parts) >= 0 Then
            y = CLng(parts(0))
            If y >= 2000 And y <= 2100 Then
                If minYear = 0 Or y < minYear Then minYear = y
                If maxYear = 0 Or y > maxYear Then maxYear = y
            End If
        End If
    Next k
End Sub
