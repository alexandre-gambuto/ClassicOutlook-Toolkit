'----------------------------------------------------------------------------
' RunEmailAnalysis - point d'entree (bouton "Statistiques emails" du ruban).
' V25.3 (F9-05) : garde de reentrance + modDefer.BeginHeavy / EndHeavy.
' Le parcours rend la main par DoEvents et la fenetre de progression est non
' modale : sans garde, un second clic relancait une analyse IMBRIQUEE qui
' remettait a zero les dictionnaires partages de la premiere (chiffres
' faux sans aucun signe), et les taches lourdes differees (reindexation,
' archivage...) partaient au milieu. Nom et signature inchanges (ruban).
'----------------------------------------------------------------------------
Public Sub RunEmailAnalysis()
    If m_Running Then
        modUi.Inform T("Une analyse des mails est d~ej~a en cours.", _
                       "A mail analysis is already running."), AnalyticsTitle()
        Exit Sub
    End If
    m_Running = True
    modDefer.BeginHeavy
    On Error Resume Next
    RunAnalysisCore
    modDefer.EndHeavy
    m_Running = False
End Sub

Private Sub RunAnalysisCore()
    DebugLog String(70, "=")
    DebugLog "=== LANCEMENT EMAIL ANALYTICS " & APP_VERSION & " (OPTIMISE) ==="
    DebugLog "Date/Heure : " & Format(Now, "dd/mm/yyyy hh:mm:ss")
    DebugLog String(70, "=")

    On Error GoTo ErrorHandler

    ' Reset compteurs
    m_TotalProcessed = 0
    m_TotalScanned = 0
    m_TotalIgnored = 0
    m_TotalErrors = 0
    m_ScanError = ""
    m_ReportError = ""
    m_ReportLeftOpen = False

    ' --- Configuration ---

'   StatMailboxN = fragment_nom ; mot-cle ; #RRGGBB [; email_utilisateur]

Private Sub EnsureStatConfigLoaded()
    If m_StatCfgLoaded Then Exit Sub
    m_StatCfgLoaded = True

    Set dictStatPresets = CreateObject("Scripting.Dictionary")
    m_StatPresetCount = 0
    m_StatDefaultSet = False
    m_StatDefaultColor = RGB(136, 136, 136)

    On Error Resume Next

    Dim cfgPath As String
    cfgPath = modConfig.ConfigPath()   ' V25.2 : config.ini du dossier de l'outil
    If Len(cfgPath) = 0 Then Exit Sub
    If Len(Dir$(cfgPath)) = 0 Then Exit Sub

    Dim f As Integer, raw As String
    f = FreeFile
    Open cfgPath For Input As #f
    raw = Input$(LOF(f), f)
    Close #f
    If Len(raw) = 0 Then Exit Sub

    raw = Replace(raw, vbCrLf, vbLf)
    raw = Replace(raw, vbCr, vbLf)

    Dim arrLines() As String
    arrLines = Split(raw, vbLf)

    Dim i As Long, s As String, eq As Long
    Dim keyName As String, keyVal As String
    Dim parts() As String, colorOK As Boolean, colorVal As Long

    For i = LBound(arrLines) To UBound(arrLines)
        s = Trim$(arrLines(i))
        If Len(s) > 0 Then
            If Left$(s, 1) <> ";" And Left$(s, 1) <> "[" Then
                eq = InStr(s, "=")
                If eq > 1 Then
                    keyName = LCase$(Trim$(Left$(s, eq - 1)))
                    keyVal = Trim$(Mid$(s, eq + 1))

                    If keyName = "statmailboxdefault" Then
                        colorVal = HexToColor(keyVal, colorOK)
                        If colorOK Then
                            m_StatDefaultColor = colorVal
                            m_StatDefaultSet = True
                        End If
                    ElseIf Left$(keyName, 11) = "statmailbox" And Len(keyName) > 11 Then
                        If IsNumeric(Mid$(keyName, 12)) Then
                            parts = Split(keyVal, ";")
                            If UBound(parts) >= 0 Then
                                Dim fragment As String
                                fragment = Trim$(parts(0))
                                If Len(fragment) > 0 Then
                                    Dim p As Object
                                    Set p = CreateObject("Scripting.Dictionary")
                                    p("Fragment") = fragment
                                    p("Keyword") = ""
                                    p("EmailUser") = ""
                                    p("ColorSet") = False
                                    p("ColorRGB") = 0
                                    p("DateMinSet") = False
                                    If UBound(parts) >= 1 Then p("Keyword") = Trim$(parts(1))
                                    If UBound(parts) >= 2 Then
                                        colorVal = HexToColor(Trim$(parts(2)), colorOK)
                                        If colorOK Then
                                            p("ColorRGB") = colorVal
                                            p("ColorSet") = True
                                        End If
                                    End If
                                    If UBound(parts) >= 3 Then p("EmailUser") = Trim$(parts(3))
                                    If UBound(parts) >= 4 Then
                                        Dim dParsed As Date
                                        If ParseIsoDate(Trim$(parts(4)), dParsed) Then
                                            p("DateMin") = dParsed
                                            p("DateMinSet") = True
                                        End If
                                    End If

                                    ' Domain derive : email prioritaire, sinon
                                    ' fragment s'il ressemble a un domaine
                                    Dim dom As String
                                    dom = ""
                                    If InStr(CStr(p("EmailUser")), "@") > 0 Then
                                        dom = Mid$(CStr(p("EmailUser")), InStr(CStr(p("EmailUser")), "@") + 1)
                                    ElseIf InStr(fragment, ".") > 0 And InStr(fragment, "@") = 0 And InStr(fragment, " ") = 0 Then
                                        dom = fragment
                                    End If
                                    p("Domain") = LCase$(dom)

                                    m_StatPresetCount = m_StatPresetCount + 1
                                    dictStatPresets.Add CStr(m_StatPresetCount), p
                                End If
                            End If
                        End If
                    End If
                End If
            End If
        End If
    Next i
End Sub

' Renvoie le preset (Dictionary) de la premiere ligne StatMailboxN dont le

    Next i
End Function

' V23.9 : parse "YYYY-MM-DD" (ou toute date reconnue par IsDate) -> Date.
' Renvoie False si la chaine n'est pas une date valide.
Private Function ParseIsoDate(ByVal s As String, ByRef outDate As Date) As Boolean
    ParseIsoDate = False
    s = Trim$(s)
    If Len(s) = 0 Then Exit Function
    If Len(s) = 10 Then
        If Mid$(s, 5, 1) = "-" And Mid$(s, 8, 1) = "-" Then
            Dim yy As Long, mo As Long, dd As Long
            yy = Val(Left$(s, 4)): mo = Val(Mid$(s, 6, 2)): dd = Val(Mid$(s, 9, 2))
            If yy >= 1900 And mo >= 1 And mo <= 12 And dd >= 1 And dd <= 31 Then
                On Error Resume Next
                outDate = DateSerial(yy, mo, dd)
                If Err.Number = 0 Then ParseIsoDate = True
                On Error GoTo 0
                Exit Function
            End If
        End If
    End If
    If IsDate(s) Then
        outDate = CDate(s)
        ParseIsoDate = True
    End If
End Function

' "#RRGGBB" ou "RRGGBB" -> Long RGB. ok=False si format invalide.
Private Function HexToColor(ByVal h As String, ByRef ok As Boolean) As Long
    On Error GoTo Bad
    ok = False
    HexToColor = 0
    h = Trim$(h)
    If Left$(h, 1) = "#" Then h = Mid$(h, 2)
    If Len(h) <> 6 Then Exit Function
    Dim r As Long, g As Long, b As Long
    r = CLng("&H" & Mid$(h, 1, 2))
    g = CLng("&H" & Mid$(h, 3, 2))
    b = CLng("&H" & Mid$(h, 5, 2))
    HexToColor = RGB(r, g, b)
    ok = True
    Exit Function
Bad:
    ok = False
End Function

' ============================================================================
' V22 : CALCUL DU TEMPS ECOULE (gere le passage par minuit)
' ============================================================================
Private Function GetElapsedSeconds() As Double
    Dim currentTimer As Double
    currentTimer = Timer

    If Date > m_StartDate Then
        GetElapsedSeconds = (86400 - m_StartTime) + currentTimer + ((Date - m_StartDate - 1) * 86400)
    Else
        GetElapsedSeconds = currentTimer - m_StartTime
    End If

    If GetElapsedSeconds < 0 Then GetElapsedSeconds = 0
End Function

'----------------------------------------------------------------------------
' BuildDateFilter (V25.3, F9-03) - filtre Items.Restrict sur la date de
' reception quand la periode est restreinte (choix "Depuis", periode
' personnalisee, mise a jour delta). Avant : tous les elements de toutes les
' boites etaient ouverts, le filtre de date n'intervenant qu'apres, en VBA
' (une mise a jour d'un mois coutait autant qu'une analyse complete).
' Bornes au JOUR pres (format court du poste, modCommon.FormatDaslDate :
' regle du pack v4) ; la comparaison exacte a la seconde reste dans
' ProcessEmail. "" = pas de filtre (periode "Tout").
'----------------------------------------------------------------------------
Private Function BuildDateFilter() As String
    On Error Resume Next
    BuildDateFilter = ""
    If g_DeltaMode Or g_PeriodStart > DateSerial(2000, 1, 1) Or _
       Int(CDbl(g_PeriodEnd)) < Int(CDbl(Now)) Then
        BuildDateFilter = "[ReceivedTime] >= '" & _
            modCommon.FormatDaslDate(CDate(Int(CDbl(g_PeriodStart)))) & _
            "' AND [ReceivedTime] < '" & _
            modCommon.FormatDaslDate(CDate(Int(CDbl(g_PeriodEnd)) + 1)) & "'"
    End If
End Function

'----------------------------------------------------------------------------
' PeriodItems (V25.3, F9-03) - elements du dossier, restreints a la periode

Private Function PeriodItems(ByVal folder As Object) As Object
    On Error Resume Next
    Set PeriodItems = Nothing
    If Len(m_DateFilter) > 0 Then Set PeriodItems = folder.Items.Restrict(m_DateFilter)
    If PeriodItems Is Nothing Then Set PeriodItems = folder.Items
End Function

    Dim kinds As Variant, i As Long, fld As Object, eid As String
    kinds = Array(olFolderDeletedItems, olFolderJunk, olFolderSyncIssues, olFolderConflicts, _
                  olFolderLocalFailures, olFolderServerFailures, olFolderOutbox, _
                  olFolderDrafts, olFolderRssFeeds)
    For i = LBound(kinds) To UBound(kinds)
        Set fld = Nothing
        Set fld = store.GetDefaultFolder(CLng(kinds(i)))
        If Not fld Is Nothing Then
            eid = ""
            eid = CStr(fld.EntryID)
            If Len(eid) > 0 Then
                m_ExclIDs(eid) = True
                m_StoreIdsOk = True
            End If
        End If
    Next i

    Set fld = Nothing
    Set fld = store.GetDefaultFolder(olFolderSentMail)
    If Not fld Is Nothing Then
        m_SentID = CStr(fld.EntryID)
        If Len(m_SentID) > 0 Then m_StoreIdsOk = True
    End If
    Set fld = Nothing
    Err.Clear

    If Not m_StoreIdsOk Then Heartbeat "  dossiers systeme non resolus : repli sur les noms"
End Sub

'----------------------------------------------------------------------------
' BuildStoreIdentities (V25.3, F9-08) - adresses qui designent l'utilisateur
' dans chaque boite analysee :
'   - adresse de la boite (4e champ de StatMailboxN / champ Email) ;
'   - si cette adresse est l'une des miennes, ou si la boite principale n'a
'     pas d'adresse renseignee : toutes mes adresses (modSmtp.MyAddresses :
'     comptes du profil + alias Exchange).
' Dictionnaire vide = aucune adresse connue : repli sur le mot-cle.
' Avant : seul le mot-cle (nom de famille) servait (tous les prereglages
' forcent ce mode) - un homonyme etait compte comme "moi".
'----------------------------------------------------------------------------
Private Sub BuildStoreIdentities(ByVal selectedStores As Collection)
    On Error Resume Next
    Set dictStoreIdentity = CreateObject("Scripting.Dictionary")

    Dim defName As String
    defName = ""
    defName = CStr(Application.Session.DefaultStore.DisplayName)

    Dim mine As Object
    Set mine = Nothing
    Set mine = modSmtp.MyAddresses()

    Dim sn As Variant, ids As Object, userEmail As String, k As Variant, useMine As Boolean
    For Each sn In selectedStores
        Set ids = CreateObject("Scripting.Dictionary")
        ids.CompareMode = 1   ' TextCompare
        userEmail = ""
        If dictStoreEmailUser.Exists(CStr(sn)) Then userEmail = LCase$(Trim$(CStr(dictStoreEmailUser(CStr(sn)))))
        If InStr(userEmail, "@") > 0 Then
            ids(userEmail) = True
            useMine = modSmtp.IsMyAddress(userEmail)
        Else
            useMine = (StrComp(CStr(sn), defName, vbTextCompare) = 0)
        End If
        If useMine And Not mine Is Nothing Then
            For Each k In mine.Keys
                ids(LCase$(CStr(k))) = True
            Next k
        End If
        Set dictStoreIdentity(CStr(sn)) = ids
        modDebugLog.DDebug MOD_NAME, "Identite d'une boite : " & ids.Count & " adresse(s)" & _
            IIf(ids.Count = 0, " (repli mot-cle)", "")
    Next sn
End Sub

Private Function StoreUsesAddress(ByVal storeName As String) As Boolean

    StoreUsesAddress = (ids.Count > 0)
End Function

' ============================================================================
' ANALYSE DES EMAILS
' ============================================================================
Private Function AnalyzeEmails(selectedStores As Collection) As Boolean
    On Error GoTo ErrorHandler

    Dim olNS As Object
    Dim store As Object
    Dim rootFolder As Object
    Dim storeName As Variant
    Dim storeIndex As Long
    Dim totalEmails As Long

    Set olNS = Application.GetNamespace("MAPI")

    If fProgress Is Nothing Then
        AnalyzeEmails = False
        Exit Function
    End If

    fProgress.UpdateProgress 0, 0, T("Comptage...", "Counting..."), 0, selectedStores.Count

    ' V25.3 : filtre de date (F9-03) et identite par boite (F9-08), une fois
    ' pour toute l'analyse.
    m_DateFilter = BuildDateFilter()
    If Len(m_DateFilter) > 0 Then Heartbeat "Filtre de date : " & m_DateFilter
    Call BuildStoreIdentities(selectedStores)

    ' Comptage des emails
    totalEmails = 0
    For Each storeName In selectedStores
        DebugLog "    Comptage d'une boite"   ' V25.3 : nom de boite (souvent une adresse) hors de debug.log
        Heartbeat "COMPTAGE " & storeName
        For Each store In olNS.Stores
            If store.DisplayName = CStr(storeName) Then
                Call PrepareStoreFolders(store)       ' V25.3 (F9-06)
                totalEmails = totalEmails + CountEmailsInStore(store)
                DebugLog "      -> " & totalEmails & " emails cumules"
                Exit For
            End If
        Next store
    Next storeName

    DebugLog "   -> Total a traiter: " & Format(totalEmails, "#,##0")
    Heartbeat "Total a traiter: " & totalEmails
    m_TotalToProcess = totalEmails
    m_TotalProcessed = 0
    m_TotalScanned = 0

    ' Analyse de chaque boite
    storeIndex = 0
    For Each storeName In selectedStores
        storeIndex = storeIndex + 1
        DebugLog "   Boite " & storeIndex & "/" & selectedStores.Count
        Heartbeat "STORE [" & storeIndex & "/" & selectedStores.Count & "] " & storeName

        For Each store In olNS.Stores
            If store.DisplayName = CStr(storeName) Then
                Call PrepareStoreFolders(store)       ' V25.3 (F9-06)
                Set rootFolder = store.GetRootFolder
                Call ProcessFolder(rootFolder, CStr(storeName), storeIndex, selectedStores.Count)
                Exit For
            End If
        Next store

        If fProgress.Cancelled Then Exit For
    Next storeName

    AnalyzeEmails = Not fProgress.Cancelled
    Exit Function

ErrorHandler:
    ' V25.3 (F9-04) : erreur gardee pour le message final et tracee.
    m_ScanError = Err.Number & " - " & Err.Description
    Heartbeat "!!! AnalyzeEmails erreur " & m_ScanError
    modDebugLog.DError MOD_NAME, "Parcours des boites interrompu : " & m_ScanError
    AnalyzeEmails = False
End Function

Private Function CountEmailsInStore(store As Object) As Long
    On Error Resume Next
    Dim rootFolder As Object
    Dim count As Long

    Set rootFolder = store.GetRootFolder
    count = CountEmailsInFolder(rootFolder)

    If g_TestMode Then count = count \ TEST_SAMPLING
    CountEmailsInStore = count
End Function

Private Function CountEmailsInFolder(folder As Object) As Long
    On Error Resume Next
    Dim count As Long
    Dim subFolder As Object

    If IsExcludedFolder(folder) Then
        CountEmailsInFolder = 0
        Exit Function
    End If

    ' V25.3 (F9-03) : meme sous-ensemble que le parcours (periode), pour une
    ' progression juste.
    count = -1
    count = PeriodItems(folder).Count
    If count < 0 Then count = 0

    For Each subFolder In folder.Folders
        count = count + CountEmailsInFolder(subFolder)
    Next subFolder

    CountEmailsInFolder = count
End Function

' ============================================================================
' V22 : IsExcludedFolder OPTIMISE avec cache dictionnaire
' V25.3 (F9-06) : par IDENTITE d'abord, valable dans toutes les langues -
'   1. dossier systeme de la boite (PrepareStoreFolders : EntryID) ;
'   2. dossier qui ne contient pas de mails (DefaultItemType : calendrier,
'      contacts, taches, notes, journal) ;
'   3. nom EXACT d'un dossier systeme sans constante Outlook (historique
'      des conversations, PersonMetadata...) ;
'   4. anciens motifs (sous-chaines du nom) seulement si aucun dossier
'      systeme n'a pu etre resolu pour cette boite.
' Avant : sous-chaines du nom seulement - "Courrier indesirable" (accent)
' n'etait pas reconnu, et un dossier de travail contenant "notes",
' "contacts", "journal", "historique"... etait ignore avec ses sous-dossiers.
' Cache par EntryID.
' ============================================================================
Private Function IsExcludedFolder(ByVal folder As Object) As Boolean
    On Error Resume Next
    Dim eid As String
    eid = ""
    eid = CStr(folder.EntryID)
    If Len(eid) > 0 Then
        If dictExcludedFolderCache.Exists(eid) Then
            IsExcludedFolder = dictExcludedFolderCache(eid)
            Exit Function
        End If
    End If

    Dim result As Boolean
    result = False

    If Len(eid) > 0 And Not m_ExclIDs Is Nothing Then
        If m_ExclIDs.Exists(eid) Then result = True
    End If

    If Not result Then
        Dim itemKind As Long
        itemKind = olMailItem
        itemKind = folder.DefaultItemType
        If itemKind <> olMailItem Then result = True
    End If

    Dim lowerName As String
    lowerName = LCase$(CStr(folder.Name))
    If Not result Then result = IsSystemFolderName(lowerName)
    If Not result And Not m_StoreIdsOk Then result = IsExcludedFolderName(lowerName)

    If Len(eid) > 0 Then dictExcludedFolderCache(eid) = result
    IsExcludedFolder = result
End Function

'----------------------------------------------------------------------------
' IsSystemFolderName (V25.3, F9-06) - nom EXACT (minuscules) d'un dossier
' systeme d'Outlook / Exchange, FR ou EN. Toujours applique : un nom entier
' ne touche pas un dossier de travail (contrairement aux sous-chaines).
'----------------------------------------------------------------------------
Private Function IsSystemFolderName(ByVal lowerName As String) As Boolean
    Dim e As String
    e = Chr(233)   ' e accent aigu
    Select Case lowerName
        Case "courrier ind" & e & "sirable", "junk email", "junk e-mail", _
             e & "l" & e & "ments supprim" & e & "s", "deleted items", _
             "probl" & Chr(232) & "mes de synchronisation", "sync issues", _
             "conflits", "conflicts", _
             e & "checs locaux", "local failures", _
             e & "checs du serveur", "server failures", _
             "bo" & Chr(238) & "te d'envoi", "outbox", "brouillons", "drafts", _
             "flux rss", "rss feeds", "rss subscriptions", _
             "historique des conversations", "conversation history", _
             "personmetadata", "quick step settings", _
             "social activity notifications", "recoverable items"
            IsSystemFolderName = True
    End Select
End Function

'----------------------------------------------------------------------------
' IsExcludedFolderName - anciens motifs (V22), en REPLI seulement (boite dont
' aucun dossier systeme n'est accessible). V25.3 : variantes accentuees
' ajoutees (indesirable, probleme, echecs).
'----------------------------------------------------------------------------
Private Function IsExcludedFolderName(ByVal lowerName As String) As Boolean
    Dim result As Boolean
    result = False

    ' Une seule passe de verification, regroupee par patterns
    Select Case True
        ' Corbeille / supprimes
        Case InStr(lowerName, "supprim") > 0, _
             InStr(lowerName, "deleted") > 0, _
             InStr(lowerName, "junk") > 0, _
             InStr(lowerName, "spam") > 0, _
             InStr(lowerName, "indesirable") > 0, _
             InStr(lowerName, "ind" & Chr(233) & "sirable") > 0
            result = True
        ' Synchronisation / Conflits
        Case InStr(lowerName, "sync issues") > 0, _
             InStr(lowerName, "probleme") > 0 And InStr(lowerName, "sync") > 0, _
             InStr(lowerName, "probl" & Chr(232) & "me") > 0 And InStr(lowerName, "sync") > 0, _
             InStr(lowerName, "personmetadata") > 0, _
             InStr(lowerName, "conflits") > 0, _
             InStr(lowerName, "conflicts") > 0, _
             InStr(lowerName, "local failures") > 0, _
             InStr(lowerName, "server failures") > 0, _
             InStr(lowerName, Chr(233) & "checs") > 0
            result = True
        ' Recuperables / Historique
        Case InStr(lowerName, "recoverable") > 0, _
             InStr(lowerName, "recuperable") > 0, _
             InStr(lowerName, "conversation history") > 0, _
             InStr(lowerName, "historique") > 0
            result = True
        ' RSS et autres
        Case InStr(lowerName, "rss") > 0, _
             InStr(lowerName, "quick step") > 0, _
             InStr(lowerName, "social activity") > 0, _
             InStr(lowerName, "suggested contacts") > 0
            result = True
        ' Calendrier, Contacts, Taches, Notes
        Case InStr(lowerName, "calendrier") > 0, _
             InStr(lowerName, "calendar") > 0, _
             InStr(lowerName, "contacts") > 0, _
             InStr(lowerName, "taches") > 0, _
             InStr(lowerName, "tasks") > 0, _
             InStr(lowerName, "notes") > 0, _
             InStr(lowerName, "journal") > 0
            result = True
    End Select

    IsExcludedFolderName = result
End Function

' ============================================================================
' V22 : BOUCLE PRINCIPALE ULTRA-OPTIMISEE
' - Boucle par index (plus rapide que For Each sur items Outlook)
' - MessageClass au lieu de TypeName
' - Cache des objets Items
' - Early exit sur exclusions
' ============================================================================
Private Sub ProcessFolder(folder As Object, storeName As String, storeIndex As Long, storeCount As Long, _
                          Optional ByVal parentIsSent As Boolean = False)
    On Error Resume Next

    If IsExcludedFolder(folder) Then Exit Sub
    If fProgress.Cancelled Then Exit Sub

    ' Cache du flag "dossier d'envoi"
    ' V25.3 (F9-06) : Elements envoyes reconnu par son EntryID, et ses
    ' sous-dossiers avec lui ; le nom ne sert qu'en repli (boite dont le
    ' dossier n'est pas accessible).
    Dim isSentFolder As Boolean
    isSentFolder = parentIsSent
    If Not isSentFolder Then
        If Len(m_SentID) > 0 Then
            isSentFolder = (CStr(folder.EntryID) = m_SentID)
        Else
            isSentFolder = IsSentFolderName(folder.Name)
        End If
    End If

    ' V23.4 : trace AVANT l'acces aux Items (un dossier casse peut bloquer ici)
    Heartbeat "DOSSIER " & folder.FolderPath

    ' V22 : Cache des Items dans une variable locale
    ' V25.3 (F9-03) : restreints a la periode si un filtre de date est actif.
    Dim folderItems As Object
    Set folderItems = PeriodItems(folder)

    Dim itemCount As Long
    itemCount = 0
    If Not folderItems Is Nothing Then itemCount = folderItems.Count

    If itemCount > 0 Then
        Heartbeat "  items=" & itemCount & " | deja examines=" & m_TotalScanned
        Dim item As Object
        Dim counter As Long: counter = 0
        Dim localProcessed As Long: localProcessed = 0
        Dim i As Long
        Dim msgClass As String

        ' V22 : Boucle par INDEX (plus rapide et plus robuste)
        For i = 1 To itemCount
            If fProgress.Cancelled Then Exit Sub

            ' V23.4 : position courante (index BRUT, y compris items ignores)
            If i Mod PROGRESS_UPDATE_FREQUENCY = 0 Then
                Heartbeat "  idx " & i & " / " & itemCount & " | total=" & m_TotalScanned
            End If

            Set item = Nothing
            Set item = folderItems.Item(i)

            If Not item Is Nothing Then
                ' V22 : Test via MessageClass (plus rapide que TypeName)
                msgClass = ""
                msgClass = item.MessageClass

                ' IPM.Note = Email standard, IPM.Note.* = variantes d'emails
                If Left$(msgClass, 8) = "IPM.Note" Then
                    counter = counter + 1

                    If (Not g_TestMode) Or (counter Mod TEST_SAMPLING = 0) Then
                        ' V25.3 (F9-03) : seuls les mails RETENUS comptent comme
                        ' analyses ; la progression suit les mails examines.
                        If ProcessEmail(item, storeName, isSentFolder) Then
                            m_TotalProcessed = m_TotalProcessed + 1
                        End If
                        m_TotalScanned = m_TotalScanned + 1
                        localProcessed = localProcessed + 1

                        ' V22 : Deux frequences separees
                        ' - Progress update plus frequent (250)
                        ' - DoEvents moins frequent (500) car plus couteux
                        If localProcessed Mod PROGRESS_UPDATE_FREQUENCY = 0 Then
                            fProgress.UpdateProgress m_TotalScanned, m_TotalToProcess, storeName, storeIndex, storeCount

                            Dim elapsed As Double, remaining As Long
                            elapsed = GetElapsedSeconds()
                            If elapsed > 0 And m_TotalScanned > 0 Then
                                remaining = CLng((elapsed / m_TotalScanned) * (m_TotalToProcess - m_TotalScanned))
                                fProgress.SetTimeInfo CLng(elapsed), remaining
                            End If
                        End If

                        If localProcessed Mod DOEVENTS_FREQUENCY = 0 Then
                            DoEvents
                        End If
                    End If
                Else
                    m_TotalIgnored = m_TotalIgnored + 1
                End If
            End If
        Next i
    End If

    Set folderItems = Nothing

    ' Recursion sur sous-dossiers (V25.3 : un sous-dossier des Elements
    ' envoyes est un dossier d'envoi)
    Dim subFolder As Object
    For Each subFolder In folder.Folders
        Call ProcessFolder(subFolder, storeName, storeIndex, storeCount, isSentFolder)
    Next subFolder

    On Error GoTo 0
End Sub

Private Function IsSentFolderName(folderName As String) As Boolean
    If dictSentFolderCache.Exists(folderName) Then
        IsSentFolderName = dictSentFolderCache(folderName)
        Exit Function
    End If

    Dim lowerName As String
    lowerName = LCase(folderName)

    Dim result As Boolean
    result = False

    Select Case lowerName
        Case "elements envoyes", "envoyes", "sent items", "sent", "sent mail", _

             Chr(233) & "l" & Chr(233) & "ments envoy" & Chr(233) & "s"  ' "Elements envoyes"
            result = True

    ' plus qu'en repli, quand les Elements envoyes de la boite ne sont pas
    ' accessibles ; la reconnaissance par adresse (IsSentByUser) fait le
    ' reste.

    dictSentFolderCache(folderName) = result
    IsSentFolderName = result
End Function

' ============================================================================
' V22 : TRAITEMENT D'UN EMAIL - VERSION OPTIMISEE
' - Cache de TOUTES les proprietes COM en variables locales
' - Pre-calcul des cles temporelles via Year/Month/Day (pas Format)
' - Gestion erreur fine
' V25.3 (F9-03) : Function - True si le mail est RETENU (compte dans les
' statistiques), False s'il est ignore (sans objet ni expediteur, journal de
' synchronisation, hors periode). Le message final ne compte plus comme
' "analyses" les mails hors periode.
' ============================================================================
Private Function ProcessEmail(mail As Object, storeName As String, Optional isSentFolder As Boolean = False) As Boolean
    On Error GoTo ErrHandler
    ProcessEmail = False

    ' === V22 : Cache des proprietes COM en variables locales ===
    Dim subj As String
    Dim senderName As String, senderAddr As String, senderEmailType As String
    Dim emailDate As Date
    Dim bodyText As String

    subj = Trim$(CStr(mail.Subject))
    senderName = CStr(mail.SenderName)
    senderAddr = CStr(mail.SenderEmailAddress)
    senderEmailType = CStr(mail.SenderEmailType)
    emailDate = mail.ReceivedTime

    ' === EXCLUSION : Emails vides ou sans sujet ===
    If Len(subj) = 0 Then
        If Len(Trim$(senderName)) = 0 And Len(Trim$(senderAddr)) = 0 Then
            m_TotalIgnored = m_TotalIgnored + 1
            Exit Function
        End If
    End If

    ' === EXCLUSION : Journaux de synchronisation (verification UNIQUE avec LCase cache) ===
    Dim subjLower As String
    subjLower = LCase$(subj)
    If InStr(subjLower, "journal de synchronisation") > 0 Or _
       InStr(subjLower, "sync issues") > 0 Or _
       InStr(subjLower, "synchronization log") > 0 Or _
       InStr(subjLower, "probleme de synchronisation") > 0 Then
        m_TotalIgnored = m_TotalIgnored + 1
        Exit Function
    End If

    ' === FILTRES DATES ===
    ' V25.3 (F9-02) : en delta, g_PeriodStart est la fin EXACTE de l'etat
    ' repris, borne INCLUSE de ses compteurs : seuls les mails strictement
    ' posterieurs sont comptes (aucun mail compte deux fois).
    Dim outOfPeriod As Boolean
    If g_DeltaMode Then
        outOfPeriod = (emailDate <= g_PeriodStart)
    Else
        outOfPeriod = (emailDate < g_PeriodStart)
    End If
    If outOfPeriod Or emailDate > g_PeriodEnd Then
        m_TotalIgnored = m_TotalIgnored + 1
        Exit Function
    End If

    Dim yearNum As Long
    yearNum = Year(emailDate)
    If yearNum < 2000 Then
        m_TotalIgnored = m_TotalIgnored + 1
        Exit Function
    End If

    ' Date minimum par boite
    If dictStoreDateMin.Exists(storeName) Then
        Dim dateMin As Date
        dateMin = dictStoreDateMin(storeName)
        If dateMin > DateSerial(1990, 1, 1) Then
            If emailDate < dateMin Then
                m_TotalIgnored = m_TotalIgnored + 1
                Exit Function
            End If
        End If
    End If

    ' Le mail est retenu (une erreur plus bas ne change pas ce verdict).
    ProcessEmail = True

    ' === Adresse SMTP de l'expediteur ===
    ' V25.3 (F9-07) : modSmtp (PR_SENDER_SMTP_ADDRESS lue sur l'element, puis
    ' annuaire), avec cache par adresse Exchange. Avant, sans la case
    ' "Resolution Exchange", l'adresse d'un collegue restait au format /o=...
    ' et le mail partait en "inconnu".
    senderAddr = SenderSmtpCached(mail, senderAddr, senderEmailType)

    ' Detection Sent/Received
    Dim isSent As Boolean
    If isSentFolder Then
        isSent = True
    Else
        isSent = IsSentByUser(senderAddr, senderName, storeName)
    End If

    Dim direction As String
    If isSent Then direction = "Sent" Else direction = "Received"

    ' === STATS PAR BOITE ===
    IncrementDict dictStoreStats, storeName & "|" & direction
    IncrementDict dictStoreStats, storeName & "|Total"

    ' V24 : A / Cc / Indirect - uniquement sur les mails RECUS (sur un envoi
    ' la question n'a pas de sens : l'utilisateur est l'expediteur).
    ' V25.3 : role garde pour le taux de reponse (F9-15).
    Dim recipRole As String
    If Not isSent Then
        recipRole = RecipientRole(mail, storeName)
        IncrementDict dictToCc, storeName & "|" & Format$(emailDate, "YYYY-MM") & _
                                "|" & recipRole
    End If

    ' === V22 : Pre-calcul OPTIMISE des cles temporelles ===
    ' Format() est tres lent ; Year/Month/Day + concatenation sont 10x plus rapides
    Dim monthNum As Long, dayNum As Long, hourNum As Long, dowNum As Long
    monthNum = Month(emailDate)
    dayNum = Day(emailDate)
    hourNum = Hour(emailDate)
    dowNum = Weekday(emailDate, vbMonday)

    Dim yearStr As String, monthStr As String, dayStr As String, hourStr As String
    yearStr = CStr(yearNum)
    monthStr = Format$(monthNum, "00")
    dayStr = Format$(dayNum, "00")
    hourStr = Format$(hourNum, "00")

    Dim monthKey As String, dayKey As String
    monthKey = yearStr & "-" & monthStr
    dayKey = monthKey & "-" & dayStr

    Dim weekKey As String, yearKey As String, quarterKey As String, semesterKey As String
    yearKey = yearStr
    weekKey = yearStr & "-W" & Format$(DatePart("ww", emailDate, vbMonday), "00")
    quarterKey = yearStr & "-Q" & ((monthNum - 1) \ 3 + 1)
    If monthNum <= 6 Then
        semesterKey = yearStr & "-S1"
    Else
        semesterKey = yearStr & "-S2"
    End If

    Dim hourKey As String, dowKey As String
    hourKey = hourStr
    dowKey = CStr(dowNum)

    ' === STATS TEMPORELLES ===
    IncrementDict dictMonthly, monthKey & "|" & storeName & "|" & direction
    IncrementDict dictMonthly, monthKey & "|" & storeName & "|Total"
    IncrementDict dictMonthly, monthKey & "|_TOTAL|" & direction
    IncrementDict dictMonthly, monthKey & "|_TOTAL|Total"

    IncrementDict dictWeekly, weekKey & "|" & direction
    IncrementDict dictWeekly, weekKey & "|Total"

    IncrementDict dictDaily, dayKey & "|" & direction
    IncrementDict dictDaily, dayKey & "|Total"

    IncrementDict dictQuarterly, quarterKey & "|" & direction
    IncrementDict dictQuarterly, quarterKey & "|Total"

    IncrementDict dictYearly, yearKey & "|" & direction
    IncrementDict dictYearly, yearKey & "|Total"

    IncrementDict dictSemesterly, semesterKey & "|" & storeName & "|" & direction
    IncrementDict dictSemesterly, semesterKey & "|" & storeName & "|Total"

    IncrementDict dictHourly, hourKey & "|" & direction
    IncrementDict dictHourly, hourKey & "|Total"

    IncrementDict dictDayOfWeek, dowKey & "|" & direction
    IncrementDict dictDayOfWeek, dowKey & "|Total"

    ' === HEATMAPS ===
    Dim heatKey As String
    heatKey = dowKey & "|" & hourNum
    If isSent Then
        IncrementDict dictHeatmapSent, heatKey
    Else
        IncrementDict dictHeatmapReceived, heatKey
    End If
    IncrementDict dictHeatmapTotal, heatKey

    ' ============================================================================
    ' LOGIQUE INTERNE / EXTERNE / INCONNU
    ' ============================================================================
    Dim storeDomain As String
    storeDomain = ""
    If dictStoreDomains.Exists(storeName) Then storeDomain = LCase$(CStr(dictStoreDomains(storeName)))

    Dim emailClassification As String
    emailClassification = "Unknown"

    Dim displayRecip As String  ' V22 : pour raw data en fin
    Dim displaySender As String
    Dim recipName As String, recipAddr As String

    If isSent Then
        ' === EMAIL ENVOYE : Analyser les destinataires ===
        Dim recipients As Object
        Set recipients = mail.Recipients

        Dim recipCount As Long: recipCount = 0
        Dim hasExternal As Boolean: hasExternal = False
        Dim hasInternal As Boolean: hasInternal = False
        Dim allUnknown As Boolean: allUnknown = True

        Dim recip As Object

        ' V25.3 (F9-07) : adresse via modSmtp (cache), verdict par
        ' ClassifyAddress (domaines internes de config.ini pour une boite
        ' interne). Plus de case "Resolution Exchange".
        For Each recip In recipients
            recipCount = recipCount + 1

            recipName = CStr(recip.Name)
            recipAddr = RecipientSmtpCached(recip)

            displayRecip = GetBestContactDisplay(recipAddr, recipName)

            If Len(displayRecip) > 0 Then
                IncrementDict dictSentTo, displayRecip
            End If

            Select Case ClassifyAddress(recipAddr, storeDomain)
                Case "Internal"
                    allUnknown = False
                    hasInternal = True
                    If Len(displayRecip) > 0 Then IncrementDict dictSentToInternal, displayRecip
                Case "External"
                    allUnknown = False
                    hasExternal = True
                    If Len(displayRecip) > 0 Then IncrementDict dictSentToExternal, displayRecip
                Case Else
                    If Len(displayRecip) > 0 Then IncrementDict dictSentToUnknown, displayRecip
            End Select
        Next recip

        Set recipients = Nothing

        If allUnknown Then
            emailClassification = "Unknown"
        ElseIf hasExternal Then
            emailClassification = "External"
        Else
            emailClassification = "Internal"
        End If

        If recipCount > 0 Then
            IncrementDict dictNbRecipients, CStr(recipCount)
        End If

        ' === Length message envoye (V20/V22) ===
        If g_AnalyzeLength Then
            bodyText = CStr(mail.Body)
            Dim msgLen As Long
            msgLen = ExtractCurrentMessageLength(bodyText)
            If msgLen > 0 Then
                Dim lenKey As String
                ' V22 : Cle numerique simple au lieu de Format(...) - plus rapide
                lenKey = CStr(dictMsgLength.Count + 1)
                dictMsgLength.Add lenKey, msgLen

                If Not dictMsgLengthByStore.Exists(storeName) Then
                    Dim newStoreDict As Object
                    Set newStoreDict = CreateObject("Scripting.Dictionary")
                    Set dictMsgLengthByStore(storeName) = newStoreDict
                End If
                Dim storeDict As Object
                Set storeDict = dictMsgLengthByStore(storeName)
                storeDict.Add CStr(storeDict.Count + 1), msgLen
            End If
        End If

    Else
        ' === EMAIL RECU : Analyser l'expediteur ===
        displaySender = GetBestContactDisplay(senderAddr, senderName)

        If Len(displaySender) > 0 Then
            IncrementDict dictReceivedFrom, displaySender
        End If

        ' V25.3 (F9-07) : senderAddr est deja l'adresse SMTP (modSmtp).
        emailClassification = ClassifyAddress(senderAddr, storeDomain)
        Select Case emailClassification
            Case "Internal"
                If Len(displaySender) > 0 Then IncrementDict dictReceivedFromInternal, displaySender
            Case "External"
                If Len(displaySender) > 0 Then IncrementDict dictReceivedFromExternal, displaySender
            Case Else
                If Len(displaySender) > 0 Then IncrementDict dictReceivedFromUnknown, displaySender
        End Select

        ' V25.3 (F9-15) : reponse faite depuis Outlook, et en combien de temps.
        IncrementDict dictReply, storeName & "|" & monthKey & "|" & recipRole & "|" & _
                                 ReplyBucket(mail, emailDate)
    End If

    ' Stats Internal/External
    IncrementDict dictInternalExternal, storeName & "|" & emailClassification
    IncrementDict dictInternalExternal, "_TOTAL|" & emailClassification

    ' === Raw data (5000 premiers) ===
    If dictRawData.Count < MAX_RAW_DATA Then
        Dim rawKey As String, rawVal As String
        rawKey = CStr(dictRawData.Count + 1)  ' V22 : plus de Format()

        Dim dateStr As String
        ' V22 : Concatenation directe au lieu de Format - beaucoup plus rapide
        dateStr = yearStr & "-" & monthStr & "-" & dayStr & " " & hourStr & ":" & Format$(Minute(emailDate), "00")

        rawVal = storeName & "|" & dateStr & "|" & direction & "|"
        If isSent Then
            rawVal = rawVal & Left$(GetBestContactDisplay(recipAddr, recipName), 50)
        Else
            rawVal = rawVal & Left$(displaySender, 50)
        End If
        rawVal = rawVal & "|" & Left$(subj, 100)
        dictRawData.Add rawKey, rawVal
    End If

    Exit Function

ErrHandler:
    ' V22 : Erreur sur un email = on passe au suivant, on n'interrompt pas
    ' V25.3 (F9-12) : 10 premieres erreurs dans debug.log et le journal de bord.
    m_TotalErrors = m_TotalErrors + 1
    If m_TotalErrors <= 10 Then
        Dim errTxt As String
        errTxt = Err.Number & " - " & Err.Description   ' avant Heartbeat (On Error Resume Next remet Err a zero)
        Heartbeat "  ProcessEmail erreur " & errTxt & " (email ignore)"
        modDebugLog.DError MOD_NAME, "Mail ignore sur erreur " & errTxt
    End If
End Function

' Adresse SMTP (minuscules) de l'expediteur, "" si introuvable.
'   rawAddr / addrType : SenderEmailAddress / SenderEmailType deja lus.
Private Function SenderSmtpCached(ByVal mail As Object, ByVal rawAddr As String, _
                                  ByVal addrType As String) As String
    On Error Resume Next
    SenderSmtpCached = ""
    If addrType = "SMTP" And InStr(rawAddr, "@") > 0 Then
        SenderSmtpCached = LCase$(Trim$(rawAddr))
        Exit Function
    End If
    If Len(rawAddr) > 0 And Not dictResolveCache Is Nothing Then
        If dictResolveCache.Exists(rawAddr) Then
            SenderSmtpCached = CStr(dictResolveCache(rawAddr))
            Exit Function
        End If
    End If
    Dim s As String
    s = LCase$(modSmtp.SenderSmtp(mail))
    If Len(rawAddr) > 0 And Not dictResolveCache Is Nothing Then dictResolveCache(rawAddr) = s
    SenderSmtpCached = s
End Function

' Adresse SMTP (minuscules) d'un destinataire, "" si introuvable.
Private Function RecipientSmtpCached(ByVal recip As Object) As String
    On Error Resume Next
    RecipientSmtpCached = ""
    Dim raw As String
    raw = ""
    raw = CStr(recip.Address)
    If InStr(raw, "@") > 0 And Left$(UCase$(raw), 3) <> "/O=" Then
        RecipientSmtpCached = LCase$(Trim$(raw))
        Exit Function
    End If
    If Len(raw) > 0 And Not dictResolveCache Is Nothing Then
        If dictResolveCache.Exists(raw) Then
            RecipientSmtpCached = CStr(dictResolveCache(raw))
            Exit Function
        End If
    End If
    Dim s As String
    s = LCase$(modSmtp.RecipientSmtp(recip))
    If Len(raw) > 0 And Not dictResolveCache Is Nothing Then dictResolveCache(raw) = s
    RecipientSmtpCached = s
End Function

'----------------------------------------------------------------------------
' ClassifyAddress (V25.3, F9-07) - "Internal" / "External" / "Unknown".
' Boite interne (domaine vide ou l'un des InternalDomains de config.ini) :
' interne = l'un des domaines internes, sous-domaines compris
' (modSmtp.IsInternalDomain). Autre boite : meme domaine que la boite.
' Avant : un seul domaine par boite - les collegues du second domaine
' interne etaient comptes comme externes.
'----------------------------------------------------------------------------
Private Function ClassifyAddress(ByVal addr As String, ByVal storeDomain As String) As String
    On Error Resume Next
    ClassifyAddress = "Unknown"
    If Not IsValidSMTPAddress(addr) Then Exit Function
    Dim dom As String
    dom = LCase$(ExtractDomain(addr))
    If Len(dom) = 0 Then Exit Function
    Dim isInternal As Boolean
    If Len(storeDomain) = 0 Then
        isInternal = modSmtp.IsInternalDomain(dom)
    ElseIf modSmtp.IsInternalDomain(storeDomain) Then
        isInternal = modSmtp.IsInternalDomain(dom)
    Else
        isInternal = (dom = storeDomain)
    End If
    If isInternal Then
        ClassifyAddress = "Internal"
    Else
        ClassifyAddress = "External"
    End If
End Function

'----------------------------------------------------------------------------
' ReplyBucket (V25.3, F9-15) - reponse faite depuis Outlook a un mail RECU,
' lue sur le mail lui-meme : Exchange y pose la derniere action executee
' (PR_LAST_VERB_EXECUTED : 102 repondre, 103 repondre a tous, 104
' transferer) et son heure UTC (PR_LAST_VERB_EXECUTION_TIME). Une seule
' lecture (GetProperties) par mail recu, aucun parcours des envoyes.
' Tranches ADDITIVES (compatibles avec _ETAT et le delta) :
'   H1 < 1 h, H4 1-4 h, H24 4-24 h, D3 1-3 j, D3P > 3 j,
'   Unknown = repondu, heure illisible ; None = pas de reponse (un
'   transfert ne compte pas comme une reponse).
'----------------------------------------------------------------------------
Private Function ReplyBucket(ByVal mail As Object, ByVal receivedAt As Date) As String
    On Error Resume Next
    ReplyBucket = "None"

    Dim pa As Object, v As Variant, verb As Long
    Set pa = Nothing
    Set pa = mail.PropertyAccessor
    If pa Is Nothing Then Exit Function
    v = Empty
    v = pa.GetProperties(Array(PR_LAST_VERB, PR_LAST_VERB_TIME))
    If Not IsArray(v) Then Exit Function
    If IsError(v(LBound(v))) Or Not IsNumeric(v(LBound(v))) Then Exit Function
    verb = 0
    verb = CLng(v(LBound(v)))
    If verb <> 102 And verb <> 103 Then Exit Function

    ReplyBucket = "Unknown"
    If VarType(v(LBound(v) + 1)) <> vbDate Then Exit Function
    Dim doneAt As Date
    doneAt = 0
    doneAt = pa.UTCToLocalTime(CDate(v(LBound(v) + 1)))
    If doneAt = 0 Then Exit Function

    Dim hrs As Double
    hrs = (CDbl(doneAt) - CDbl(receivedAt)) * 24#
    If hrs < 1 Then
        ReplyBucket = "H1"
    ElseIf hrs < 4 Then
        ReplyBucket = "H4"
    ElseIf hrs < 24 Then
        ReplyBucket = "H24"
    ElseIf hrs < 72 Then
        ReplyBucket = "D3"
    Else
        ReplyBucket = "D3P"
    End If
End Function

'============================================================================
' V24 - ROLE DU DESTINATAIRE : A / Cc / Indirect
'
' Sur un mail RECU, dit si l'utilisateur figure en A, en Cc, ou dans aucun
' des deux ("Indirect" : liste de diffusion, boite partagee, Cci).
'
' V25.3 (F9-08) : quand la boite connait l'adresse de l'utilisateur
' (BuildStoreIdentities), on parcourt la collection Recipients (Type 1 = A,
' 2 = Cc) et on compare les ADRESSES (modSmtp, cache par adresse Exchange).
' Avant : les chaines .To / .CC (des NOMS affiches) etaient comparees au
' mot-cle (nom de famille) - un homonyme en A alors que l'utilisateur
' etait en Cc donnait "A". Le mot-cle sur .To / .CC ne sert plus qu'en
' repli, pour une boite sans adresse connue. Un panier "Indirect"
' anormalement gros signalerait des reperes mal regles.
'
' Les trois categories totalisent 100 % : tout mail recu tombe dans une et
' une seule.
'============================================================================
Private Function RecipientRole(mail As Object, storeName As String) As String
    On Error Resume Next
    RecipientRole = "Indirect"

    Dim ids As Object
    Set ids = StoreIdentity(storeName)
    If Not ids Is Nothing Then
        If ids.Count > 0 Then
            Dim rcps As Object, r As Object, i As Long, n As Long, typ As Long, addr As String
            Set rcps = Nothing
            Set rcps = mail.Recipients
            If rcps Is Nothing Then Exit Function
            n = 0
            n = rcps.Count
            For i = 1 To n
                Set r = Nothing
                Set r = rcps.Item(i)
                If Not r Is Nothing Then
                    typ = 0
                    typ = r.Type
                    If typ = olTo Or typ = olCC Then
                        addr = RecipientSmtpCached(r)
                        If Len(addr) > 0 Then
                            If ids.Exists(addr) Then
                                If typ = olTo Then
                                    RecipientRole = "To"
                                    Exit Function
                                End If
                                RecipientRole = "Cc"   ' on cherche encore un A
                            End If
                        End If
                    End If
                End If
            Next i
            Exit Function
        End If
    End If

    ' Repli (boite sans adresse connue) : mot-cle dans les chaines .To / .CC
    Dim toStr As String, ccStr As String
    toStr = LCase$(CStr(mail.To))
    ccStr = LCase$(CStr(mail.CC))
    If Len(toStr) = 0 And Len(ccStr) = 0 Then Exit Function

    Dim motCle As String
    If dictStoreKeyword.Exists(storeName) Then motCle = LCase$(CStr(dictStoreKeyword(storeName)))

    If MatchesIdentity(toStr, "", motCle) Then
        RecipientRole = "To"
    ElseIf MatchesIdentity(ccStr, "", motCle) Then
        RecipientRole = "Cc"
    End If
End Function

Private Function MatchesIdentity(hay As String, userEmail As String, motCle As String) As Boolean
    MatchesIdentity = False
    If Len(hay) = 0 Then Exit Function
    If Len(userEmail) > 0 Then
        If InStr(hay, userEmail) > 0 Then MatchesIdentity = True: Exit Function
    End If
    If Len(motCle) > 0 Then
        If InStr(hay, motCle) > 0 Then MatchesIdentity = True
    End If
End Function

' ============================================================================
' EXTRACTION LONGUEUR MESSAGE - optimise V22
' ============================================================================
Private Function ExtractCurrentMessageLength(bodyText As String) As Long
    On Error Resume Next
    ExtractCurrentMessageLength = 0
    If Len(bodyText) = 0 Then Exit Function

    Dim msgText As String
    msgText = bodyText

    ' V22 : Test les marqueurs les plus frequents en premier
    Dim minPos As Long: minPos = Len(msgText) + 1
    Dim pos As Long

    ' Marqueur 1 : "-----Original Message-----" / "-----Message d'origine-----"
    pos = InStr(1, msgText, "-----", vbTextCompare)
    If pos > 10 And pos < minPos Then
        ' Verifier que c'est bien un marqueur d'en-tete
        Dim checkStr As String
        checkStr = Mid$(msgText, pos, 30)
        If InStr(1, checkStr, "Message", vbTextCompare) > 0 Or InStr(1, checkStr, "Original", vbTextCompare) > 0 Then
            minPos = pos
        End If
    End If

    ' Marqueur 2 : "________________________________" (32 underscores)
    pos = InStr(1, msgText, "________________________________", vbTextCompare)
    If pos > 10 And pos < minPos Then minPos = pos

    ' Marqueur 3 : "From:" en debut de ligne
    pos = InStr(1, msgText, vbCrLf & "From:", vbTextCompare)
    If pos > 10 And pos < minPos Then minPos = pos

    ' Marqueur 4 : "De :" en debut de ligne
    pos = InStr(1, msgText, vbCrLf & "De :", vbTextCompare)
    If pos > 10 And pos < minPos Then minPos = pos

    ' Marqueur 5 : "Le " suivi de date (debut de reponse francaise)
    ' V25.3 (F9-09) : seulement si la ligne (ou la suivante, l'en-tete etant
    ' souvent coupe en deux) contient "a ecrit" ou "wrote" - "Le 12/11/2026
    ' a 10:02, X a ecrit :". Avant : la premiere ligne du message commencant
    ' par l'article "Le" coupait le message a cet endroit.
    Dim segEnd As Long, seg As String
    pos = InStr(1, msgText, vbCrLf & "Le ", vbTextCompare)
    Do While pos > 0 And pos < minPos
        If pos > 10 Then
            segEnd = InStr(pos + 2, msgText, vbCrLf)
            If segEnd > 0 Then segEnd = InStr(segEnd + 2, msgText, vbCrLf)
            If segEnd = 0 Then segEnd = Len(msgText) + 1
            If segEnd - pos > 400 Then segEnd = pos + 400
            seg = Mid$(msgText, pos, segEnd - pos)
            If InStr(1, seg, "a " & Chr(233) & "crit", vbTextCompare) > 0 Or _
               InStr(1, seg, "a ecrit", vbTextCompare) > 0 Or _
               InStr(1, seg, "wrote", vbTextCompare) > 0 Then
                minPos = pos
                Exit Do
            End If
        End If
        pos = InStr(pos + 2, msgText, vbCrLf & "Le ", vbTextCompare)
    Loop

    If minPos <= Len(msgText) Then
        msgText = Left$(msgText, minPos - 1)
    End If

    ' Nettoyer
    msgText = Replace(msgText, vbCrLf, " ")
    msgText = Replace(msgText, vbCr, " ")
    msgText = Replace(msgText, vbLf, " ")
    msgText = Trim$(msgText)

    ExtractCurrentMessageLength = Len(msgText)
End Function

' ----------------------------------------------------------------------------
' ResolveDesktopPath (V23.2)
' Real Desktop folder, OneDrive Known-Folder-Move aware. USERPROFILE\Desktop
' is WRONG on redirected machines (invisible legacy folder, or missing).
' First existing candidate wins; falls back to the user profile root.
' ----------------------------------------------------------------------------
Private Function ResolveDesktopPath() As String
    On Error Resume Next
    Dim cands As Variant
    Dim i As Long
    Dim p As String
    cands = Array( _
        Environ("OneDriveCommercial") & "\Bureau", _
        Environ("OneDriveCommercial") & "\Desktop", _
        Environ("OneDrive") & "\Bureau", _
        Environ("OneDrive") & "\Desktop", _
        Environ("USERPROFILE") & "\OneDrive\Bureau", _
        Environ("USERPROFILE") & "\OneDrive\Desktop", _
        Environ("USERPROFILE") & "\Bureau", _
        Environ("USERPROFILE") & "\Desktop")
    For i = LBound(cands) To UBound(cands)
        p = CStr(cands(i))
        ' Skip candidates whose environment variable was empty ("\Bureau").
        If Left$(p, 1) <> "\" Then
            If Dir(p, vbDirectory) <> "" Then
                ResolveDesktopPath = p
                Exit Function
            End If
        End If
    Next i
    ResolveDesktopPath = Environ("USERPROFILE")
End Function

' lui-meme, a cote de config.ini et de debug.log). Ce module tourne dans le
' projet VBA d'Outlook, a cote de modConfig (il en utilise deja T() / A()).
' Repli sur la racine du Bureau si le dossier est inaccessible.
'============================================================================
Private Function ReportFolderPath() As String
    On Error Resume Next
    Dim d As String, fso As Object
    d = ""
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not (fso Is Nothing) Then
        d = modConfig.MacroVBADir()
        If Len(d) > 0 Then
            If Not fso.FolderExists(d) Then fso.CreateFolder d
            If Not fso.FolderExists(d) Then d = ""
        End If
    End If
    If Len(d) = 0 Then d = ResolveDesktopPath()
    ReportFolderPath = d
End Function

' BuildSheet - construit UNE feuille, isole ses erreurs.
' Une erreur dans un constructeur sans gestionnaire remonte ici (et plus
' jusqu'a GenerateExcelReport) : elle est tracee, la generation continue.
' ----------------------------------------------------------------------------
Private Sub BuildSheet(ByRef xlWB As Object, ByVal which As String)
    On Error GoTo Fail
    Select Case which
        Case "State":            CreateSheetState xlWB
        Case "Dashboard":        CreateSheetDashboard xlWB
        Case "Resume":           CreateSheetResume xlWB
        Case "ParHeure":         CreateSheetParHeure xlWB
        Case "ParJourSemaine":   CreateSheetParJourSemaine xlWB
        Case "ParMois":          CreateSheetParMois xlWB
        Case "ParAnnee":         CreateSheetParAnnee xlWB
        Case "Evolutions":       CreateSheetEvolutions xlWB
        Case "Heatmaps":         CreateSheetHeatmaps xlWB
        Case "Contacts":         CreateSheetContacts xlWB
        Case "ContactsIntExt":   CreateSheetContactsInternalExternal xlWB
        Case "NbRecipients":     CreateSheetNbRecipients xlWB
        Case "LongueurMessages": CreateSheetLongueurMessages xlWB
        Case "ToCc":             CreateSheetToCc xlWB
        Case "Replies":          CreateSheetReplies xlWB
        Case "DATA":             CreateSheetDATA xlWB
        Case "FillState":        FillStateSheet xlWB
        Case "Polish":           PolishWorkbook xlWB
    End Select
    Exit Sub
Fail:
    Dim errTxt As String
    errTxt = Err.Number & " - " & Err.Description   ' avant Heartbeat (remet Err a zero)
    Heartbeat "!!! FEUILLE " & which & " : erreur " & errTxt
    modDebugLog.DError MOD_NAME, "Feuille " & which & " : erreur " & errTxt
End Sub

' ============================================================================
' UTILITAIRE : Tri decroissant
' ============================================================================
Public Function GetSortedKeysByValueDesc(dict As Object) As Variant
    Dim keys As Variant, vals As Variant
    Dim i As Long, j As Long
    Dim tempK As Variant, tempV As Variant, count As Long

    count = dict.Count
    If count = 0 Then
        GetSortedKeysByValueDesc = Array()
        Exit Function
    End If

    ReDim keys(0 To count - 1)
    ReDim vals(0 To count - 1)

    i = 0
    Dim k As Variant
    For Each k In dict.Keys
        keys(i) = k
        vals(i) = dict(k)
        i = i + 1
    Next k

    ' Tri par insertion pour petites listes, tri rapide pour grandes
    If count < 30 Then
        For i = 0 To count - 2
            For j = i + 1 To count - 1
                If vals(i) < vals(j) Then
                    tempK = keys(i): tempV = vals(i)
                    keys(i) = keys(j): vals(i) = vals(j)
                    keys(j) = tempK: vals(j) = tempV
                End If
            Next j
        Next i
    Else
        Call QuickSortByValueDesc(keys, vals, 0, count - 1)
    End If

    GetSortedKeysByValueDesc = keys
End Function

' V22 : QuickSort pour tri rapide des grandes listes
Private Sub QuickSortByValueDesc(ByRef keys As Variant, ByRef vals As Variant, _
                                   ByVal low As Long, ByVal high As Long)
    Dim i As Long, j As Long
    Dim pivot As Variant, tempK As Variant, tempV As Variant

    If low < high Then
        pivot = vals((low + high) \ 2)
        i = low
        j = high

        Do
            Do While vals(i) > pivot: i = i + 1: Loop
            Do While vals(j) < pivot: j = j - 1: Loop

            If i <= j Then
                tempK = keys(i): tempV = vals(i)
                keys(i) = keys(j): vals(i) = vals(j)
                keys(j) = tempK: vals(j) = tempV
                i = i + 1
                j = j - 1
            End If
        Loop Until i > j

        If low < j Then Call QuickSortByValueDesc(keys, vals, low, j)
        If i < high Then Call QuickSortByValueDesc(keys, vals, i, high)
    End If
End Sub

' V22 : Ecriture des donnees d'evolution en bloc (tres rapide)
Private Function WriteEvolutionDataBlock(ws As Object, startRow As Long, sortedKeys As Variant, _
                                           boitesList As Variant, dict As Object, suffix As String, _
                                           lastCol As Long) As Long
    Dim nRows As Long
    If IsArray(sortedKeys) Then
        nRows = UBound(sortedKeys) - LBound(sortedKeys) + 1
    Else
        nRows = 0
    End If

    If nRows = 0 Then
        WriteEvolutionDataBlock = startRow - 1
        Exit Function
    End If

    Dim nMailboxes As Long
    nMailboxes = UBound(boitesList) - LBound(boitesList) + 1

    Dim dataArr() As Variant
    ReDim dataArr(1 To nRows, 1 To nMailboxes + 1)

    Dim i As Long, j As Long
    For i = 0 To nRows - 1
        dataArr(i + 1, 1) = sortedKeys(i)
        For j = 0 To nMailboxes - 1
            dataArr(i + 1, j + 2) = GetDictValue(dict, CStr(sortedKeys(i)) & "|" & CStr(boitesList(j)) & "|" & suffix)
        Next j
    Next i

    ws.Range(ws.Cells(startRow, 1), ws.Cells(startRow + nRows - 1, lastCol)).Value = dataArr

    WriteEvolutionDataBlock = startRow + nRows - 1
End Function

                                  boitesList As Variant, chartLeft As Long, chartTop As Long, chartTitle As String, _

    On Error Resume Next

' Renvoie un tableau (1..n, 1..4) = libelle, Envoyes, Recus, Total agrege
' sur toutes les boites pour la granularite demandee. Empty si aucune donnee.
Private Function GlobalPeriodArray(ByVal kind As String) As Variant
    Dim srcDict As Object
    Select Case kind
        Case "month": Set srcDict = dictMonthly
        Case "quarter": Set srcDict = dictQuarterly
        Case "semester": Set srcDict = dictSemesterly
        Case "year": Set srcDict = dictYearly
    End Select
    If srcDict Is Nothing Then Exit Function

    Dim setKeys As Object
    Set setKeys = CreateObject("Scripting.Dictionary")
    Dim k As Variant, parts As Variant, pk As String
    For Each k In srcDict.Keys
        parts = Split(CStr(k), "|")
        pk = CStr(parts(0))
        If kind = "month" Then
            If UBound(parts) >= 2 Then
                If CStr(parts(1)) = "_TOTAL" Then
                    If Not setKeys.Exists(pk) Then setKeys.Add pk, True
                End If
            End If
        Else
            If Not setKeys.Exists(pk) Then setKeys.Add pk, True
        End If
    Next k
    If setKeys.Count = 0 Then Exit Function

    Dim sortedKeys As Variant
    sortedKeys = SortDictKeys(setKeys)
    Dim n As Long: n = UBound(sortedKeys) - LBound(sortedKeys) + 1
    Dim outArr() As Variant
    ReDim outArr(1 To n, 1 To 4)

    Dim i As Long, sentV As Long, recvV As Long
    For i = 0 To n - 1
        pk = CStr(sortedKeys(i))
        sentV = 0: recvV = 0
        Select Case kind
            Case "month"
                sentV = GetDictValue(dictMonthly, pk & "|_TOTAL|Sent")
                recvV = GetDictValue(dictMonthly, pk & "|_TOTAL|Received")
            Case "quarter"
                sentV = GetDictValue(dictQuarterly, pk & "|Sent")
                recvV = GetDictValue(dictQuarterly, pk & "|Received")
            Case "year"
                sentV = GetDictValue(dictYearly, pk & "|Sent")
                recvV = GetDictValue(dictYearly, pk & "|Received")
            Case "semester"
                For Each k In dictSemesterly.Keys
                    parts = Split(CStr(k), "|")
                    If UBound(parts) >= 2 Then
                        If CStr(parts(0)) = pk Then
                            If CStr(parts(2)) = "Sent" Then
                                sentV = sentV + CLng(dictSemesterly(k))
                            ElseIf CStr(parts(2)) = "Received" Then
                                recvV = recvV + CLng(dictSemesterly(k))
                            End If
                        End If
                    End If
                Next k
        End Select
        outArr(i + 1, 1) = pk
        outArr(i + 1, 2) = sentV
        outArr(i + 1, 3) = recvV
        outArr(i + 1, 4) = sentV + recvV
    Next i

    GlobalPeriodArray = outArr
End Function

' Fonction de couleur pour Heatmap (Vert -> Jaune -> Rouge)
Private Function GetHeatmapColor(val As Long, maxVal As Long) As Long
    Dim intensity As Double
    intensity = val / maxVal

    Dim r As Long, g As Long, b As Long

    If intensity <= 0.5 Then
        Dim t1 As Double: t1 = intensity * 2
        r = CLng(0 + t1 * 255)
        g = CLng(200 + t1 * 55)
        b = 0
    Else
        Dim t2 As Double: t2 = (intensity - 0.5) * 2
        r = 255
        g = CLng(255 - t2 * 255)
        b = 0
    End If

    GetHeatmapColor = RGB(r, g, b)
End Function

' ============================================================================
' UTILITAIRE : GetTopN - V22 avec QuickSort pour gros dictionnaires
' ============================================================================
' ----------------------------------------------------------------------------
' GetTopN
' V25.1 : dictionnaire vide -> renvoie Empty. L'ancien
' ReDim result(0 To 1, 0 To -1) leve l'erreur 9 en VBA (pas de tableau vide
' par ReDim) : c'est ce qui faisait tomber tout le rapport.
' ----------------------------------------------------------------------------
Public Function GetTopN(dict As Object, n As Long) As Variant
    Dim result() As Variant
    Dim items() As Variant
    Dim i As Long
    Dim count As Long

    GetTopN = Empty
    If dict Is Nothing Then Exit Function
    count = dict.Count
    If count = 0 Or n < 1 Then Exit Function

    ReDim items(0 To 1, 0 To count - 1)
    i = 0
    Dim k As Variant
    For Each k In dict.Keys
        items(0, i) = k
        items(1, i) = dict(k)
        i = i + 1
    Next k

    If count > 30 Then
        Call QuickSortTopN(items, 0, count - 1)
    Else
        Dim j As Long, temp1 As Variant, temp2 As Variant
        For i = 0 To count - 2
            For j = i + 1 To count - 1
                If items(1, i) < items(1, j) Then
                    temp1 = items(0, i): temp2 = items(1, i)
                    items(0, i) = items(0, j): items(1, i) = items(1, j)
                    items(0, j) = temp1: items(1, j) = temp2
                End If
            Next j
        Next i
    End If

    Dim resultCount As Long
    resultCount = n
    If resultCount > count Then resultCount = count

    ReDim result(0 To 1, 0 To resultCount - 1)
    For i = 0 To resultCount - 1
        result(0, i) = items(0, i)
        result(1, i) = items(1, i)
    Next i

    GetTopN = result
End Function

' V22 : QuickSort descendant sur items(0=key, 1=val)
Private Sub QuickSortTopN(ByRef items As Variant, ByVal low As Long, ByVal high As Long)
    Dim i As Long, j As Long
    Dim pivot As Variant, tempK As Variant, tempV As Variant

    If low < high Then
        pivot = items(1, (low + high) \ 2)
        i = low
        j = high

        Do
            Do While items(1, i) > pivot: i = i + 1: Loop
            Do While items(1, j) < pivot: j = j - 1: Loop

            If i <= j Then
                tempK = items(0, i): tempV = items(1, i)
                items(0, i) = items(0, j): items(1, i) = items(1, j)
                items(0, j) = tempK: items(1, j) = tempV
                i = i + 1
                j = j - 1
            End If
        Loop Until i > j

        If low < j Then Call QuickSortTopN(items, low, j)
        If i < high Then Call QuickSortTopN(items, i, high)
    End If
End Sub

                Call SortLongArray(storeLengths)

                Else

                storeArr(idxStore, 4) = storeMediane
                storeArr(idxStore, 5) = storeMin

            End If
        Next storeName

        End If
    End If

    ws.Columns("A:F").AutoFit
End Sub

' ============================================================================
' UTILITAIRE : Tri tableau de Long (QuickSort)
' ============================================================================
Public Sub SortLongArray(ByRef arr() As Long)
    If UBound(arr) - LBound(arr) < 1 Then Exit Sub
    Call QuickSortLong(arr, LBound(arr), UBound(arr))
End Sub

Private Sub QuickSortLong(ByRef arr() As Long, ByVal low As Long, ByVal high As Long)
    Dim pivot As Long, i As Long, j As Long, temp As Long

    If low < high Then
        Dim mid As Long: mid = (low + high) \ 2
        If arr(low) > arr(mid) Then: temp = arr(low): arr(low) = arr(mid): arr(mid) = temp
        If arr(low) > arr(high) Then: temp = arr(low): arr(low) = arr(high): arr(high) = temp
        If arr(mid) > arr(high) Then: temp = arr(mid): arr(mid) = arr(high): arr(high) = temp

        pivot = arr(mid)
        i = low
        j = high

        Do
            Do While arr(i) < pivot: i = i + 1: Loop
            Do While arr(j) > pivot: j = j - 1: Loop

            If i <= j Then
                temp = arr(i): arr(i) = arr(j): arr(j) = temp
                i = i + 1
                j = j - 1
            End If
        Loop Until i > j

        If low < j Then Call QuickSortLong(arr, low, j)
        If i < high Then Call QuickSortLong(arr, i, high)
    End If
End Sub

' ============================================================================
' UTILITAIRE : Tri des cles d'un dictionnaire
' ============================================================================
Public Function SortDictKeys(dict As Object) As Variant
    If dict Is Nothing Then
        SortDictKeys = Array()
        Exit Function
    End If

    If dict.Count = 0 Then
        SortDictKeys = Array()
        Exit Function
    End If

    Dim keys() As String
    ReDim keys(0 To dict.Count - 1)

    Dim i As Long: i = 0
    Dim k As Variant
    For Each k In dict.Keys
        keys(i) = CStr(k)
        i = i + 1
    Next k

    Call QuickSortStrings(keys, LBound(keys), UBound(keys))

    SortDictKeys = keys
End Function

Private Sub QuickSortStrings(ByRef arr() As String, ByVal low As Long, ByVal high As Long)
    Dim pivot As String, i As Long, j As Long, temp As String

    If low < high Then
        pivot = arr((low + high) \ 2)
        i = low
        j = high

        Do
            Do While arr(i) < pivot: i = i + 1: Loop
            Do While arr(j) > pivot: j = j - 1: Loop

            If i <= j Then
                temp = arr(i): arr(i) = arr(j): arr(j) = temp
                i = i + 1
                j = j - 1
            End If
        Loop Until i > j

        If low < j Then Call QuickSortStrings(arr, low, j)
        If i < high Then Call QuickSortStrings(arr, i, high)
    End If
End Sub

Private Sub CreateSheetToCc(ByRef xlWB As Object)
    On Error GoTo Fin

    Dim ws As Object
    Set ws = xlWB.Worksheets.Add(After:=xlWB.Worksheets(xlWB.Worksheets.count))
    ws.Name = T("A - Cc", "To - Cc")

    ws.Cells(1, 1).Value = T("R~EPARTITION A / Cc / INDIRECT (mails re" & Chr(231) & "us)", _
                             "TO / Cc / INDIRECT BREAKDOWN (received mail)")
    ws.Cells(1, 1).Font.Bold = True
    ws.Cells(1, 1).Font.Size = 14

    ws.Cells(2, 1).Value = T("Indirect = ni en A ni en Cc : liste de diffusion, bo~ite partag~ee, Cci.", _
                             "Indirect = neither To nor Cc: distribution list, shared mailbox, Bcc.")

    If dictToCc.count = 0 Then
        ws.Cells(4, 1).Value = T("Aucun mail re" & Chr(231) & "u sur la p~eriode.", "No received mail over the period.")
        Exit Sub
    End If

    Dim byStore As Object, byMonth As Object, months As Object, stores As Object
    Set byStore = CreateObject("Scripting.Dictionary")
    Set byMonth = CreateObject("Scripting.Dictionary")
    Set months = CreateObject("Scripting.Dictionary")
    Set stores = CreateObject("Scripting.Dictionary")

    Dim k As Variant, parts() As String, gTo As Long, gCc As Long, gIn As Long
    For Each k In dictToCc.Keys
        parts = Split(CStr(k), "|")
        If UBound(parts) = 2 Then
            IncrementDict byStore, parts(0) & "|" & parts(2), CLng(dictToCc(k))
            IncrementDict byMonth, parts(1) & "|" & parts(2), CLng(dictToCc(k))
            If Not months.Exists(parts(1)) Then months.Add parts(1), 1
            If Not stores.Exists(parts(0)) Then stores.Add parts(0), 1
            Select Case parts(2)
                Case "To":  gTo = gTo + CLng(dictToCc(k))
                Case "Cc":  gCc = gCc + CLng(dictToCc(k))
                Case Else:  gIn = gIn + CLng(dictToCc(k))
            End Select
        End If
    Next k

    ' --- Bloc 1 : par boite ---
    Dim r As Long
    r = 4
    ws.Cells(r, 1).Value = T("PAR BO~ITE", "BY MAILBOX")
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 1).Font.Size = 12

    ws.Cells(r, 6).Value = "% Cc"
    ws.Cells(r, 7).Value = T("% Indirect", "% Indirect")
    ws.Range(ws.Cells(r, 1), ws.Cells(r, 7)).Font.Bold = True
    Dim headerRowStores As Long
    headerRowStores = r

    Dim sortedStores As Variant
    sortedStores = SortDictKeys(stores)
    Dim i As Long, sn As String, a As Long, b As Long, c As Long, tot As Long
    For i = LBound(sortedStores) To UBound(sortedStores)
        sn = CStr(sortedStores(i))
        a = GetDictValue(byStore, sn & "|To")
        b = GetDictValue(byStore, sn & "|Cc")
        c = GetDictValue(byStore, sn & "|Indirect")
        tot = a + b + c
        If tot > 0 Then
            r = r + 1
            ws.Cells(r, 1).Value = sn
            ws.Cells(r, 1).Interior.Color = GetStoreColor(sn)
            ws.Cells(r, 1).Font.Color = RGB(255, 255, 255)
            ws.Cells(r, 2).Value = a
            ws.Cells(r, 3).Value = b
            ws.Cells(r, 4).Value = c
            ws.Cells(r, 5).Value = a / tot
            ws.Cells(r, 6).Value = b / tot
            ws.Cells(r, 7).Value = c / tot
        End If
    Next i
    ws.Range(ws.Cells(headerRowStores + 1, 2), ws.Cells(r, 4)).NumberFormat = "#,##0"
    ws.Range(ws.Cells(headerRowStores + 1, 5), ws.Cells(r, 7)).NumberFormat = "0.0%"

    ' --- Bloc 2 : consolide ---
    r = r + 2
    ws.Cells(r, 1).Value = T("TOUTES BO~ITES CONFONDUES", "ALL MAILBOXES COMBINED")
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 1).Font.Size = 12
    Dim gTot As Long
    gTot = gTo + gCc + gIn
    r = r + 1
    ws.Cells(r, 1).Value = "A / To":       ws.Cells(r, 2).Value = gTo
    If gTot > 0 Then ws.Cells(r, 3).Value = gTo / gTot
    r = r + 1
    ws.Cells(r, 1).Value = "Cc":           ws.Cells(r, 2).Value = gCc
    If gTot > 0 Then ws.Cells(r, 3).Value = gCc / gTot
    r = r + 1
    ws.Cells(r, 1).Value = T("Indirect", "Indirect"): ws.Cells(r, 2).Value = gIn
    If gTot > 0 Then ws.Cells(r, 3).Value = gIn / gTot
    r = r + 1
    ws.Cells(r, 1).Value = T("Total re" & Chr(231) & "u", "Total received")
    ws.Cells(r, 2).Value = gTot
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 2).Font.Bold = True
    ws.Range(ws.Cells(r - 3, 2), ws.Cells(r, 2)).NumberFormat = "#,##0"
    ws.Range(ws.Cells(r - 3, 3), ws.Cells(r - 1, 3)).NumberFormat = "0.0%"

    ' --- Bloc 3 : suivi mensuel des pourcentages ---

'----------------------------------------------------------------------------
' CreateSheetReplies (V25.3, F9-15) - taux et delai de reponse aux mails
' RECUS, d'apres la derniere action posee par Exchange sur chaque mail
' (ReplyBucket). Deux blocs : par boite (avec une ligne "toutes boites"),
' puis suivi mensuel avec la courbe des taux. "Taux en A" = mails recus ou
' l'utilisateur etait en A (plus parlant que le taux global, qui melange
' copies et listes de diffusion).
'----------------------------------------------------------------------------
Private Sub CreateSheetReplies(ByRef xlWB As Object)
    On Error GoTo Fin

    Dim ws As Object
    Set ws = xlWB.Worksheets.Add(After:=xlWB.Worksheets(xlWB.Worksheets.count))
    ws.Name = T("R~eponses", "Replies")

    ws.Cells(1, 1).Value = T("TAUX ET D~ELAI DE R~EPONSE (mails re" & Chr(231) & "us)", _
                             "REPLY RATE AND DELAY (received mail)")
    ws.Cells(1, 1).Font.Bold = True
    ws.Cells(1, 1).Font.Size = 14
    ws.Cells(2, 1).Value = T("R~eponse = R~epondre ou R~epondre ~a tous fait depuis Outlook sur le mail re" & Chr(231) & "u " & _
                             "(marque pos~ee par Exchange) ; un transfert ne compte pas. " & _
                             "D~elai = heure de la r~eponse - heure de r~eception.", _
                             "Reply = Reply or Reply All done from Outlook on the received mail " & _
                             "(flag set by Exchange); a forward does not count. " & _
                             "Delay = reply time - received time.")
    ws.Cells(2, 1).Font.Italic = True

    If dictReply Is Nothing Then Exit Sub
    If dictReply.count = 0 Then
        ws.Cells(4, 1).Value = T("Aucun mail re" & Chr(231) & "u sur la p~eriode.", "No received mail over the period.")
        Exit Sub
    End If

    ' --- Agregation : boite | mois | role | tranche -> nb ---
    Dim aggS As Object, aggM As Object, stores As Object, months As Object
    Set aggS = CreateObject("Scripting.Dictionary")
    Set aggM = CreateObject("Scripting.Dictionary")
    Set stores = CreateObject("Scripting.Dictionary")
    Set months = CreateObject("Scripting.Dictionary")

    Dim k As Variant, parts() As String, nb As Long, scopes(1 To 3) As String, sc As Long
    Dim agg As Object
    For Each k In dictReply.Keys
        parts = Split(CStr(k), "|")
        If UBound(parts) = 3 Then
            nb = CLng(dictReply(k))
            If Not stores.Exists(parts(0)) Then stores.Add parts(0), 1
            If Not months.Exists(parts(1)) Then months.Add parts(1), 1
            scopes(1) = parts(0)          ' par boite
            scopes(2) = "_ALL"            ' toutes boites
            scopes(3) = parts(1)          ' par mois
            For sc = 1 To 3
                If sc = 3 Then Set agg = aggM Else Set agg = aggS
                IncrementDict agg, scopes(sc) & "|R", nb                       ' recus
                If parts(2) = "To" Then IncrementDict agg, scopes(sc) & "|RT", nb
                If parts(3) <> "None" Then
                    IncrementDict agg, scopes(sc) & "|A", nb                   ' repondus
                    If parts(2) = "To" Then IncrementDict agg, scopes(sc) & "|AT", nb
                    IncrementDict agg, scopes(sc) & "|" & parts(3), nb         ' tranche
                End If
            Next sc
        End If
    Next k

    Dim buckets As Variant
    buckets = Array("H1", "H4", "H24", "D3", "D3P")

    ' --- Bloc 1 : par boite ---
    Dim r As Long, i As Long, j As Long, sn As String
    r = 4
    ws.Cells(r, 1).Value = T("PAR BO~ITE", "BY MAILBOX")
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 1).Font.Size = 12
    r = r + 1

    Dim sortedStores As Variant
    sortedStores = SortDictKeys(stores)
    Dim nS As Long
    nS = UBound(sortedStores) - LBound(sortedStores) + 1

    Dim blk() As Variant
    ReDim blk(1 To nS + 2, 1 To 12)
    blk(1, 1) = T("Bo~ite", "Mailbox")
    blk(1, 2) = T("Re" & Chr(231) & "us", "Received")
    blk(1, 3) = T("R~epondus", "Replied")
    blk(1, 4) = T("Taux", "Rate")
    blk(1, 5) = T("Re" & Chr(231) & "us en A", "Received as To")
    blk(1, 6) = T("R~epondus en A", "Replied as To")
    blk(1, 7) = T("Taux en A", "Rate as To")
    blk(1, 8) = T("< 1 h", "< 1 h")
    blk(1, 9) = T("1 ~a 4 h", "1 to 4 h")
    blk(1, 10) = T("4 ~a 24 h", "4 to 24 h")
    blk(1, 11) = T("1 ~a 3 j", "1 to 3 d")
    blk(1, 12) = T("> 3 j", "> 3 d")
    Dim key0 As String, recv As Long, repl As Long, recvTo As Long, replTo As Long
    For i = 1 To nS + 1
        If i <= nS Then
            key0 = CStr(sortedStores(LBound(sortedStores) + i - 1))
            blk(i + 1, 1) = key0
        Else
            key0 = "_ALL"
            blk(i + 1, 1) = T("Toutes bo~ites", "All mailboxes")
        End If
        recv = GetDictValue(aggS, key0 & "|R")
        repl = GetDictValue(aggS, key0 & "|A")
        recvTo = GetDictValue(aggS, key0 & "|RT")
        replTo = GetDictValue(aggS, key0 & "|AT")
        blk(i + 1, 2) = recv
        blk(i + 1, 3) = repl
        If recv > 0 Then blk(i + 1, 4) = repl / recv
        blk(i + 1, 5) = recvTo
        blk(i + 1, 6) = replTo
        If recvTo > 0 Then blk(i + 1, 7) = replTo / recvTo
        For j = 0 To 4
            blk(i + 1, 8 + j) = GetDictValue(aggS, key0 & "|" & CStr(buckets(j)))
        Next j
    Next i
    Dim headerRowS As Long
    headerRowS = r
    ws.Range(ws.Cells(r, 1), ws.Cells(r + nS + 1, 12)).Value = blk
    ws.Range(ws.Cells(r, 1), ws.Cells(r, 12)).Font.Bold = True
    ws.Range(ws.Cells(r + nS + 1, 1), ws.Cells(r + nS + 1, 12)).Font.Bold = True
    ws.Range(ws.Cells(r + 1, 2), ws.Cells(r + nS + 1, 3)).NumberFormat = "#,##0"
    ws.Range(ws.Cells(r + 1, 5), ws.Cells(r + nS + 1, 6)).NumberFormat = "#,##0"
    ws.Range(ws.Cells(r + 1, 8), ws.Cells(r + nS + 1, 12)).NumberFormat = "#,##0"
    ws.Range(ws.Cells(r + 1, 4), ws.Cells(r + nS + 1, 4)).NumberFormat = "0.0%"
    ws.Range(ws.Cells(r + 1, 7), ws.Cells(r + nS + 1, 7)).NumberFormat = "0.0%"
    For i = 1 To nS
        sn = CStr(sortedStores(LBound(sortedStores) + i - 1))
        ws.Cells(r + i, 1).Interior.Color = GetStoreColor(sn)
        ws.Cells(r + i, 1).Font.Color = RGB(255, 255, 255)
    Next i
    r = r + nS + 1

    ' --- Bloc 2 : suivi mensuel ---
    r = r + 2
    Dim monthHeader As Long
    monthHeader = r
    ws.Cells(r, 1).Value = T("~EVOLUTION MENSUELLE", "MONTHLY TREND")
    ws.Cells(r, 1).Font.Bold = True
    ws.Cells(r, 1).Font.Size = 12
    r = r + 1

    Dim sortedMonths As Variant
    sortedMonths = SortDictKeys(months)
    Dim nM As Long
    nM = UBound(sortedMonths) - LBound(sortedMonths) + 1

    Dim blkM() As Variant
    ReDim blkM(1 To nM + 1, 1 To 10)
    blkM(1, 1) = T("Mois", "Month")
    blkM(1, 2) = T("Re" & Chr(231) & "us", "Received")
    blkM(1, 3) = T("R~epondus", "Replied")
    blkM(1, 4) = T("Taux", "Rate")
    blkM(1, 5) = T("Taux en A", "Rate as To")
    blkM(1, 6) = T("< 1 h", "< 1 h")
    blkM(1, 7) = T("1 ~a 4 h", "1 to 4 h")
    blkM(1, 8) = T("4 ~a 24 h", "4 to 24 h")
    blkM(1, 9) = T("1 ~a 3 j", "1 to 3 d")
    blkM(1, 10) = T("> 3 j", "> 3 d")
    For i = 1 To nM
        key0 = CStr(sortedMonths(LBound(sortedMonths) + i - 1))
        recv = GetDictValue(aggM, key0 & "|R")
        repl = GetDictValue(aggM, key0 & "|A")
        recvTo = GetDictValue(aggM, key0 & "|RT")
        replTo = GetDictValue(aggM, key0 & "|AT")
        blkM(i + 1, 1) = key0
        blkM(i + 1, 2) = recv
        blkM(i + 1, 3) = repl
        If recv > 0 Then blkM(i + 1, 4) = repl / recv
        If recvTo > 0 Then blkM(i + 1, 5) = replTo / recvTo
        For j = 0 To 4
            blkM(i + 1, 6 + j) = GetDictValue(aggM, key0 & "|" & CStr(buckets(j)))
        Next j
    Next i
    Dim firstDataRow As Long
    firstDataRow = r + 1
    ' Mois ecrits en texte (sinon Excel convertit "2026-11" en date)
    ws.Range(ws.Cells(r + 1, 1), ws.Cells(r + nM, 1)).NumberFormat = "@"
    ws.Range(ws.Cells(r, 1), ws.Cells(r + nM, 10)).Value = blkM
    ws.Range(ws.Cells(r, 1), ws.Cells(r, 10)).Font.Bold = True
    ws.Range(ws.Cells(firstDataRow, 2), ws.Cells(r + nM, 3)).NumberFormat = "#,##0"
    ws.Range(ws.Cells(firstDataRow, 4), ws.Cells(r + nM, 5)).NumberFormat = "0.0%"
    ws.Range(ws.Cells(firstDataRow, 6), ws.Cells(r + nM, 10)).NumberFormat = "#,##0"

    ' Courbe des deux taux
    If nM >= 2 Then
        Dim chartObj As Object, srcAddr As String
        srcAddr = ws.Range(ws.Cells(r, 1), ws.Cells(r + nM, 1)).Address & "," & _
                  ws.Range(ws.Cells(r, 4), ws.Cells(r + nM, 5)).Address
        Set chartObj = ws.ChartObjects.Add(Left:=700, Top:=(monthHeader - 1) * 15, Width:=560, Height:=300)
        If Not chartObj Is Nothing Then
            With chartObj.Chart
                .ChartType = 4          ' xlLine
                .SetSourceData ws.Range(srcAddr)
                .HasTitle = True
                .ChartTitle.Text = T("Taux de r~eponse par mois", "Reply rate per month")
            End With

    ws.Columns("A:L").AutoFit

'----------------------------------------------------------------------------
' PolishWorkbook (V24) - finition, en un seul endroit.
'
' Placement = 3 (xlFreeFloating) sur TOUS les graphiques : par defaut ils
' suivent les cellules, si bien que toute colonne masquee ou redimensionnee
' apres coup les fait glisser - c'est ce qui superposait le 3e graphique du
' tableau de bord au tableau des boites.
'
' PageSetup n'est volontairement PAS touche : chaque propriete declenche un
' aller-retour avec le pilote d'impression, soit plusieurs secondes sur 14
' feuilles, pour un benefice cosmetique.
'----------------------------------------------------------------------------
Private Sub PolishWorkbook(ByRef xlWB As Object)
    On Error Resume Next

    Dim colors As Object
    Set colors = CreateObject("Scripting.Dictionary")
    colors(T("Tableau de bord", "Dashboard")) = RGB(0, 169, 186)
    colors(T("R~esum~e", "Summary")) = RGB(0, 169, 186)
    colors(T("A - Cc", "To - Cc")) = RGB(0, 169, 186)
    colors(T("R~eponses", "Replies")) = RGB(0, 169, 186)     ' V25.3 (F9-15)
    colors(T("Par heure", "By hour")) = RGB(45, 38, 97)
    colors(T("Par jour semaine", "By weekday")) = RGB(45, 38, 97)
    colors(T("Par mois", "By month")) = RGB(45, 38, 97)
    colors(T("Par ann~ee", "By year")) = RGB(45, 38, 97)
    colors(T("Evolution par bo~ite", "Trend by mailbox")) = RGB(45, 38, 97)
    colors(T("Heatmaps", "Heatmaps")) = RGB(45, 38, 97)
    colors(T("Contacts", "Contacts")) = RGB(216, 122, 0)
    colors(T("Contacts int-ext", "Contacts int-ext")) = RGB(216, 122, 0)
    colors(T("Nb destinataires", "Nb recipients")) = RGB(216, 122, 0)
    colors(T("Longueur messages", "Message length")) = RGB(216, 122, 0)
    colors(T("DATA", "DATA")) = RGB(128, 128, 128)

    Dim ws As Object, i As Long
    For Each ws In xlWB.Worksheets

        ' 1. Graphiques flottants
        For i = 1 To ws.ChartObjects.count
            ws.ChartObjects(i).Placement = 3
        Next i

        ' 2. Feuille d'etat : masquee, jamais coloree
        If ws.Name = STATE_SHEET Then
            ws.Visible = 0          ' xlSheetHidden
        Else
            If colors.Exists(ws.Name) Then ws.Tab.Color = colors(ws.Name)
        End If
    Next ws

    ' 3. Filtre automatique sur les feuilles tabulaires
    ApplyFilter xlWB, T("Contacts", "Contacts"), 2
    ApplyFilter xlWB, T("Contacts int-ext", "Contacts int-ext"), 2
    ApplyFilter xlWB, T("DATA", "DATA"), 2

    ' 4. Volets figes - necessite d'activer la feuille, donc en dernier.
    FreezeAt xlWB, T("Contacts", "Contacts"), 3
    FreezeAt xlWB, T("DATA", "DATA"), 3
    FreezeAt xlWB, T("A - Cc", "To - Cc"), 6
End Sub

Private Sub ApplyFilter(ByRef xlWB As Object, ByVal sheetName As String, ByVal headerRow As Long)
    On Error Resume Next
    Dim ws As Object
    Set ws = xlWB.Worksheets(sheetName)
    If ws Is Nothing Then Exit Sub
    If ws.AutoFilterMode Then ws.AutoFilterMode = False
    ws.Rows(headerRow).AutoFilter
End Sub

Private Sub FreezeAt(ByRef xlWB As Object, ByVal sheetName As String, ByVal firstFreeRow As Long)
    On Error Resume Next
    Dim ws As Object
    Set ws = xlWB.Worksheets(sheetName)
    If ws Is Nothing Then Exit Sub
    ws.Activate
    ws.Cells(firstFreeRow, 1).Select
    ws.Application.ActiveWindow.FreezePanes = True
End Sub

'----------------------------------------------------------------------------
' RunEmailAnalysisDelta - POINT D'ENTREE (bouton / Alt+F8).
'
' V25.3 (F9-01 / F9-02) :
' 1. reprend le rapport FULL ou DELTA le plus recent du dossier Sorties
'    (plus de boite de choix : Application.GetOpenFilename est un membre
'    d'EXCEL, absent d'Outlook - erreur 438 masquee, la commande ne faisait
'    rien du tout) ;
' 2. lit sa feuille _ETAT par modExcelBridge (configuration, fin exacte,
'    compteurs) ; un etat d'avant la V25.3 ou un rapport TEST est refuse ;
' 3. analyse les mails STRICTEMENT posterieurs a la fin de l'etat, jusqu'a
'    maintenant (le mois en cours n'est plus ni exclu ni compte deux fois) ;
' 4. enchaine sur RunEmailAnalysis : la fenetre de lancement nomme le
'    rapport repris.
' Aucun rapport, ou etat inutilisable : proposition de lancer une analyse
' complete dans la foulee (parcours sans rupture).
'----------------------------------------------------------------------------
Public Sub RunEmailAnalysisDelta()
    On Error GoTo ErrH

    If m_Running Then
        modUi.Inform T("Une analyse des mails est d~ej~a en cours.", _
                       "A mail analysis is already running."), AnalyticsTitle()
        Exit Sub
    End If

    g_DeltaMode = False
    g_DeltaSourcePath = ""

    Dim p As String
    p = LatestReportPath()
    If Len(p) = 0 Then
        If modUi.Ask(T("Aucun rapport pr~ec~edent (analyse compl~ete ou mise ~a jour) " & _
                       "dans le dossier :", _
                       "No previous report (full analysis or update) in the folder:") & _
                     vbCrLf & ReportFolderPath() & vbCrLf & vbCrLf & _
                     T("Lancer une analyse compl~ete maintenant ?", "Run a full analysis now?"), _
                     AnalyticsTitle()) = vbYes Then RunEmailAnalysis
        Exit Sub
    End If

    Dim whyNot As String
    If Not LoadStateFromWorkbook(p, whyNot) Then
        ' whyNot rempli = etat inutilisable (ancien format, rapport TEST...) ;
        ' vide = erreur de lecture deja signalee.
        If Len(whyNot) > 0 Then
            If modUi.Ask(whyNot & vbCrLf & vbCrLf & _
                         T("Lancer une analyse compl~ete maintenant ?", "Run a full analysis now?"), _
                         AnalyticsTitle()) = vbYes Then RunEmailAnalysis
        End If
        Exit Sub
    End If

    g_DeltaEnd = Now
    If g_DeltaStart >= g_DeltaEnd Then
        modUi.Inform T("Rien ~a ajouter : le dernier rapport va d~ej~a jusqu'~a maintenant.", _
                       "Nothing to add: the latest report already runs up to now.") & _
                     vbCrLf & vbCrLf & FileNameOf(p), AnalyticsTitle()
        Exit Sub
    End If

    g_DeltaMode = True
    g_DeltaSourcePath = p

    modDebugLog.DInfo MOD_NAME, "Mise a jour delta : base " & FileNameOf(p) & _
        ", mails apres " & Format$(g_DeltaStart, "yyyy-mm-dd hh:nn:ss")

    RunEmailAnalysis
    Exit Sub

ErrH:
    Dim errTxt As String
    errTxt = Err.Number & " - " & Err.Description
    g_DeltaMode = False
    m_DeltaRows = Empty
    modDebugLog.DError MOD_NAME, "Mise a jour delta : erreur " & errTxt
    modUi.Fail T("Erreur en mode delta : ", "Delta mode error: ") & errTxt, AnalyticsTitle(), vbCritical
End Sub

'----------------------------------------------------------------------------
' LatestReportPath (V25.3, F9-01) - rapport Stats_Emails_FULL_* ou
' Stats_Emails_DELTA_* le plus recent du dossier Sorties (horodatage du nom
' de fichier, AAAAMMJJ_HHMMSS). Les rapports TEST (1 mail sur 10) ne
' servent jamais de base. "" si aucun.
'----------------------------------------------------------------------------
Private Function LatestReportPath() As String
    On Error Resume Next
    LatestReportPath = ""

    Dim folder As String
    folder = ReportFolderPath()
    If Len(folder) = 0 Then Exit Function

    Dim f As String, u As String, stamp As String, bestStamp As String, best As String
    f = Dir$(folder & "\Stats_Emails_*.xlsx")
    Do While Len(f) > 0
        u = UCase$(f)
        If u Like "STATS_EMAILS_FULL_########_######.XLSX" Or _
           u Like "STATS_EMAILS_DELTA_########_######.XLSX" Then
            stamp = Mid$(u, Len(u) - 19, 15)          ' AAAAMMJJ_HHMMSS
            If stamp > bestStamp Then
                bestStamp = stamp
                best = f
            End If
        End If
        f = Dir$()
    Loop

    If Len(best) > 0 Then LatestReportPath = folder & "\" & best
End Function

'----------------------------------------------------------------------------
' LoadStateFromWorkbook - lit la feuille _ETAT du classeur choisi.
'
' Ouverture en LECTURE SEULE et fermeture sans enregistrer : le rapport
' precedent ne doit jamais etre modifie par une analyse delta.
' V25.3 (F9-01) : par modExcelBridge (OpenWorkbook / ReadBlock /
' CloseWorkbook) : Application.Workbooks.Open n'existe pas dans Outlook. Un
' rapport deja ouvert a l'ecran est lu sans etre referme.
' V25.3 (F9-02) : refuse un etat sans STATE_FORMAT 2 (regle du "dernier
' mois complet", qui comptait deux fois le debut du mois en cours) et un
' rapport TEST ; whyNot dit pourquoi (la question "analyse complete ?" est
' posee par l'appelant). Erreur de lecture : message ici, whyNot = "".
'
' Le bloc de compteurs est lu EN UNE FOIS dans un tableau Variant. Lu cellule
' par cellule, 100 000 lignes prendraient plusieurs minutes en COM.
'----------------------------------------------------------------------------
Private Function LoadStateFromWorkbook(ByVal path As String, ByRef whyNot As String) As Boolean
    LoadStateFromWorkbook = False
    whyNot = ""
    m_DeltaRows = Empty
    Set m_DeltaConfig = Nothing

    Dim xlApp As Object, wb As Object, ws As Object
    Dim wasOpen As Boolean, ownXl As Boolean
    Set xlApp = Nothing

    On Error GoTo ErrH
    If Not modExcelBridge.OpenWorkbook(path, True, xlApp, wb, wasOpen, ownXl) Then
        modUi.Fail T("Impossible d'ouvrir le dernier rapport :", "Cannot open the latest report:") & _
                   vbCrLf & path, AnalyticsTitle()
        GoTo Cleanup
    End If

    Set ws = Nothing
    On Error Resume Next
    Set ws = wb.Worksheets(STATE_SHEET)
    On Error GoTo ErrH

    If ws Is Nothing Then
        ' v- 2026-08-22 : STATE_SHEET hors de T() (nom de feuille = donnee).
        whyNot = FileNameOf(path) & vbCrLf & vbCrLf & _
                 T("Ce rapport n'a pas de feuille ", "This report has no ") & STATE_SHEET & _
                 T(" : il a ~et~e produit par une version ant~erieure ~a la V24.", _
                   " sheet: it was produced before V24.")
        GoTo Cleanup
    End If

    ' En-tete : lignes 12 a 28 en un seul appel (ligne L -> hdr(L - 11, c)).
    Dim hdr As Variant
    hdr = modExcelBridge.ReadBlock(ws, 12, 1, 28, 3)

    If Trim$(CStr(hdr(5, 1))) <> "STATE_FORMAT" Or Val(CStr(hdr(5, 2))) < STATE_FORMAT Then
        whyNot = FileNameOf(path) & vbCrLf & vbCrLf & _
                 T("Ce rapport a ~et~e produit avant la V25.3 : sa feuille d'~etat ne " & _
                   "permet pas une mise ~a jour exacte (le d~ebut du mois en cours y ~etait " & _
                   "compt~e deux fois).", _
                   "This report was produced before V25.3: its state sheet does not " & _
                   "allow an exact update (the start of the current month was counted " & _
                   "twice).")
        GoTo Cleanup
    End If
    If Trim$(CStr(hdr(6, 2))) = "1" Then
        whyNot = FileNameOf(path) & vbCrLf & vbCrLf & _
                 T("Ce rapport est un rapport de TEST (1 mail sur 10) : il ne peut pas " & _
                   "servir de base ~a une mise ~a jour.", _
                   "This is a TEST report (1 mail in 10): it cannot be the base of an update.")
        GoTo Cleanup
    End If

    Dim dStart As Date, dEnd As Date
    If Not CellDate(hdr(3, 2), dStart) Or Not CellDate(hdr(4, 2), dEnd) Then
        whyNot = FileNameOf(path) & vbCrLf & vbCrLf & _
                 T("La p~eriode de ce rapport est illisible.", "The period of this report is unreadable.")
        GoTo Cleanup
    End If
    m_DeltaBaseStart = dStart
    g_DeltaStart = dEnd            ' borne EXCLUE du delta (incluse dans l'etat)

    ' Configuration des boites telle qu'utilisee lors de l'analyse d'origine
    ' (lignes 20 a 28).
    Set m_DeltaConfig = CreateObject("Scripting.Dictionary")
    Dim r As Long, sn As String
    For r = 9 To 17
        sn = Trim$(CStr(hdr(r, 1)))
        If Len(sn) > 0 Then
            m_DeltaConfig(sn) = CStr(hdr(r, 2)) & "|" & CStr(hdr(r, 3))
        End If
    Next r

    Dim nRows As Long
    nRows = 0
    If IsNumeric(hdr(7, 2)) Then nRows = CLng(hdr(7, 2))
    If nRows > 0 Then
        m_DeltaRows = modExcelBridge.ReadBlock(ws, 31, 1, 30 + nRows, 3)
    Else
        m_DeltaRows = Empty
    End If

    LoadStateFromWorkbook = True

Cleanup:
    On Error Resume Next
    Set ws = Nothing
    modExcelBridge.CloseWorkbook xlApp, wb, wasOpen, ownXl, False
    Exit Function

ErrH:
    Dim errTxt As String
    errTxt = Err.Number & " - " & Err.Description
    modDebugLog.DError MOD_NAME, "Lecture de l'etat impossible : " & errTxt
    modUi.Fail T("Lecture de l'~etat impossible : ", "Cannot read state: ") & errTxt, _
               AnalyticsTitle(), vbCritical
    LoadStateFromWorkbook = False
    whyNot = ""
    Resume Cleanup
End Function

' V25.3 : date lue dans une cellule (Date, nombre de serie Excel ou texte).
Private Function CellDate(ByVal v As Variant, ByRef outDate As Date) As Boolean
    On Error GoTo Bad
    CellDate = False
    If IsEmpty(v) Or IsError(v) Then Exit Function
    If VarType(v) = vbDate Then
        outDate = v
    ElseIf IsNumeric(v) Then
        outDate = CDate(CDbl(v))
    ElseIf IsDate(v) Then
        outDate = CDate(v)
    Else
        Exit Function
    End If
    CellDate = (outDate > DateSerial(1990, 1, 1))
    Exit Function
Bad:
    CellDate = False
End Function

'----------------------------------------------------------------------------
' RehydrateFromState - recharge les compteurs lus dans _ETAT.
' Appelee APRES InitializeDictionaries, AVANT le scan.
'----------------------------------------------------------------------------
Private Sub RehydrateFromState()
    On Error GoTo Fin
    If IsEmpty(m_DeltaRows) Then Exit Sub

    Dim i As Long, dn As String, k As String, v As Long, n As Long
    For i = LBound(m_DeltaRows, 1) To UBound(m_DeltaRows, 1)
        dn = Trim$(CStr(m_DeltaRows(i, 1)))
        If Len(dn) > 0 Then
            k = CStr(m_DeltaRows(i, 2))
            v = CLng(m_DeltaRows(i, 3))
            If SeedDict(dn, k, v) Then n = n + 1
        End If
    Next i

    Heartbeat "DELTA : " & n & " compteurs recharges (etat jusqu'au " & _
              Format$(g_DeltaStart, "YYYY-MM-DD HH:NN:SS") & ")"
Fin:
    On Error Resume Next
End Sub

'----------------------------------------------------------------------------
' SeedDict - aiguillage nom -> dictionnaire. Table en dur, comme partout
' ailleurs dans ce projet : VBA ne resout pas un nom de variable a partir
' d'une chaine. histMsgLength500 est ignore ici (il ne sert qu'a recalculer
' une mediane, pas au scan).
'----------------------------------------------------------------------------
Private Function SeedDict(ByVal dictName As String, ByVal k As String, ByVal v As Long) As Boolean
    On Error Resume Next
    SeedDict = True
    Select Case dictName
        Case "dictStoreStats":            dictStoreStats(k) = v
        Case "dictMonthly":               dictMonthly(k) = v
        Case "dictWeekly":                dictWeekly(k) = v
        Case "dictDaily":                 dictDaily(k) = v
        Case "dictQuarterly":             dictQuarterly(k) = v
        Case "dictYearly":                dictYearly(k) = v
        Case "dictSemesterly":            dictSemesterly(k) = v
        Case "dictHourly":                dictHourly(k) = v
        Case "dictDayOfWeek":             dictDayOfWeek(k) = v
        Case "dictHeatmapSent":           dictHeatmapSent(k) = v
        Case "dictHeatmapReceived":       dictHeatmapReceived(k) = v
        Case "dictHeatmapTotal":          dictHeatmapTotal(k) = v
        Case "dictSentTo":                dictSentTo(k) = v
        Case "dictReceivedFrom":          dictReceivedFrom(k) = v
        Case "dictSentToInternal":        dictSentToInternal(k) = v
        Case "dictSentToExternal":        dictSentToExternal(k) = v
        Case "dictSentToUnknown":         dictSentToUnknown(k) = v
        Case "dictReceivedFromInternal":  dictReceivedFromInternal(k) = v
        Case "dictReceivedFromExternal":  dictReceivedFromExternal(k) = v
        Case "dictReceivedFromUnknown":   dictReceivedFromUnknown(k) = v
        Case "dictNbRecipients":          dictNbRecipients(k) = v
        Case "dictInternalExternal":      dictInternalExternal(k) = v
        Case "dictToCc":                  dictToCc(k) = v
        Case "dictReply":                 dictReply(k) = v       ' V25.3 (F9-15)
        Case Else:                        SeedDict = False
    End Select
End Function

'----------------------------------------------------------------------------
' DeltaStoreEmail / DeltaStoreKeyword - configuration relue dans _ETAT, pour
' que frmConfig pre-remplisse les boites a l'identique de l'analyse d'origine.
'----------------------------------------------------------------------------
Public Function DeltaStoreEmail(ByVal storeName As String) As String
    On Error Resume Next
    If m_DeltaConfig Is Nothing Then Exit Function
    If Not m_DeltaConfig.Exists(storeName) Then Exit Function
    DeltaStoreEmail = Split(CStr(m_DeltaConfig(storeName)), "|")(0)
End Function

Public Function DeltaStoreKeyword(ByVal storeName As String) As String
    On Error Resume Next
    If m_DeltaConfig Is Nothing Then Exit Function
    If Not m_DeltaConfig.Exists(storeName) Then Exit Function
    Dim p() As String
    p = Split(CStr(m_DeltaConfig(storeName)), "|")
    If UBound(p) >= 1 Then DeltaStoreKeyword = p(1)
End Function

Private Sub DetectYearRange(ByRef minYear As Long, ByRef maxYear As Long)
    On Error Resume Next
    minYear = 0
    maxYear = 0

    If dictYearly Is Nothing Then Exit Sub
    If dictYearly.Count = 0 Then Exit Sub

    Dim k As Variant, parts As Variant, y As Long
    For Each k In dictYearly.Keys
        parts = Split(CStr(k), "|")
        If UBound(parts) >= 0 Then
            y = CLng(parts(0))
            If y >= 2000 And y <= 2100 Then
                If minYear = 0 Or y < minYear Then minYear = y
                If maxYear = 0 Or y > maxYear Then maxYear = y
            End If
        End If
    Next k
End Sub
