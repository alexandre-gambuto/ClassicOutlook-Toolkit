'=============================================================================
' OBTENIR UNE INSTANCE EXCEL (reutilise si possible)
'=============================================================================
Private Function GetExcelApp() As Object

    ' v5.2 : test de vitalite REEL. Lire .Version pouvait reussir sur une
    ' instance morte (valeur servie localement par le proxy COM) alors que
    ' la premiere vraie ecriture echouait ensuite en erreur 462. On force
    ' un aller-retour complet (ExcelAlive) ; en cas d'echec on jette le
    ' cache et on repart proprement.
    If Not mXlApp Is Nothing Then
        If ExcelAlive(mXlApp) Then
            Set GetExcelApp = mXlApp
            Exit Function
        End If
        ResetExcelCache
    End If

    ' v6.7 (F4-12) : instance obtenue par modExcelBridge.GetExcelApp, qui
    ' reessaie GetObject quand Excel est occupe (avant : un seul essai, puis
    ' une SECONDE instance creee qui trouvait le classeur verrouille). Memes
    ' regles qu'en v5.2 / v5.9 : instance fantome ecartee ; instance
    ' INVISIBLE adoptee (jamais celle de l'utilisateur : quittee a la
    ' liberation si elle n'heberge plus de classeur) ; sinon CreateObject.
    Dim createdByUs As Boolean
    createdByUs = False
    Set mXlApp = modExcelBridge.GetExcelApp(createdByUs)
    mXlCreatedByUs = createdByUs
    If mXlApp Is Nothing Then
        modDebugLog.DError MOD_NAME, "GetExcelApp : aucune instance Excel disponible"
    End If

    Set GetExcelApp = mXlApp

End Function

'=============================================================================
' OBTENIR LE CLASSEUR TRACKER (le retrouve ouvert, ou l'ouvre)
' v6.7 (F4-12) : classeur deja ouvert retrouve par modExcelBridge
' (FindOpenWorkbookByRef : chemin, ou NOM sous OneDrive ; comptage avec
' essais repetes si Excel est occupe) ; ouverture avec essais repetes.
' v6.7 (F4-17) : classeur absent -> message et Nothing. Plus de creation
' d'un classeur vide (SaveAs refusait l'extension .xlsm et laissait un
' classeur fantome dans l'instance invisible ; il n'aurait eu ni ruban ni
' code). quiet = appel differe : toast au lieu d'une fenetre.
' mTrackerFailReported : True si l'utilisateur a deja ete informe.
'=============================================================================
Private Function GetTrackerWorkbook(xlApp As Object, Optional ByVal quiet As Boolean = False) As Object

    mTrackerFailReported = False
    Dim filePath As String
    filePath = GetTrackerFilePath()

    Dim wb As Object
    Set wb = Nothing
    If Not xlApp Is Nothing Then Set wb = modExcelBridge.FindOpenWorkbookByRef(xlApp, filePath)

    If Not wb Is Nothing Then
        ' v5.9 : trois cas pour un classeur DEJA ouvert.
        If mWbDepth > 0 Then
            ' 1) Appel IMBRIQUE : une routine de ce module tient deja le
            '    classeur (differe/evenement survenu pendant sa MsgBox ou
            '    son DoEvents). On partage ; seul le dernier sorti referme.
            mWbDepth = mWbDepth + 1
            mWbOpenedHere = True
            modDebugLog.DDebug MOD_NAME, "GetTrackerWorkbook : appel imbrique, profondeur " & mWbDepth
        ElseIf IsBackgroundOrphan(xlApp, wb) Then
            ' 2) ORPHELIN : classeur masque, ou instance Excel invisible.
            '    Personne ne le regarde ; il a ete laisse la par une sortie
            '    anticipee ou une session precedente. On l'adopte pour le
            '    refermer proprement a la liberation.
            mWbDepth = 1
            mWbOpenedHere = True
            modDebugLog.DInfo MOD_NAME, "GetTrackerWorkbook : classeur orphelin (masque/invisible) adopte"
        Else
            ' 3) Classeur VISIBLE chez l'utilisateur (tableau de bord ouvert,
            '    ou ouvert a la main) : on ne le refermera pas.
            mWbOpenedHere = False
        End If
        PrepareWorkbook wb               ' v6.3 : gabarit + _Schema, meme deja ouvert
        If Not LayoutReady(wb) Then      ' v6.3 : jamais d'ecriture dans un gabarit inconnu
            ReleaseBackground xlApp, wb
            mTrackerFailReported = True  ' LayoutReady a prevenu (toast)
            Set GetTrackerWorkbook = Nothing
            Exit Function
        End If
        Set GetTrackerWorkbook = wb
        Exit Function
    End If

    ' Classeur absent de l'instance : une profondeur residuelle serait un
    ' reliquat (Ctrl+S VBE, fermeture externe). On repart a zero.
    If mWbDepth > 0 Then
        modDebugLog.DDebug MOD_NAME, "GetTrackerWorkbook : profondeur residuelle " & mWbDepth & " remise a 0"
        mWbDepth = 0
    End If
    If xlApp Is Nothing Then Exit Function

    If Dir(filePath) = "" Then
        ' v6.1 : le CHEMIN reste hors de T() (A() decoderait un "~e" du
        ' chemin et afficherait un chemin faux).
        modDebugLog.DError MOD_NAME, "Classeur de suivi introuvable : " & filePath
        If quiet Then
            modNotify.Notify T("Suivi des mails : classeur de suivi introuvable (voir debug.log)", _
                               "Mail tracking: tracking workbook not found (see debug.log)")
        Else
            modUi.Fail T("Classeur de suivi introuvable :", "Tracking workbook not found:") & vbCrLf & _
                       filePath, T("Suivi Mails", "Mail Tracking")
        End If
        mTrackerFailReported = True
        Exit Function
    End If

    Set wb = OpenTrackerFile(xlApp, filePath)
    If wb Is Nothing Then Exit Function      ' journalise ; l'appelant decide du message
    PrepareWorkbook wb

    mWbOpenedHere = True               ' on vient de l'ouvrir : a refermer
    mWbDepth = 1
    If Not LayoutReady(wb) Then          ' v6.3 : jamais d'ecriture dans un gabarit inconnu
        ReleaseBackground xlApp, wb
        mTrackerFailReported = True
        Set GetTrackerWorkbook = Nothing
        Exit Function
    End If
    Set GetTrackerWorkbook = wb

End Function

'=============================================================================
' v6.7 (F4-12) - Ouverture du classeur de suivi avec essais repetes (Excel
' occupe : cellule en saisie, boite ouverte -> erreur 1004 "Open"). Meme
' cadence que modExcelBridge.OpenWorkbook. Nothing apres 5 echecs (journal).
'=============================================================================
Private Function OpenTrackerFile(ByVal xlApp As Object, ByVal filePath As String) As Object
    Set OpenTrackerFile = Nothing
    Dim wb As Object, tentative As Long, e As Long, eDesc As String
    For tentative = 1 To 5
        Set wb = Nothing
        Err.Clear
        On Error Resume Next
        Set wb = xlApp.Workbooks.Open(filePath)
        e = Err.Number
        eDesc = Err.Description
        On Error GoTo 0
        If e = 0 And Not wb Is Nothing Then
            Set OpenTrackerFile = wb
            Exit Function
        End If
        modDebugLog.DDebug MOD_NAME, "Ouverture du classeur de suivi : err " & e & ", essai " & tentative
        modExcelBridge.PauseMs 200 * tentative
    Next tentative
    modDebugLog.DError MOD_NAME, "Ouverture du classeur de suivi impossible apres 5 essais (err " & _
        e & " - " & eDesc & ")"
End Function

'=============================================================================
' v5.9 - Le classeur de suivi deja ouvert est-il un ORPHELIN d'arriere-plan ?
' Oui si sa fenetre est masquee (seul ce module la masque) ou si l'instance
' Excel est invisible (aucun utilisateur ne regarde une instance invisible).
' En cas de doute (erreur COM), on repond Non : on ne referme jamais un
' classeur qui pourrait etre sous les yeux de l'utilisateur.
'=============================================================================
Private Function IsBackgroundOrphan(xlApp As Object, wb As Object) As Boolean
    On Error Resume Next
    IsBackgroundOrphan = False
    Dim appVisible As Boolean, winVisible As Boolean
    appVisible = True
    winVisible = True
    appVisible = CBool(xlApp.Visible)
    winVisible = CBool(wb.Windows(1).Visible)
    If Err.Number <> 0 Then
        Err.Clear
        Exit Function
    End If
    IsBackgroundOrphan = (Not appVisible) Or (Not winVisible)
End Function

'=============================================================================
' RESOUDRE LE MAIL CIBLE SELON LE CONTEXTE D'APPEL
' Renvoie True si un mail a ete trouve, avec isComposing positionne.
' Priorite : Inspector actif > Selection explorateur
'=============================================================================
Private Function ResolveTargetMail(ByRef oMail As Outlook.mailItem, _
                                    ByRef isComposing As Boolean) As Boolean

    On Error Resume Next

    ResolveTargetMail = False
    isComposing = False
    Set oMail = Nothing

    ' --- La FENETRE ACTIVE decide ---
    ' Corrige le cas : un mail ouvert en grand pendant qu'on navigue dans
    ' l'explorateur. On regarde quelle fenetre est reellement au premier
    ' plan (Application.ActiveWindow), pas "y a-t-il un inspecteur ouvert".
    Dim actWin As Object
    Set actWin = Application.ActiveWindow

    If TypeOf actWin Is Outlook.Inspector Then
        ' Couvre : composition ET lecture d'un mail envoye deja ouvert
        Dim inspItem As Object
        Set inspItem = actWin.CurrentItem
        If Not inspItem Is Nothing Then
            If inspItem.Class = OL_MAIL_CLASS Then
                Set oMail = inspItem
                ' .Sent : True = deja envoye (suivi direct) ; False = brouillon
                isComposing = Not oMail.Sent
                ResolveTargetMail = True
                Exit Function
            End If
        End If
        ' Inspecteur actif mais pas un mail -> on ne devine pas.
        Exit Function
    End If

    ' --- Explorateur actif : selection dans la liste ---
    Dim olSelection As Outlook.Selection
    Set olSelection = Application.ActiveExplorer.Selection

    If Not olSelection Is Nothing Then
        If olSelection.count > 0 Then
            If olSelection.Item(1).Class = OL_MAIL_CLASS Then
                Set oMail = olSelection.Item(1)
                ' Dans l'explorateur, on ne traite que les mails envoyes
                If oMail.Sent Then
                    isComposing = False
                    ResolveTargetMail = True
                Else
                    ' Brouillon selectionne dans l'explorateur : on refuse
                    ' (il faudrait l'ouvrir en composition)
                    Set oMail = Nothing
                    ResolveTargetMail = False
                End If
                Exit Function
            End If
        End If
    End If

    On Error GoTo 0

End Function

'=============================================================================
' POINT D'ENTREE 1 : MARQUER UN MAIL POUR SUIVI
'=============================================================================
Public Sub MarkMailForTracking()

    On Error GoTo ErrorHandler

    Dim oMail As Outlook.mailItem
    Dim isComposing As Boolean
    Dim foundMail As Boolean

    foundMail = ResolveTargetMail(oMail, isComposing)

    If Not foundMail Then
        MsgBox T("Veuillez :" & vbCrLf & _
               "  - ouvrir un mail en cours de r" & Chr(233) & "daction, ou" & vbCrLf & _
               "  - ouvrir un mail d" & Chr(233) & "j" & Chr(224) & " envoy" & Chr(233) & " dans sa fen" & Chr(234) & "tre, ou" & vbCrLf & _
               "  - s" & Chr(233) & "lectionner un mail envoy" & Chr(233) & " dans la liste.", _
               "Please:" & vbCrLf & _
               "  - open a mail being composed, or" & vbCrLf & _
               "  - open an already-sent mail in its window, or" & vbCrLf & _
               "  - select a sent mail in the list."), _
               vbExclamation, T("Suivi Mails", "Mail Tracking")
        Exit Sub
    End If

    ' --- Verifier qu'il y a des destinataires TO ---
    Dim recipientCount As Long
    recipientCount = CountTORecipients(oMail)

    If recipientCount = 0 Then
        MsgBox T("Ce mail n'a aucun destinataire en TO.", "This mail has no TO recipient."), vbExclamation, T("Suivi Mails", "Mail Tracking")
        Exit Sub
    End If

    ' --- Lire l'objet du mail ---
    Dim mailSubject As String
    mailSubject = oMail.subject
    If Len(Trim(mailSubject)) = 0 Then mailSubject = T("(sans objet)", "(no subject)")

    ' --- Demander le delai ---
    Dim contextLabel As String
    If isComposing Then
        contextLabel = T("Mail en cours de r" & Chr(233) & "daction", "Mail being composed")
    Else
        contextLabel = T("Mail d" & Chr(233) & "j" & Chr(224) & " envoy" & Chr(233), "Already-sent mail")
    End If

    Dim delayStr As String
    ' v6 : le SUJET et le CONTEXTE sont concatenes A L'EXTERIEUR de T().
    ' T() applique A() a ses deux branches : un objet de mail contenant
    ' "~e", "~a" ou "~o" etait decode en accent et affiche faux - or c'est
    ' le seul element qui permet d'identifier de quel mail on parle.
    delayStr = modUi.AskText( _
        T("R~eponse attendue sous combien de jours ?", _
          "Reply expected within how many days?") & vbCrLf & vbCrLf & _
        T("Contexte : ", "Context: ") & contextLabel & vbCrLf & _
        T("Mail : ", "Mail: ") & Left(mailSubject, 80) & vbCrLf & _
        T("Destinataires TO : ", "TO recipients: ") & recipientCount, _
        T("Suivi Mails - D~elai", "Mail Tracking - Delay"), CStr(modConfig.ReplyDelayDays()))

    If delayStr = "" Then Exit Sub

    If Not IsNumeric(delayStr) Then
        MsgBox T("Veuillez entrer un nombre de jours valide.", "Please enter a valid number of days."), vbExclamation, T("Suivi Mails", "Mail Tracking")
        Exit Sub
    End If

    Dim delayDays As Long
    delayDays = CLng(delayStr)

    ' --- Recuperer les infos du mail ---
    Dim mailRecipients As String
    Dim mailConvID As String
    Dim mailEntryID As String
    Dim mailDate As Date

    mailRecipients = GetTORecipientsList(oMail)

    If isComposing Then
        oMail.Save
        mailConvID = oMail.ConversationID
        mailEntryID = oMail.entryID
        mailDate = Now
    Else
        mailConvID = oMail.ConversationID
        mailEntryID = oMail.entryID
        ' Pour un mail deja envoye, utiliser SentOn (pas Now)
        If oMail.SentOn > #1/1/2000# Then
            mailDate = oMail.SentOn
        Else
            mailDate = Now
        End If
    End If

    ' v6.3 : dossier du mail deja envoye. Un brouillon n'a pas encore sa
    ' place definitive : la colonne sera remplie au prochain recalcul.
    Dim mailFolder As String
    mailFolder = ""
    If Not isComposing Then mailFolder = ItemFolderName(oMail)

    ' --- Enregistrer dans Excel ---
    ' v5.2 : pas de confirmation si l'ecriture a echoue (l'echec a deja
    ' ete signale par SaveTrackingEntry).
    If Not SaveTrackingEntry(mailSubject, mailRecipients, recipientCount, _
                             delayDays, mailDate, mailConvID, mailEntryID, mailFolder) Then Exit Sub

    ' --- Message de confirmation ---
    If isComposing Then
        modUi.Toast T("Mail marqu~e pour suivi (", "Mail flagged for tracking (") & _
               delayDays & T(" jours).", " days).") & vbLf & _
               T("Le suivi sera actif d~fs l'envoi.", _
                 "Tracking will start as soon as you send."), uiSuccess, True
    Else
        modUi.Toast T("Mail marqu~e pour suivi (", "Mail flagged for tracking (") & _
               delayDays & T(" jours).", " days).") & vbLf & _
               T("Destinataires suivis : ", "Tracked recipients: ") & recipientCount & vbLf & _
               T("Date d'envoi : ", "Sent date: ") & Format(mailDate, "dd/mm/yyyy hh:mm"), uiSuccess, True
    End If

    Exit Sub

ErrorHandler:
    MsgBox T("Erreur ", "Error ") & Err.Number & " : " & Err.Description, _
           vbCritical, T("Suivi Mails - Erreur", "Mail Tracking - Error")
End Sub

' Suivi SANS AUCUNE fenetre : delai = ReplyDelayDays (config, repli 7 j),

' MsgBox). (RecordSentMail, variante avec saisie du delai, retiree en v6.7 :
' plus d'appelant depuis modReplySuspicion v4.1.)
'=============================================================================
Public Sub RecordSentMailQuick(ByVal oMail As Outlook.mailItem)
    On Error GoTo ErrorHandler

    If oMail Is Nothing Then Exit Sub

    Dim recipientCount As Long
    recipientCount = CountTORecipients(oMail)
    If recipientCount = 0 Then
        modDebugLog.DInfo MOD_NAME, "RecordSentMailQuick : aucun destinataire TO, suivi ignore"
        modNotify.Notify T("Suivi impossible : aucun destinataire TO", _
                           "Tracking impossible: no TO recipient")
        Exit Sub
    End If

    Dim delayDays As Long
    delayDays = modConfig.ReplyDelayDays()
    If delayDays <= 0 Then delayDays = 7

    Dim mailSubject As String
    mailSubject = oMail.subject
    If Len(Trim(mailSubject)) = 0 Then mailSubject = T("(sans objet)", "(no subject)")

    Dim mailRecipients As String
    Dim mailConvID As String
    Dim mailEntryID As String
    Dim mailDate As Date

    mailRecipients = GetTORecipientsList(oMail)
    mailConvID = oMail.ConversationID
    mailEntryID = oMail.entryID
    If oMail.SentOn > #1/1/2000# Then
        mailDate = oMail.SentOn
    Else
        mailDate = Now
    End If

    If Not SaveTrackingEntry(mailSubject, mailRecipients, recipientCount, _
                             delayDays, mailDate, mailConvID, mailEntryID, _
                             ItemFolderName(oMail), True) Then Exit Sub

    Exit Sub

ErrorHandler:

        Err.Number & " - " & Err.Description
End Sub

'=============================================================================
' v5.6 - SUIVI AUTOMATIQUE DES MAILS "URGENT"
' EnqueueUrgentTracking : capture SYNCHRONE des champs du mail (aucun COM
'   Excel ici). La date limite est deja validee cote ThisOutlookSession.
' FlushUrgentTracking : ecriture differee (modDefer "UrgentFlush"). UPSERT
'   par conversation (ligne existante REMPLACEE = nouveau depart).
'=============================================================================
Public Sub EnqueueUrgentTracking(ByVal oMail As Outlook.mailItem, ByVal dueDate As Date)
    On Error GoTo ErrH
    If oMail Is Nothing Then Exit Sub

    Dim recipCount As Long
    recipCount = CountTORecipients(oMail)

    Dim subj As String
    subj = oMail.subject
    If Len(Trim(subj)) = 0 Then subj = T("(sans objet)", "(no subject)")

    Dim recips As String
    recips = GetTORecipientsList(oMail)

    Dim convID As String
    convID = oMail.ConversationID

    ' delai = (date limite - aujourd'hui), plancher 0 (une urgence datee
    ' d'aujourd'hui ou anterieure reste due aujourd'hui).
    Dim delayDays As Long
    delayDays = DateDiff("d", Int(Now), Int(dueDate))
    If delayDays < 0 Then delayDays = 0

    If mUrgentQueue Is Nothing Then Set mUrgentQueue = New Collection
    ' EntryID laisse vide : le brouillon change d'EntryID a l'envoi ; la
    ' ligne est re-identifiee (objet + date d'envoi, RelinkRows) a la
    ' premiere actualisation complete.
    mUrgentQueue.Add Array(subj, recips, recipCount, delayDays, Now, dueDate, convID, "")

    modDebugLog.DInfo MOD_NAME, "EnqueueUrgentTracking : '" & modDebugLog.LogClip(subj) & _
        "' due " & Format(dueDate, "dd/mm/yyyy") & " (delai " & delayDays & " j)"
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "EnqueueUrgentTracking err " & Err.Number & " - " & Err.Description
End Sub

' v6.7 (F4-12) : les elements restent dans la file tant que le classeur
' n'est pas ENREGISTRE (avant : retires avant l'ecriture, donc perdus sur
' une erreur). Echec (Excel occupe, enregistrement refuse) : nouvel essai
' differe (RetryUrgentFlush), puis toast d'alerte. Chaque ligne est ecrite
' en une affectation (PutTrackRow) ; filtre leve avant (F4-01).
Public Sub FlushUrgentTracking()
    On Error GoTo ErrH
    If mUrgentQueue Is Nothing Then Exit Sub
    If mUrgentQueue.count = 0 Then Exit Sub
    ' v6.7 (F4-06) : un autre traitement du suivi tient le classeur -> plus tard.
    If TrackerBusy() Then
        modDefer.ArmUnique "UrgentFlush", BUSY_RETRY_SEC
        Exit Sub
    End If
    BeginTrackerWork "write"

    Dim why As String
    why = ""
    Dim xlApp As Object, wb As Object, ws As Object
    Set xlApp = GetExcelApp()
    If xlApp Is Nothing Then
        why = "Excel indisponible"
        GoTo Failed
    End If
    xlApp.DisplayAlerts = False
    Set wb = GetTrackerWorkbook(xlApp, True)
    If wb Is Nothing Then
        why = "classeur de suivi indisponible"
        GoTo Failed
    End If
    Set ws = wb.Sheets(modConfig.MailTrackingSheet())
    modExcelBridge.ClearSheetFilter ws        ' F4-01 : la ligne ecrite reste visible

    Dim n As Long, i As Long
    n = mUrgentQueue.count
    For i = 1 To n
        Dim it As Variant
        it = mUrgentQueue.Item(i)

        Dim subj As String, recips As String, convID As String, entryID As String
        Dim recipCount As Long, delayDays As Long
        Dim sentDate As Date, dueDate As Date
        subj = CStr(it(0)): recips = CStr(it(1)): recipCount = CLng(it(2))
        delayDays = CLng(it(3)): sentDate = CDate(it(4)): dueDate = CDate(it(5))
        convID = CStr(it(6)): entryID = CStr(it(7))

        Dim r As Long
        r = 0
        If Len(convID) > 0 Then r = FindConvRow(ws, convID, False)

        ' Bloc FOLDER..ACK_COUNT de la ligne : identifiant, fil et dossier
        ' d'une ligne existante conserves ; (re)ecriture du reste = nouveau
        ' depart.
        Dim vals As Variant
        If r = 0 Then
            r = TrackLastRow(ws) + 1
            vals = RowValues(ws, r)
            SetV vals, COL_ID, NewTrackID()
            SetV vals, COL_CONV_ID, convID
        Else
            vals = RowValues(ws, r)
        End If
        SetV vals, COL_SUBJECT, subj
        SetV vals, COL_RECIPIENTS, recips
        SetV vals, COL_RESPONSE_COUNT, 0
        SetV vals, COL_RESPONDERS, ""
        SetV vals, COL_SENT_DATE, sentDate
        SetV vals, COL_DELAY_DAYS, delayDays
        SetV vals, COL_DUE_DATE, DateAdd("d", delayDays, sentDate)
        SetV vals, COL_LAST_FOLLOWUP, ""
        SetV vals, COL_FOLLOWUP_COUNT, 0
        SetV vals, COL_ACK_COUNT, 0
        If Len(entryID) > 0 Then SetV vals, COL_ENTRY_ID, entryID
        PutTrackRow ws, r, vals, "En attente"
    Next i

    WriteDueCache ws
    If Not SaveWorkbookVisible(xlApp, wb) Then
        why = "enregistrement du classeur refuse"
        GoTo Failed
    End If
    ReleaseBackground xlApp, wb
    For i = 1 To n
        mUrgentQueue.Remove 1
    Next i
    mUrgentFails = 0
    EndTrackerWork

    If n > 0 Then
        modNotify.Notify T("Suivi urgent enregistr~e", "Urgent tracking saved")
        modDebugLog.DInfo MOD_NAME, "FlushUrgentTracking : " & n & " ligne(s) ecrite(s)/actualisee(s)"
    End If
    Exit Sub
ErrH:
    why = "err " & Err.Number & " - " & Err.Description
Failed:
    ' v5.9 : ne jamais laisser le classeur ouvert masque apres une erreur.
    On Error Resume Next
    If Not wb Is Nothing Then ReleaseBackground xlApp, wb
    EndTrackerWork
    RetryUrgentFlush why
End Sub

' v6.7 (F4-12) - Echec de FlushUrgentTracking : la file est conservee ;
' nouvel essai dans FLUSH_RETRY_SEC s (FLUSH_MAX_TRIES fois), puis toast
' d'alerte. Classeur absent ou gabarit inconnu (deja signale) : pas
' d'essai inutile, la file attend le prochain envoi urgent.
Private Sub RetryUrgentFlush(ByVal why As String)
    On Error Resume Next
    If mTrackerFailReported Then
        modDebugLog.DError MOD_NAME, "FlushUrgentTracking : " & why & " - suivi(s) urgent(s) garde(s) en file"
        mUrgentFails = 0
        Exit Sub
    End If
    mUrgentFails = mUrgentFails + 1
    If mUrgentFails <= FLUSH_MAX_TRIES Then
        modDebugLog.DError MOD_NAME, "FlushUrgentTracking : " & why & " - nouvel essai dans " & _
            FLUSH_RETRY_SEC & " s (" & mUrgentFails & "/" & FLUSH_MAX_TRIES & ")"
        modDefer.ArmUnique "UrgentFlush", FLUSH_RETRY_SEC
    Else
        modDebugLog.DError MOD_NAME, "FlushUrgentTracking : " & why & " - abandon apres " & _
            FLUSH_MAX_TRIES & " essais, suivi(s) garde(s) en file"
        modUi.Toast T("Suivi urgent NON enregistr~e : le classeur de suivi ne r~epond pas.", _
                      "Urgent tracking NOT saved: the tracking workbook does not respond.") & vbLf & _
                    T("Nouvel essai au prochain envoi urgent ; sinon utilisez Marquer pour suivi.", _
                      "Retried with the next urgent mail; otherwise use Mark for tracking."), uiError, True
        mUrgentFails = 0
    End If
End Sub

'=============================================================================
' v6.7 - Premiere ligne dont le ConversationID correspond (0 si aucune).
' Lecture en bloc (avant : un appel Excel par ligne), toutes les lignes, y
' compris celles masquees par un filtre. activeOnly = ne retenir qu'une
' ligne EN COURS (statut ni Repondu, ni Repondu ?, ni Traite). Une erreur
' Excel remonte a l'appelant.
'=============================================================================
Private Function FindConvRow(ByVal ws As Object, ByVal convID As String, ByVal activeOnly As Boolean) As Long
    FindConvRow = 0
    If Len(convID) = 0 Then Exit Function
    Dim lastRow As Long
    lastRow = TrackLastRow(ws)
    If lastRow < 2 Then Exit Function
    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)
    Dim r As Long
    For r = 1 To lastRow - 1
        If CStr(SafeVal(blk, r, COL_CONV_ID)) = convID Then
            If Not activeOnly Then
                FindConvRow = r + 1
                Exit Function
            End If
            If Not IsTerminalStatus(CStr(SafeVal(blk, r, COL_STATUS))) Then
                FindConvRow = r + 1
                Exit Function
            End If
        End If
    Next r
End Function

' v6.7 - Statut termine : ni recalcule ni relance (Repondu, Repondu ?,
' Traite). Valeurs du contrat de donnees, jamais traduites.
Private Function IsTerminalStatus(ByVal st As String) As Boolean
    st = Trim$(st)
    IsTerminalStatus = (st = "R" & Chr(233) & "pondu" Or st = "R" & Chr(233) & "pondu ?" Or _
                        st = "Trait" & Chr(233))
End Function

'=============================================================================
' v6.7 (F4-12 (5)) - Ecriture d'une ligne de suivi en UNE affectation.
' RowValues : bloc FOLDER..ACK_COUNT de la ligne r (tableau 1 x n, valeurs
' actuelles, vides pour une ligne neuve) ; SetV : une valeur par indice de
' colonne ; PutTrackRow : ecrit le bloc d'un coup (reussit ou echoue en
' bloc : plus de ligne a moitie ecrite), puis le statut, les formats de
' date et la formule d'echeance. Les erreurs remontent a l'appelant.
'=============================================================================
Private Function RowValues(ByVal ws As Object, ByVal r As Long) As Variant
    RowValues = modExcelBridge.ReadBlock(ws, r, COL_FOLDER, r, COL_ACK_COUNT)
End Function

Private Sub SetV(ByRef vals As Variant, ByVal col As Long, ByVal v As Variant)
    vals(1, col - COL_FOLDER + 1) = v
End Sub

Private Sub PutTrackRow(ByVal ws As Object, ByVal r As Long, ByRef vals As Variant, ByVal status As String)
    ws.Range(ws.Cells(r, COL_FOLDER), ws.Cells(r, COL_ACK_COUNT)).Value = vals
    ws.Cells(r, COL_STATUS).Value = status
    ws.Cells(r, COL_SENT_DATE).NumberFormat = "dd/mm/yyyy hh:mm"
    ws.Cells(r, COL_DUE_DATE).NumberFormat = "dd/mm/yyyy"
    FormatStatusCell ws.Cells(r, COL_STATUS)
    UpdateDeadlineCell ws, r
End Sub

' Identifiant de suivi unique (meme en rafale dans la meme seconde).
Private Function NewTrackID() As String
    Static lastStamp As String
    Static sameSec As Long
    Dim stamp As String
    stamp = Format(Now, "yyyymmdd-hhnnss")
    If stamp = lastStamp Then
        sameSec = sameSec + 1
    Else
        sameSec = 0
        lastStamp = stamp
    End If
    NewTrackID = "ST-" & stamp
    If sameSec > 0 Then NewTrackID = NewTrackID & "-" & sameSec
End Function

'=============================================================================
' v6.7 - Corps du tableau de bord.
'   checkLate = True (FollowUpOverdue, F4-16 / F4-09) : apres le recalcul,
'   les suivis EN COURS dont la date limite est passee (jours entiers)
'   sont COCHES, les autres decoches ; le bouton Relancer du classeur
'   prepare ensuite leurs relances. Une ligne "Repondu ?" (reponse recue, a
'   valider) n'est jamais cochee.
'   F4-06 : evenements et affichage d'Excel coupes pendant le travail
'   (retablis sur toutes les sorties) ; chaque mail relu une seule fois
'   (cache EntryID -> dossier partage par RelinkRows et RefreshFolderNames).
'   F4-07 : fichier des echeances ecrit avant l'enregistrement.
'=============================================================================
Private Sub OpenDashboardCore(ByVal checkLate As Boolean)

    On Error GoTo ErrorHandler

    Dim trackingFile As String
    trackingFile = GetTrackerFilePath()

    If Dir(trackingFile) = "" Then
        MsgBox T("Aucun suivi en cours." & vbCrLf & _
               "Utilisez 'Marquer Suivi' sur un mail envoy" & Chr(233) & " pour commencer.", _
               "No tracking in progress." & vbCrLf & _
               "Use 'Mark for tracking' on a sent mail to start."), _
               vbInformation, T("Suivi Mails", "Mail Tracking")
        Exit Sub
    End If

    ' v6.7 : un seul traitement du suivi a la fois.
    If TrackerBusy() Then
        NotifyTrackerBusy
        Exit Sub
    End If
    BeginTrackerWork "dashboard"

    Dim xlApp As Object, wb As Object, ws As Object
    Dim evSaved As Boolean, suSaved As Boolean, quietOn As Boolean
    quietOn = False
    Set xlApp = GetExcelApp()
    If xlApp Is Nothing Then GoTo NoWorkbook
    xlApp.DisplayAlerts = False

    Set wb = GetTrackerWorkbook(xlApp)
    If wb Is Nothing Then GoTo NoWorkbook

    Set ws = wb.Sheets(modConfig.MailTrackingSheet())
    XlQuiet xlApp, evSaved, suSaved
    quietOn = True

    StartFolderCache
    UpdateAllTrackingStatuses ws

    ' v6.3 : noms de dossier relus, tri, controle de coherence (APRES le
    ' tri : ses commentaires citent des numeros de ligne), mise en page.
    RefreshFolderNames ws
    StopFolderCache
    SortDashboard ws
    Dim nNoId As Long, nDup As Long
    CheckConsistency ws, nNoId, nDup
    ApplyDashboardLayout ws
    Dim nLate As Long
    nLate = 0
    If checkLate Then nLate = MarkLateRows(ws)

    WriteDueCache ws
    SaveWorkbookVisible xlApp, wb
    XlRestore xlApp, evSaved, suSaved
    quietOn = False

    ' --- Afficher et forcer le premier plan ---
    xlApp.Visible = True

    ' v5.1 : re-afficher la fenetre du classeur. Un rappel de demarrage a pu
    ' la masquer (HideBackgroundWorkbook) sans refermeture (fuite corrigee
    ' par ailleurs, mais un classeur herite d'une session precedente peut
    ' encore etre cache) : sans ce geste, Excel apparaissait VIDE.
    On Error Resume Next
    wb.Windows(1).Visible = True
    On Error GoTo ErrorHandler

    ' v5.9 : le classeur est maintenant sous les yeux de l'utilisateur ; il
    ' lui appartient. Les ecritures d'arriere-plan ne le refermeront pas.
    HandOverWorkbook
    EndTrackerWork

    ' v6.7 : compteurs du ruban du classeur ; recalcul complet = recalcul du
    ' jour (F4-08) ; toast du matin s'il etait en attente.
    NotifyExcelRefreshDone xlApp, wb
    MorningAfterRefresh

    ' v6.3 : resultat du controle de coherence, une fois le classeur affiche.
    NotifyConsistency nNoId, nDup
    If checkLate Then
        If nLate > 0 Then
            modUi.Toast nLate & T(" suivi(s) en retard coch~e(s).", " overdue item(s) checked.") & vbLf & _
                        T("Bouton Relancer du classeur : brouillons de relance ~a relire.", _
                          "Workbook Follow up button: follow-up drafts to review."), uiInfo, True
        Else
            modUi.Toast T("Aucun suivi en retard.", "No overdue item."), uiInfo
        End If
    End If

    ' Forcer l'affichage : plein ecran + premier plan
    On Error Resume Next
    xlApp.WindowState = -4137                 ' xlMaximized
    wb.Activate
    ws.Activate
    AppActivate xlApp.Caption
    SetForegroundWindow xlApp.hwnd
    On Error GoTo ErrorHandler

    xlApp.DisplayAlerts = True                ' v5.5

    Exit Sub

NoWorkbook:
    EndTrackerWork
    If Not mTrackerFailReported Then
        MsgBox T("Impossible d'ouvrir le fichier de suivi.", "Cannot open the tracking file."), vbCritical, T("Suivi Mails", "Mail Tracking")
    End If
    Exit Sub

ErrorHandler:
    Dim dashErrNum As Long, dashErrDesc As String
    dashErrNum = Err.Number
    dashErrDesc = Err.Description
    ' v5.9 : erreur AVANT l'affichage -> le classeur ouvert en arriere-plan
    ' est encore le notre : le liberer (sinon Excel fantome). Apres
    ' l'affichage (HandOverWorkbook deja passe), il n'y a rien a refermer.
    On Error Resume Next
    If quietOn Then XlRestore xlApp, evSaved, suSaved
    StopFolderCache
    If mWbDepth > 0 Then
        If Not wb Is Nothing Then ReleaseBackground xlApp, wb
    End If
    EndTrackerWork
    MsgBox T("Erreur ", "Error ") & dashErrNum & " : " & dashErrDesc, _
           vbCritical, T("Suivi Mails - Erreur", "Mail Tracking - Error")
End Sub

'=============================================================================
' v6.7 (F4-06) - Evenements du classeur (chaque ecriture declenchait
' Workbook_SheetChange) et affichage d'Excel coupes pendant un traitement ;
' XlRestore remet les valeurs d'avant, sur toutes les sorties.
'=============================================================================
Private Sub XlQuiet(ByVal xlApp As Object, ByRef evSaved As Boolean, ByRef suSaved As Boolean)
    On Error Resume Next
    evSaved = True
    suSaved = True
    evSaved = CBool(xlApp.EnableEvents)
    suSaved = CBool(xlApp.ScreenUpdating)
    xlApp.EnableEvents = False
    xlApp.ScreenUpdating = False
End Sub

Private Sub XlRestore(ByVal xlApp As Object, ByVal evSaved As Boolean, ByVal suSaved As Boolean)
    On Error Resume Next
    xlApp.EnableEvents = evSaved
    xlApp.ScreenUpdating = suSaved
End Sub

    On Error Resume Next
    Set mEidFolder = CreateObject("Scripting.Dictionary")

    Set mEidFolder = Nothing
End Sub

' v6.7 (F4-16 / F4-09) - Coche les suivis EN COURS (ni Repondu, ni Repondu ?,
' ni Traite) dont la date limite est passee (le lendemain de la date
' limite, comme la colonne Echeance), decoche les autres. Une affectation
' pour toute la colonne. Renvoie le nombre de lignes cochees.
'=============================================================================
Private Function MarkLateRows(ByVal ws As Object) As Long
    MarkLateRows = 0
    Dim lastRow As Long
    lastRow = TrackLastRow(ws)
    If lastRow < 2 Then Exit Function
    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)
    Dim n As Long, r As Long
    n = lastRow - 1
    Dim marks() As Variant
    ReDim marks(1 To n, 1 To 1)
    For r = 1 To n
        marks(r, 1) = ""
        If Len(CStr(SafeVal(blk, r, COL_ID))) > 0 Then
            If IsLateValue(CStr(SafeVal(blk, r, COL_STATUS)), SafeVal(blk, r, COL_DUE_DATE)) Then
                marks(r, 1) = ChrW$(MARK_GLYPH)
                MarkLateRows = MarkLateRows + 1
            End If
        End If
    Next r
    ws.Range(ws.Cells(2, COL_MARK), ws.Cells(lastRow, COL_MARK)).Value = marks
End Function

' v6.7 (F4-15) - Suivi en retard : statut en cours ET date limite passee en
' JOURS ENTIERS (en retard le lendemain de la date limite, comme la colonne
' Echeance et le rappel du matin ; avant : des l'heure d'envoi + delai).
Private Function IsLateValue(ByVal st As String, ByVal dueV As Variant) As Boolean
    IsLateValue = False
    If IsTerminalStatus(st) Then Exit Function
    Dim due As Date
    due = ToDateOrZero(dueV)
    If due <= 0 Then Exit Function
    IsLateValue = (Date > Int(CDbl(due)))
End Function

' v6.7 - Valeur de cellule -> date (0 si vide, illisible ou erreur). Une
' date sans format de date est lue comme un nombre : acceptee.
Private Function ToDateOrZero(ByVal v As Variant) As Date
    On Error Resume Next
    ToDateOrZero = 0
    If IsEmpty(v) Or IsError(v) Then Exit Function
    If VarType(v) = vbString Then
        If Len(Trim$(CStr(v))) = 0 Then Exit Function
        If Not IsDate(v) Then Exit Function
    End If
    ToDateOrZero = CDate(v)
End Function

Private Sub ReleaseBackground(xlApp As Object, wb As Object)

    If Not mWbOpenedHere Then Exit Sub

' etait avale : l'appelant croyait le suivi enregistre). Les appelants qui
' ne testent pas le resultat l'appellent comme avant.
'=============================================================================
Private Function SaveWorkbookVisible(ByVal xlApp As Object, ByVal wb As Object) As Boolean
    SaveWorkbookVisible = False
    On Error Resume Next
    Dim w As Object
    Set w = wb.Windows(1)
    Dim wasHidden As Boolean
    wasHidden = False
    If Not w Is Nothing Then wasHidden = Not CBool(w.Visible)
    If wasHidden Then
        xlApp.ScreenUpdating = False
        w.Visible = True
    End If
    Err.Clear
    wb.Save
    Dim errNum As Long, errDesc As String
    errNum = Err.Number
    errDesc = Err.Description
    If wasHidden Then
        w.Visible = False
        xlApp.ScreenUpdating = True
    End If
    If errNum <> 0 Then
        modDebugLog.DError MOD_NAME, "Enregistrement du classeur de suivi refuse (err " & errNum & " - " & errDesc & ")"
        Exit Function
    End If
    SaveWorkbookVisible = True
End Function

'=============================================================================
' AddrInExpected (v5.5) : l'adresse appartient-elle a la liste attendue
' (elements separes par ";") par EGALITE stricte, insensible a la casse ?
' Remplace l'ancien test InStr par sous-chaine : "paul@x.fr" validait
' "jean-paul@x.fr" comme repondant.
'=============================================================================
Private Function AddrInExpected(ByVal expectedList As String, ByVal addr As String) As Boolean
    AddrInExpected = False
    If Len(addr) = 0 Then Exit Function
    Dim parts() As String
    parts = Split(expectedList, ";")
    Dim k As Long
    For k = LBound(parts) To UBound(parts)
        If StrComp(Trim$(parts(k)), addr, vbTextCompare) = 0 Then
            AddrInExpected = True
            Exit Function
        End If
    Next k
End Function

Private Function GetTORecipientsList(oMail As Outlook.mailItem) As String
    Dim result As String
    Dim i As Long
    Dim recip As Outlook.Recipient

    For i = 1 To oMail.recipients.count
        Set recip = oMail.recipients.Item(i)
        If recip.Type = olTo Then
            Dim smtpAddr As String

            result = result & smtpAddr
        End If
    Next i

    GetTORecipientsList = result
End Function

'=============================================================================
' SAUVEGARDER UNE ENTREE DE SUIVI
' v5.2 : Function Boolean. Un echec (instance Excel fantome -> erreur 462)
' declenche UNE nouvelle tentative sur instance neuve ; l'erreur n'est
' montree qu'en cas d'echec definitif. Les appelants ne confirment le
' suivi QUE si le retour est True (avant : erreur 462 PUIS message de
' succes, alors qu'aucune ligne n'avait ete ecrite).
' v6.7 : quiet = appel differe (RecordSentMailQuick, F4-12) : echec
' definitif annonce par un toast, jamais de MsgBox. errNum < 0 = deja
' signale a l'utilisateur, sans nouvel essai : classeur introuvable ou
' gabarit inconnu (-1), suivi en cours de mise a jour (-1), fil deja suivi
' (-2, F4-03).
'=============================================================================
Private Function SaveTrackingEntry(subject As String, _
                              recipients As String, recipientCount As Long, _
                              delayDays As Long, sentDate As Date, _
                              convID As String, entryID As String, _
                              Optional ByVal folderName As String = "", _
                              Optional ByVal quiet As Boolean = False) As Boolean

    Dim errNum As Long, errDesc As String
    Dim attempt As Long
    For attempt = 1 To 2
        errNum = 0: errDesc = ""
        If TrySaveTrackingEntry(subject, recipients, recipientCount, _
                                delayDays, sentDate, convID, entryID, folderName, _
                                errNum, errDesc) Then
            SaveTrackingEntry = True
            Exit Function
        End If
        If errNum < 0 Then Exit Function     ' deja signale (cf. en-tete)
        ' Echec probable sur instance morte : purger le cache et retenter
        ' une seule fois sur une instance propre.
        modDebugLog.DInfo MOD_NAME, "SaveTrackingEntry : tentative " & _
            attempt & " KO (err " & errNum & "), purge du cache Excel"
        ResetExcelCache
    Next attempt

    If quiet Then
        modUi.Toast T("Suivi NON enregistr~e (erreur ", "Tracking NOT saved (error ") & errNum & ")." & vbLf & _
                    T("Le classeur de suivi ne r~epond pas : voir debug.log.", _
                      "The tracking workbook does not respond: see debug.log."), uiError, True
        Exit Function
    End If
    MsgBox T("Erreur lors de la sauvegarde du suivi :", "Error while saving the tracking:") & vbCrLf & _
           T("Erreur ", "Error ") & errNum & " : " & errDesc & vbCrLf & vbCrLf & _
           T("Le suivi n'a PAS " & Chr(233) & "t" & Chr(233) & " enregistr" & Chr(233) & ".", _
             "The tracking was NOT recorded."), _
           vbCritical, T("Suivi Mails", "Mail Tracking")
End Function

'=============================================================================
' v6.7 : fil deja suivi (F4-03), filtre leve et derniere ligne comptee sous
' les lignes masquees (F4-01), ligne ecrite en une affectation (F4-12),
' identifiant par NewTrackID (F4-17 : le calcul etait recopie ici avec ses
' propres compteurs ; deux suivis crees dans la meme seconde par les deux
' chemins recevaient le meme identifiant), fichier des echeances (F4-07).
'=============================================================================
Private Function TrySaveTrackingEntry(subject As String, _
                              recipients As String, recipientCount As Long, _
                              delayDays As Long, sentDate As Date, _
                              convID As String, entryID As String, _
                              ByVal folderName As String, _
                              ByRef errNum As Long, ByRef errDesc As String) As Boolean

    On Error GoTo SaveError

    ' v6.7 : un autre traitement du suivi tient le classeur.
    If TrackerBusy() Then
        errNum = -1
        NotifyTrackerBusy
        Exit Function
    End If
    BeginTrackerWork "write"

    Dim xlApp As Object
    Set xlApp = GetExcelApp()
    If xlApp Is Nothing Then
        errNum = 429
        errDesc = "Excel indisponible"
        EndTrackerWork
        Exit Function
    End If
    xlApp.DisplayAlerts = False

    Dim wb As Object
    Set wb = GetTrackerWorkbook(xlApp)

    If wb Is Nothing Then
        If mTrackerFailReported Then
            errNum = -1                      ' message deja affiche, ne pas retenter
        Else
            errNum = 1004
            errDesc = "ouverture du classeur de suivi impossible"
        End If
        EndTrackerWork
        Exit Function
    End If

    Dim ws As Object
    Set ws = wb.Sheets(modConfig.MailTrackingSheet())

    ' F4-03 : fil deja suivi par une ligne EN COURS (marquage pendant la
    ' redaction puis "Suivre" apres l'envoi, double marquage...) : rien
    ' n'est ajoute.
    Dim dupRow As Long
    dupRow = FindConvRow(ws, convID, True)
    If dupRow > 0 Then
        ReleaseBackground xlApp, wb
        EndTrackerWork
        errNum = -2
        modDebugLog.DInfo MOD_NAME, "Suivi non ajoute : fil deja suivi (ligne " & dupRow & ")"
        modUi.Toast T("Ce fil est d~ej~a suivi (ligne ", "This thread is already tracked (row ") & dupRow & ")." & vbLf & _
                    T("Rien n'a ~et~e ajout~e.", "Nothing was added."), uiInfo, True
        Exit Function
    End If

    ' F4-01 : filtre leve (la ligne ecrite reste visible) ; la derniere
    ' ligne est cherchee AUSSI sous les lignes masquees.
    modExcelBridge.ClearSheetFilter ws
    Dim NextRow As Long
    NextRow = TrackLastRow(ws) + 1

    Dim vals As Variant
    vals = RowValues(ws, NextRow)
    SetV vals, COL_FOLDER, folderName                 ' v6.3
    SetV vals, COL_SUBJECT, subject
    SetV vals, COL_RECIPIENTS, recipients
    SetV vals, COL_RESPONSE_COUNT, 0
    SetV vals, COL_RESPONDERS, ""
    SetV vals, COL_SENT_DATE, sentDate
    SetV vals, COL_DELAY_DAYS, delayDays
    SetV vals, COL_DUE_DATE, DateAdd("d", delayDays, sentDate)
    SetV vals, COL_LAST_FOLLOWUP, ""
    SetV vals, COL_FOLLOWUP_COUNT, 0
    SetV vals, COL_ID, NewTrackID()
    SetV vals, COL_CONV_ID, convID
    SetV vals, COL_ENTRY_ID, entryID
    SetV vals, COL_ACK_COUNT, 0
    PutTrackRow ws, NextRow, vals, "En attente"

    WriteDueCache ws
    If Not SaveWorkbookVisible(xlApp, wb) Then
        If mWbOpenedHere Then
            ' Classeur ouvert ici : referme sans enregistrer -> rien n'est
            ' garde, nouvel essai par SaveTrackingEntry.
            errNum = 1004
            errDesc = "enregistrement du classeur refuse"
            ReleaseBackground xlApp, wb
            EndTrackerWork
            Exit Function
        End If
        ' Classeur ouvert chez l'utilisateur : la ligne y est, il
        ' l'enregistrera.
        modDebugLog.DError MOD_NAME, "SaveTrackingEntry : ligne ecrite, classeur ouvert non enregistre"
    End If
    ReleaseBackground xlApp, wb
    EndTrackerWork

    TrySaveTrackingEntry = True
    Exit Function

SaveError:
    errNum = Err.Number
    errDesc = Err.Description
    modDebugLog.DError MOD_NAME, "SaveTrackingEntry err " & _
        errNum & " - " & errDesc
    ' v5.9 : liberer le classeur si l'erreur est survenue apres son ouverture.
    On Error Resume Next
    If Not wb Is Nothing Then ReleaseBackground xlApp, wb
    EndTrackerWork
End Function

'=============================================================================
' EnsureSchemaSheet (pack v3 - A11) : ecrit la feuille masquee _Schema
' (A = nom logique, B = indice de colonne) si absente ou d'une autre

Private Sub EnsureSchemaSheet(ByVal wb As Object)
    On Error Resume Next
    If wb Is Nothing Then Exit Sub
    Dim ws As Object
    Set ws = Nothing
    Set ws = wb.Worksheets(SCHEMA_SHEET)
    Err.Clear
    If Not ws Is Nothing Then
        If CStr(ws.Cells(1, 2).Value) = SCHEMA_VERSION Then Exit Sub
    Else
        Set ws = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.count))
        If Err.Number <> 0 Then Err.Clear: Exit Sub
        ws.name = SCHEMA_SHEET
    End If
    ws.Cells.ClearContents
    Dim names As Variant, cols As Variant, i As Long
    names = Array("SchemaVersion", "MARK", "STATUS", "DEADLINE", "FOLDER", "SUBJECT", "RECIPIENTS", _
                  "RESPONSE_COUNT", "RESPONDERS", "SENT_DATE", "DELAY_DAYS", "DUE_DATE", "LAST_FOLLOWUP", _

                 COL_RESPONSE_COUNT, COL_RESPONDERS, COL_SENT_DATE, COL_DELAY_DAYS, COL_DUE_DATE, COL_LAST_FOLLOWUP, _
                 COL_FOLLOWUP_COUNT, COL_ID, COL_CONV_ID, COL_ENTRY_ID, COL_ACK_COUNT, COL_ARCHIVED_AT)
    For i = LBound(names) To UBound(names)
        ws.Cells(i + 1, 1).Value = names(i)
        ws.Cells(i + 1, 2).Value = cols(i)
    Next i
    ws.Visible = 0                     ' xlSheetHidden (visible via Afficher)
    modDebugLog.DInfo MOD_NAME, "Feuille _Schema ecrite (" & SCHEMA_VERSION & ")"
    Err.Clear
End Sub

'=============================================================================
' v6.7 (F4-06) - Couleurs de la colonne Statut par SIX regles de mise en

' toute la colonne : un statut ecrit en bloc prend sa couleur sans aucune

' colonne n'a pas deja ses regles (2 appels Excel dans le cas courant).

Private Sub EnsureStatusRules(ByVal ws As Object)
    On Error GoTo ErrH
    If Not IsSubjectHeader(ws.Cells(1, COL_SUBJECT).Value) Then Exit Sub
    Dim nExisting As Long
    nExisting = ws.Cells(2, COL_STATUS).FormatConditions.count
    If nExisting >= 6 Then Exit Sub

    rng.FormatConditions.Delete
    AddStatusRule rng, "R" & Chr(233) & "pondu", RGB(198, 239, 206), RGB(0, 97, 0)
    AddStatusRule rng, "En retard", RGB(255, 174, 174), RGB(156, 0, 0)
    AddStatusRule rng, "R" & Chr(233) & "pondu ?", RGB(255, 217, 179), RGB(197, 90, 17)

    AddStatusRule rng, "En attente", RGB(230, 230, 230), RGB(89, 89, 89)
    AddStatusRule rng, "Trait" & Chr(233), RGB(217, 217, 217), RGB(64, 64, 64)
    modDebugLog.DInfo MOD_NAME, "Couleurs de statut posees (mise en forme conditionnelle, feuille " & ws.name & ")"
    Exit Sub
ErrH:

End Sub

Private Sub AddStatusRule(ByVal rng As Object, ByVal status As String, ByVal fill As Long, ByVal fore As Long)
    Dim fc As Object
    ' Type 1 = xlCellValue, Operator 3 = xlEqual
    Set fc = rng.FormatConditions.Add(Type:=1, Operator:=3, Formula1:="=""" & status & """")
    fc.Interior.Color = fill
    fc.Font.Color = fore
    fc.Font.Bold = True
End Sub

'=============================================================================
' v6.7 (F8-22) - Charte commune des en-tetes (findings F8, choix Arial),

    Set wsS = Nothing

    Set wsS = wb.Worksheets(SCHEMA_SHEET)
    On Error GoTo ErrH
    If wsS Is Nothing Then Exit Sub

    Dim blk As Variant, i As Long, styleRow As Long, lastUsed As Long
    blk = modExcelBridge.ReadBlock(wsS, 1, 1, 40, 2)
    styleRow = 0
    lastUsed = 0
    For i = 1 To 40
        If Len(CStr(SafeVal(blk, i, 1))) > 0 Then lastUsed = i
        If CStr(SafeVal(blk, i, 1)) = STYLE_KEY Then styleRow = i
    Next i
    If styleRow > 0 Then
        If CStr(SafeVal(blk, styleRow, 2)) = STYLE_VERSION Then Exit Sub
    Else

    End If

    Dim ws As Object
    Set ws = Nothing
    On Error Resume Next
    Set ws = wb.Worksheets(modConfig.MailTrackingSheet())
    On Error GoTo ErrH

    Set ws = Nothing
    On Error Resume Next
    Set ws = wb.Worksheets(ARCHIVE_SHEET)
    On Error GoTo ErrH
    If Not ws Is Nothing Then ApplyHeaderCharter ws, COL_ARCHIVED_AT

    wsS.Cells(styleRow, 1).Value = STYLE_KEY
    wsS.Cells(styleRow, 2).Value = STYLE_VERSION
    modDebugLog.DInfo MOD_NAME, "Charte des en-tetes appliquee (" & STYLE_VERSION & ")"
    Exit Sub

End Sub

'   - Objet en 5 : gabarit courant, rien a faire ;

'   - Objet en 4 : gabarit intermediaire du 05/10 matin, jamais installe
'     (dossier en 3, liens en 13-17).
' Migration : retrait des colonnes de liens (en-tetes VERIFIES avant
' suppression), insertion de la colonne coche en 1 et, au besoin, de la
' colonne dossier en 4. Feuille "Archives" (creee par le code du classeur
' avec le meme en-tete, liens masques et sans en-tete) traitee de meme : son
' ancienne colonne "Archive le" (21) devient la colonne ARCHIVED_AT (18).
' Gabarit inconnu : on ne touche a rien (LayoutReady refusera d'ecrire).

    Set ws = wb.Worksheets(modConfig.MailTrackingSheet())

        End If
    End If
    Dim wsA As Object

ErrH:

End Sub

' True si la feuille vient d'etre migree. Une erreur remonte a EnsureLayout.
Private Function MigrateSheetLayout(ByVal ws As Object, ByVal isArchive As Boolean) As Boolean
    MigrateSheetLayout = False
    Dim subjAt As Long
    subjAt = FindHeaderCol(ws, "objet", "subject")
    If subjAt = 0 Then Exit Function                  ' feuille vide ou inconnue
    If subjAt = COL_SUBJECT Then Exit Function        ' gabarit courant

    Dim actFirst As Long, actLast As Long, hasFolder As Boolean
    Select Case subjAt
        Case 3: actFirst = 12: actLast = 16: hasFolder = False
        Case 4: actFirst = 13: actLast = 17: hasFolder = True
        Case Else
            modDebugLog.DError MOD_NAME, "Gabarit inconnu (feuille " & ws.name & _
                ", Objet en colonne " & subjAt & ") : pas de migration"
            Exit Function
    End Select
    If hasFolder Then
        If Not HeaderIs(ws, 3, "nom du dossier", "folder name") Then
            modDebugLog.DError MOD_NAME, "Gabarit inconnu (feuille " & ws.name & _
                ", Objet en 4 sans dossier en 3) : pas de migration"
            Exit Function
        End If
    End If

    ' Controle des colonnes de liens AVANT suppression (jamais a l'aveugle).
    Dim okLinks As Boolean
    If isArchive Then
        ' Archives : en-tetes de liens deja vides (EnsureArchiveSheet v6).
        okLinks = (Len(Trim$(CStr(ws.Cells(1, actFirst).Value))) = 0 And _
                   Len(Trim$(CStr(ws.Cells(1, actLast).Value))) = 0)
    Else
        okLinks = HeaderIs(ws, actFirst, "relancer", "follow up") And _
                  HeaderIs(ws, actLast, "supprimer", "delete")
    End If
    If Not okLinks Then
        modDebugLog.DError MOD_NAME, "Gabarit inconnu (feuille " & ws.name & _
            ", colonnes de liens " & actFirst & "-" & actLast & " non reconnues) : pas de migration"
        Exit Function
    End If

    ws.Range(ws.Columns(actFirst), ws.Columns(actLast)).Delete
    ws.Columns(COL_MARK).Insert
    If Not hasFolder Then ws.Columns(COL_FOLDER).Insert

    ' En-tetes des colonnes nouvelles, au format de l'en-tete Statut.
    ws.Cells(1, COL_STATUS).Copy ws.Cells(1, COL_MARK)
    ws.Cells(1, COL_MARK).Value = ChrW$(10003)
    If Not hasFolder Then
        ws.Cells(1, COL_STATUS).Copy ws.Cells(1, COL_FOLDER)
        ws.Cells(1, COL_FOLDER).Value = T("Nom du dossier", "Folder name")
        ws.Columns(COL_FOLDER).ColumnWidth = 28
    End If
    FormatMarkColumn ws
    If isArchive Then
        ws.Cells(1, COL_ARCHIVED_AT).Value = T("Archiv~e le", "Archived on")
        ws.Columns(COL_ARCHIVED_AT).ColumnWidth = 16
    End If
    MigrateSheetLayout = True
End Function

'=============================================================================
' v6.3 - Filtre automatique. v6.4 : sur TOUTES les colonnes (1 a LAST_COL),
' colonnes techniques masquees comprises. En v6.3 il s'arretait a la
' derniere colonne VISIBLE : un tri lance depuis une fleche du filtre
' deplacait les lignes visibles sans leurs colonnes techniques (ID, fil,
' mail, acquittement), et chaque objet se retrouvait relie au mail d'une
' autre ligne. Un filtre pose sur une autre plage (colonnes OU nombre de
' lignes) est remplace. Meme plage que modMailTracker.EnsureFilterCoversAll
' (code du classeur, v1.1).
'=============================================================================
Private Sub EnsureAutoFilter(ws As Object)
    On Error Resume Next
    Dim lastRow As Long
    lastRow = TrackLastRow(ws)          ' v6.7 (F4-01) : lignes filtrees comprises
    If lastRow < 2 Then Exit Sub
    Dim mode As Boolean, needReset As Boolean
    mode = False
    mode = CBool(ws.AutoFilterMode)
    needReset = True
    If mode Then
        Dim c1 As Long, nc As Long, nr As Long
        c1 = 0: nc = 0: nr = 0
        c1 = ws.AutoFilter.Range.Column
        nc = ws.AutoFilter.Range.Columns.count
        nr = ws.AutoFilter.Range.Rows.count
        needReset = (c1 <> 1 Or nc <> LAST_COL Or nr <> lastRow)
        If needReset Then ws.AutoFilterMode = False
    End If
    If needReset Then ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, LAST_COL)).AutoFilter
End Sub

    ws.Columns(COL_ENTRY_ID).Hidden = True

'=============================================================================
' v6.3 - Formule de la colonne Echeance (R1C1 relatif : survit au tri).
' Vide pour Repondu, Repondu ? et Traite ; sinon J-x / Aujourd'hui /
' Retard x j, d'apres la date limite et TODAY(). Les statuts compares sont
' le contrat de donnees (jamais traduits) ; les textes passent par T().
'=============================================================================
Private Function DeadlineFormulaR1C1() As String
    Dim st As String, du As String
    st = "RC" & COL_STATUS
    du = "RC" & COL_DUE_DATE
    DeadlineFormulaR1C1 = "=IF(OR(" & st & "=""R" & Chr(233) & "pondu""," & _
        st & "=""R" & Chr(233) & "pondu ?""," & st & "=""Trait" & Chr(233) & """," & du & "=""""),""""," & _
        "IFERROR(IF(INT(" & du & ")-TODAY()>0,""" & T("J-", "D-") & """&(INT(" & du & ")-TODAY())," & _
        "IF(INT(" & du & ")-TODAY()=0,""" & T("Aujourd'hui", "Today") & """,""" & _
        T("Retard ", "Overdue ") & """&(TODAY()-INT(" & du & "))&""" & T(" j", " d") & """)),""""))"
End Function

'=============================================================================
' v6.4 - Le mail relu (objet mailSubj) peut-il etre celui d'une ligne d'objet
' rowSubj ? Objets normalises (prefixes RE:/TR:... retires) egaux, ou l'un
' contenu dans l'autre (reponse dont l'objet a ete complete). Un objet vide
' ne prouve rien : True.
'=============================================================================
Private Function SameSubjectFamily(ByVal rowSubj As String, ByVal mailSubj As String) As Boolean
    Dim a1 As String, b1 As String
    SameSubjectFamily = True
    a1 = NormalizeSubject(rowSubj)
    b1 = NormalizeSubject(mailSubj)
    If Len(a1) = 0 Or Len(b1) = 0 Then Exit Function
    If a1 = b1 Then Exit Function
    If InStr(1, a1, b1, vbBinaryCompare) > 0 Then Exit Function
    If InStr(1, b1, a1, vbBinaryCompare) > 0 Then Exit Function
    SameSubjectFamily = False
End Function

'=============================================================================
' v6.3 - RefreshFolderNames : colonne "Nom du dossier" tenue a jour a chaque
' recalcul. Pour chaque ligne, le mail de reference (EntryID, repare juste
' avant par UpdateAllTrackingStatuses pour les lignes actives) est relu et
' le nom de son dossier ACTUEL ecrit s'il a change : couvre le renommage du
' dossier ET le deplacement du mail. Mail introuvable (ancrage perime d'une
' ligne close, mail supprime) : valeur connue conservee, jamais effacee.
' Lecture en bloc, ecriture des seules cellules modifiees, budget
' FOLDER_BUDGET_SEC secondes.
' v6.4 : le mail relu doit porter l'objet de la ligne (SameSubjectFamily).
' Sinon la ligne est DESALIGNEE (colonnes techniques d'une autre ligne) :
' aucun dossier ecrit, journal + toast invitant a lancer RepairTrackerRows.
'=============================================================================
Private Sub RefreshFolderNames(ws As Object)
    On Error GoTo ErrH
    Dim lastRow As Long
    lastRow = TrackLastRow(ws)
    If lastRow < 2 Then Exit Sub

    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)

    Dim olNS As Outlook.NameSpace
    Set olNS = Application.GetNamespace("MAPI")

    Dim t0 As Double
    t0 = Timer
    Dim r As Long, eid As String, fresh As String, cached As String
    Dim nChanged As Long, nMissing As Long, nMismatch As Long, nCached As Long
    For r = 1 To lastRow - 1
        eid = CStr(SafeVal(blk, r, COL_ENTRY_ID))
        If Len(eid) > 0 Then
            ' v6.7 (F4-06) : mail deja relu par RelinkRows pendant cette
            ' actualisation -> resultat repris, pas de seconde lecture.
            cached = ""
            If Not mEidFolder Is Nothing Then
                If mEidFolder.Exists(eid) Then cached = CStr(mEidFolder(eid))
            End If
            If Len(cached) > 0 Then
                nCached = nCached + 1
                If cached = FOLDER_MISSING Then
                    nMissing = nMissing + 1
                ElseIf cached = FOLDER_MISMATCH Then
                    nMismatch = nMismatch + 1
                ElseIf StrComp(cached, CStr(SafeVal(blk, r, COL_FOLDER)), vbBinaryCompare) <> 0 Then
                    ws.Cells(r + 1, COL_FOLDER).Value = cached
                    nChanged = nChanged + 1
                End If
            Else
                Dim itm As Object
                Set itm = Nothing
                On Error Resume Next
                Set itm = olNS.GetItemFromID(eid)
                On Error GoTo ErrH
                If itm Is Nothing Then
                    nMissing = nMissing + 1
                ElseIf Not SameSubjectFamily(CStr(SafeVal(blk, r, COL_SUBJECT)), ItemSubject(itm)) Then
                    ' v6.4 : le mail de cette ligne n'est pas le sien -> pas de
                    ' dossier faux ecrit.
                    nMismatch = nMismatch + 1
                Else
                    fresh = ItemFolderName(itm)
                    If Len(fresh) > 0 Then
                        If StrComp(fresh, CStr(SafeVal(blk, r, COL_FOLDER)), vbBinaryCompare) <> 0 Then
                            ws.Cells(r + 1, COL_FOLDER).Value = fresh
                            nChanged = nChanged + 1
                        End If
                    End If
                End If
            End If
        End If
        If (r Mod 25) = 0 Then
            DoEvents
            If ElapsedSec(t0) > FOLDER_BUDGET_SEC Then
                modDebugLog.DInfo MOD_NAME, "Noms de dossier : budget de " & FOLDER_BUDGET_SEC & " s atteint"
                Exit For
            End If
        End If
    Next r
    modDebugLog.DInfo MOD_NAME, "Noms de dossier : " & nChanged & " mis a jour, " & _
        nMissing & " mail(s) introuvable(s), " & nCached & " deja relu(s), en " & _
        Format$(ElapsedSec(t0), "0.0") & " s"
    If nMismatch > 0 Then
        modDebugLog.DError MOD_NAME, "Noms de dossier : " & nMismatch & _
            " ligne(s) dont le mail ne porte pas l'objet (desalignement) - lancer RepairTrackerRows"
        modNotify.Notify T("Suivi des mails : ", "Mail tracking: ") & nMismatch & _
            T(" ligne(s) d~esalign~ee(s) - lancez la macro RepairTrackerRows", _
              " misaligned row(s) - run the RepairTrackerRows macro")
    End If
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "RefreshFolderNames err " & Err.Number & " - " & Err.Description
End Sub

' v6.3 - Secondes ecoulees depuis un Timer de depart (passage de minuit gere).
Private Function ElapsedSec(ByVal since As Double) As Double
    ElapsedSec = Timer - since
    If ElapsedSec < 0 Then ElapsedSec = ElapsedSec + 86400#
End Function

'=============================================================================
' v6.3 - CheckConsistency : controle de coherence, APRES le tri (ses
' commentaires citent des numeros de ligne).
'   - ligne sans identifiant de mail (ni ConversationID ni EntryID) :
'     Ouvrir et Relancer y echouent ;
'   - doublon : meme ConversationID qu'une ligne precedente, ou meme objet
'     et meme date d'envoi (a la minute) ; la seconde occurrence est marquee.
' Rendu : objet surligne (orange clair) + commentaire de cellule qui dit
' pourquoi. Les marques posees ici (commentaire commencant par FLAG_PREFIX)
' sont effacees avant chaque controle. Nombres renvoyes pour un toast.
' v6.7 (F4-06) : effacement par la liste des commentaires de la feuille
' (ClearAllFlags) au lieu de 2 a 4 appels Excel par ligne.
'=============================================================================
Private Sub CheckConsistency(ws As Object, ByRef nNoId As Long, ByRef nDup As Long)
    On Error Resume Next
    nNoId = 0: nDup = 0
    Dim lastRow As Long
    lastRow = TrackLastRow(ws)
    If lastRow < 2 Then Exit Sub
    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)

    Dim dConv As Object, dSubj As Object
    Set dConv = CreateObject("Scripting.Dictionary")
    Set dSubj = CreateObject("Scripting.Dictionary")

    ClearAllFlags ws
    Dim r As Long, trackID As String, convID As String, eid As String, key As String
    Dim flagged As Boolean
    For r = 1 To lastRow - 1
        trackID = CStr(SafeVal(blk, r, COL_ID))
        If Len(trackID) > 0 Then
            flagged = False
            convID = CStr(SafeVal(blk, r, COL_CONV_ID))
            eid = CStr(SafeVal(blk, r, COL_ENTRY_ID))
            If Len(convID) = 0 And Len(eid) = 0 Then
                FlagRow ws, r + 1, T("Aucun identifiant de mail : Ouvrir et Relancer sont impossibles sur cette ligne.", _
                                     "No mail identifier: Open and Follow up cannot work on this row.")
                nNoId = nNoId + 1
                flagged = True
            End If
            If Len(convID) > 0 Then
                If dConv.Exists(convID) Then
                    If Not flagged Then
                        FlagRow ws, r + 1, T("Doublon probable de la ligne ", "Probable duplicate of row ") & _
                            dConv(convID) & T(" (m~xme conversation).", " (same conversation).")
                        nDup = nDup + 1
                        flagged = True
                    End If
                Else
                    dConv.Add convID, CStr(r + 1)
                End If
            End If
            key = NormalizeSubject(CStr(SafeVal(blk, r, COL_SUBJECT))) & "|" & DateKey(SafeVal(blk, r, COL_SENT_DATE))
            If Len(key) > 1 Then
                If dSubj.Exists(key) Then
                    If Not flagged Then
                        FlagRow ws, r + 1, T("Doublon probable de la ligne ", "Probable duplicate of row ") & _
                            dSubj(key) & T(" (m~xme objet, m~xme date d'envoi).", " (same subject, same sent date).")
                        nDup = nDup + 1
                        flagged = True
                    End If
                Else
                    dSubj.Add key, CStr(r + 1)
                End If
            End If
        End If
    Next r
    modDebugLog.DInfo MOD_NAME, "Controle de coherence : " & nNoId & " ligne(s) sans identifiant, " & nDup & " doublon(s)"
End Sub

' v6.7 (F4-06) - Efface les marques du controle (commentaire commencant par
' FLAG_PREFIX + surlignage) en parcourant les commentaires de la feuille,
' en general peu nombreux, au lieu de lire chaque ligne. Les cellules sont
' relevees d'abord, effacees ensuite (ne pas retirer un commentaire de la
' collection pendant son parcours).
Private Sub ClearAllFlags(ws As Object)
    On Error Resume Next
    Dim cms As Object, cm As Object, txt As String
    Dim targets As Collection
    Set targets = New Collection
    Set cms = Nothing
    Set cms = ws.Comments
    If cms Is Nothing Then Exit Sub
    For Each cm In cms
        txt = ""
        txt = CStr(cm.Text)
        If Left$(txt, Len(FLAG_PREFIX)) = FLAG_PREFIX Then targets.Add cm.Parent
    Next cm
    Dim c As Variant
    For Each c In targets
        c.ClearComments
        c.Interior.ColorIndex = -4142                     ' xlNone
    Next c
End Sub

' Objet normalise : minuscules, prefixes RE:/TR:/FW:/FWD:/AW:/WG: retires.
Private Function NormalizeSubject(ByVal s As String) As String
    Dim p As Variant, changed As Boolean, i As Long
    s = LCase$(Trim$(s))
    p = Array("re:", "tr:", "fw:", "fwd:", "aw:", "wg:", "re :", "tr :")
    Do
        changed = False
        For i = LBound(p) To UBound(p)
            If Left$(s, Len(p(i))) = p(i) Then
                s = Trim$(Mid$(s, Len(p(i)) + 1))
                changed = True
            End If
        Next i
    Loop While changed
    NormalizeSubject = s
End Function

' Cle de date a la minute ("" si pas une date).

    On Error Resume Next
    DateKey = ""
    If IsDate(v) Then DateKey = Format$(CDate(v), "yyyymmddhhnn")
End Function

'=============================================================================
' v6.6 - RelinkRows : RE-IDENTIFICATION des lignes dont le mail n'est plus
' joignable. Remplace RepairOrphans (v6.3), qui ne traitait que les lignes
' sans ConversationID, cherchait d'abord dans les Elements envoyes (or les
' mails suivis sont ranges ailleurs par QuickMove / ReplyInPlace), puis
' parcourait la boite MAIL PAR MAIL en 20 s maximum, et bloquait tout nouvel
' essai pendant 7 jours apres un echec.
'
' Lignes visees (feuilles transmises : Suivi, et Archives pour Reparer) :
' identifiant de fil vide, OU mail de reference introuvable (un mail deplace
' change d'identifiant), OU mail de reference dont l'objet ne correspond pas
' a la ligne - quel que soit le statut (Repondu ? et Repondu compris : c'est
' la que "Ouvrir le mail" echouait).
'
' Recherche : UN passage sur tous les dossiers de courrier de la boite par
' defaut, par tables Outlook filtrees sur la date d'envoi (fenetres de
' +/- 1 jour autour des dates des lignes visees, fusionnees) : seules les
' lignes de la table sont lues (objet, date d'envoi, expediteur), aucun mail
' n'est ouvert pendant le parcours. Correspondance : mail ENVOYE PAR SOI,
' meme objet normalise (RE:/TR:... retires, casse ignoree), date d'envoi a un
' jour pres ; le plus proche gagne (la date d'une ligne est en general la
' date d'envoi exacte : ecart de quelques secondes). Un meme mail ne sert
' qu'a une ligne. Trouve : ConversationID, EntryID et nom du dossier
' reecrits.
'
' force = True (bouton Reparer) : memoire des echecs ignoree, budget long,
' "(mail introuvable)" ecrit dans la colonne dossier des lignes non
' retrouvees. force = False (Actualiser) : budget court, une ligne non
' retrouvee n'est recherchee de nouveau qu'apres REPAIR_RETRY_DAYS jours.
'
' v6.7 (audit 11/2026, F4-05 / F10-11) :
'   - budget teste a CHAQUE dossier et juste apres chaque lecture de table
'     (avant : tous les 20 dossiers et toutes les 500 lignes : 64 s pour un
'     budget de 20 s) ; le dossier ou le budget est atteint est journalise ;
'   - dossiers ou le mail se trouve presque toujours lus d'abord : Elements
'     envoyes, puis les dossiers recents de QuickMove ;
'   - REPRISE : une recherche interrompue par le budget memorise le rang du
'     dernier dossier lu et les lignes cherchees (registre
'     MailResponseTracker\Relink) ; la suivante saute ces dossiers tant
'     qu'aucune NOUVELLE ligne n'est a chercher (des lignes retrouvees entre
'     temps ne l'empechent pas). Parcours complet : reprise oubliee ;
'   - F4-02 : "envoye par soi" = l'une de SES adresses (modSmtp) ;
'   - F4-10 : fil et mail d'une ligne ecrits en une seule affectation ;
'   - F4-06 : les mails relus pour le diagnostic remplissent le cache
'     EntryID -> dossier de RefreshFolderNames (une lecture par mail).
'=============================================================================
Private Sub RelinkRows(ByVal sheets As Collection, ByVal force As Boolean, ByVal budgetSec As Long, _
                       ByRef nFound As Long, ByRef nLost As Long, ByRef relinkAborted As Boolean)
    On Error GoTo ErrH
    nFound = 0
    nLost = 0
    relinkAborted = False
    Dim heavyOn As Boolean
    heavyOn = False
    Dim olNS As Outlook.NameSpace
    Set olNS = Application.GetNamespace("MAPI")

    ' --- 1. Lignes a re-identifier : Array(feuille, ligne, objet normalise, date, ID)
    Dim cand As Collection
    Set cand = New Collection
    Dim healthy As Object                    ' EntryID des mails deja rattaches a une ligne saine
    Set healthy = CreateObject("Scripting.Dictionary")
    Dim t0 As Double
    t0 = Timer
    Dim sh As Variant, ws As Object, lastRow As Long, blk As Variant, r As Long
    Dim trackID As String, eid As String, subjRaw As String, subjN As String
    Dim sd As Variant, v As String, itm As Object, needs As Boolean
    For Each sh In sheets
        Set ws = sh
        lastRow = TrackLastRow(ws)
        If lastRow >= 2 Then
            blk = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)
            For r = 1 To lastRow - 1
                trackID = CStr(SafeVal(blk, r, COL_ID))
                subjRaw = CStr(SafeVal(blk, r, COL_SUBJECT))
                subjN = NormalizeSubject(subjRaw)
                sd = SafeVal(blk, r, COL_SENT_DATE)
                If Len(trackID) > 0 And Len(subjN) > 0 And IsDate(sd) Then
                    ' Actualiser : une ligne deja cherchee sans succes recemment
                    ' est sautee AVANT tout acces Outlook.
                    Dim skipLine As Boolean
                    skipLine = False
                    If Not force Then
                        v = GetSetting(REG_TRACK_APP, REG_REPAIR_SECTION, "orphan:" & trackID, "")
                        If Len(v) > 0 Then
                            If CDbl(Date) - Val(v) < REPAIR_RETRY_DAYS Then skipLine = True
                        End If
                    End If
                    If Not skipLine Then
                        needs = (Len(CStr(SafeVal(blk, r, COL_CONV_ID))) = 0)
                        If Not needs Then
                            eid = CStr(SafeVal(blk, r, COL_ENTRY_ID))
                            Set itm = Nothing
                            If Len(eid) > 0 Then
                                On Error Resume Next
                                Set itm = olNS.GetItemFromID(eid)
                                Err.Clear
                                On Error GoTo ErrH
                            End If
                            If itm Is Nothing Then
                                needs = True
                                CacheFolder eid, FOLDER_MISSING
                            ElseIf Not SameSubjectFamily(subjRaw, ItemSubject(itm)) Then
                                needs = True
                                CacheFolder eid, FOLDER_MISMATCH
                            Else
                                healthy(CStr(itm.entryID)) = True
                                CacheFolder eid, ItemFolderName(itm)
                            End If
                        End If
                        If needs Then cand.Add Array(ws, r + 1, subjN, CDate(sd), trackID)
                    End If
                End If
            Next r
        End If
    Next sh
    If cand.count = 0 Then
        ClearRelinkCursor                    ' plus rien a chercher : reprise sans objet
        Exit Sub
    End If
    If ElapsedSec(t0) > budgetSec Then
        modDebugLog.DInfo MOD_NAME, "Re-identification : budget epuise avant la recherche (" & cand.count & " ligne(s) en attente)"
        Exit Sub
    End If

    ' --- 2. Index objet normalise -> "i1;i2", filtre par dates, reprise
    Dim dSubj As Object
    Set dSubj = CreateObject("Scripting.Dictionary")
    Dim i As Long, c As Variant, key As String, idsNow As String
    idsNow = ";"
    For i = 1 To cand.count
        c = cand(i)
        key = CStr(c(2))
        If dSubj.Exists(key) Then
            dSubj(key) = dSubj(key) & ";" & i
        Else
            dSubj.Add key, CStr(i)
        End If
        idsNow = idsNow & CStr(c(4)) & ";"
    Next i
    Dim flt As String
    flt = RelinkDateFilter(cand)
    Dim resumeFrom As Long
    resumeFrom = RelinkResumePoint(idsNow)
    modDebugLog.DInfo MOD_NAME, "Re-identification : " & cand.count & " ligne(s) a rechercher" & _
        IIf(force, " (reparation)", "") & IIf(resumeFrom > 0, " - reprise apres le dossier " & resumeFrom, "")

    ' --- 3. Parcours : dossiers probables d'abord, puis toute la boite
    Dim myName As String, deletedId As String
    myName = ""
    deletedId = ""
    On Error Resume Next
    myName = CStr(olNS.CurrentUser.Name)
    deletedId = CStr(olNS.GetDefaultFolder(olFolderDeletedItems).entryID)
    Err.Clear
    On Error GoTo ErrH

    Dim pairs As Collection                  ' Array(indice ligne, EntryID, ecart)
    Set pairs = New Collection
    Dim nFolders As Long, aborted As Boolean, budgetHit As Boolean
    Dim nTables As Long, nTableErr As Long, walkIdx As Long, lastDone As Long
    Dim doneIds As Object                    ' EntryID des dossiers deja lus en priorite
    Set doneIds = CreateObject("Scripting.Dictionary")
    aborted = False
    budgetHit = False
    walkIdx = 0
    lastDone = 0
    modDefer.BeginHeavy
    heavyOn = True

    ' 3a. Elements envoyes, puis dossiers recents de QuickMove.
    Dim pri As Collection, pf As Variant, pfId As String
    Set pri = PriorityFolders(olNS, deletedId)
    For Each pf In pri
        If aborted Then Exit For
        pfId = ""
        On Error Resume Next
        pfId = CStr(pf.entryID)
        On Error GoTo ErrH
        If Len(pfId) > 0 Then doneIds(pfId) = True
        nFolders = nFolders + 1
        RelinkScanOne pf, flt, dSubj, cand, myName, pairs, t0, budgetSec, nTables, nTableErr, aborted, budgetHit
    Next pf

    ' 3b. Toute la boite par defaut (dossiers de 3a et, en reprise, dossiers
    ' deja lus par la recherche interrompue : sautes).
    Dim root As Object
    Set root = Nothing
    On Error Resume Next
    Set root = olNS.DefaultStore.GetRootFolder
    On Error GoTo ErrH
    If Not root Is Nothing And Not aborted Then
        RelinkScanFolder root, True, deletedId, flt, dSubj, cand, myName, pairs, t0, budgetSec, _
                         doneIds, resumeFrom, walkIdx, lastDone, nFolders, nTables, nTableErr, aborted, budgetHit
    End If
    ' Aucune table lisible : recherche impossible (filtre refuse...), pas un
    ' "introuvable".
    If nTables = 0 Then aborted = True
    modDefer.EndHeavy
    heavyOn = False

    ' Reprise : memorisee si le budget a coupe le parcours, oubliee si le
    ' parcours est alle au bout.
    If budgetHit Then
        If lastDone < resumeFrom Then lastDone = resumeFrom
        SaveRelinkCursor lastDone, idsNow
    ElseIf Not aborted Then
        ClearRelinkCursor
    End If

    ' --- 4. Attribution : ecart le plus faible d'abord, un mail par ligne
    Dim n As Long, ordIdx() As Long, gaps() As Double, j As Long, tmpL As Long
    n = pairs.count
    Dim usedRow As Object, usedMail As Object
    Set usedRow = CreateObject("Scripting.Dictionary")
    Set usedMail = CreateObject("Scripting.Dictionary")
    Dim hk As Variant
    For Each hk In healthy.keys
        usedMail(CStr(hk)) = True                ' jamais pris a une ligne saine
    Next hk
    If n > 0 Then
        ReDim ordIdx(1 To n)
        ReDim gaps(1 To n)
        For i = 1 To n
            ordIdx(i) = i
            c = pairs(i)
            gaps(i) = CDbl(c(2))
        Next i
        For i = 2 To n
            tmpL = ordIdx(i)
            j = i - 1
            Do While j >= 1
                If gaps(ordIdx(j)) <= gaps(tmpL) Then Exit Do
                ordIdx(j + 1) = ordIdx(j)
                j = j - 1
            Loop
            ordIdx(j + 1) = tmpL
        Next i
        For i = 1 To n
            c = pairs(ordIdx(i))
            If Not usedRow.Exists(CStr(c(0))) And Not usedMail.Exists(CStr(c(1))) Then
                usedRow.Add CStr(c(0)), CStr(c(1))
                usedMail(CStr(c(1))) = True
            End If
        Next i
    End If

    ' --- 5. Ecriture. v6.7 (F4-10) : fil et mail (colonnes CONV_ID et
    ' ENTRY_ID, contigues) en UNE affectation : jamais une ligne avec le fil
    ' d'un mail et l'identifiant d'un autre.
    Dim rowRef As Variant, wsT As Object, rr As Long, found As Object, fname As String
    Dim ce(1 To 1, 1 To 2) As Variant
    For i = 1 To cand.count
        rowRef = cand(i)
        Set wsT = rowRef(0)
        rr = CLng(rowRef(1))
        trackID = CStr(rowRef(4))
        Set found = Nothing
        If usedRow.Exists(CStr(i)) Then
            On Error Resume Next
            Set found = olNS.GetItemFromID(CStr(usedRow(CStr(i))))
            Err.Clear
            On Error GoTo ErrH
        End If
        If Not found Is Nothing Then
            ce(1, 1) = CStr(found.ConversationID)
            ce(1, 2) = CStr(found.entryID)
            wsT.Range(wsT.Cells(rr, COL_CONV_ID), wsT.Cells(rr, COL_ENTRY_ID)).Value = ce
            fname = ItemFolderName(found)
            wsT.Cells(rr, COL_FOLDER).Value = fname
            CacheFolder CStr(found.entryID), fname
            DeleteOrphanMemory trackID
            nFound = nFound + 1
        Else
            nLost = nLost + 1
            ' Recherche interrompue (budget, tables illisibles) : la ligne n'est
            ' pas declaree introuvable, elle sera recherchee de nouveau.
            If Not aborted Then
                SaveSetting REG_TRACK_APP, REG_REPAIR_SECTION, "orphan:" & trackID, Str$(CLng(Date))
                If force Then
                    If Len(Trim$(CStr(wsT.Cells(rr, COL_FOLDER).Value))) = 0 Then
                        wsT.Cells(rr, COL_FOLDER).Value = T("(mail introuvable)", "(mail not found)")
                    End If
                End If
            End If
        End If
    Next i

    modDebugLog.DInfo MOD_NAME, "Re-identification : " & nFound & " retrouvee(s), " & nLost & _
        " introuvable(s), " & nFolders & " dossier(s) lu(s) sur " & walkIdx & " parcouru(s) (" & _
        nTableErr & " table(s) illisible(s)) en " & Format$(ElapsedSec(t0), "0.0") & " s" & _
        IIf(aborted, " - RECHERCHE INCOMPLETE (budget " & budgetSec & " s)", "") & _
        IIf(budgetHit, " - reprise memorisee apres le dossier " & lastDone, "")
    relinkAborted = aborted
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "RelinkRows err " & Err.Number & " - " & Err.Description
    relinkAborted = True
    On Error Resume Next
    If heavyOn Then modDefer.EndHeavy
End Sub

' v6.7 (F4-06) - Note le dossier d'un mail relu (cache de l'actualisation
' en cours ; sans effet hors actualisation).
Private Sub CacheFolder(ByVal eid As String, ByVal folderName As String)
    On Error Resume Next
    If mEidFolder Is Nothing Then Exit Sub
    If Len(eid) = 0 Or Len(folderName) = 0 Then Exit Sub
    mEidFolder(eid) = folderName
End Sub

'=============================================================================
' v6.7 (F4-05) - Reprise de la re-identification (registre HKCU
' MailResponseTracker\Relink : Cursor = rang du dernier dossier lu dans
' l'ordre du parcours, Ids = ";ID1;ID2;" des lignes alors cherchees).
' RelinkResumePoint : rang a partir duquel reprendre, 0 = depuis le debut
' (aucune reprise, ou une ligne nouvelle a chercher).
'=============================================================================
Private Function RelinkResumePoint(ByVal idsNow As String) As Long
    On Error Resume Next
    RelinkResumePoint = 0
    Dim saved As String, cur As Long
    saved = GetSetting(REG_TRACK_APP, REG_RELINK_SECTION, "Ids", "")
    cur = CLng(Val(GetSetting(REG_TRACK_APP, REG_RELINK_SECTION, "Cursor", "0")))
    If Len(saved) = 0 Or cur <= 0 Then Exit Function
    Dim parts() As String, k As Long
    parts = Split(idsNow, ";")
    For k = LBound(parts) To UBound(parts)
        If Len(parts(k)) > 0 Then
            If InStr(1, saved, ";" & parts(k) & ";", vbBinaryCompare) = 0 Then Exit Function
        End If
    Next k
    RelinkResumePoint = cur
End Function

Private Sub SaveRelinkCursor(ByVal cursor As Long, ByVal idsNow As String)
    On Error Resume Next
    SaveSetting REG_TRACK_APP, REG_RELINK_SECTION, "Cursor", CStr(cursor)
    SaveSetting REG_TRACK_APP, REG_RELINK_SECTION, "Ids", idsNow
End Sub

Private Sub ClearRelinkCursor()
    On Error Resume Next
    DeleteSetting REG_TRACK_APP, REG_RELINK_SECTION
End Sub

'=============================================================================
' v6.7 (F4-05) - Dossiers ou un mail envoye suivi se trouve presque
' toujours : Elements envoyes, puis les dossiers recents de QuickMove
' (index charge seulement, boite par defaut seulement, Elements supprimes
' exclus). Lus avant le parcours complet. Jamais d'erreur.
'=============================================================================
Private Function PriorityFolders(ByVal olNS As Outlook.NameSpace, ByVal deletedId As String) As Collection
    Dim res As Collection
    Set res = New Collection
    Set PriorityFolders = res
    On Error Resume Next
    Dim seen As Object
    Set seen = CreateObject("Scripting.Dictionary")
    Dim f As Object, fid As String, defStore As String
    defStore = ""
    defStore = CStr(olNS.DefaultStore.StoreID)

    Set f = Nothing
    Set f = olNS.GetDefaultFolder(olFolderSentMail)
    If Not f Is Nothing Then
        fid = ""
        fid = CStr(f.entryID)
        If Len(fid) > 0 Then
            seen(fid) = True
            res.Add f
        End If
    End If

    If Not modQuickMove.IsIndexLoaded() Then Exit Function
    Dim i As Long, idx As Long, nRecent As Long, fStore As String
    nRecent = 0
    nRecent = modQuickMove.mRecentCount
    For i = 1 To nRecent
        idx = 0
        idx = modQuickMove.mRecentFolders(i)
        Set f = Nothing
        If idx > 0 Then Set f = modQuickMove.IndexFolderObject(idx)
        If Not f Is Nothing Then
            fid = ""
            fid = CStr(f.entryID)
            fStore = ""
            fStore = CStr(f.StoreID)
            If Len(fid) > 0 And fid <> deletedId And fStore = defStore Then
                If Not seen.Exists(fid) Then
                    seen(fid) = True
                    res.Add f
                End If
            End If
        End If
    Next i
End Function

' v6.6 - Filtre Jet sur la date d'envoi : fenetres de +/- 1 jour autour des
' dates des lignes visees, triees et fusionnees ; au-dela de 30 fenetres,
' une seule plage du plus ancien au plus recent.
Private Function RelinkDateFilter(ByVal cand As Collection) As String
    Dim n As Long, i As Long, j As Long, tmp As Double, c As Variant
    n = cand.count
    Dim ds() As Double
    ReDim ds(1 To n)
    For i = 1 To n
        c = cand(i)
        ds(i) = CDbl(CDate(c(3)))
    Next i
    For i = 2 To n
        tmp = ds(i)
        j = i - 1
        Do While j >= 1
            If ds(j) <= tmp Then Exit Do
            ds(j + 1) = ds(j)
            j = j - 1
        Loop
        ds(j + 1) = tmp
    Next i

    Dim parts As Collection, lo As Double, hi As Double
    Set parts = New Collection
    lo = ds(1) - 1#
    hi = ds(1) + 1#
    For i = 2 To n
        If ds(i) - 1# <= hi Then
            If ds(i) + 1# > hi Then hi = ds(i) + 1#
        Else
            parts.Add Array(lo, hi)
            lo = ds(i) - 1#
            hi = ds(i) + 1#
        End If
    Next i
    parts.Add Array(lo, hi)

    Dim s As String, p As Variant
    If parts.count > 30 Then
        RelinkDateFilter = "[SentOn] >= '" & modCommon.FormatDaslDate(CDate(ds(1) - 1#)) & _
                           "' AND [SentOn] <= '" & modCommon.FormatDaslDate(CDate(ds(n) + 1#)) & "'"
        Exit Function
    End If
    For Each p In parts
        If Len(s) > 0 Then s = s & " OR "
        s = s & "([SentOn] >= '" & modCommon.FormatDaslDate(CDate(p(0))) & _
                "' AND [SentOn] <= '" & modCommon.FormatDaslDate(CDate(p(1))) & "')"
    Next p
    RelinkDateFilter = s
End Function


'=============================================================================
' v6.6 - Un dossier : table filtree sur les dates, puis sous-dossiers. Les
' dossiers qui ne sont pas du courrier sont elagues avec leur descendance
' (la racine de la boite est toujours parcourue).
' v6.7 (F4-05) : budget teste a CHAQUE dossier (DoEvents tous les 20) ;
' walkIdx = rang du dossier dans le parcours ; table non lue pour un
' dossier deja lu en priorite (doneIds) ou deja lu par la recherche
' interrompue precedente (rang <= resumeFrom) ; lastDone = rang du dernier
' dossier traite jusqu'au bout.
'=============================================================================
Private Sub RelinkScanFolder(ByVal fld As Object, ByVal isRoot As Boolean, ByVal deletedId As String, _
                             ByVal flt As String, ByVal dSubj As Object, _
                             ByVal cand As Collection, ByVal myName As String, _
                             ByVal pairs As Collection, ByVal t0 As Double, ByVal budgetSec As Long, _
                             ByVal doneIds As Object, ByVal resumeFrom As Long, _
                             ByRef walkIdx As Long, ByRef lastDone As Long, _
                             ByRef nFolders As Long, ByRef nTables As Long, ByRef nTableErr As Long, _
                             ByRef aborted As Boolean, ByRef budgetHit As Boolean)
    On Error Resume Next
    If aborted Then Exit Sub
    Dim itemType As Long
    itemType = -1
    itemType = CLng(fld.DefaultItemType)
    If itemType <> olMailItem And Not isRoot Then Exit Sub
    ' Elements supprimes (et sa descendance) : une copie y disparait a la
    ' prochaine purge.
    Dim fid As String
    fid = ""
    fid = CStr(fld.entryID)
    If Len(deletedId) > 0 Then
        If fid = deletedId Then Exit Sub
    End If
    Err.Clear

    walkIdx = walkIdx + 1
    If (walkIdx Mod 20) = 0 Then DoEvents
    If ElapsedSec(t0) > budgetSec Then
        aborted = True
        budgetHit = True
        Exit Sub
    End If

    Dim readIt As Boolean
    readIt = (walkIdx > resumeFrom)
    If readIt And Len(fid) > 0 Then
        If doneIds.Exists(fid) Then readIt = False
    End If
    If readIt Then
        nFolders = nFolders + 1
        If Not RelinkScanOne(fld, flt, dSubj, cand, myName, pairs, t0, budgetSec, _
                             nTables, nTableErr, aborted, budgetHit) Then Exit Sub
    End If
    lastDone = walkIdx
    If aborted Then Exit Sub
    Err.Clear

    Dim sub_ As Object
    For Each sub_ In fld.Folders
        If aborted Then Exit Sub
        RelinkScanFolder sub_, False, deletedId, flt, dSubj, cand, myName, pairs, t0, budgetSec, _
                         doneIds, resumeFrom, walkIdx, lastDone, nFolders, nTables, nTableErr, aborted, budgetHit
    Next sub_
End Sub

'=============================================================================
' v6.7 (F4-05, F10-11) - Lit la table filtree d'UN dossier et note les mails
' candidats. True si le dossier a ete lu jusqu'au bout (ou sa table
' refusee), False si la lecture a ete coupee par le budget en cours de
' table. Budget re-teste juste apres GetTable : depasse, la table deja
' obtenue est quand meme lue (sinon la reprise relirait toujours ce meme
' dossier lent), puis la recherche s'arrete ; le dossier et la duree de la
' table sont journalises.
'=============================================================================
Private Function RelinkScanOne(ByVal fld As Object, ByVal flt As String, ByVal dSubj As Object, _
                               ByVal cand As Collection, ByVal myName As String, _
                               ByVal pairs As Collection, ByVal t0 As Double, ByVal budgetSec As Long, _
                               ByRef nTables As Long, ByRef nTableErr As Long, _
                               ByRef aborted As Boolean, ByRef budgetHit As Boolean) As Boolean
    RelinkScanOne = True
    On Error Resume Next
    Dim tbl As Object, tq As Double, overAfterGet As Boolean
    Set tbl = Nothing
    Err.Clear
    tq = Timer
    Set tbl = fld.GetTable(flt)
    If Err.Number <> 0 Or tbl Is Nothing Then
        nTableErr = nTableErr + 1
        If nTableErr = 1 Then
            modDebugLog.DError MOD_NAME, "Re-identification : table refusee (" & Err.Number & " - " & _
                Err.Description & ") dossier " & CStr(fld.FolderPath) & " filtre " & Left$(flt, 200)
        End If
        Err.Clear
        Exit Function
    End If
    nTables = nTables + 1
    overAfterGet = (ElapsedSec(t0) > budgetSec)
    If overAfterGet Then
        modDebugLog.DInfo MOD_NAME, "Re-identification : budget atteint au dossier " & CStr(fld.FolderPath) & _
            " (table obtenue en " & Format$(ElapsedSec(tq), "0.0") & " s) ; table lue puis arret"
    End If

    Err.Clear
    tbl.Columns.Add "SentOn"
    tbl.Columns.Add "SenderName"
    tbl.Columns.Add PR_SENDER_SMTP
    tbl.Columns.Add PR_SENT_REP_SMTP
    Err.Clear
    Dim rw As Object, key As String, sent As Date, idxs() As String, k As Long
    Dim ci As Long, c As Variant, gap As Double, guard As Long
    guard = 0
    Do Until tbl.EndOfTable
        guard = guard + 1
        If (guard Mod 500) = 0 Then
            DoEvents
            If Not overAfterGet Then
                If ElapsedSec(t0) > budgetSec Then
                    aborted = True
                    budgetHit = True
                    RelinkScanOne = False
                    Exit Function
                End If
            End If
        End If
        If guard > 50000 Then
            modDebugLog.DError MOD_NAME, "Re-identification : dossier trop volumineux, lecture arretee : " & _
                CStr(fld.FolderPath)
            aborted = True
            RelinkScanOne = False
            Exit Function
        End If
        Set rw = Nothing
        Set rw = tbl.GetNextRow
        If rw Is Nothing Then Exit Do
        key = ""
        key = NormalizeSubject(CStr(rw("Subject")))
        If Len(key) > 0 Then
            If dSubj.Exists(key) Then
                If IsPlainMail(rw) And RowIsFromMe(rw, myName) Then
                    sent = 0
                    sent = CDate(rw("SentOn"))
                    If sent > 0 Then
                        idxs = Split(CStr(dSubj(key)), ";")
                        For k = LBound(idxs) To UBound(idxs)
                            ci = CLng(Val(idxs(k)))
                            c = cand(ci)
                            gap = Abs(CDbl(sent) - CDbl(CDate(c(3))))
                            If gap <= 1# Then pairs.Add Array(ci, CStr(rw("EntryID")), gap)
                        Next k
                    End If
                End If
            End If
        End If
        Err.Clear
    Loop
    Err.Clear
    If overAfterGet Then
        aborted = True
        budgetHit = True
    End If
End Function

' v6.6 - Ligne de table = un mail (pas une invitation, un accuse...).
' v6.7 : meme regle que modSmtp.MessageClassKind (reponses automatiques
' et rapports exclus).
Private Function IsPlainMail(ByVal rw As Object) As Boolean
    On Error Resume Next
    Dim mc As String
    mc = ""
    mc = CStr(rw("MessageClass"))
    IsPlainMail = (modSmtp.MessageClassKind(mc) = modSmtp.MSG_KIND_NORMAL)
End Function

' v6.6 - Ligne de table envoyee par soi ? v6.7 (F4-02) : adresse SMTP de
' l'expediteur (ou du "de la part de") parmi TOUTES ses adresses
' (modSmtp.IsMyAddress : comptes du profil, alias Exchange) ; le nom n'est
' compare que si la ligne n'a aucune adresse lisible ou si l'ensemble des
' adresses est vide.
Private Function RowIsFromMe(ByVal rw As Object, ByVal myName As String) As Boolean
    On Error Resume Next
    RowIsFromMe = False
    Dim a1 As String, a2 As String, nm As String, nMine As Long
    a1 = ""
    a2 = ""
    a1 = LCase$(Trim$(CStr(rw(PR_SENDER_SMTP))))
    a2 = LCase$(Trim$(CStr(rw(PR_SENT_REP_SMTP))))
    If Len(a1) > 0 Then
        If modSmtp.IsMyAddress(a1) Then
            RowIsFromMe = True
            Exit Function
        End If
    End If
    If Len(a2) > 0 Then
        If modSmtp.IsMyAddress(a2) Then
            RowIsFromMe = True
            Exit Function
        End If
    End If
    nMine = 0
    nMine = modSmtp.MyAddresses().count
    If (nMine = 0 Or (Len(a1) = 0 And Len(a2) = 0)) And Len(myName) > 0 Then
        nm = ""
        nm = CStr(rw("SenderName"))
        RowIsFromMe = (StrComp(nm, myName, vbTextCompare) = 0)
    End If
End Function

'=============================================================================
' v6.3 - ACTUALISATION DEMANDEE PAR LE CLASSEUR (bouton Actualiser du ruban
' Excel). RequestDashboardRefresh (appele par
' ThisOutlookSession.RefreshMailTracking, donc depuis Excel) memorise le
' classeur transmis et DIFFERE le travail (modDefer "DashRefresh") : Excel
' reprend la main tout de suite, Outlook n'ecrit dans le classeur qu'une
' fois l'appel d'Excel termine (sinon appel imbrique, refus possible de COM).
' RefreshDashboard fait le meme travail que le tableau de bord sans afficher
' Excel (il l'est deja), puis rappelle le classeur (Application.Run
' "AfterOutlookRefresh") pour ses compteurs. Sans classeur transmis :
' chemin habituel (GetExcelApp / GetTrackerWorkbook), classeur libere apres.
' v6.4 : sans classeur transmis (repli du bouton Actualiser, modMailTracker
' v1.1), si le classeur trouve est celui de l'utilisateur (deja ouvert et
' visible), il est rappele aussi : ses compteurs de ruban suivent.
' v6.7 (F10-11, F4-19) : une demande recue alors qu'une actualisation ou
' une reparation est deja en file ou en cours est IGNOREE (avant : mise en
' file et rejouee - trois "Reparation demandee" en 43 s le 09/10, deux
' traitements imbriques) ; Reparer repete dans les 30 s qui suivent la fin
' d'une reparation aussi. Une demande venue par la note de commande est
' acquittee dans la note (AckCommand) : le classeur sait qu'Outlook l'a
' recue, ou qu'une actualisation tournait deja.
'=============================================================================
Public Sub RequestDashboardRefresh(ByVal wb As Object)
    On Error Resume Next
    Dim viaNote As Boolean
    viaNote = (wb Is Nothing)
    If RefreshPendingOrRunning() Then
        modDebugLog.DInfo MOD_NAME, "Actualisation demandee par le classeur : ignoree (deja en file ou en cours)"
        If viaNote Then
            AckCommand "REFRESH", "BUSY"
        Else
            modUi.Toast T("Actualisation du suivi d~ej~a en cours.", "Tracking refresh already in progress."), uiInfo
        End If
        Exit Sub
    End If
    Set mRefreshWb = wb
    mLightRequested = False              ' une actualisation complete remplace le recalcul leger
    mRefreshQueuedMs = modCommon.TickMs()
    modDefer.ArmUnique "DashRefresh", 1
    If viaNote Then AckCommand "REFRESH", "QUEUED"
    modDebugLog.DInfo MOD_NAME, "Actualisation demandee par le classeur"
End Sub

' v6.5 - Reparation demandee par le classeur (bouton Reparer, canal de
' commandes). Meme differe que l'actualisation ; RefreshDashboard voit le
' drapeau et lance la reparation avant le recalcul.
' v6.7 : une actualisation seulement EN FILE devient une reparation (qui
' recalcule aussi) ; en cours, ou reparation finie depuis moins de 30 s :
' demande ignoree.
Public Sub RequestTrackerRepair()
    On Error Resume Next
    Dim ignoreIt As Boolean
    ignoreIt = False
    If mRepairRequested Then ignoreIt = True
    If TrackerBusy() Then
        If mWorkKind = "refresh" Or mWorkKind = "repair" Or mWorkKind = "dashboard" Then ignoreIt = True
    End If
    If mLastRepairEndMs > 0 Then
        If modCommon.ElapsedMs(mLastRepairEndMs) < REPAIR_REPEAT_MS Then ignoreIt = True
    End If
    If ignoreIt Then
        modDebugLog.DInfo MOD_NAME, "Reparation demandee par le classeur : ignoree (deja en file, en cours ou tout juste faite)"
        AckCommand "REPAIR", "BUSY"
        Exit Sub
    End If
    mRepairRequested = True
    mLightRequested = False
    Set mRefreshWb = Nothing
    mRefreshQueuedMs = modCommon.TickMs()
    modDefer.ArmUnique "DashRefresh", 1
    AckCommand "REPAIR", "QUEUED"
    modDebugLog.DInfo MOD_NAME, "Reparation demandee par le classeur"
End Sub

' v6.7 (F10-11) - Une actualisation ou une reparation est-elle deja en file
' (DashRefresh arme depuis moins de 10 min) ou en cours ?
Private Function RefreshPendingOrRunning() As Boolean
    RefreshPendingOrRunning = False
    If mRefreshQueuedMs > 0 Then
        If modCommon.ElapsedMs(mRefreshQueuedMs) < BUSY_MAX_MS Then
            RefreshPendingOrRunning = True
            Exit Function
        End If
        mRefreshQueuedMs = 0               ' file perimee (minuterie perdue)
    End If
    If TrackerBusy() Then
        If mWorkKind = "refresh" Or mWorkKind = "repair" Or mWorkKind = "dashboard" Then
            RefreshPendingOrRunning = True
        End If
    End If
End Function

'=============================================================================
' v6.7 (F4-19) - Accuse de reception d'une commande du classeur : la note
' de commande (Notes\ClassicOutlook-Toolkit) qui porte encore
' "TOOLKIT|<cmd>|..." est reecrite en "ACK|<horodatage>|<cmd>|<etat>"
' (QUEUED = prise en compte ; BUSY = ignoree, un traitement est deja en
' file ou en cours). Le classeur relit la note quelques secondes apres
' l'envoi ; sans accuse, il previent que la demande n'a pas ete recue
' (Outlook n'ecoute plus le dossier : Reinit). La reecriture redeclenche
' ItemChange, que ThisOutlookSession ignore (la note ne commence plus par
' TOOLKIT). Jamais d'erreur.
'=============================================================================
Private Sub AckCommand(ByVal cmd As String, ByVal state As String)
    On Error Resume Next
    Dim fld As Object, itm As Object, body As String
    Set fld = Nothing
    Set fld = Application.Session.GetDefaultFolder(olFolderNotes).Folders(CMD_FOLDER_NAME)
    If fld Is Nothing Then Exit Sub
    Set itm = Nothing
    Set itm = fld.items.GetFirst
    If itm Is Nothing Then Exit Sub
    body = ""
    body = CStr(itm.body)
    If InStr(1, body, "TOOLKIT|" & cmd & "|", vbTextCompare) <> 1 Then Exit Sub
    Err.Clear
    itm.body = "ACK|" & Format$(Now, "yyyymmddhhnnss") & "|" & cmd & "|" & state
    itm.Save
    If Err.Number <> 0 Then
        modDebugLog.DDebug MOD_NAME, "Accuse de reception non ecrit (err " & Err.Number & ")"
    End If
End Sub

Public Sub RefreshDashboard()
    ' v6.7 : jamais a l'interieur d'un autre traitement du suivi (DoEvents,
    ' appel COM qui pompe les messages) : nouvel essai dans quelques s.
    If TrackerBusy() Then
        modDefer.ArmUnique "DashRefresh", BUSY_RETRY_SEC
        Exit Sub
    End If
    mRefreshQueuedMs = 0
    ' v6.5 : reparation demandee par le classeur -> elle passe d'abord ; elle
    ' rappelle RefreshDashboard une fois les lignes remises en place.
    If mRepairRequested Then
        mRepairRequested = False
        mLightRequested = False
        RunTrackerRepair True
        Exit Sub
    End If
    ' v6.7 (F4-08) : recalcul leger du jour, programme par RemindOverdue.
    If mLightRequested Then
        LightRefresh
        Exit Sub
    End If
    On Error GoTo ErrH
    BeginTrackerWork "refresh"
    Dim xlApp As Object, wb As Object, ws As Object
    Dim viaExcel As Boolean, userWb As Boolean
    Dim evSaved As Boolean, suSaved As Boolean, quietOn As Boolean
    viaExcel = False
    userWb = False
    quietOn = False
    Set wb = mRefreshWb
    Set mRefreshWb = Nothing
    If Not wb Is Nothing Then
        On Error Resume Next
        Set xlApp = wb.Application
        On Error GoTo ErrH
        If Not xlApp Is Nothing Then viaExcel = True
    End If

    If viaExcel Then
        PrepareWorkbook wb
        If Not LayoutReady(wb) Then GoTo Bail
    Else
        Set xlApp = GetExcelApp()
        If xlApp Is Nothing Then GoTo Bail
        xlApp.DisplayAlerts = False
        Set wb = GetTrackerWorkbook(xlApp, True)
        If wb Is Nothing Then GoTo Bail
        userWb = Not mWbOpenedHere          ' v6.4 : classeur deja ouvert chez l'utilisateur
    End If
    Set ws = wb.Sheets(modConfig.MailTrackingSheet())
    XlQuiet xlApp, evSaved, suSaved         ' v6.7 (F4-06)
    quietOn = True

    modCommon.SetStatus MOD_NAME, T("Mise ~a jour des suivis...", "Updating tracking...")
    StartFolderCache
    UpdateAllTrackingStatuses ws
    RefreshFolderNames ws
    StopFolderCache
    SortDashboard ws
    Dim nNoId As Long, nDup As Long
    CheckConsistency ws, nNoId, nDup
    ApplyDashboardLayout ws
    WriteDueCache ws
    SaveWorkbookVisible xlApp, wb
    XlRestore xlApp, evSaved, suSaved
    quietOn = False

    If viaExcel Or userWb Then NotifyExcelRefreshDone xlApp, wb
    If Not viaExcel Then ReleaseBackground xlApp, wb
    On Error Resume Next
    xlApp.DisplayAlerts = True
    EndTrackerWork
    modCommon.SetStatus MOD_NAME, T("Suivis actualis~es.", "Tracking updated.")
    NotifyConsistency nNoId, nDup
    MorningAfterRefresh
    modDebugLog.DInfo MOD_NAME, "RefreshDashboard OK"
    Exit Sub
Bail:
    ' Classeur indisponible ou gabarit inconnu (deja signale si
    ' mTrackerFailReported).
    EndTrackerWork
    modDebugLog.DError MOD_NAME, "RefreshDashboard : classeur de suivi indisponible"
    If Not mTrackerFailReported Then
        modNotify.Notify T("Actualisation du suivi impossible (voir debug.log)", "Tracking refresh failed (see debug.log)")
    End If
    FinishMorning
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "RefreshDashboard err " & Err.Number & " - " & Err.Description
    On Error Resume Next
    If quietOn Then XlRestore xlApp, evSaved, suSaved
    StopFolderCache
    If Not viaExcel Then
        If Not wb Is Nothing Then ReleaseBackground xlApp, wb
    End If
    EndTrackerWork
    modNotify.Notify T("Actualisation du suivi impossible (voir debug.log)", "Tracking refresh failed (see debug.log)")
    FinishMorning
End Sub

' Rappel du classeur : compteurs du ruban, barre d'etat (macro publique

    On Error Resume Next
    xlApp.Run "'" & wb.name & "'!AfterOutlookRefresh"
End Sub

'=============================================================================
' v6.7 (F4-08, choix A) - RECALCUL LEGER DU JOUR, programme par
' RemindOverdue (cle modDefer lourde "DashRefresh", via RefreshDashboard).
' Attend un "moment calme" (LIGHT_IDLE_SEC secondes sans clavier ni souris ;
' nouvel essai toutes les LIGHT_RETRY_SEC s, LIGHT_MAX_TRIES fois au plus,
' sinon pas de recalcul ce jour-la). Puis : fils des lignes EN COURS relus
' par l'index de conversations seulement (ni re-identification, ni scan de
' reparation, ni tri, ni mise en page), valeurs changees ecrites, fichier
' des echeances ecrit, classeur enregistre et rendu (Excel ne surgit pas).
' Enfin le toast du matin, a jour.
'=============================================================================
Private Sub LightRefresh()
    On Error GoTo ErrH
    If UserIdleSec() < LIGHT_IDLE_SEC Then
        mLightTries = mLightTries + 1
        If mLightTries <= LIGHT_MAX_TRIES Then
            modDebugLog.DTrace MOD_NAME, "Recalcul leger : utilisateur actif, nouvel essai dans " & LIGHT_RETRY_SEC & " s"
            modDefer.ArmUnique "DashRefresh", LIGHT_RETRY_SEC
            Exit Sub
        End If
        mLightRequested = False
        modDebugLog.DInfo MOD_NAME, "Recalcul leger du jour abandonne : aucun moment calme en " & _
            (LIGHT_MAX_TRIES * LIGHT_RETRY_SEC \ 60) & " min"
        FinishMorning
        Exit Sub
    End If
    mLightRequested = False
    If Dir(GetTrackerFilePath()) = "" Then
        FinishMorning
        Exit Sub
    End If

    BeginTrackerWork "refresh"
    Dim xlApp As Object, wb As Object, ws As Object
    Dim userWb As Boolean, evSaved As Boolean, suSaved As Boolean, quietOn As Boolean
    Dim t0 As Double
    t0 = Timer
    quietOn = False
    Set xlApp = GetExcelApp()
    If xlApp Is Nothing Then GoTo Bail
    xlApp.DisplayAlerts = False
    Set wb = GetTrackerWorkbook(xlApp, True)
    If wb Is Nothing Then GoTo Bail
    userWb = Not mWbOpenedHere
    If mWbOpenedHere Then HideBackgroundWorkbook xlApp, wb
    Set ws = wb.Sheets(modConfig.MailTrackingSheet())
    XlQuiet xlApp, evSaved, suSaved
    quietOn = True

    UpdateAllTrackingStatuses ws, True
    WriteDueCache ws
    SaveWorkbookVisible xlApp, wb
    XlRestore xlApp, evSaved, suSaved
    quietOn = False
    If userWb Then NotifyExcelRefreshDone xlApp, wb
    ReleaseBackground xlApp, wb
    EndTrackerWork
    SaveSetting REG_TRACK_APP, REG_TRACK_SECTION, REG_KEY_LASTLIGHT, Format$(Date, "yyyy-mm-dd")
    modDebugLog.DInfo MOD_NAME, "Recalcul leger du jour : " & Format$(ElapsedSec(t0), "0.0") & " s"
    FinishMorning
    Exit Sub
Bail:
    EndTrackerWork
    modDebugLog.DError MOD_NAME, "Recalcul leger du jour : classeur de suivi indisponible"
    FinishMorning
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "Recalcul leger du jour err " & Err.Number & " - " & Err.Description
    On Error Resume Next
    If quietOn Then XlRestore xlApp, evSaved, suSaved
    If Not wb Is Nothing Then ReleaseBackground xlApp, wb
    EndTrackerWork
    FinishMorning
End Sub

' v6.7 (F4-08) - Secondes depuis la derniere frappe ou le dernier mouvement
' de souris (GetLastInputInfo, compteur 32 bits compare aux 32 bits bas de
' modCommon.TickMs : juste meme apres 49 jours d'allumage). En cas d'echec :
' grande valeur (considere comme calme).
Private Function UserIdleSec() As Double
    On Error GoTo Bad
    Dim lii As LASTINPUTINFO
    lii.cbSize = Len(lii)
    If GetLastInputInfo(lii) = 0 Then GoTo Bad
    Dim nowLow As Double, lastIn As Double, d As Double
    nowLow = modCommon.TickMs()
    nowLow = nowLow - Int(nowLow / 4294967296#) * 4294967296#
    lastIn = CDbl(lii.dwTime)
    If lastIn < 0 Then lastIn = lastIn + 4294967296#
    d = nowLow - lastIn
    If d < 0 Then d = d + 4294967296#
    UserIdleSec = d / 1000#
    Exit Function
Bad:
    UserIdleSec = 99999#
End Function

' v6.7 (F4-08) - Apres un recalcul COMPLET reussi (tableau de bord,
' Actualiser) : il vaut recalcul du jour ; toast du matin s'il attendait.
Private Sub MorningAfterRefresh()
    On Error Resume Next
    SaveSetting REG_TRACK_APP, REG_TRACK_SECTION, REG_KEY_LASTLIGHT, Format$(Date, "yyyy-mm-dd")
    ' Recalcul leger encore programme (tableau de bord ouvert entre-temps) :
    ' devenu inutile, sa minuterie est retiree (sauf actualisation demandee).
    If mLightRequested And mRefreshQueuedMs = 0 Then modDefer.Cancel "DashRefresh"
    mLightRequested = False
    FinishMorning
End Sub

' v6.7 (F4-07) - Toast du matin s'il attendait la fin d'un recalcul (reussi
' ou non : il lit alors le dernier fichier des echeances).
Private Sub FinishMorning()
    On Error Resume Next
    If Not mMorningPending Then Exit Sub
    mMorningPending = False
    ShowMorningToast
End Sub

'=============================================================================
' v6.4 - RepairTrackerRows : REPARATION UNIQUE d'un classeur desaligne.
' A lancer une fois depuis Outlook (Alt+F8, ou fenetre Execution du VBE).
'
' Symptome : l'objet d'une ligne ne correspond plus a son dossier, a son
' statut ni au mail ouvert par "Ouvrir le mail". Cause : tri lance depuis une
' fleche du filtre automatique, qui ne couvrait que les colonnes visibles
' (corrige v6.4) : les colonnes visibles ont bouge, pas les colonnes
' techniques masquees (ID, ConversationID, EntryID, AckCount).
'
' Methode, feuille Suivi puis Archives :
'   1. copie de sauvegarde du classeur (meme dossier, suffixe date-heure) ;
'   2. chaque bloc technique est rattache a SA ligne : le mail de son EntryID
'      est relu dans Outlook (objet + date d'envoi) puis cherche parmi les
'      lignes : meme objet normalise et date d'envoi a un jour pres (la plus
'      proche) ; a defaut, meme objet s'il n'y a qu'une ligne candidate. Un
'      bloc deja sur la bonne ligne ne bouge pas ;
'   3. bloc dont le mail est introuvable, reste a sa place : conserve si son
'      identifiant de suivi a ete cree au plus un jour avant l'envoi de la
'      ligne (plausible), sinon traite comme les autres restes ;
'   4. restes : appaires dans l'ordre, ConversationID et EntryID EFFACES ->
'      RelinkRows re-identifie la ligne par objet + date
'      d'envoi, comme une ligne ancienne ;
'   5. lignes modifiees : nom de dossier efface (relu au recalcul), statut
'      "Repondu ?" remis a "En attente" (calcule sur le mauvais mail ; le
'      recalcul le repose si une reponse non validee existe). Statuts
'      Repondu et Traite (choix de l'utilisateur) conserves.
' Puis le tableau de bord s'ouvre (recalcul complet). Une feuille au gabarit
' inconnu, ou dont une ligne n'a pas d'objet ou pas d'identifiant, n'est pas
' touchee.
'=============================================================================
Public Sub RepairTrackerRows()
    RunTrackerRepair False
End Sub

' v6.5 - corps de la reparation. fromExcel = demande du bouton Reparer du
' classeur : pas de fenetre Outlook (elle s'ouvrirait derriere Excel),
' toasts, puis recalcul du classeur ouvert (RefreshDashboard). Sinon
' (macro) : fenetre de compte rendu puis tableau de bord, comme en v6.4.
' v6.7 (F4-10) : une feuille dont la remise en place a echoue (Excel occupe
' : cellule en saisie, boite ouverte) -> RIEN n'est enregistre (avant :
' l'etat a moitie permute etait enregistre) et l'utilisateur le sait.
' Evenements ET affichage d'Excel coupes pendant les ecritures (XlQuiet).
' Un seul traitement du suivi a la fois (macro lancee pendant une
' actualisation : refusee par un toast).
Private Sub RunTrackerRepair(ByVal fromExcel As Boolean)
    On Error GoTo ErrH
    Dim xlApp As Object, wb As Object, ws As Object, wsA As Object
    Dim evSaved As Boolean, suSaved As Boolean, quietOn As Boolean
    Dim backupPath As String
    quietOn = False
    backupPath = ""

    If Dir(GetTrackerFilePath()) = "" Then
        RepairNotice T("Fichier de suivi introuvable.", "Tracking file not found."), fromExcel
        Exit Sub
    End If
    If TrackerBusy() Then
        NotifyTrackerBusy
        Exit Sub
    End If
    BeginTrackerWork "repair"

    Set xlApp = GetExcelApp()
    If xlApp Is Nothing Then GoTo Bail
    xlApp.DisplayAlerts = False
    Set wb = GetTrackerWorkbook(xlApp, fromExcel)
    If wb Is Nothing Then GoTo Bail
    Set ws = wb.Sheets(modConfig.MailTrackingSheet())

    modCommon.SetStatus MOD_NAME, T("R~eparation du suivi des mails...", "Repairing mail tracking...")

    backupPath = BackupTrackerCopy(xlApp, wb)
    If Len(backupPath) = 0 Then
        ReleaseBackground xlApp, wb
        EndTrackerWork
        RepairNotice T("R~eparation annul~ee : copie de sauvegarde impossible, rien n'a ~et~e modifi~e.", _
                       "Repair cancelled: backup copy failed, nothing was changed."), fromExcel
        Exit Sub
    End If

    ' Evenements du classeur coupes pendant les ecritures (son code reagit
    ' aux modifications de cellules) ; v6.7 : affichage aussi.
    XlQuiet xlApp, evSaved, suSaved
    quietOn = True

    Dim nRows As Long, nMoved As Long, nCleared As Long, okTrack As Boolean, failTrack As Boolean
    okTrack = RealignSheet(ws, True, nRows, nMoved, nCleared, failTrack)

    Dim aRows As Long, aMoved As Long, aCleared As Long, okArch As Boolean, failArch As Boolean
    okArch = True
    failArch = False
    Set wsA = Nothing
    On Error Resume Next
    Set wsA = wb.Worksheets(ARCHIVE_SHEET)
    On Error GoTo ErrH
    If Not wsA Is Nothing Then okArch = RealignSheet(wsA, False, aRows, aMoved, aCleared, failArch)

    ' v6.7 (F4-10) : remise en place interrompue -> rien n'est enregistre.
    If failTrack Or failArch Then
        XlRestore xlApp, evSaved, suSaved
        quietOn = False
        ReleaseBackground xlApp, wb
        Set wb = Nothing
        EndTrackerWork
        mLastRepairEndMs = modCommon.TickMs()
        modDebugLog.DError MOD_NAME, "RepairTrackerRows : remise en place interrompue, classeur non enregistre (copie " & backupPath & ")"
        RepairNotice T("R~eparation interrompue : Excel a refus~e une ~ecriture. Rien n'a ~et~e enregistr~e.", _
                       "Repair interrupted: Excel refused a write. Nothing was saved.") & vbLf & _
                     T("Ne touchez pas au classeur pendant la r~eparation, puis relancez R~eparer.", _
                       "Do not touch the workbook during the repair, then run Repair again."), fromExcel
        Exit Sub
    End If

    ' v6.6 : lignes sans mail joignable -> re-identification par objet + date
    ' d'envoi, Suivi et Archives, sans tenir compte des echecs passes.
    Dim relinkShs As Collection
    Set relinkShs = New Collection
    If okTrack Then relinkShs.Add ws
    If okArch And Not wsA Is Nothing Then relinkShs.Add wsA
    Dim nRelinked As Long, nRelinkLost As Long, relinkCut As Boolean
    modCommon.SetStatus MOD_NAME, T("R~eparation : recherche des mails dans tous les dossiers...", _
                                    "Repair: searching all folders for the mails...")
    RelinkRows relinkShs, True, RELINK_REPAIR_BUDGET_SEC, nRelinked, nRelinkLost, relinkCut

    WriteDueCache ws
    SaveWorkbookVisible xlApp, wb
    XlRestore xlApp, evSaved, suSaved
    quietOn = False
    ReleaseBackground xlApp, wb
    Set wb = Nothing
    EndTrackerWork
    mLastRepairEndMs = modCommon.TickMs()

    modDebugLog.DInfo MOD_NAME, "RepairTrackerRows : suivi " & nMoved & " retablie(s), " & _
        nCleared & " a re-identifier sur " & nRows & " ; archives " & aMoved & " retablie(s), " & _
        aCleared & " effacee(s) sur " & aRows & " ; re-identifiees " & nRelinked & ", introuvables " & _
        nRelinkLost & " ; copie " & backupPath

    If fromExcel Then
        Dim note As String
        note = T("R~eparation : ", "Repair: ") & nMoved & T(" ligne(s) r~etablie(s), ", " row(s) realigned, ") & _
               nRelinked & T(" mail(s) retrouv~e(s)", " mail(s) found again")
        If nRelinkLost > 0 Then
            If relinkCut Then
                note = note & ", " & nRelinkLost & T(" non trouv~e(s) - recherche incompl~fte, relancez R~eparer", _
                                                     " not found - search incomplete, run Repair again")
            Else
                note = note & ", " & nRelinkLost & T(" introuvable(s)", " not found")
            End If
        End If
        If aMoved + aCleared > 0 Then
            note = note & T(" (archives : ", " (archives: ") & (aMoved + aCleared) & ")"
        End If
        If Not (okTrack And okArch) Then
            note = note & T(" - une feuille non trait~ee, voir debug.log", " - a sheet was not processed, see debug.log")
        End If
        note = note & vbLf & T("Copie : ", "Backup: ") & Mid$(backupPath, InStrRev(backupPath, "\") + 1)
        modNotify.Notify note
        RefreshDashboard
        Exit Sub
    End If

    Dim msg As String
    msg = T("Suivi : ", "Tracking: ") & nRows & T(" ligne(s), ", " row(s), ") & _
          nMoved & T(" r~etablie(s), ", " realigned, ") & _
          nCleared & T(" ~a re-identifier.", " to re-identify.") & vbCrLf & _
          T("Mails retrouv~es dans les dossiers : ", "Mails found again in the folders: ") & nRelinked & _
          T(", introuvables : ", ", not found: ") & nRelinkLost & _
          IIf(relinkCut, T(" (recherche incompl~fte : relancez)", " (search incomplete: run again)"), "") & vbCrLf
    If aRows > 0 Then
        msg = msg & T("Archives : ", "Archives: ") & aRows & T(" ligne(s), ", " row(s), ") & _
              aMoved & T(" r~etablie(s), ", " realigned, ") & _
              aCleared & T(" sans mail retrouv~e.", " without mail found.") & vbCrLf
    End If
    If Not (okTrack And okArch) Then
        msg = msg & vbCrLf & T("Une feuille n'a pas ~et~e trait~ee (gabarit ou lignes incompl~ftes) : voir debug.log.", _
                               "A sheet was not processed (layout or incomplete rows): see debug.log.") & vbCrLf
    End If
    msg = msg & vbCrLf & T("Copie de sauvegarde :", "Backup copy:") & vbCrLf & backupPath & vbCrLf & vbCrLf & _
          T("Le tableau de bord va s'ouvrir et tout recalculer.", _
            "The dashboard will now open and recompute everything.")
    MsgBox msg, vbInformation, T("R~eparation du suivi", "Tracking repair")

    OpenTrackingDashboard
    Exit Sub

Bail:
    ' Excel ou classeur indisponible (deja signale si mTrackerFailReported).
    EndTrackerWork
    If Not mTrackerFailReported Then
        RepairNotice T("R~eparation impossible : classeur de suivi indisponible (voir debug.log).", _
                       "Repair impossible: tracking workbook unavailable (see debug.log)."), fromExcel
    End If
    Exit Sub

ErrH:
    Dim errNum As Long, errDesc As String
    errNum = Err.Number
    errDesc = Err.Description
    On Error Resume Next
    If quietOn Then XlRestore xlApp, evSaved, suSaved
    If Not wb Is Nothing Then ReleaseBackground xlApp, wb
    EndTrackerWork
    mLastRepairEndMs = modCommon.TickMs()
    modDebugLog.DError MOD_NAME, "RepairTrackerRows err " & errNum & " - " & errDesc
    If fromExcel Then
        modNotify.Notify T("R~eparation du suivi impossible (erreur ", "Tracking repair failed (error ") & errNum & _
                         T(") - voir debug.log", ") - see debug.log")
    ElseIf Len(backupPath) > 0 Then
        MsgBox T("Erreur ", "Error ") & errNum & " : " & errDesc & vbCrLf & vbCrLf & _
               T("Copie de sauvegarde :", "Backup copy:") & vbCrLf & backupPath, _
               vbCritical, T("R~eparation du suivi", "Tracking repair")
    Else
        MsgBox T("Erreur ", "Error ") & errNum & " : " & errDesc, _
               vbCritical, T("R~eparation du suivi", "Tracking repair")
    End If
End Sub

'=============================================================================
' v6.4 - Copie de sauvegarde du classeur avant reparation, a cote du fichier
' de suivi : <nom>_avant-reparation_<aaaammjj-hhnnss>.<ext>. Chemin local
' (modConfig) et non wb.FullName, qui peut etre une adresse web sous
' OneDrive. Fenetre re-affichee pendant la copie si elle etait masquee
' (Excel enregistre sa visibilite dans le fichier). "" si echec.
'=============================================================================
Private Function BackupTrackerCopy(ByVal xlApp As Object, ByVal wb As Object) As String
    On Error Resume Next
    BackupTrackerCopy = ""
    Dim full As String, p As Long, target As String
    full = GetTrackerFilePath()
    p = InStrRev(full, ".")
    If p <= 1 Then Exit Function
    target = Left$(full, p - 1) & "_avant-reparation_" & Format$(Now, "yyyymmdd-hhnnss") & Mid$(full, p)

    Dim wasHidden As Boolean
    wasHidden = False
    wasHidden = Not CBool(wb.Windows(1).Visible)
    If wasHidden Then
        xlApp.ScreenUpdating = False
        wb.Windows(1).Visible = True
    End If
    Err.Clear
    wb.SaveCopyAs target
    Dim errNum As Long, errDesc As String
    errNum = Err.Number
    errDesc = Err.Description
    If wasHidden Then
        wb.Windows(1).Visible = False
        xlApp.ScreenUpdating = True
    End If
    If errNum <> 0 Then
        modDebugLog.DError MOD_NAME, "Copie de sauvegarde impossible (" & errNum & " - " & errDesc & ")"
        Exit Function
    End If
    If Dir(target) = "" Then Exit Function
    BackupTrackerCopy = target
    PurgeOldBackups Left$(full, p - 1) & "_avant-reparation_", Mid$(full, p)
End Function

'=============================================================================
' v6.7 (F4-19) - Ne garde que les BACKUP_KEEP copies "_avant-reparation_"
' les plus recentes (une copie par clic sur Reparer, jamais purgee avant).
' Le nom porte la date (aaaammjj-hhnnss) : ordre alphabetique = ordre
' chronologique. Silencieux ; un fichier verrouille est laisse en place.
'=============================================================================
Private Sub PurgeOldBackups(ByVal prefix As String, ByVal ext As String)
    On Error Resume Next
    Dim folder As String, pat As String, nm As String
    folder = Left$(prefix, InStrRev(prefix, "\"))
    pat = Mid$(prefix, InStrRev(prefix, "\") + 1)
    Dim names() As String, n As Long, i As Long, j As Long, tmp As String
    n = 0
    ReDim names(1 To 1)
    nm = Dir(prefix & "*" & ext)
    Do While Len(nm) > 0
        If StrComp(Left$(nm, Len(pat)), pat, vbTextCompare) = 0 Then
            n = n + 1
            ReDim Preserve names(1 To n)
            names(n) = nm
        End If
        nm = Dir()
    Loop
    If n <= BACKUP_KEEP Then Exit Sub
    For i = 2 To n
        tmp = names(i)
        j = i - 1
        Do While j >= 1
            If StrComp(names(j), tmp, vbTextCompare) <= 0 Then Exit Do
            names(j + 1) = names(j)
            j = j - 1
        Loop
        names(j + 1) = tmp
    Next i
    Dim nKilled As Long
    For i = 1 To n - BACKUP_KEEP
        Err.Clear
        Kill folder & names(i)
        If Err.Number = 0 Then nKilled = nKilled + 1
    Next i
    If nKilled > 0 Then
        modDebugLog.DInfo MOD_NAME, "Copies de sauvegarde : " & nKilled & " ancienne(s) supprimee(s), " & _
            BACKUP_KEEP & " gardee(s)"
    End If
End Sub

'=============================================================================
' v6.4 - RealignSheet : remet les blocs techniques (ID, ConversationID,
' EntryID, AckCount) en face de leur ligne sur UNE feuille. Methode : cf.
' en-tete de RepairTrackerRows. isTrack = feuille Suivi (statuts "Repondu ?"
' remis a "En attente" sur les lignes modifiees). False si la feuille n'a pas
' ete traitee (gabarit inconnu, ligne sans objet ou sans identifiant, erreur).
'=============================================================================
Private Function RealignSheet(ByVal ws As Object, ByVal isTrack As Boolean, _
                              ByRef nRows As Long, ByRef nMoved As Long, _
                              ByRef nCleared As Long, ByRef failed As Boolean) As Boolean
    On Error GoTo ErrH
    RealignSheet = False
    failed = False
    nRows = 0: nMoved = 0: nCleared = 0

    If Not IsSubjectHeader(ws.Cells(1, COL_SUBJECT).Value) Then
        modDebugLog.DError MOD_NAME, "Realign (" & ws.name & ") : gabarit non reconnu, feuille ignoree"
        Exit Function
    End If

    ' v6.7 (F4-01) : dernieres lignes cherchees AUSSI sous les lignes
    ' masquees par un filtre.
    Dim lastRow As Long, lr As Long
    lastRow = TrackLastRow(ws)
    lr = modExcelBridge.DataLastRow(ws, COL_SUBJECT, COL_SUBJECT)
    If lr > lastRow Then lastRow = lr
    If lastRow < 2 Then
        RealignSheet = True
        Exit Function
    End If

    Dim n As Long, i As Long, j As Long, k As Long
    n = lastRow - 1
    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)

    ' Garde-fou : chaque ligne doit avoir un objet ET un identifiant de suivi.
    ' Sinon les deux listes n'ont pas la meme longueur : on ne devine pas.
    For i = 1 To n
        If Len(Trim$(CStr(SafeVal(blk, i, COL_SUBJECT)))) = 0 Or _
           Len(Trim$(CStr(SafeVal(blk, i, COL_ID)))) = 0 Then
            modDebugLog.DError MOD_NAME, "Realign (" & ws.name & ") : ligne " & (i + 1) & _
                " sans objet ou sans identifiant, feuille ignoree"
            Exit Function
        End If
    Next i
    nRows = n

    ' Lignes : objet normalise + date d'envoi.
    Dim vSubj() As String, vSent() As Date
    ReDim vSubj(1 To n)
    ReDim vSent(1 To n)
    For j = 1 To n
        vSubj(j) = NormalizeSubject(CStr(SafeVal(blk, j, COL_SUBJECT)))
        vSent(j) = 0
        If IsDate(SafeVal(blk, j, COL_SENT_DATE)) Then vSent(j) = CDate(SafeVal(blk, j, COL_SENT_DATE))
    Next j

    ' Blocs techniques : objet + date d'envoi du mail de leur EntryID.
    Dim olNS As Outlook.NameSpace
    Set olNS = Application.GetNamespace("MAPI")
    Dim tSubj() As String, tSent() As Date, tOk() As Boolean
    ReDim tSubj(1 To n)
    ReDim tSent(1 To n)
    ReDim tOk(1 To n)
    Dim eid As String, itm As Object
    For k = 1 To n
        eid = CStr(SafeVal(blk, k, COL_ENTRY_ID))
        If Len(eid) > 0 Then
            Set itm = Nothing
            On Error Resume Next
            Set itm = olNS.GetItemFromID(eid)
            On Error GoTo ErrH
            If Not itm Is Nothing Then
                tSubj(k) = NormalizeSubject(ItemSubject(itm))
                tSent(k) = ItemSentDate(itm)
                tOk(k) = (Len(tSubj(k)) > 0)
            End If
        End If
        If (k Mod 20) = 0 Then DoEvents
    Next k

    ' rowOf(k) = ligne du bloc k ; tupOf(j) = bloc de la ligne j (0 = libre).
    Dim rowOf() As Long, tupOf() As Long, cleared() As Boolean
    ReDim rowOf(1 To n)
    ReDim tupOf(1 To n)
    ReDim cleared(1 To n)
    Dim best As Long, bestGap As Double, gap As Double, nCand As Long

    ' Passe 1 : bloc deja sur sa ligne -> il reste en place.
    For k = 1 To n
        If tOk(k) Then
            If MailFitsRow(tSubj(k), tSent(k), vSubj(k), vSent(k)) Then
                rowOf(k) = k
                tupOf(k) = k
            End If
        End If
    Next k

    ' Passe 2 : meme objet, date d'envoi a un jour pres (la plus proche).
    For k = 1 To n
        If tOk(k) And rowOf(k) = 0 Then
            best = 0
            bestGap = 2#
            For j = 1 To n
                If tupOf(j) = 0 Then
                    If MailFitsRow(tSubj(k), tSent(k), vSubj(j), vSent(j)) Then
                        gap = DateGap(tSent(k), vSent(j))
                        If gap < bestGap Then
                            bestGap = gap
                            best = j
                        End If
                    End If
                End If
            Next j
            If best > 0 Then
                rowOf(k) = best
                tupOf(best) = k
            End If
        End If
    Next k

    ' Passe 3 : meme objet, une seule ligne candidate (l'EntryID peut etre
    ' celui d'un autre mail du fil, d'une autre date).
    For k = 1 To n
        If tOk(k) And rowOf(k) = 0 Then
            best = 0
            nCand = 0
            For j = 1 To n
                If tupOf(j) = 0 Then
                    If tSubj(k) = vSubj(j) Then
                        nCand = nCand + 1
                        best = j
                    End If
                End If
            Next j
            If nCand = 1 Then
                rowOf(k) = best
                tupOf(best) = k
            End If
        End If
    Next k

    ' Reste 1 : bloc et ligne libres a la MEME position, mail du bloc
    ' introuvable -> conserve si l'identifiant de suivi est plausible.
    For j = 1 To n
        If tupOf(j) = 0 And rowOf(j) = 0 And Not tOk(j) Then
            If StampPlausible(CStr(SafeVal(blk, j, COL_ID)), vSent(j)) Then
                rowOf(j) = j
                tupOf(j) = j
            End If
        End If
    Next j

    ' Reste 2 : blocs libres -> lignes libres, dans l'ordre ; fil et mail
    ' effaces (re-identification par RelinkRows).
    k = 1
    For j = 1 To n
        If tupOf(j) = 0 Then
            Do While k <= n
                If rowOf(k) = 0 Then Exit Do
                k = k + 1
            Loop
            If k > n Then Exit For                  ' impossible : autant de blocs que de lignes
            rowOf(k) = j
            tupOf(j) = k
            cleared(j) = True
            k = k + 1
        End If
    Next j

    ' Ecriture. v6.7 (F4-10) : nouvelles valeurs construites EN MEMOIRE puis
    ' ecrites en UNE affectation par colonne ou bloc de colonnes, sur la
    ' seule plage des lignes modifiees : d'abord les colonnes techniques
    ' ID..ACK_COUNT (contigues : c'est la permutation ; une affectation
    ' reussit ou echoue en bloc, plus jamais de feuille a moitie permutee),
    ' puis la colonne dossier et la colonne statut. Couleur du statut : mise
    ' en forme conditionnelle (EnsureStatusRules).
    Dim nTech As Long, cc As Long, st As String, jMin As Long, jMax As Long, anyStat As Boolean
    nTech = COL_ACK_COUNT - COL_ID + 1
    Dim tech() As Variant, fold() As Variant, stat() As Variant
    ReDim tech(1 To n, 1 To nTech)
    ReDim fold(1 To n, 1 To 1)
    ReDim stat(1 To n, 1 To 1)
    For j = 1 To n
        For cc = 1 To nTech
            tech(j, cc) = SafeVal(blk, j, COL_ID + cc - 1)
        Next cc
        fold(j, 1) = SafeVal(blk, j, COL_FOLDER)
        stat(j, 1) = SafeVal(blk, j, COL_STATUS)
    Next j
    jMin = 0
    jMax = 0
    anyStat = False
    For j = 1 To n
        k = tupOf(j)
        If k > 0 Then
            If k <> j Or cleared(j) Then
                If jMin = 0 Then jMin = j
                jMax = j
                tech(j, 1) = SafeVal(blk, k, COL_ID)
                If cleared(j) Then
                    tech(j, COL_CONV_ID - COL_ID + 1) = ""
                    tech(j, COL_ENTRY_ID - COL_ID + 1) = ""
                    tech(j, COL_ACK_COUNT - COL_ID + 1) = 0
                    nCleared = nCleared + 1
                Else
                    tech(j, COL_CONV_ID - COL_ID + 1) = SafeVal(blk, k, COL_CONV_ID)
                    tech(j, COL_ENTRY_ID - COL_ID + 1) = SafeVal(blk, k, COL_ENTRY_ID)
                    tech(j, COL_ACK_COUNT - COL_ID + 1) = SafeVal(blk, k, COL_ACK_COUNT)
                    nMoved = nMoved + 1
                End If
                fold(j, 1) = ""
                If isTrack Then
                    st = CStr(SafeVal(blk, j, COL_STATUS))
                    If st = "R" & Chr(233) & "pondu ?" Then
                        stat(j, 1) = "En attente"
                        anyStat = True
                    End If
                End If
            End If
        End If
    Next j

    If jMin > 0 Then
        ws.Range(ws.Cells(jMin + 1, COL_ID), ws.Cells(jMax + 1, COL_ACK_COUNT)).Value = SliceRows(tech, jMin, jMax)
        For j = 1 To n
            If cleared(j) Then DeleteOrphanMemory CStr(SafeVal(blk, tupOf(j), COL_ID))
        Next j
        ws.Range(ws.Cells(jMin + 1, COL_FOLDER), ws.Cells(jMax + 1, COL_FOLDER)).Value = SliceRows(fold, jMin, jMax)
        If anyStat Then
            ws.Range(ws.Cells(jMin + 1, COL_STATUS), ws.Cells(jMax + 1, COL_STATUS)).Value = SliceRows(stat, jMin, jMax)
        End If
    End If

    modDebugLog.DInfo MOD_NAME, "Realign (" & ws.name & ") : " & n & " ligne(s), " & _
        nMoved & " bloc(s) remis en place, " & nCleared & " ligne(s) a re-identifier"
    RealignSheet = True
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "Realign (" & ws.name & ") err " & Err.Number & " - " & Err.Description
    RealignSheet = False
    failed = True
End Function

' v6.7 - Lignes lo..hi d'un tableau 2D (1..n, 1..m) -> nouveau tableau
' (1..hi-lo+1, 1..m), pour ecrire une plage partielle en une affectation.
Private Function SliceRows(ByRef src() As Variant, ByVal lo As Long, ByVal hi As Long) As Variant
    Dim m As Long, i As Long, c As Long
    m = UBound(src, 2)
    Dim out() As Variant
    ReDim out(1 To hi - lo + 1, 1 To m)
    For i = lo To hi
        For c = 1 To m
            out(i - lo + 1, c) = src(i, c)
        Next c
    Next i
    SliceRows = out
End Function

' v6.4 - Le mail (objet normalise mSubj, envoye le mSent) est-il celui d'une
' ligne (objet normalise rSubj, envoyee le rSent) ? Meme objet et date a un
' jour pres ; une date inconnue ne contredit rien.
Private Function MailFitsRow(ByVal mSubj As String, ByVal mSent As Date, _
                             ByVal rSubj As String, ByVal rSent As Date) As Boolean
    MailFitsRow = False
    If Len(mSubj) = 0 Then Exit Function
    If mSubj <> rSubj Then Exit Function
    MailFitsRow = (DateGap(mSent, rSent) <= 1#)
End Function

' v6.4 - Ecart en jours entre deux dates (1 si l'une est inconnue).
Private Function DateGap(ByVal d1 As Date, ByVal d2 As Date) As Double
    If d1 = 0 Or d2 = 0 Then
        DateGap = 1#
    Else
        DateGap = Abs(CDbl(d1) - CDbl(d2))
    End If
End Function

' v6.4 - Date d'envoi d'un item (reception a defaut ; 0 si illisible).
Private Function ItemSentDate(ByVal itm As Object) As Date
    On Error Resume Next
    Dim d As Date
    d = 0
    d = itm.SentOn
    If d = 0 Or Year(d) > 4000 Then
        d = 0
        d = itm.ReceivedTime
    End If
    If Year(d) > 4000 Then d = 0
    ItemSentDate = d
End Function

' v6.4 - Identifiant de suivi "ST-aaaammjj-hhnnss" plausible pour une ligne
' envoyee le sentDate : cree au plus un jour AVANT l'envoi (un suivi pose
' apres coup est normal). Identifiant illisible ou date inconnue : True.
Private Function StampPlausible(ByVal trackID As String, ByVal sentDate As Date) As Boolean
    On Error GoTo Bad
    StampPlausible = True
    If sentDate = 0 Then Exit Function
    If Len(trackID) < 18 Then Exit Function
    If Left$(trackID, 3) <> "ST-" Then Exit Function
    If Not IsNumeric(Mid$(trackID, 4, 8)) Or Not IsNumeric(Mid$(trackID, 13, 6)) Then Exit Function
    Dim stamp As Date
    stamp = DateSerial(CInt(Mid$(trackID, 4, 4)), CInt(Mid$(trackID, 8, 2)), CInt(Mid$(trackID, 10, 2))) + _
            TimeSerial(CInt(Mid$(trackID, 13, 2)), CInt(Mid$(trackID, 15, 2)), CInt(Mid$(trackID, 17, 2)))
    StampPlausible = (CDbl(stamp) >= CDbl(sentDate) - 1#)
    Exit Function
Bad:
    StampPlausible = True
End Function

' v6.4 - Oublie l'echec de re-identification memorise pour un ID de suivi
' (RelinkRows le recherchera au prochain recalcul).
Private Sub DeleteOrphanMemory(ByVal trackID As String)
    On Error Resume Next
    If Len(trackID) = 0 Then Exit Sub
    DeleteSetting REG_TRACK_APP, REG_REPAIR_SECTION, "orphan:" & trackID
End Sub

'   item hors fenetre de cache Exchange ajoutait un aller-retour reseau.
'   Maintenant : on ne parcourt plus la boite. Pour chaque suivi actif, on
'   resout le mail d'ancrage via l'EntryID stocke (colonne ENTRY_ID), puis
'   MailItem.GetConversation -> Conversation.GetTable enumere UNIQUEMENT les
'   items du fil (l'index de conversations Exchange fait le travail).
'   Complexite O(suivis x taille de fil), independante de la taille de la
'   boite et du nombre de dossiers.
'   Ancrage perime (l'EntryID change quand le mail d'origine est deplace par
'   QuickMove/ReplyInPlace) : scan de REPARATION unique facon v4.3, limite
'   aux fils concernes, depuis la RACINE du store (dossiers non-mail
'   elagues), avec arret des que tout est repare. Des qu'un mail du fil est
'   croise, bascule immediate sur le chemin rapide ci-dessus et reecriture
'   d'un EntryID frais (colonne ENTRY_ID : auto-guerison, la ligne repasse
'   par le chemin rapide aux ouvertures suivantes). Secours inline v4.3
'   conserve si l'index de conversations est indisponible sur le store.
'   Effets de bord assumes : la detection couvre desormais le fil ENTIER du
'   store (plus seulement le sous-arbre Boite de reception) ; la date d'un
'   repondant est sa PREMIERE reponse (min ReceivedTime), plus celle du
'   premier mail croise au hasard du parcours des dossiers.
'
' v6.7 (audit 11/2026) :
'   - F4-03 : plusieurs lignes en cours d'un meme fil (mails a des
'     personnes differentes, ou doublon ancien) : le fil est lu UNE fois et
'     toutes les reponses gardees avec leur date ; chaque ligne compte
'     ensuite les reponses de SES destinataires recues apres SA date de
'     reference (avant : la seconde ligne recevait les repondants de la
'     premiere, et la reponse de ses propres destinataires n'etait jamais
'     vue) ;
'   - F4-02 : reponses automatiques, accuses et rapports ecartes ; "soi" =
'     toutes ses adresses ;
'   - F4-06 : phase 3 calculee en memoire puis ecrite PAR COLONNE (une
'     affectation par colonne modifiee, aucune si rien ne change ; couleurs
'     par mise en forme conditionnelle) ; feuille relue juste avant
'     l'ecriture (la phase 2 laisse passer d'autres evenements) et rien
'     n'est ecrit si des lignes ont bouge entre-temps (tri a la main) ;
'   - F4-15 : "En retard" le lendemain de la date limite (jours entiers) ;
'   - fil sans donnee fraiche (ancrage perime non repare, ou reparation
'     sautee) : reponses de la ligne CONSERVEES, seuls la date limite et le
'     retard sont recalcules (avant : remises a zero) ;
'   - lightMode (F4-08, recalcul leger du jour) : ni re-identification
'     (phase 0) ni scan de reparation (phase 2b).
'=============================================================================
Private Sub UpdateAllTrackingStatuses(ws As Object, _
                                      Optional ByVal lightMode As Boolean = False)

    Dim lastRow As Long
    lastRow = TrackLastRow(ws)

    If lastRow < 2 Then Exit Sub

    ' --- Phase 0 (v6.6) : lignes sans mail joignable (fil vide, mail
    ' introuvable ou d'un autre objet) re-identifiees par objet + date
    ' d'envoi (RelinkRows). Pas en mode leger (v6.7).
    If Not lightMode Then
        Dim relinkShs As Collection
        Set relinkShs = New Collection
        relinkShs.Add ws
        Dim nRelinked As Long, nRelinkLost As Long, relinkCut As Boolean
        RelinkRows relinkShs, False, RELINK_AUTO_BUDGET_SEC, nRelinked, nRelinkLost, relinkCut
    End If

    ' --- Phase 1 : fils des lignes en cours ---
    ' Dictionary PARALLELES indexes par convID (un Dictionary imbrique via
    ' "Set dict(key) = obj" est fragile en late binding, cf. v4.3) :
    '   dExpected(convID)    -> destinataires attendus de TOUTES ses lignes en
    '                           cours (lower, separes par ";")
    '   dRows(convID)        -> liste des lignes "r1;r2;..."
    '   dRespondents(convID) -> Dictionary (adresse -> dates des reponses)
    '   dAnchor(convID)      -> EntryID d'ancrage (premiere ligne qui en a un)
    '   dFloor(convID)       -> plus ancienne date de reference des lignes
    Dim dExpected As Object: Set dExpected = CreateObject("Scripting.Dictionary")
    Dim dRows As Object:     Set dRows = CreateObject("Scripting.Dictionary")
    Dim dRespondents As Object: Set dRespondents = CreateObject("Scripting.Dictionary")
    Dim dAnchor As Object:   Set dAnchor = CreateObject("Scripting.Dictionary")
    Dim dFloor As Object:    Set dFloor = CreateObject("Scripting.Dictionary")

    ' Pack v3 (B3) : la feuille est lue en UN bloc (lastRow x LAST_COL).
    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)
    Dim nR As Long
    nR = lastRow - 1
    Dim ids1() As String
    ReDim ids1(1 To nR)

    Dim row As Long, convID As String, currentStatut As String
    Dim rowExp As String, rFloor As Date, anchor As String
    For row = 2 To lastRow
        ids1(row - 1) = CStr(SafeVal(blk, row - 1, COL_ID))
        convID = CStr(SafeVal(blk, row - 1, COL_CONV_ID))
        If Len(convID) = 0 Then GoTo NextCollect

        currentStatut = CStr(SafeVal(blk, row - 1, COL_STATUS))
        ' Statuts terminaux : ni recalcules, ni relances.
        If IsTerminalStatus(currentStatut) Then GoTo NextCollect

        rowExp = LCase$(CStr(SafeVal(blk, row - 1, COL_RECIPIENTS)))
        rFloor = RowFloorFromBlock(blk, row - 1)
        anchor = CStr(SafeVal(blk, row - 1, COL_ENTRY_ID))
        If Not dExpected.Exists(convID) Then
            dExpected(convID) = rowExp
            dRows(convID) = CStr(row)
            dAnchor(convID) = anchor
            dFloor(convID) = rFloor
            ' sous-Dictionary repondants : ajout via .Add (syntaxe sure pour objets)
            Dim newResp As Object
            Set newResp = CreateObject("Scripting.Dictionary")
            dRespondents.Add convID, newResp
        Else
            ' v6.7 (F4-03) : autre ligne du meme fil. Le fil sera lu une
            ' fois pour toutes : destinataires reunis, plancher le plus
            ' ancien ; chaque ligne retrouve SES reponses en phase 3.
            dRows(convID) = dRows(convID) & ";" & CStr(row)
            dExpected(convID) = dExpected(convID) & ";" & rowExp
            If rFloor < CDate(dFloor(convID)) Then dFloor(convID) = rFloor
            If Len(CStr(dAnchor(convID))) = 0 Then dAnchor(convID) = anchor
        End If
NextCollect:
    Next row

    If dExpected.count = 0 Then Exit Sub

    Dim olNS As Outlook.NameSpace
    Set olNS = Application.GetNamespace("MAPI")

    ' --- Phase 2a : CHEMIN RAPIDE par fil (index de conversations) ---
    ' dDead : convID dont l'ancrage est mort / fil indisponible -> reparation
    Dim dDead As Object:     Set dDead = CreateObject("Scripting.Dictionary")
    Dim dRepaired As Object: Set dRepaired = CreateObject("Scripting.Dictionary")

    Dim tPhase0 As Double
    tPhase0 = Timer

    Dim convKey As Variant
    Dim respD As Object
    For Each convKey In dExpected.keys
        Set respD = dRespondents(convKey)
        Dim ok As Boolean
        ok = CollectFromConversation(olNS, CStr(dAnchor(convKey)), _
                                     CStr(convKey), CStr(dExpected(convKey)), respD, _
                                     CDate(dFloor(convKey)))
        If Not ok Then dDead(convKey) = True
    Next convKey

    ' v6.2 : duree au journal (INFO).
    Dim tPhase As Double
    tPhase = Timer - tPhase0
    If tPhase < 0 Then tPhase = tPhase + 86400#
    modDebugLog.DInfo MOD_NAME, "MAJ statuts" & IIf(lightMode, " (leger)", "") & " : " & _
        (dExpected.count - dDead.count) & "/" & dExpected.count & _
        " fil(s) via index de conversations en " & Format$(tPhase, "0.0") & " s"

    ' v6.7 : fils sans donnee fraiche (leurs lignes gardent leurs reponses).
    Dim dNoData As Object: Set dNoData = CreateObject("Scripting.Dictionary")
    Dim dk As Variant
    For Each dk In dDead.keys
        dNoData(CStr(dk)) = True
    Next dk

    ' --- Phase 2b : REPARATION (uniquement si necessaire, jamais en mode
    ' leger) ---
    ' Scan d'arbre unique facon v4.3, limite aux fils morts, depuis la
    ' racine du store (l'ancrage d'origine a pu etre range n'importe ou,
    ' y compris hors du sous-arbre Reception). Des qu'un mail d'un fil
    ' mort est croise, on rebascule sur le chemin rapide et on memorise
    ' son EntryID comme nouvel ancrage. Arret des que dDead est vide.
    ' v6.2 (pack v4) : scan BORNE a REPAIR_BUDGET_SEC secondes, et un fil
    ' deja cherche sans succes n'est plus recherche avant REPAIR_RETRY_DAYS
    ' jours. Avant : le MEME fil introuvable relancait a chaque mise a jour
    ' un parcours de TOUTE la boite, mail par mail (9 min 22 s le 17/09,
    ' Outlook ferme de force le 22/09, 9 min 57 s le 24/09).
    If lightMode Then
        If dDead.count > 0 Then
            modDebugLog.DInfo MOD_NAME, "MAJ statuts (leger) : " & dDead.count & _
                " fil(s) sans ancrage valide, laisse(s) en l'etat"
        End If
    Else
        If dDead.count > 0 Then
            Dim skippedRepair As Long
            skippedRepair = SkipRecentRepairFailures(dDead)
            If skippedRepair > 0 Then
                modDebugLog.DInfo MOD_NAME, "MAJ statuts : " & skippedRepair & _
                    " fil(s) introuvable(s) deja cherche(s) depuis moins de " & _
                    REPAIR_RETRY_DAYS & " j, reparation sautee"
            End If
        End If

        If dDead.count > 0 Then
            modDebugLog.DInfo MOD_NAME, "MAJ statuts : " & dDead.count & _
                " ancrage(s) perime(s), scan de reparation (" & REPAIR_BUDGET_SEC & " s max)"

            Dim rootFolder As Outlook.MAPIFolder
            Set rootFolder = Nothing
            On Error Resume Next
            Set rootFolder = olNS.GetDefaultFolder(olFolderInbox).Store.GetRootFolder
            On Error GoTo 0
            If rootFolder Is Nothing Then Set rootFolder = olNS.GetDefaultFolder(olFolderInbox)

            mRepairT0 = Timer
            mRepairAborted = False
            modDefer.BeginHeavy
            RepairScanTree rootFolder, olNS, dDead, dExpected, dRespondents, dRepaired, dFloor
            modDefer.EndHeavy

            ' Fils toujours introuvables : memorises (pas de nouvel essai avant
            ' REPAIR_RETRY_DAYS jours) ; fils repares : oublies.
            RecordRepairFailures dDead
            ClearRepairFailures dRepaired

            If mRepairAborted Then
                modDebugLog.DInfo MOD_NAME, "Reparation : " & dRepaired.count & _
                    " ancrage(s) retrouve(s), " & dDead.count & _
                    " fil(s) non retrouve(s) - arret au bout de " & REPAIR_BUDGET_SEC & " s"
            Else
                modDebugLog.DInfo MOD_NAME, "Reparation : " & dRepaired.count & _
                    " ancrage(s) retrouve(s), " & dDead.count & " fil(s) introuvable(s)"
            End If
        End If
        For Each dk In dRepaired.keys
            If dNoData.Exists(CStr(dk)) Then dNoData.Remove CStr(dk)
        Next dk
    End If

    ' --- Phase 3 (v6.7, F4-06) : calcul en memoire, ecriture par colonnes ---
    ' Feuille relue : si des lignes ont bouge depuis la phase 1 (tri fait a
    ' la main pendant la phase 2), rien n'est ecrit ; la prochaine
    ' actualisation refera le calcul.
    Dim blk2 As Variant, i As Long
    blk2 = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)
    For i = 1 To nR
        If CStr(SafeVal(blk2, i, COL_ID)) <> ids1(i) Then
            modDebugLog.DError MOD_NAME, "MAJ statuts : lignes deplacees pendant le calcul (ligne " & _
                (i + 1) & "), statuts non ecrits"
            Exit Sub
        End If
    Next i

    Dim cCount() As Variant, cResp() As Variant, cDue() As Variant
    Dim cStat() As Variant, cAck() As Variant, cEntry() As Variant
    ReDim cCount(1 To nR, 1 To 1)
    ReDim cResp(1 To nR, 1 To 1)
    ReDim cDue(1 To nR, 1 To 1)
    ReDim cStat(1 To nR, 1 To 1)
    ReDim cAck(1 To nR, 1 To 1)
    ReDim cEntry(1 To nR, 1 To 1)
    For i = 1 To nR
        cCount(i, 1) = SafeVal(blk2, i, COL_RESPONSE_COUNT)
        cResp(i, 1) = SafeVal(blk2, i, COL_RESPONDERS)
        cDue(i, 1) = SafeVal(blk2, i, COL_DUE_DATE)
        cStat(i, 1) = SafeVal(blk2, i, COL_STATUS)
        cAck(i, 1) = SafeVal(blk2, i, COL_ACK_COUNT)
        cEntry(i, 1) = SafeVal(blk2, i, COL_ENTRY_ID)
    Next i
    Dim chCount As Boolean, chResp As Boolean, chDue As Boolean
    Dim chStat As Boolean, chAck As Boolean, chEntry As Boolean

    Dim parts() As String, pi As Long, r As Long, noData As Boolean
    Dim responseCount As Long, respondents As String
    Dim delayDays As Long, dueDate As Date, followCount As Long, ackCount As Long
    Dim status As String, newAck As Variant
    For Each convKey In dExpected.keys
        Set respD = dRespondents(convKey)
        noData = dNoData.Exists(CStr(convKey))
        parts = Split(dRows(convKey), ";")
        For pi = LBound(parts) To UBound(parts)
            If Len(parts(pi)) > 0 Then
                r = CLng(parts(pi)) - 1                 ' indice dans les tableaux

                ' Reponses de CETTE ligne : ses destinataires, apres SA date
                ' de reference (derniere relance, sinon date d'envoi).
                If noData Then
                    responseCount = CLng(Val(CStr(cCount(r, 1))))
                    respondents = CStr(cResp(r, 1))
                Else
                    RowReplies respD, LCase$(CStr(SafeVal(blk2, r, COL_RECIPIENTS))), _
                               RowFloorFromBlock(blk2, r), responseCount, respondents
                End If

                ' v4.6 : date limite DERIVEE du delai a chaque refresh
                ' (editer le delai + rouvrir => date limite recalculee).
                delayDays = CLng(Val(CStr(SafeVal(blk2, r, COL_DELAY_DAYS))))
                dueDate = DateAdd("d", delayDays, RowFloorFromBlock(blk2, r))
                followCount = CLng(Val(CStr(SafeVal(blk2, r, COL_FOLLOWUP_COUNT))))

                ' v5.7 : plus d'auto-cloture. ackCount = nb de reponses deja
                ' acquittees. -1 = ligne rouverte -> acquitter le nb actuel.
                ' v6.7 (F4-15) : en retard le LENDEMAIN de la date limite.
                ackCount = CLng(Val(CStr(SafeVal(blk2, r, COL_ACK_COUNT))))
                newAck = cAck(r, 1)
                If ackCount = -1 Then
                    newAck = responseCount
                    status = CalculateStatus(False, (Date > Int(CDbl(dueDate))), followCount)
                ElseIf responseCount > ackCount Then
                    status = "R" & Chr(233) & "pondu ?"
                Else
                    status = CalculateStatus(False, (Date > Int(CDbl(dueDate))), followCount)
                End If

                SetCellValue cCount, r, responseCount, chCount
                SetCellValue cResp, r, respondents, chResp
                SetCellValue cDue, r, dueDate, chDue
                SetCellValue cStat, r, status, chStat
                SetCellValue cAck, r, newAck, chAck
                ' Ancrage repare : EntryID frais (auto-guerison).
                If dRepaired.Exists(convKey) Then SetCellValue cEntry, r, CStr(dRepaired(convKey)), chEntry
            End If
        Next pi
    Next convKey

    If chCount Then WriteColumn ws, COL_RESPONSE_COUNT, lastRow, cCount
    If chResp Then WriteColumn ws, COL_RESPONDERS, lastRow, cResp
    If chDue Then
        WriteColumn ws, COL_DUE_DATE, lastRow, cDue
        ws.Range(ws.Cells(2, COL_DUE_DATE), ws.Cells(lastRow, COL_DUE_DATE)).NumberFormat = "dd/mm/yyyy"
    End If
    If chStat Then WriteColumn ws, COL_STATUS, lastRow, cStat
    If chAck Then WriteColumn ws, COL_ACK_COUNT, lastRow, cAck
    If chEntry Then WriteColumn ws, COL_ENTRY_ID, lastRow, cEntry
    modDebugLog.DDebug MOD_NAME, "MAJ statuts : colonnes ecrites (nb rep., repondants, limite, statut, acq., mail) = " & _
        IIf(chCount, "1", "0") & IIf(chResp, "1", "0") & IIf(chDue, "1", "0") & _
        IIf(chStat, "1", "0") & IIf(chAck, "1", "0") & IIf(chEntry, "1", "0")

End Sub

' v6.7 (F4-06) - Nouvelle valeur d'une cellule du tableau de colonne ;
' changed passe a True si elle differe de l'ancienne.
Private Sub SetCellValue(ByRef arr() As Variant, ByVal r As Long, ByVal v As Variant, ByRef changed As Boolean)
    If SameCellValue(arr(r, 1), v) Then Exit Sub
    arr(r, 1) = v
    changed = True
End Sub

' v6.7 (F4-06) - Valeur lue dans la cellule (a) = valeur calculee (b) ?
' Dates et nombres compares en valeur (ecart < 0,1 s pour une date).
Private Function SameCellValue(ByVal a As Variant, ByVal b As Variant) As Boolean
    On Error GoTo Diff
    SameCellValue = False
    If IsError(a) Then Exit Function
    Select Case VarType(b)
        Case vbString
            If IsEmpty(a) Then
                SameCellValue = (Len(b) = 0)
            ElseIf VarType(a) = vbString Then
                SameCellValue = (CStr(a) = CStr(b))
            End If
        Case vbDate, vbDouble, vbSingle, vbLong, vbInteger, vbCurrency, vbByte
            If IsEmpty(a) Or VarType(a) = vbString Then Exit Function
            If Not IsNumeric(a) And Not IsDate(a) Then Exit Function
            SameCellValue = (Abs(CDbl(a) - CDbl(b)) < 0.000001)
        Case vbEmpty
            SameCellValue = IsEmpty(a)
    End Select
    Exit Function
Diff:
    SameCellValue = False
End Function

' v6.7 (F4-06) - Colonne col, lignes 2..lastRow, ecrite en UNE affectation.
Private Sub WriteColumn(ByVal ws As Object, ByVal col As Long, ByVal lastRow As Long, ByRef arr() As Variant)
    ws.Range(ws.Cells(2, col), ws.Cells(lastRow, col)).Value = arr
End Sub

' v6.7 (F4-03) - Reponses d'UNE ligne : adresses de ses destinataires
' attendus, premiere reponse STRICTEMENT apres sa date de reference.
' cnt = nombre de repondants ; txt = "adresse (jj/mm); ...".
Private Sub RowReplies(ByVal respDict As Object, ByVal expected As String, ByVal floor As Date, _
                       ByRef cnt As Long, ByRef txt As String)
    cnt = 0
    txt = ""
    Dim k As Variant, first As Double
    For Each k In respDict.keys
        If AddrInExpected(expected, CStr(k)) Then
            first = FirstTimeAfter(CStr(respDict(k)), CDbl(floor))
            If first > 0 Then
                cnt = cnt + 1
                If Len(txt) > 0 Then txt = txt & "; "
                txt = txt & CStr(k) & " (" & Format$(CDate(first), "dd/mm") & ")"
            End If
        End If
    Next k
End Sub

' v6.7 (F4-03) - Dates des reponses d'une adresse : " d1 d2 ..." (Str$ :
' point decimal quelle que soit la langue du poste).
Private Sub AddReplyTime(ByVal respDict As Object, ByVal addr As String, ByVal rcv As Date)
    If respDict.Exists(addr) Then
        respDict(addr) = respDict(addr) & Str$(CDbl(rcv))
    Else
        respDict.Add addr, Str$(CDbl(rcv))
    End If
End Sub

' Plus petite date de la liste strictement posterieure a floor (0 si aucune).
Private Function FirstTimeAfter(ByVal times As String, ByVal floor As Double) As Double
    FirstTimeAfter = 0
    Dim ps() As String, i As Long, v As Double
    ps = Split(Trim$(times), " ")
    For i = LBound(ps) To UBound(ps)
        If Len(ps(i)) > 0 Then
            v = Val(ps(i))
            If v > floor Then
                If FirstTimeAfter = 0 Or v < FirstTimeAfter Then FirstTimeAfter = v
            End If
        End If
    Next i
End Function

'=============================================================================
' CHEMIN RAPIDE (v4.5) : collecte les reponses d'UN fil via l'index de
' conversations Exchange, sans parcourir les dossiers.
'   anchorEID : EntryID d'un mail quelconque du fil (colonne ENTRY_ID) -
'               GetItemFromID echoue si l'EntryID est perime (Move change
'               l'EntryID).
'   convID    : ConversationID attendu (controle de coherence)
'   expected  : destinataires attendus de toutes les lignes du fil (lower)
'   respDict  : Dictionary (adresse -> dates des reponses), mute
'   floor     : on ne garde que les reponses recues STRICTEMENT apres
' v6.7 (F4-02) : classe de message lue sur la ligne de la table du fil
' AVANT d'ouvrir l'element (reponses automatiques d'absence, accuses et
' rapports ecartes : modSmtp.MessageClassKind), puis controle de l'element
' ouvert (CountReply : modSmtp.IsAutoReplyItem, modSmtp.IsFromMe sur TOUTES
' ses adresses ; avant : une seule adresse ecartee). Toutes les reponses
' sont gardees avec leur date (F4-03).
' Renvoie False si ancrage mort / fil indisponible -> passe de reparation.
'=============================================================================
Private Function CollectFromConversation(ByVal olNS As Outlook.NameSpace, _
                                         ByVal anchorEID As String, _
                                         ByVal convID As String, _
                                         ByVal expected As String, _
                                         ByVal respDict As Object, _
                                         ByVal floor As Date) As Boolean
    On Error GoTo Dead
    CollectFromConversation = False

    If Len(anchorEID) = 0 Then Exit Function

    ' --- Resolution de l'ancrage ---
    Dim anchorMail As Object
    Set anchorMail = olNS.GetItemFromID(anchorEID)
    If anchorMail Is Nothing Then Exit Function
    If anchorMail.Class <> OL_MAIL_CLASS Then Exit Function

    ' Controle de coherence : l'item resolu doit appartenir au bon fil
    If StrComp(CStr(anchorMail.ConversationID), convID, vbBinaryCompare) <> 0 Then Exit Function

    Dim conv As Object   ' Outlook.Conversation
    Set conv = anchorMail.GetConversation
    If conv Is Nothing Then Exit Function   ' store sans index de conversations

    Dim storeID As String
    storeID = anchorMail.Parent.StoreID

    ' --- Table legere du fil : colonnes par defaut (dont EntryID et
    ' MessageClass), pas d'hydratation d'objet pendant l'enumeration. On ne
    ' resout en item complet que les mails ordinaires DU FIL. ---
    Dim tbl As Object    ' Outlook.Table
    Set tbl = conv.GetTable

    ' Items individuels : un item capricieux ne doit pas tuer le fil entier.
    ' Boucle bornee (garde anti-boucle si EndOfTable devient illisible).
    On Error Resume Next
    Dim guard As Long, tblRow As Object, itm As Object, cls As Long
    Dim mcRaw As String, mcKind As Long
    guard = 0
    Do Until tbl.EndOfTable
        guard = guard + 1
        If guard > 500 Then Exit Do

        Set tblRow = Nothing
        Set tblRow = tbl.GetNextRow
        If tblRow Is Nothing Then Exit Do

        ' Classe illisible : l'element est ouvert, CountReply la controle.
        mcKind = modSmtp.MSG_KIND_NORMAL
        mcRaw = ""
        Err.Clear
        mcRaw = CStr(tblRow("MessageClass"))
        If Err.Number = 0 Then mcKind = modSmtp.MessageClassKind(mcRaw)
        Err.Clear
        If mcKind = modSmtp.MSG_KIND_NORMAL Then
            Set itm = Nothing
            Set itm = olNS.GetItemFromID(CStr(tblRow("EntryID")), storeID)
            If Not itm Is Nothing Then
                cls = 0
                cls = itm.Class
                If cls = OL_MAIL_CLASS Then CountReply itm, expected, respDict, floor
            End If
        End If
    Loop
    On Error GoTo 0

    CollectFromConversation = True
    Exit Function

Dead:
    modDebugLog.DDebug MOD_NAME, "Fil indisponible via ancrage (err " & _
        Err.Number & ") convID=" & Left(convID, 16) & "..."
    CollectFromConversation = False
End Function

' v6.7 (F4-02) - Un mail du fil compte-t-il comme reponse ? Pas une reponse
' automatique (modSmtp.IsAutoReplyItem : classe, en-tetes, objet), pas un
' de ses propres mails (modSmtp.IsFromMe), expediteur parmi les
' destinataires attendus (AddrInExpected : egalite d'adresse), recu
' STRICTEMENT apres floor (v4.6). Oui : sa date est notee. Jamais d'erreur.
Private Sub CountReply(ByVal itm As Object, ByVal expected As String, _
                       ByVal respDict As Object, ByVal floor As Date)
    On Error Resume Next
    If modSmtp.IsAutoReplyItem(itm) Then Exit Sub
    If modSmtp.IsFromMe(itm) Then Exit Sub
    Dim senderAddr As String
    senderAddr = ""
    senderAddr = LCase$(modSmtp.SenderSmtp(itm))
    If Len(senderAddr) = 0 Then Exit Sub
    If Not AddrInExpected(expected, senderAddr) Then Exit Sub
    Dim rcv As Date
    rcv = 0
    rcv = itm.ReceivedTime
    If rcv <= floor Then Exit Sub
    AddReplyTime respDict, senderAddr, rcv
End Sub

'=============================================================================
' REPARATION (v4.5) : scan recursif unique, limite aux fils dont l'ancrage
' EntryID est perime. Reprend la mecanique v4.3 (test en memoire du
' ConversationID, aucun Restrict), avec trois differences :
'   - depart de la RACINE du store (l'ancrage a pu etre range hors du
'     sous-arbre Reception), dossiers non-mail elagues (Calendrier,
'     Contacts, Taches...) ;
'   - des qu'un mail d'un fil mort est croise, bascule IMMEDIATE sur
'     CollectFromConversation avec cet ancrage frais (n'importe quel
'     membre du fil convient pour GetConversation), memorisation du
'     nouvel EntryID dans dRepaired, retrait du fil de dDead ;
'   - arret total des que dDead est vide.
' Si l'index de conversations est indisponible (Collect echoue meme avec
' un ancrage frais en main), collecte inline v4.3 en secours ultime.
' v6.7 : memes regles de reponse que le chemin rapide (CountReply).
'=============================================================================
Private Sub RepairScanTree(folder As Outlook.MAPIFolder, _
                           ByVal olNS As Outlook.NameSpace, _
                           ByRef dDead As Object, _
                           ByRef dExpected As Object, _
                           ByRef dRespondents As Object, _
                           ByRef dRepaired As Object, _
                           ByRef dFloor As Object)
    On Error Resume Next

    If dDead.count = 0 Then Exit Sub
    If RepairOverBudget() Then Exit Sub       ' v6.2

    ' Elagage : dossiers non-mail ignores avec toute leur descendance.
    ' (Si la lecture de DefaultItemType echoue, le If est saute sous
    ' On Error Resume Next : on scanne, comportement v4.3 par defaut.)
    If folder.DefaultItemType <> olMailItem Then Exit Sub

    RepairScanFolder folder, olNS, dDead, dExpected, dRespondents, dRepaired, dFloor

    Dim sub_ As Outlook.MAPIFolder
    For Each sub_ In folder.Folders
        DoEvents
        If RepairOverBudget() Then Exit Sub   ' v6.2
        RepairScanTree sub_, olNS, dDead, dExpected, dRespondents, dRepaired, dFloor
        If dDead.count = 0 Then Exit Sub
    Next sub_

    On Error GoTo 0
End Sub

'=============================================================================
' Scanne UN dossier (passe de reparation). Pour chaque mail dont le
' ConversationID figure parmi les fils a reparer : tentative de bascule sur
' le chemin rapide, sinon collecte inline (cf. en-tete RepairScanTree).
'=============================================================================
Private Sub RepairScanFolder(folder As Outlook.MAPIFolder, _
                             ByVal olNS As Outlook.NameSpace, _
                             ByRef dDead As Object, _
                             ByRef dExpected As Object, _
                             ByRef dRespondents As Object, _
                             ByRef dRepaired As Object, _
                             ByRef dFloor As Object)
    On Error Resume Next

    Dim items As Outlook.items
    Set items = folder.items

    Dim cnt As Long
    cnt = items.count
    If cnt = 0 Then Exit Sub

    Dim i As Long
    For i = 1 To cnt
        ' v6.2 : respiration + budget (un dossier peut compter des milliers
        ' de mails ; avant, DoEvents seulement entre deux dossiers).
        If (i Mod 200) = 0 Then
            DoEvents
            If RepairOverBudget() Then Exit Sub
        End If
        Dim Item As Object
        Set Item = Nothing
        Set Item = items.Item(i)
        If Item Is Nothing Then GoTo NextItem
        If Item.Class <> OL_MAIL_CLASS Then GoTo NextItem

        Dim oMail As Outlook.mailItem
        Set oMail = Item

        Dim cID As String
        cID = ""
        cID = oMail.ConversationID
        If Len(cID) = 0 Then GoTo NextItem
        If Not dDead.Exists(cID) Then GoTo NextItem

        Dim respD As Object
        Set respD = dRespondents(cID)

        ' --- Bascule immediate sur le chemin rapide avec cet ancrage frais ---
        If CollectFromConversation(olNS, CStr(oMail.entryID), cID, _
                                   CStr(dExpected(cID)), respD, _
                                   CDate(dFloor(cID))) Then
            dRepaired(cID) = CStr(oMail.entryID)
            dDead.Remove cID
            If dDead.count = 0 Then Exit Sub
            GoTo NextItem
        End If

        ' --- Secours ultime (index de conversations indisponible) :
        '     collecte inline v4.3 sur ce mail. Le fil reste dans dDead
        '     (chaque mail du fil croise sera examine, comme en v4.3). ---
        ' v6.7 (F4-02) : memes regles que le chemin rapide (reponses
        ' automatiques et propres mails ecartes).
        CountReply oMail, CStr(dExpected(cID)), respD, CDate(dFloor(cID))
NextItem:
    Next i

    On Error GoTo 0
End Sub

'=============================================================================
' v6.2 (pack v4) : BUDGET ET MEMOIRE DE LA REPARATION
'   RepairOverBudget         : True au-dela de REPAIR_BUDGET_SEC secondes.
'   SkipRecentRepairFailures : retire de dDead les fils deja cherches sans
'                              succes depuis moins de REPAIR_RETRY_DAYS jours.
'   RecordRepairFailures     : memorise les fils non retrouves (registre).
'   ClearRepairFailures      : oublie les fils repares.
' Registre HKCU : MailResponseTracker\RepairFail, valeur = ConversationID,
' donnee = numero de jour (Str$ : independant de la locale).
'=============================================================================
Private Function RepairOverBudget() As Boolean
    Dim dt As Double
    dt = Timer - mRepairT0
    If dt < 0 Then dt = dt + 86400#
    If dt > REPAIR_BUDGET_SEC Then mRepairAborted = True
    RepairOverBudget = mRepairAborted
End Function

Private Function SkipRecentRepairFailures(ByVal dDead As Object) As Long
    On Error Resume Next
    SkipRecentRepairFailures = 0
    If dDead.count = 0 Then Exit Function
    Dim ks As Variant, i As Long, v As String, n As Long
    ks = dDead.keys
    For i = LBound(ks) To UBound(ks)
        v = GetSetting(REG_TRACK_APP, REG_REPAIR_SECTION, CStr(ks(i)), "")
        If Len(v) > 0 Then
            If CDbl(Date) - Val(v) < REPAIR_RETRY_DAYS Then
                dDead.Remove ks(i)
                n = n + 1
            End If
        End If
    Next i
    SkipRecentRepairFailures = n
End Function

Private Sub RecordRepairFailures(ByVal dDead As Object)
    On Error Resume Next
    Dim k As Variant
    For Each k In dDead.keys
        SaveSetting REG_TRACK_APP, REG_REPAIR_SECTION, CStr(k), Str$(CLng(Date))
    Next k
End Sub

Private Sub ClearRepairFailures(ByVal dRepaired As Object)
    On Error Resume Next
    Dim k As Variant
    For Each k In dRepaired.keys
        DeleteSetting REG_TRACK_APP, REG_REPAIR_SECTION, CStr(k)
    Next k
End Sub

Private Function CalculateStatus(hasReply As Boolean, isLate As Boolean, _
                                 followupCount As Long) As String
    ' v4.6 : modele BINAIRE. On ne distingue plus partielle/complete : des
    ' qu'au moins une reponse (apres t0, hors soi-meme) est arrivee, le
    ' suivi est "Repondu". Sinon : "En retard" si echeance depassee,
    ' "Relance" si une relance est partie et qu'on attend de nouveau,
    ' "En attente" sinon.
    If hasReply Then
        CalculateStatus = "R" & Chr(233) & "pondu"
    ElseIf isLate Then
        CalculateStatus = "En retard"
    ElseIf followupCount > 0 Then
        CalculateStatus = "Relanc" & Chr(233)
    Else
        CalculateStatus = "En attente"
    End If
End Function

Public Sub RemindOverdue()

' v6.7 (F4-07, F4-08) - TOAST DU MATIN, lu dans le fichier des echeances
' (aucun Excel) : suivis en retard (en cours, date limite passee en jours
' entiers ou statut "En retard"), avec les 3 premiers objets, et reponses a
' valider ("Repondu ?"). Rien a dire : rien (journal). Marque le rappel du
' jour (une fois par jour).
'=============================================================================
Private Sub ShowMorningToast()
    On Error GoTo ErrH
    Dim path As String
    path = modConfig.DueCachePath()
    If Len(path) = 0 Then Exit Sub
    If Len(Dir(path)) = 0 Then
        modDebugLog.DInfo MOD_NAME, "RemindOverdue : fichier des echeances absent (cree au prochain enregistrement du suivi)"
        Exit Sub
    End If

    Dim f As Integer, txt As String
    f = FreeFile
    Open path For Input As #f
    txt = Input$(LOF(f), #f)
    Close #f
    f = 0

    Dim lines() As String, i As Long, ln As String, p() As String
    Dim nLate As Long, nValidate As Long, preview As String
    Dim dueV As Date, st As String, subj As String
    lines = Split(Replace(txt, vbCr, ""), vbLf)
    For i = LBound(lines) To UBound(lines)
        ln = lines(i)
        If Len(ln) > 0 And Left$(ln, 1) <> "#" Then
            p = Split(ln, vbTab)
            If UBound(p) >= 2 Then
                st = Trim$(p(1))
                subj = p(2)
                dueV = ParseIsoDate(p(0))
                If st = "R" & Chr(233) & "pondu ?" Then
                    nValidate = nValidate + 1
                ElseIf Not IsTerminalStatus(st) Then
                    If st = "En retard" Or (dueV > 0 And Date > dueV) Then
                        nLate = nLate + 1
                        If nLate <= 3 Then
                            If Len(subj) > 55 Then subj = Left$(subj, 55) & "..."
                            preview = preview & "  - " & subj & vbLf
                        End If
                    End If
                End If
            End If
        End If
    Next i

    ' Rappel du jour marque (qu'il y ait des retards ou non).
    SaveSetting REG_TRACK_APP, REG_TRACK_SECTION, REG_KEY_LASTPOP, Format$(Date, "yyyy-mm-dd")
    If nLate + nValidate = 0 Then
        modDebugLog.DDebug MOD_NAME, "RemindOverdue : 0 retard, 0 reponse a valider"
        Exit Sub
    End If

    Dim msg As String
    msg = ""
    If nLate > 0 Then
        msg = nLate & T(" mail(s) en attente de r" & Chr(233) & "ponse, " & _
              "d" & Chr(233) & "lai d" & Chr(233) & "pass" & Chr(233) & " :", _
              " mail(s) awaiting a reply, overdue:") & vbLf & preview
        If nLate > 3 Then
            msg = msg & T("  ... et ", "  ... and ") & (nLate - 3) & T(" autre(s)", " other(s)") & vbLf
        End If
    End If
    If nValidate > 0 Then
        msg = msg & nValidate & T(" r~eponse(s) ~a valider.", " reply(ies) to confirm.") & vbLf
    End If
    ' v6 : le toast informe, il ne demande rien (toasts non cliquables).
    msg = msg & T("Tableau de bord : bouton Suivi Mails du ruban.", _
                  "Dashboard: Mail Tracking button on the ribbon.")
    modDebugLog.DInfo MOD_NAME, "RemindOverdue : " & nLate & " retard(s), " & nValidate & " reponse(s) a valider"
    modUi.Toast msg, uiSuccess, True
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "Toast du matin : err " & Err.Number & " - " & Err.Description
    On Error Resume Next
    If f > 0 Then Close #f
End Sub

' "aaaa-mm-jj" -> date (0 si illisible).
Private Function ParseIsoDate(ByVal s As String) As Date
    On Error GoTo Bad
    ParseIsoDate = 0
    s = Trim$(s)
    If Len(s) <> 10 Then Exit Function
    If Mid$(s, 5, 1) <> "-" Or Mid$(s, 8, 1) <> "-" Then Exit Function
    ParseIsoDate = DateSerial(CInt(Left$(s, 4)), CInt(Mid$(s, 6, 2)), CInt(Mid$(s, 9, 2)))
    Exit Function
Bad:
    ParseIsoDate = 0
End Function

' v6.7 (F4-07, F10-09) - FICHIER DES ECHEANCES <Sorties>\suivi-echeances.txt
' (modConfig.DueCachePath, nom fixe) : une ligne par suivi de la feuille
' Suivi qui n'est ni "Repondu" ni "Traite" ("Repondu ?" compris : reponses
' a valider) : "date limite aaaa-mm-jj <TAB> statut <TAB> objet". Ecrit
' avant chaque enregistrement du classeur par Outlook, et par le classeur
' lui-meme apres chaque enregistrement (modMailTracker.WriteDueCache, meme
' format) ; lu par le rappel du matin sans ouvrir Excel. Silencieux : un
' echec est journalise.
'=============================================================================
Private Sub WriteDueCache(ByVal ws As Object)
    On Error GoTo ErrH
    Dim path As String
    path = modConfig.DueCachePath()
    If Len(path) = 0 Then Exit Sub
    Dim lastRow As Long
    lastRow = TrackLastRow(ws)
    Dim txt As String, r As Long, st As String, dueV As Date, n As Long
    txt = "# ClassicOutlook-Toolkit suivi-echeances v1 " & Format$(Now, "yyyy-mm-dd hh:nn:ss") & vbCrLf
    If lastRow >= 2 Then
        Dim blk As Variant
        blk = modExcelBridge.ReadBlock(ws, 2, 1, lastRow, LAST_COL)
        For r = 1 To lastRow - 1
            If Len(CStr(SafeVal(blk, r, COL_ID))) > 0 Then
                st = Trim$(CStr(SafeVal(blk, r, COL_STATUS)))
                If st <> "R" & Chr(233) & "pondu" And st <> "Trait" & Chr(233) Then
                    dueV = ToDateOrZero(SafeVal(blk, r, COL_DUE_DATE))
                    If dueV > 0 Then txt = txt & Format$(dueV, "yyyy-mm-dd")
                    txt = txt & vbTab & st & vbTab & CacheText(CStr(SafeVal(blk, r, COL_SUBJECT))) & vbCrLf
                    n = n + 1
                End If
            End If
        Next r
    End If
    Dim f As Integer
    f = FreeFile
    Open path For Output As #f
    Print #f, txt;
    Close #f
    f = 0
    modDebugLog.DDebug MOD_NAME, "Fichier des echeances ecrit (" & n & " suivi(s))"
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "Fichier des echeances non ecrit (err " & Err.Number & " - " & Err.Description & ")"
    On Error Resume Next
    If f > 0 Then Close #f
End Sub

' Texte d'une cellule pour le fichier des echeances : sur une ligne, sans
' tabulation, 120 caracteres au plus.
Private Function CacheText(ByVal s As String) As String
    s = Replace(Replace(Replace(s, vbTab, " "), vbCr, " "), vbLf, " ")
    If Len(s) > 120 Then s = Left$(s, 120)
    CacheText = s
End Function

'=============================================================================
' Appele (differe "FollowupSent") par Application_ItemSend quand une relance
' preparee par le classeur (UserProperty FollowupConvID) a ete REELLEMENT
' envoyee. Retrouve la ligne par ConversationID (une ligne en cours d'abord)
' et lui applique la mise a jour de relance (t0, date limite, compteur,
' statut).
' v6.7 (F4-12) : argument "convID" ou "convID|essai". Suivi en cours de mise
' a jour : nouvel essai dans quelques secondes. Echec (Excel occupe,
' enregistrement refuse) : nouvel essai differe (FLUSH_MAX_TRIES fois), puis
' toast d'alerte (avant : sortie silencieuse, la relance n'etait jamais
' comptee). Ligne lue et ecrite en bloc ; fichier des echeances a jour.
'=============================================================================
Public Sub RegisterFollowupSent(ByVal convArg As String)
    On Error GoTo ErrH
    Dim convID As String, attempt As Long, p As Long, why As String
    convID = convArg
    attempt = 0
    why = ""
    p = InStr(1, convArg, "|")
    If p > 0 Then
        convID = Left$(convArg, p - 1)
        attempt = CLng(Val(Mid$(convArg, p + 1)))
    End If
    If Len(convID) = 0 Then Exit Sub
    If TrackerBusy() Then
        modDefer.Arm "FollowupSent", BUSY_RETRY_SEC, convArg
        Exit Sub
    End If
    BeginTrackerWork "write"

    Dim xlApp As Object, wb As Object, ws As Object
    Set xlApp = GetExcelApp()
    If xlApp Is Nothing Then
        why = "Excel indisponible"
        GoTo Failed
    End If
    xlApp.DisplayAlerts = False
    Set wb = GetTrackerWorkbook(xlApp, True)
    If wb Is Nothing Then
        why = "classeur de suivi indisponible"
        GoTo Failed
    End If
    Set ws = wb.Sheets(modConfig.MailTrackingSheet())

    Dim r As Long
    r = FindConvRow(ws, convID, True)
    If r = 0 Then r = FindConvRow(ws, convID, False)
    If r > 0 Then
        ApplyFollowupToRow ws, r
    Else
        modDebugLog.DInfo MOD_NAME, "RegisterFollowupSent : aucune ligne pour ce fil (archivee ?)"
    End If

    WriteDueCache ws
    If Not SaveWorkbookVisible(xlApp, wb) Then
        If mWbOpenedHere Then
            why = "enregistrement du classeur refuse"
            GoTo Failed
        End If
        ' Classeur ouvert chez l'utilisateur : la ligne y est a jour.
    End If
    ReleaseBackground xlApp, wb
    EndTrackerWork
    If r > 0 Then modDebugLog.DInfo MOD_NAME, "RegisterFollowupSent : relance comptee (ligne " & r & ")"
    Exit Sub
ErrH:
    why = "err " & Err.Number & " - " & Err.Description
Failed:
    ' v5.9 : liberer le classeur sur erreur.
    On Error Resume Next
    If Not wb Is Nothing Then ReleaseBackground xlApp, wb
    EndTrackerWork
    If mTrackerFailReported Then
        modDebugLog.DError MOD_NAME, "RegisterFollowupSent : " & why & " - relance non comptee"
        Exit Sub
    End If
    If attempt < FLUSH_MAX_TRIES Then
        modDebugLog.DError MOD_NAME, "RegisterFollowupSent : " & why & " - nouvel essai dans " & _
            FLUSH_RETRY_SEC & " s (" & (attempt + 1) & "/" & FLUSH_MAX_TRIES & ")"
        modDefer.Arm "FollowupSent", FLUSH_RETRY_SEC, convID & "|" & CStr(attempt + 1)
    Else
        modDebugLog.DError MOD_NAME, "RegisterFollowupSent : " & why & " - abandon, relance non comptee"
        modUi.Toast T("Relance envoy~ee mais NON compt~ee dans le suivi : le classeur ne r~epond pas.", _
                      "Follow-up sent but NOT counted in the tracking: the workbook does not respond.") & vbLf & _
                    T("Mettez la ligne ~a jour ~a la main (Derni~fre relance, Nb relances).", _
                      "Update the row by hand (Last follow-up, # follow-ups)."), uiError, True
    End If
End Sub

'=============================================================================
' Met la ligne a jour apres relance : t0 = maintenant, date limite =
' maintenant + delai, +1 relance, statut "Relance".
' v6.7 (F4-12) : ligne lue et ecrite en UNE affectation ; les erreurs
' remontent a RegisterFollowupSent (nouvel essai) ; une relance deja
' comptee il y a moins de 10 min n'est pas recomptee (nouvel essai apres
' un echec d'enregistrement dans le classeur ouvert de l'utilisateur).
'=============================================================================
Private Sub ApplyFollowupToRow(ws As Object, ByVal r As Long)
    Dim vals As Variant
    vals = RowValues(ws, r)
    Dim lastFu As Date
    lastFu = ToDateOrZero(vals(1, COL_LAST_FOLLOWUP - COL_FOLDER + 1))
    If lastFu > 0 And lastFu <= Now Then
        If Now - lastFu < TimeSerial(0, 10, 0) Then
            modDebugLog.DInfo MOD_NAME, "Relance deja comptee il y a moins de 10 min (ligne " & r & ")"
            Exit Sub
        End If
    End If

    Dim delayDays As Long, fc As Long, rc As Long
    delayDays = CLng(Val(CStr(vals(1, COL_DELAY_DAYS - COL_FOLDER + 1))))
    fc = CLng(Val(CStr(vals(1, COL_FOLLOWUP_COUNT - COL_FOLDER + 1))))
    rc = CLng(Val(CStr(vals(1, COL_RESPONSE_COUNT - COL_FOLDER + 1))))
    SetV vals, COL_LAST_FOLLOWUP, Now
    SetV vals, COL_FOLLOWUP_COUNT, fc + 1
    SetV vals, COL_DUE_DATE, DateAdd("d", delayDays, Now)
    ' v5.7 : une relance vaut "ces reponses ne suffisent pas" -> acquitter
    ' le nb actuel (pas de "Repondu ?" immediat au prochain recalcul).
    SetV vals, COL_ACK_COUNT, rc
    PutTrackRow ws, r, vals, "Relanc" & Chr(233)
    ws.Cells(r, COL_LAST_FOLLOWUP).NumberFormat = "dd/mm/yyyy hh:mm"
End Sub

'=============================================================================
' v5.7 - TRI DU TABLEAU DE BORD (a l'ouverture)
' Ordre : Repondu ? (haut, action requise) / En retard / En attente /
' Relance / Repondu / Traite, puis date d'envoi croissante dans chaque groupe.
' Rang ecrit dans une colonne temporaire hors donnees, puis efface.
' v6.7 (F4-06) : statuts lus en un bloc, rangs ecrits en UNE affectation
' (avant : deux appels Excel par ligne).
'=============================================================================
Private Sub SortDashboard(ws As Object)
    On Error Resume Next
    Dim lastRow As Long
    lastRow = TrackLastRow(ws)
    If lastRow < 3 Then Exit Sub

    Dim helperCol As Long
    helperCol = SORT_HELPER_COL    ' v6.3 : hors des colonnes de donnees et d'archive
    Dim sts As Variant, ranks() As Variant, r As Long, n As Long
    n = lastRow - 1
    sts = modExcelBridge.ReadBlock(ws, 2, COL_STATUS, lastRow, COL_STATUS)
    ReDim ranks(1 To n, 1 To 1)
    For r = 1 To n
        ranks(r, 1) = StatusRank(CStr(SafeVal(sts, r, 1)))
    Next r
    ws.Range(ws.Cells(2, helperCol), ws.Cells(lastRow, helperCol)).Value = ranks

    Dim rng As Object
    Set rng = ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, helperCol))
    rng.Sort Key1:=ws.Cells(1, helperCol), Order1:=1, _
             Key2:=ws.Cells(1, COL_SENT_DATE), Order2:=1, _
             Header:=1
    ws.Columns(helperCol).ClearContents
End Sub

Private Function StatusRank(ByVal st As String) As Long
    Select Case st
        Case "R" & Chr(233) & "pondu ?":  StatusRank = 0
        Case "En retard":                 StatusRank = 1
        Case "En attente":                StatusRank = 2
        Case "Relanc" & Chr(233):         StatusRank = 3
        Case "R" & Chr(233) & "pondu":    StatusRank = 4
        Case "Trait" & Chr(233):          StatusRank = 5
        Case Else:                        StatusRank = 6
    End Select
End Function
