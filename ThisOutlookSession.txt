Private Sub Application_Startup()
    ' v10.20 (F10-03 / F1-13) : heure de la derniere ligne du journal de la
    ' session precedente, lue AVANT toute ligne de cette session (LoadConfig
    ' ecrit deja au journal).
    Dim prevLastLine As Date
    prevLastLine = modDebugLog.LastLineTime()

    ' Charge la configuration (config.ini -> variables globales modConfig.g_*)
    ' AVANT tout module, car plusieurs en dependent (tag, coloration, etc.).
    modConfig.LoadConfig

    ' v10.8 : marqueur de session (apres LoadConfig : le niveau de log est
    ' charge). v10.20 : la ligne "Fermeture precedente : ..." ecrite juste
    ' apres dit elle-meme si la session precedente s'est terminee proprement.
    modDebugLog.DLogSection "SESSION START"

    ' v10.10 : detection REDEMARRAGE RAPIDE. Les rattrapages de demarrage
    ' (couleurs +15 s, envoyes +25 s, archivage +120 s) ne servent qu'a
    ' couvrir ce qui s'est synchronise pendant une VRAIE fermeture ; les
    ' relancer a chaque redemarrage de la journee (VBE, mise a jour,
    ' plantage) etait du travail a vide. Horodatage pose par
    ' Application_Quit (Str$/Val : point decimal quelle que soit la
    ' locale). v10.20 (F1-13) : la marque est CONSOMMEE des sa lecture ; une
    ' session qui plante ne la reecrit pas, le demarrage suivant voit donc 0
    ' et les rattrapages tournent normalement (avant : la marque d'une
    ' fermeture propre plus ancienne faisait sauter les rattrapages apres un
    ' plantage survenu dans la demi-heure).
    Dim quickRestart As Boolean
    quickRestart = False
    On Error Resume Next
    Dim lastEndSerial As Double
    lastEndSerial = Val(GetSetting("Toolkit", "Session", "LastEnd", "0"))
    If lastEndSerial > 0 Then DeleteSetting "Toolkit", "Session", "LastEnd"
    Err.Clear
    If lastEndSerial > 0 Then
        If DateDiff("n", CDate(lastEndSerial), Now) < QUICK_RESTART_MIN Then quickRestart = True
    End If
    On Error GoTo 0
    LogPreviousShutdown prevLastLine, lastEndSerial
    If quickRestart Then
        modDebugLog.DInfo "Session", "Redemarrage rapide (< " & _
            QUICK_RESTART_MIN & " min) : rattrapages couleurs/envoyes/archivage sautes"
    End If

    ' v10.20 (F10-14) : demarrage prudent apres SAFE_START_AFTER arrets
    ' anormaux de suite (taches de demarrage non armees, evenements gardes).
    Dim safeStart As Boolean
    safeStart = RegisterStartupAttempt()

    Set mSentItems = Application.GetNamespace("MAPI").GetDefaultFolder(olFolderSentMail).items

    ' v10.12 : alertes nouveau mail par boite (watchers ItemAdd sur les
    ' boites de reception configurees). Sans effet si InboxAlertEnabled=0.
    On Error Resume Next
    modInboxAlert.InboxAlertStart
    On Error GoTo 0

    ' v10.16 : canal de commandes du classeur Suivi des mails.
    BindCommandChannel

    ' v10.20 (F6-11) : etat MEMORISE de la question d'etiquetage automatique
    ' (registre, modProjectTag.AutoAskEnabled), lu a chaque evenement par
    ' CheckAssignment / CheckReTag ; PROJECT_AUTO_ASK n'en est que le reflet.
    ' (Avant : remis a True a chaque demarrage - un arret ne tenait pas.)
    modProjectTag.PROJECT_AUTO_ASK = modProjectTag.AutoAskEnabled()

    Dim calFolder As Outlook.MAPIFolder
    Set calFolder = Application.Session.GetDefaultFolder(olFolderCalendar)
    Set CalendarItems = calFolder.items
    bTaggingBusy = False
    bCalendarEventsSuspended = False
    mCalCurrentEID = ""

    ' v10.20 (F1-03) : ecoute Inspectors / volet de lecture creee AVANT les
    ' watchers de fenetres (meme ordre que modDevTools.DoReinit) : le role
    ' courrier y relie l'ecoute du volet de lecture (RebindInlineExplorer),
    ' sur la vraie fenetre courrier et non sur la fenetre active.
    modNewMeetingDefault.InitEvents

    ' v10 : une fenetre COURRIER et une fenetre CALENDRIER a role stable, chacune
    ' surveillee par un watcher. v10.20 (F1-03) : chaque fenetre ouverte recoit
    ' le role de ce qu'elle affiche (avant : la fenetre active devenait la
    ' fenetre courrier, meme si Outlook s'ouvrait sur le calendrier). La
    ' fenetre manquante sera creee (et surveillee) au premier basculement.
    AttachRoleWatchers

    modReplyInPlace.EnsureInit

    mQuickMovePromptBusy = False
    Set mPendingSentEntryIDs = New Collection

    ' === NIVEAU DE LOG PILOTE PAR config.ini (cle DefaultLogLevel) ===
    ' Le niveau est applique par modConfig.LoadConfig au demarrage
    ' (defaut INFO = 1, pour conserver le comportement historique).
    ' Pour changer durablement : editer DefaultLogLevel dans config.ini
    ' puis Maintenance > Configuration > Recharger.
    ' Pour un ajustement temporaire en session : fenetre Execution VBE ->
    ' modDebugLog.LogDebugMode / LogTraceMode / LogOff (perdu au prochain
    ' redemarrage si pas mis dans config.ini).
    ' (Ancien appel en dur "modDebugLog.LogInfoMode" supprime : la valeur
    '  vient desormais de la config.)

    ' v10.20 : taches differees du demarrage armees par ArmStartupTasks
    ' (aussi appele par ResumeStartupTasks), sauf demarrage prudent (F10-14).
    If safeStart Then
        modDebugLog.DError "Session", "Demarrage prudent : " & SAFE_START_AFTER & _
            " arrets anormaux de suite -> taches de demarrage suspendues (ResumeStartupTasks pour les relancer)"
        modUi.Toast T("D~emarrage prudent : t~Aches de d~emarrage suspendues apr~fs ", _
                      "Safe start: startup tasks suspended after ") & SAFE_START_AFTER & _
                    T(" arr~xts anormaux d'Outlook.", " abnormal Outlook stops.") & vbLf & _
                    T("Pour les relancer : macro ResumeStartupTasks (Alt+F8).", _
                      "To restart them: macro ResumeStartupTasks (Alt+F8)."), uiWarn, True
    Else
        ArmStartupTasks quickRestart
    End If

    ' v10.20 (F10-14) : la session qui vit encore STARTUP_CONFIRM_SEC
    ' secondes confirme le demarrage (compteur remis a zero).
    On Error Resume Next
    modDefer.ArmUnique "StartupConfirm", STARTUP_CONFIRM_SEC
    On Error GoTo 0

    ' v4 (2026-06-10) : health-check des bindings WithEvents. Silencieux si
    ' tout est OK ; MsgBox d'alerte sinon (les bindings morts non detectes
    ' etaient la 1re cause des regressions "la fonction ne marche plus").
    ' v10.20 (F1-07) : seulement si une fenetre courrier existe. Outlook
    ' lance sans fenetre (par un classeur) ou ouvert sur le calendrier : les
    ' liaisons de la fenetre courrier sont legitimement absentes (elles se
    ' font a l'ouverture de la fenetre, OnFirstExplorer / premier
    ' basculement) - plus de boite d'alerte cachee au demarrage.
    If mMailExplorer Is Nothing Then
        modDebugLog.DInfo MOD_NAME, "Demarrage sans fenetre courrier : controle des liaisons reporte"
    Else
        On Error Resume Next
        modMaintenance.AssertBindings True
        On Error GoTo 0
    End If

    modDebugLog.DInfo MOD_NAME, "Application_Startup termine"
End Sub

'=============================================================================
' ArmStartupTasks (v10.20) - arme les taches differees du demarrage (cles
' modDefer). Appele par Application_Startup (sauf demarrage prudent, F10-14)
' et par ResumeStartupTasks. quickRestart : rattrapages couleurs / envoyes /
' archivage sautes (redemarrage rapide, v10.10). ArmUnique partout : un
' second appel (ResumeStartupTasks) n'empile pas les minuteries. Delais
' courts (tous sous ~4 min, regle d'edition VBE : PrepareForEdit avant toute
' edition dans les premieres minutes).
'=============================================================================
Private Sub ArmStartupTasks(ByVal quickRestart As Boolean)
    On Error Resume Next

    ' v10.12 (2026-09-15) : guide de demarrage rapide, propose +8 s apres le
    ' demarrage si aucun config.ini n'existe (conditions dans
    ' modQuickStart.AutoWanted : marqueur registre "never" / "done" / "later
    ' aujourd'hui" respectes). Jamais de boite depuis Application_Startup.
    If modQuickStart.AutoWanted() Then modDefer.ArmUnique "QuickStart", 8

    ' Rattrapage des RDV synchronises pendant qu'Outlook etait ferme :
    ' acceptation iOS sans Outlook PC ouvert -> aucun ItemAdd/Change
    ' n'est emis. Scan +15 s pour laisser la synchro initiale OST se
    ' terminer avant de balayer les RDV non colores.
    If Not quickRestart Then modDefer.ArmUnique "CatchUpColor", 15

    ' v3.1 (2026-05-28) : rappel des suivis de mail en retard, differe +20 s
    ' (apres le rattrapage coloration, pour ne pas empiler les gels au
    ' demarrage). Garde-fou "un rappel par jour" cote module.
    modDefer.ArmUnique "RemindOverdue", 20

    ' v8 (2026-06-14) : rattrapage des courriels ENVOYES depuis iOS pendant
    ' qu'Outlook etait ferme (la synchro differee n'emet aucun ItemAdd).
    ' +25 s : apres les autres rattrapages. Range ceux dont le fil est deja
    ' classe ; un seul toast resume a la fin.
    If Not quickRestart Then modDefer.ArmUnique "SentCatchUp", 25

    ' v10.9 : le prechargement du cache projets (InitProjectTag) ouvrait
    ' Excel + le referentiel en SYNCHRONE a t=0, sur le chemin critique du
    ' lancement (1-3 s). Differe a +45 s via modDefer ("TagPreload"). Un tag
    ' dans les 45 premieres secondes charge le cache a la demande.
    modDefer.ArmUnique "TagPreload", 45

    ' v10.14 (pack v4) : surveillance des creations / renommages /
    ' suppressions de dossiers (liste QuickMove a jour sans re-scan).
    modDefer.ArmUnique "QMWatch", 60

    ' v5 (2026-06-10) : archivage differe a +2 min (P1). AutoArchiveMove
    ' parcourait toute l'arborescence configuree de maniere SYNCHRONE au
    ' demarrage : Outlook est utilisable immediatement, l'archivage tourne
    ' 2 minutes plus tard.
    If Not quickRestart Then modDefer.ArmUnique "AutoArchive", 120

    ' v10.20 (F3-16) : retour des mails reportes (_Snoozed, toutes les
    ' boites) dont le rappel a ete rate ou ferme, au creneau de l'ancienne
    ' purge mensuelle des drapeaux (supprimee : F1-06, choix B). Toast
    ' seulement si un mail revient.
    modDefer.ArmUnique "SnoozeCatchUp", 180

    ' v10.9 : backup mensuel differe a +210 s (la 1re session du mois
    ' copiait OTM + classeurs + registre en SYNCHRONE au demarrage). Garde
    ' mensuelle interne : les autres jours, l'appel differe ressort
    ' immediatement.
    modDefer.ArmUnique "MonthlyBackup", 210

    ' Pack v3 (C6) / v10.14 (pack v4) : reindexation QuickMove seulement si
    ' le dernier parcours COMPLET a plus de 7 jours (ou n'a jamais eu lieu),
    ' PAR PAQUETS (Outlook reste utilisable), a +240 s : apres les autres
    ' rattrapages.
    If modQuickMove.CacheNeedsRefresh() Then modDefer.ArmUnique "Reindex", 240

    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' RegisterStartupAttempt (v10.20, F10-14) - compte ce demarrage parmi les
' demarrages non confirmes (registre Toolkit\Session\PendingStarts) et
' renvoie True si les SAFE_START_AFTER demarrages precedents n'ont ete ni
' fermes proprement (Application_Quit) ni confirmes (StartupConfirm) :
' Outlook s'est arrete en boucle peu apres son lancement.
'-----------------------------------------------------------------------------
Private Function RegisterStartupAttempt() As Boolean
    On Error Resume Next
    RegisterStartupAttempt = False
    Dim prevCount As Long
    prevCount = 0
    prevCount = CLng(Val(GetSetting("Toolkit", "Session", "PendingStarts", "0")))
    If prevCount < 0 Then prevCount = 0
    SaveSetting "Toolkit", "Session", "PendingStarts", CStr(prevCount + 1)
    If prevCount > 0 Then
        modDebugLog.DInfo "Session", prevCount & " demarrage(s) precedent(s) non confirme(s) (arret avant " & _
            STARTUP_CONFIRM_SEC & " s, sans fermeture propre)"
    End If
    RegisterStartupAttempt = (prevCount >= SAFE_START_AFTER)
End Function

'-----------------------------------------------------------------------------
' LogPreviousShutdown (v10.20, F10-03 / F1-13) - dit au journal comment la
' session precedente s'est terminee. La marque LastEnd est posee par
' Application_Quit juste AVANT la ligne "SESSION END" (meme seconde) :
'   - marque presente, derniere ligne du journal a la marque (2 s pres) :
'     fermeture propre ;
'   - marque presente mais derniere ligne plus ancienne : fermeture propre
'     dont la ligne "SESSION END" n'a pas ete ecrite ;
'   - pas de marque : arret anormal (plantage, arret force, arret de
'     Windows sans fermer Outlook), heure de la derniere ligne ecrite.
'-----------------------------------------------------------------------------
Private Sub LogPreviousShutdown(ByVal prevLastLine As Date, ByVal lastEndSerial As Double)
    On Error Resume Next
    If lastEndSerial > 0 Then
        If prevLastLine >= CDate(lastEndSerial) - TimeSerial(0, 0, 2) Then
            modDebugLog.DInfo "Session", "Fermeture precedente : propre"
        Else
            modDebugLog.DInfo "Session", "Fermeture precedente : propre (non journalisee)"
        End If
    ElseIf prevLastLine = 0 Then
        modDebugLog.DInfo "Session", "Fermeture precedente : inconnue (journal vide)"
    Else
        modDebugLog.DInfo "Session", "Fermeture precedente : anormale (derniere ligne du journal a " & _
            Format$(prevLastLine, "yyyy-mm-dd hh:nn:ss") & ")"
    End If
End Sub

' apres une remise a zero du projet VBA, la ligne "SESSION END" n'est donc
' plus perdue (F10-03). La fermeture propre confirme aussi le demarrage
' (compteur du demarrage prudent remis a zero, F10-14).
'=============================================================================
Private Sub Application_Quit()
    On Error Resume Next
    ' v10.10 : horodatage de fin de session pour la detection de
    ' redemarrage rapide (Str$ : point decimal independant de la locale).
    SaveSetting "Toolkit", "Session", "LastEnd", Str$(CDbl(Now))
    SaveSetting "Toolkit", "Session", "PendingStarts", "0"
    modDebugLog.DLogSection "SESSION END (fermeture propre)"
End Sub

Public Sub RebindEvents()
    Set mSentItems = Application.GetNamespace("MAPI").GetDefaultFolder(olFolderSentMail).items

    Dim calFolder As Outlook.MAPIFolder
    Set calFolder = Application.Session.GetDefaultFolder(olFolderCalendar)
    Set CalendarItems = calFolder.items
    bTaggingBusy = False
    bCalendarEventsSuspended = False
    mCalCurrentEID = ""

    ' v10 : reconstruire les watchers de fenetres a role. Apres un reset
    ' d'etat (Ctrl+S VBE) la collection et les WithEvents sont morts.
    ' v10.20 (F1-03) : chaque fenetre ouverte reprend le role de ce qu'elle
    ' affiche (avant : la fenetre ACTIVE devenait la fenetre courrier ; si
    ' c'etait le calendrier, la vraie fenetre courrier perdait l'ecoute du
    ' volet de lecture et une troisieme fenetre finissait par s'ouvrir).
    AttachRoleWatchers
    If mPendingSentEntryIDs Is Nothing Then Set mPendingSentEntryIDs = New Collection

    ' v10.20 (F6-05) : evenements calendrier encore en file -> rejeu.
    On Error Resume Next
    If Not mCalPending Is Nothing Then
        If mCalPending.count > 0 Then modDefer.ArmUnique "CalReplay", CAL_REPLAY_DELAY_SEC
    End If
    On Error GoTo 0

    ' v3 (2026-05-27) : invalider le cache des EntryID systeme dans
    ' modReplyInPlace. Il sera repeuple au prochain appel d'EnsureInit.
    On Error Resume Next
    modReplyInPlace.InvalidateSysCache
    On Error GoTo 0

    ' v10.12 : rearmer les watchers d'alertes nouveau mail (WithEvents perdus).
    On Error Resume Next
    modInboxAlert.InboxAlertStart
    On Error GoTo 0

    ' v10.14 (pack v4) : surveillants de dossiers QuickMove (WithEvents
    ' perdus aussi a chaque reinitialisation du projet VBA).
    On Error Resume Next
    modQuickMove.FolderWatchStart
    On Error GoTo 0

    ' v10.16 : canal de commandes (WithEvents perdu aussi).
    BindCommandChannel

    modDebugLog.DInfo MOD_NAME, "RebindEvents OK"
End Sub

Private Sub Application_Reminder(ByVal Item As Object)
    On Error Resume Next

    ' v6 : point de passage watchdog.
    CheckBindingsWatchdog

    ' Pack v3 (C6) : un eventuel RDV technique "QuickMove_BackgroundReindex"
    ' herite d'une version anterieure est simplement supprime (la
    ' reindexation passe par modDefer "Reindex").
    If TypeOf Item Is Outlook.AppointmentItem Then
        If Item.subject = "QuickMove_BackgroundReindex" Then
            Item.Delete
            Exit Sub
        End If
    End If

    If TypeOf Item Is Outlook.mailItem Then
        Dim handled As Boolean
        handled = modSnooze.HandleSnoozeReminder(Item)
    End If

    On Error GoTo 0
End Sub

' COURRIER ET CALENDRIER DANS DEUX FENETRES A ROLE STABLE  (v10 - 2026-06-16)
'
' Principe : chaque fenetre a role (courrier / calendrier) est surveillee par
' une instance de clsExplorerWatcher. Quand une fenetre derive hors de son
' role (clic sur le module de l'autre type), HandleExplorerSwitch la rapatrie
' sur son dossier et active/cree la fenetre de l'autre role. Bilan : une
' fenetre porte toujours le courrier, une autre toujours le calendrier.
'
' Anti-rebond (v10.2) : SelectFolder ET Activate relancent chacun un
' FolderSwitch, mais APRES la fin de cette procedure (au cycle de message
' suivant). Un drapeau synchrone ne les couvre donc pas. On utilise une GARDE
' TEMPORELLE : pendant SWITCH_GUARD_MS apres une intervention, tout
' FolderSwitch est ignore, le temps qu'Outlook emette et epuise ses rebonds.
'
' Ordre des operations (v10.2) : on rapatrie d'ABORD la fenetre qui a derive
' sur son propre dossier, et on active la fenetre de l'autre role EN DERNIER.
' Ainsi le dernier geste de focus est le bon ; sinon les deux fenetres se
' disputent le premier plan (symptome : le courrier passe devant puis
' "disparait").
'
' Fenetres libres : une fenetre ouverte volontairement par l'utilisateur (clic
' droit dossier -> "Ouvrir dans une nouvelle fenetre") n'a PAS de watcher,
' donc n'est jamais interceptee.
'=============================================================================

'-----------------------------------------------------------------------------
' HandleExplorerSwitch - appele par un watcher quand SA fenetre change de
' dossier. role = role de la fenetre qui a switche ("mail" ou "calendar").
'-----------------------------------------------------------------------------
Public Sub HandleExplorerSwitch(ByVal exp As Outlook.Explorer, ByVal role As String)
    If CAL_SEPARATE_WINDOW_ENABLED <> 1 Then Exit Sub
    If exp Is Nothing Then Exit Sub

    ' Garde temporelle : on ignore les FolderSwitch declenches par nos propres
    ' SelectFolder/Activate (rebonds) pendant la fenetre de garde.
    If SwitchGuardActive() Then Exit Sub

    On Error GoTo CleanSwitch

    Dim switchedFolder As Outlook.folder
    Set switchedFolder = exp.CurrentFolder
    If switchedFolder Is Nothing Then Exit Sub

    Dim isCalendarFolder As Boolean
    isCalendarFolder = (switchedFolder.DefaultItemType = olAppointmentItem)

    If role = ROLE_MAIL Then
        ' Fenetre courrier : on n'intervient QUE si elle derive vers le
        ' calendrier. Toute autre navigation courrier (entre dossiers de mail)
        ' est laissee libre.
        If Not isCalendarFolder Then Exit Sub

        ArmSwitchGuard                              ' couvre le rapatriement + rebonds
        SendExplorerToMail exp                      ' rapatrie CETTE fenetre sur le courrier
        ' Activation de la fenetre calendrier DIFFEREE (~300 ms) : si on
        ' l'active en synchrone, le rapatriement ci-dessus (mis en file par
        ' Outlook) reprend le focus juste apres et annule l'activation -
        ' surtout en fenetres superposees sur le meme ecran. Differer garantit
        ' que l'activation est le DERNIER geste, apres l'application du
        ' rapatriement et des rebonds. Voir ActivateRoleDeferred.
        modDefer.ArmMs "ActivateRole", 300, ROLE_CALENDAR
        Exit Sub

    ElseIf role = ROLE_CALENDAR Then
        ' Fenetre calendrier : on n'intervient QUE si elle derive HORS du
        ' calendrier (clic Message / Contacts / Taches...). Rester dans le
        ' calendrier (changer de vue, de date) est laisse libre.
        If isCalendarFolder Then Exit Sub

        ArmSwitchGuard                              ' couvre le rapatriement + rebonds
        SendExplorerToCalendar exp                  ' rapatrie CETTE fenetre sur le calendrier
        ' Activation de la fenetre courrier DIFFEREE (cf. commentaire ci-dessus).
        modDefer.ArmMs "ActivateRole", 300, ROLE_MAIL
        Exit Sub
    End If

    Exit Sub

CleanSwitch:
End Sub

'-----------------------------------------------------------------------------
' ActivateRoleDeferred - appele par modDefer (cle "ActivateRole") ~300 ms
' apres un rapatriement, pour activer (ou creer) la fenetre du role cible.
' Differe expres : en synchrone, le rapatriement reprend le focus et annule
' l'activation (fenetres superposees, meme ecran). Ici le rapatriement et les
' rebonds sont deja appliques, donc cette activation est le dernier geste et
' tient. On re-arme la garde car Activate peut declencher un FolderSwitch.
' argText = ROLE_MAIL ou ROLE_CALENDAR.
'-----------------------------------------------------------------------------
Public Sub ActivateRoleDeferred(ByVal targetRole As String)
    On Error Resume Next
    ArmSwitchGuard

    If targetRole = ROLE_MAIL Then
        EnsureMailExplorer
    ElseIf targetRole = ROLE_CALENDAR Then
        EnsureCalendarExplorer
    End If

    ' v10.5 : rollback v10.4 - plus de remontee forcee par titre exact
    ' (fragile : HWND introuvable des que le titre reel differe de
    ' Explorer.Caption ; fenetre minimisee jamais restauree). Ensure* fait
    ' deja Explorer.Activate ; on re-arme la garde car Activate peut
    ' declencher un FolderSwitch.
    ArmSwitchGuard
    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' Garde temporelle anti-rebond (v10.2).
'-----------------------------------------------------------------------------
Private Sub ArmSwitchGuard()
    On Error Resume Next
    mSwitchGuardUntilMs = modCommon.TickMs() + SWITCH_GUARD_MS
End Sub

Private Function SwitchGuardActive() As Boolean
    On Error Resume Next
    ' v10.20 (F10-05) : compteur 64 bits (modCommon.TickMs) : ne reboucle
    ' jamais. Avant : GetTickCount 32 bits signe, negatif apres 24,9 jours
    ' d'allumage du PC -> garde toujours active toute la session (echeance
    ' remise a 0), fenetre calendrier separee et vue anniversaires muettes.
    SwitchGuardActive = (modCommon.TickMs() < mSwitchGuardUntilMs)
End Function

'-----------------------------------------------------------------------------
' SendExplorerToMail - remet une fenetre sur la boite de reception et bascule
' son volet de navigation sur le module Courrier.
'-----------------------------------------------------------------------------
Private Sub SendExplorerToMail(ByVal exp As Outlook.Explorer)
    On Error Resume Next
    exp.SelectFolder Application.Session.GetDefaultFolder(olFolderInbox)
    Dim np As Outlook.NavigationPane
    Set np = exp.NavigationPane
    If Not np Is Nothing Then
        Set np.CurrentModule = np.Modules.GetNavigationModule(olModuleMail)
    End If
    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' SendExplorerToCalendar - remet une fenetre sur le calendrier par defaut et
' bascule son volet de navigation sur le module Calendrier.
'-----------------------------------------------------------------------------
Private Sub SendExplorerToCalendar(ByVal exp As Outlook.Explorer)
    On Error Resume Next
    exp.SelectFolder Application.Session.GetDefaultFolder(olFolderCalendar)
    Dim np As Outlook.NavigationPane
    Set np = exp.NavigationPane
    If Not np Is Nothing Then
        Set np.CurrentModule = np.Modules.GetNavigationModule(olModuleCalendar)
    End If
    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' EnsureCalendarExplorer - garantit une fenetre calendrier vivante :
'   - si mCalendarExplorer existe encore, l'active. v10.20 (F1-11) : sans
'     reappliquer la position memorisee - une fenetre deplacee ou agrandie
'     pendant la session garde sa place et sa taille ;
'   - sinon en cree une, lui applique la position memorisee (si elle tombe
'     sur un ecran branche) et l'attache a un watcher de role "calendar".
'-----------------------------------------------------------------------------
Private Sub EnsureCalendarExplorer()
    On Error GoTo ErrCreate

    If IsExplorerAlive(mCalendarExplorer) Then
        mCalendarExplorer.Activate
        ArmBirthdayViewSetup
        Exit Sub
    End If

    Set mCalendarExplorer = Nothing

    Dim calFolder As Outlook.folder
    Set calFolder = Application.Session.GetDefaultFolder(olFolderCalendar)

    mCreatingRoleWindow = True                       ' v10.20 (F1-07) : NewExplorer de CETTE fenetre ignore
    Set mCalendarExplorer = Application.Explorers.Add(calFolder, olFolderDisplayNormal)
    mCalendarExplorer.Display
    DoEvents
    ApplyRememberedPosition mCalendarExplorer        ' position memorisee : calendrier UNIQUEMENT, a la creation
    mCalendarExplorer.Activate
    AttachWatcher mCalendarExplorer, ROLE_CALENDAR
    mCreatingRoleWindow = False
    ArmBirthdayViewSetup
    Exit Sub

ErrCreate:
    mCreatingRoleWindow = False
    modDebugLog.DError MOD_NAME, "EnsureCalendarExplorer err " & Err.Number & " - " & Err.Description
    Set mCalendarExplorer = Nothing
End Sub

'-----------------------------------------------------------------------------
' ArmBirthdayViewSetup (v10.13) - arme la mise en place DIFFEREE de la vue
' calendrier des anniversaires (calendrier dans "Mes calendriers", coche, en
' superposition, les autres decoches, vue Semaine). Pilote par config.ini
' (BirthdayCalendarAutoShow) ; garde cote modContactBirthday : sans effet une
' fois la vue VERIFIEE. Pourquoi differer : juste apres Explorers.Add/Display
' la fenetre n'est pas construite, son volet n'est pas le vrai, et Outlook
' recharge ses calendriers coches apres coup (symptome : coche qui flashe
' cote courrier puis fenetre calendrier decochee). ArmUniqueMs : un seul
' timer en attente meme si Ensure* est appele plusieurs fois de suite.
' NE PEUT PAS se faire dans Application_Startup : NavigationFolder.IsSelected
' n'est modifiable que si le module Calendrier est le module COURANT du volet.
'-----------------------------------------------------------------------------
Private Sub ArmBirthdayViewSetup()
    On Error Resume Next
    If mCalendarExplorer Is Nothing Then Exit Sub
    If Not modContactBirthday.ContactBirthdayViewPending() Then Exit Sub
    modDefer.ArmUniqueMs "BirthdayView", BIRTHDAY_VIEW_DELAY_MS, "1"
    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' SetupBirthdayViewDeferred (v10.13) - cible modDefer (cle "BirthdayView").
' Applique la vue sur mCalendarExplorer (deux fenetres a role : ActiveExplorer
' serait ambigu), puis, si modContactBirthday n'a pas pu VERIFIER le coche,
' re-arme une tentative +BIRTHDAY_VIEW_RETRY_MS (plafond cote module). La
' garde anti-rebond est re-armee : CurrentModule peut emettre un FolderSwitch.
' Silencieux, aucune erreur visible.
'-----------------------------------------------------------------------------
Public Sub SetupBirthdayViewDeferred(ByVal attemptText As String)
    On Error Resume Next
    If Not IsExplorerAlive(mCalendarExplorer) Then Exit Sub
    If Not modContactBirthday.ContactBirthdayViewPending() Then Exit Sub

    ArmSwitchGuard
    modContactBirthday.ContactBirthdaySetupView mCalendarExplorer
    If Err.Number <> 0 Then
        modDebugLog.DDebug MOD_NAME, "SetupBirthdayViewDeferred err " & Err.Number
        Err.Clear
    End If
    ArmSwitchGuard

    If modContactBirthday.ContactBirthdayViewPending() Then
        Dim nextAttempt As Long
        nextAttempt = Val(attemptText) + 1
        modDebugLog.DDebug MOD_NAME, "Vue anniversaires non confirmee, re-armement (tentative " & nextAttempt & ")"
        modDefer.ArmUniqueMs "BirthdayView", BIRTHDAY_VIEW_RETRY_MS, CStr(nextAttempt)
    End If
    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' AttachRoleWatchers (v10.20, F1-03) - repart de zero (demarrage,
' RebindEvents) : chaque fenetre ouverte recoit le role de ce qu'elle
' affiche. La fenetre ACTIVE d'abord (comportement historique quand elle
' affiche le courrier), puis les autres fenetres pour le role encore libre.
' Une fenetre calendrier n'est plus prise pour la fenetre courrier. Aucune
' fenetre lisible : repli historique (fenetre active = courrier ; sans
' fenetre du tout, rien n'est attache).
'-----------------------------------------------------------------------------
Private Sub AttachRoleWatchers()
    On Error Resume Next
    Set mExplorerWatchers = New Collection
    mExplWatchSeq = 0
    Set mMailExplorer = Nothing
    Set mCalendarExplorer = Nothing
    mSwitchGuardUntilMs = 0

    Dim act As Outlook.Explorer
    Set act = Nothing
    Set act = Application.ActiveExplorer
    If Not act Is Nothing Then AttachByContent act

    Dim ex As Outlook.Explorer
    For Each ex In Application.Explorers
        If mMailExplorer Is Nothing Or mCalendarExplorer Is Nothing Then
            If Not ex Is act Then AttachByContent ex
        End If
    Next ex
    Err.Clear

    If mMailExplorer Is Nothing And mCalendarExplorer Is Nothing Then
        AttachWatcher act, ROLE_MAIL                 ' repli historique (act = Nothing : sans effet)
    End If
    modDebugLog.DDebug MOD_NAME, "Fenetres a role : courrier=" & (Not mMailExplorer Is Nothing) & _
        ", calendrier=" & (Not mCalendarExplorer Is Nothing)
    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' AttachByContent (v10.20) - donne a exp le role de ce qu'il affiche
' (calendrier ou courrier), si ce role est encore libre.
'-----------------------------------------------------------------------------
Private Sub AttachByContent(ByVal exp As Outlook.Explorer)
    On Error Resume Next
    If exp Is Nothing Then Exit Sub
    Dim isCal As Boolean
    isCal = False
    isCal = (exp.CurrentFolder.DefaultItemType = olAppointmentItem)
    Err.Clear
    If isCal Then
        If mCalendarExplorer Is Nothing Then AttachWatcher exp, ROLE_CALENDAR
    Else
        If mMailExplorer Is Nothing Then AttachWatcher exp, ROLE_MAIL
    End If
End Sub

'-----------------------------------------------------------------------------
' OnFirstExplorer (v10.20, F1-07) - appele par clsAppEvents a chaque nouvelle
' fenetre (Explorers.NewExplorer). N'agit que si AUCUNE fenetre a role
' n'existe : cas d'Outlook lance sans fenetre par un classeur, puis fenetre
' ouverte par l'utilisateur. La fenetre recoit le role de ce qu'elle affiche
' et, en role courrier, l'ecoute du volet de lecture (reponse rangee,
' salutation, Cci masque) y est reliee par AttachWatcher. Ignore les
' fenetres creees par Ensure*Explorer et les fenetres ouvertes a la main
' quand les roles existent deja (elles restent libres).
'-----------------------------------------------------------------------------
Public Sub OnFirstExplorer(ByVal exp As Outlook.Explorer)
    On Error Resume Next
    If mCreatingRoleWindow Then Exit Sub
    If exp Is Nothing Then Exit Sub
    If Not WatcherForRole(ROLE_MAIL) Is Nothing Then Exit Sub
    If Not WatcherForRole(ROLE_CALENDAR) Is Nothing Then Exit Sub
    AttachByContent exp
    If Not mMailExplorer Is Nothing Then
        modDebugLog.DInfo MOD_NAME, "Premiere fenetre de la session : role courrier attribue"
    ElseIf Not mCalendarExplorer Is Nothing Then
        modDebugLog.DInfo MOD_NAME, "Premiere fenetre de la session : role calendrier attribue"
    End If
End Sub

'-----------------------------------------------------------------------------
' AttachWatcher - cree un clsExplorerWatcher sur une fenetre, le memorise dans
' mExplorerWatchers (sinon GC) et met a jour la reference de role.
' Tolere exp = Nothing (cas demarrage sans fenetre active).
'-----------------------------------------------------------------------------
Private Sub AttachWatcher(ByVal exp As Outlook.Explorer, ByVal role As String)
    On Error Resume Next
    If exp Is Nothing Then
        modDebugLog.DTrace MOD_NAME, "AttachWatcher : pas de fenetre (role=" & role & ")"
        Exit Sub
    End If
    If mExplorerWatchers Is Nothing Then Set mExplorerWatchers = New Collection

    mExplWatchSeq = mExplWatchSeq + 1
    Dim key As String
    key = "exp_" & mExplWatchSeq

    Dim w As clsExplorerWatcher
    Set w = New clsExplorerWatcher
    w.Init exp, role, key
    mExplorerWatchers.Add w, key

    If role = ROLE_MAIL Then
        Set mMailExplorer = exp
        ' Pack v3 (A5) : clsAppEvents ecoutait l'InlineResponse de la SEULE
        ' fenetre active a l'init. Une fenetre courrier fermee puis recreee
        ' (EnsureMailExplorer) perdait ReplyInPlace inline, ReplyGreeting et
        ' HideBcc jusqu'au prochain Reinit. On re-lie a chaque attache.
        modNewMeetingDefault.RebindInlineExplorer exp
    ElseIf role = ROLE_CALENDAR Then
        Set mCalendarExplorer = exp
    End If

    modDebugLog.DTrace MOD_NAME, "Watcher attache (role=" & role & ", key=" & key & ")"
    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' OnExplorerWatcherClosed - appele par un watcher quand SA fenetre se ferme.
' On retire le watcher de la collection et on oublie la reference de role.
'-----------------------------------------------------------------------------
Public Sub OnExplorerWatcherClosed(ByVal key As String, ByVal role As String)
    On Error Resume Next
    If Not mExplorerWatchers Is Nothing Then mExplorerWatchers.Remove key

    If role = ROLE_MAIL Then
        Set mMailExplorer = Nothing
    ElseIf role = ROLE_CALENDAR Then
        Set mCalendarExplorer = Nothing
    End If

    modDebugLog.DTrace MOD_NAME, "Watcher libere (role=" & role & ", key=" & key & ")"
    On Error GoTo 0
End Sub

'-----------------------------------------------------------------------------
' WatcherForRole - renvoie le watcher VIVANT du role demande, ou Nothing.
' Nettoie au passage les watchers dont la fenetre est morte.
'-----------------------------------------------------------------------------
Private Function WatcherForRole(ByVal role As String) As clsExplorerWatcher
    On Error Resume Next
    Set WatcherForRole = Nothing
    If mExplorerWatchers Is Nothing Then Exit Function

    Dim w As clsExplorerWatcher
    Dim i As Long
    For i = mExplorerWatchers.count To 1 Step -1
        Set w = mExplorerWatchers.Item(i)
        If w Is Nothing Then
            mExplorerWatchers.Remove i
        ElseIf Not w.IsAlive Then
            mExplorerWatchers.Remove i
        ElseIf w.Role = role Then
            Set WatcherForRole = w
            Exit Function
        End If
    Next i
    On Error GoTo 0
End Function

'-----------------------------------------------------------------------------
' IsExplorerAlive - True si l'Explorer reference est encore ouvert (la lecture
' d'une propriete sur une fenetre fermee leve une erreur).
'-----------------------------------------------------------------------------
Private Function IsExplorerAlive(ByVal exp As Outlook.Explorer) As Boolean
    On Error Resume Next
    IsExplorerAlive = False
    If exp Is Nothing Then Exit Function
    Dim dummy As Long
    dummy = exp.Left
    If Err.Number = 0 Then IsExplorerAlive = True
    Err.Clear
    On Error GoTo 0
End Function


'-----------------------------------------------------------------------------
' ApplyRememberedPosition - position memorisee (RememberCalendarPosition)
' appliquee a la fenetre calendrier qui vient d'etre creee (v10.20, F1-11 :
' plus a chaque reactivation). Une position qui tomberait hors des ecrans
' branches (memorisee sur l'ecran externe, portable seul) est ignoree : la
' fenetre garde la place donnee par Outlook (choix B).
'-----------------------------------------------------------------------------
Private Sub ApplyRememberedPosition(exp As Outlook.Explorer)
    On Error GoTo ErrPos

    Dim sL As String, sT As String, sW As String, sH As String
    sL = GetSetting("Outlook_Calendrier", "Position", "Left", "")
    sT = GetSetting("Outlook_Calendrier", "Position", "Top", "")
    sW = GetSetting("Outlook_Calendrier", "Position", "Width", "")
    sH = GetSetting("Outlook_Calendrier", "Position", "Height", "")

    If Len(sL) = 0 Or Len(sT) = 0 Or Len(sW) = 0 Or Len(sH) = 0 Then Exit Sub

    If Not modNotify.IsRectVisibleOnScreen(CLng(sL), CLng(sT), CLng(sW), CLng(sH)) Then
        modDebugLog.DInfo MOD_NAME, "Position memorisee du calendrier hors ecran : ignoree"
        Exit Sub
    End If

    exp.WindowState = olNormalWindow
    exp.Left = CLng(sL)
    exp.Top = CLng(sT)
    exp.Width = CLng(sW)
    exp.Height = CLng(sH)
    Exit Sub
ErrPos:
End Sub

Public Sub RememberCalendarPosition()
    On Error GoTo ErrMem

    Dim expToSave As Outlook.Explorer
    Set expToSave = Nothing

    If Not mCalendarExplorer Is Nothing Then
        On Error Resume Next
        Dim testLeft As Long
        testLeft = mCalendarExplorer.Left
        If Err.Number = 0 Then Set expToSave = mCalendarExplorer
        Err.Clear
        On Error GoTo ErrMem
    End If

    If expToSave Is Nothing Then
        Dim actExp As Outlook.Explorer
        Set actExp = Application.ActiveExplorer
        If Not actExp Is Nothing Then
            If Not actExp.CurrentFolder Is Nothing Then
                If actExp.CurrentFolder.DefaultItemType = olAppointmentItem Then
                    Set expToSave = actExp
                End If
            End If
        End If
    End If

    If expToSave Is Nothing Then
        MsgBox T("Aucune fen~xtre calendrier active trouv~ee.", "No active calendar window found."), vbExclamation, T("M~emorisation position", "Save position")
        Exit Sub
    End If

    If expToSave.WindowState = olMaximized Then
        MsgBox T("La fen~xtre calendrier est maximis~ee. Restaure-la d'abord.", "The calendar window is maximized. Restore it first."), _
               vbInformation, T("M~emorisation position", "Save position")
        Exit Sub
    End If

    Dim leftV As Long, topV As Long, widthV As Long, heightV As Long
    leftV = expToSave.Left
    topV = expToSave.Top
    widthV = expToSave.Width
    heightV = expToSave.Height

    SaveSetting "Outlook_Calendrier", "Position", "Left", CStr(leftV)
    SaveSetting "Outlook_Calendrier", "Position", "Top", CStr(topV)
    SaveSetting "Outlook_Calendrier", "Position", "Width", CStr(widthV)
    SaveSetting "Outlook_Calendrier", "Position", "Height", CStr(heightV)

    ' v10.20 (F1-11) : confirmation de routine -> toast (plus de MsgBox).
    modUi.Toast T("Position de la fen~xtre calendrier m~emoris~ee.", "Calendar window position saved."), uiSuccess
    Exit Sub
ErrMem:
    MsgBox T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, vbExclamation, T("M~emorisation position", "Save position")
End Sub

Public Sub ForgetCalendarPosition()
    On Error Resume Next
    DeleteSetting "Outlook_Calendrier", "Position"
    On Error GoTo 0
    ' v10.20 (F1-11) : toast au lieu de la MsgBox de routine.
    modUi.Toast T("Position de la fen~xtre calendrier oubli~ee.", "Calendar window position cleared."), uiSuccess
End Sub

Private Function IsTechnicalEvent(ByVal Item As Object) As Boolean
    IsTechnicalEvent = False
    If Not TypeOf Item Is Outlook.AppointmentItem Then
        IsTechnicalEvent = True
        Exit Function
    End If
    Dim appt As Outlook.AppointmentItem
    Set appt = Item

    ' v10.7 : lecture des proprietes sous garde d'erreur. Un item en cours
    ' de suppression ou de synchro Exchange (typiquement le RDV technique
    ' QuickMove qui vient d'etre supprime) leve MAPI_E_NOT_FOUND (8004010F)
    ' sur .subject ou .Categories. Comme cette fonction est appelee AVANT
    ' le On Error GoTo des handlers, l'erreur remontait a l'utilisateur.
    ' Un item illisible n'a rien a faire dans le pipeline => True (skip).
    Dim subj As String, cats As String
    On Error Resume Next
    subj = appt.subject
    cats = appt.Categories
    If Err.Number <> 0 Then
        modDebugLog.DInfo MOD_NAME, "IsTechnicalEvent: item illisible (err " & _
            Err.Number & "), skip"
        Err.Clear
        On Error GoTo 0
        IsTechnicalEvent = True
        Exit Function
    End If
    On Error GoTo 0

    If InStr(1, subj, "QuickMove_", vbTextCompare) > 0 Then
        IsTechnicalEvent = True
        Exit Function
    End If
    If InStr(1, cats, "QuickMove", vbTextCompare) > 0 Then
        IsTechnicalEvent = True
        Exit Function
    End If
End Function

Private Function SubjectAlreadyTagged(ByVal subjectText As String) As Boolean
    SubjectAlreadyTagged = False
    Dim p1 As Long
    p1 = InStr(subjectText, "[")
    If p1 = 0 Then Exit Function
    Dim p2 As Long
    p2 = InStr(p1, subjectText, "]")
    If p2 > p1 + 1 Then SubjectAlreadyTagged = True
End Function

' NB (v10.6) : "ProjectAsk" est une UserProperty FONCTIONNELLE volontairement
' FIXE (hors config, contrairement a UpTag) : elle ne marque que "deja
' propose" et n'a pas besoin d'etre configurable ni migrable.
Private Function AlreadyProposed(ByVal appt As Outlook.AppointmentItem) As Boolean
    AlreadyProposed = False
    On Error Resume Next
    Dim up As Outlook.UserProperty
    Set up = appt.UserProperties.Find("ProjectAsk")
    If Not up Is Nothing Then AlreadyProposed = True
    On Error GoTo 0
End Function

Private Sub MarkProposed(ByVal appt As Outlook.AppointmentItem)
    On Error Resume Next
    Dim up As Outlook.UserProperty
    Set up = appt.UserProperties.Find("ProjectAsk")
    If up Is Nothing Then
        Set up = appt.UserProperties.Add("ProjectAsk", olText, False)
    End If
    up.Value = "1"
    appt.Save
    On Error GoTo 0
End Sub

' IsExternalSync - filtre les ItemAdd/ItemChange provoques par la synchro
' MAPI/Exchange d'invitations deja acceptees ailleurs (ex : Outlook for iOS).
'
' Probleme : quand on accepte une invitation depuis iOS, l'evenement arrive
' sur Exchange avec ResponseStatus=Accepted. Quand on ouvre Outlook Desktop
' ensuite, MAPI synchronise l'item local, ce qui declenche un ItemAdd puis
' tres souvent un ItemChange (propagation des proprietes body/MIME).
' L'ItemChange franchissait le filtre ResponseStatus=olResponseAccepted et
' declenchait la popup Projet d'affectation alors qu'il s'agit d'une simple
' convergence d'etat serveur, pas d'une action utilisateur locale.
'
' Heuristique : un RDV NON-ORGANISE dont LastModificationTime est anterieur
' de plus de 120 s au moment actuel est un sync, pas une action recente.
'  - Un clic "Accepter" depuis Desktop produit un ItemChange dans la seconde
'    qui suit, donc reste sous la marge.
'  - Un RDV organise par moi-meme : Organized => retourne False, pas d'impact
'    sur le flow normal de creation de RDV.
'  - Coloration, anti-chevauchement, private-auto continuent a tourner :
'    le filtre s'applique UNIQUEMENT autour de CheckAssignment.
'=============================================================================
Private Function IsExternalSync(ByVal appt As Outlook.AppointmentItem) As Boolean
    IsExternalSync = False
    On Error Resume Next
    If appt.ResponseStatus = olResponseOrganized Then Exit Function
    Dim ageSec As Long
    ageSec = DateDiff("s", appt.LastModificationTime, Now)
    If ageSec > 120 Then
        IsExternalSync = True
        modDebugLog.DInfo MOD_NAME, "Sync MAPI detectee (skip popup projet) '" & _
            Left(appt.subject, 60) & "' age=" & ageSec & "s status=" & appt.ResponseStatus
    End If
    On Error GoTo 0
End Function

'=============================================================================
' Evenements du calendrier (ItemAdd / ItemChange).
' v10.20 (F6-05, F1-09) : la question d'etiquetage (modProjectTag.
' CheckAssignment) est une boite modale ouverte DEPUIS le gestionnaire ;
' Outlook continue pendant ce temps de livrer les evenements des AUTRES
' rendez-vous. Ils etaient perdus (sortie sur bTaggingBusy) : ils sont
' maintenant mis en file (QueueCalendarReplay) puis rejoues par le differe
' "CalReplay" une fois la question fermee (CalendarReplayDeferred).
' Pendant SuspendCalendarEvents (boucles des modules qui enregistrent
' eux-memes des rendez-vous), les evenements restent ignores : c'est le but
' de cette suspension.
'=============================================================================
Private Sub CalendarItems_ItemAdd(ByVal Item As Object)
    If bTaggingBusy Then
        QueueCalendarReplay Item
        Exit Sub
    End If
    If bCalendarEventsSuspended Then Exit Sub
    If IsTechnicalEvent(Item) Then Exit Sub

    Dim appt As Outlook.AppointmentItem
    Set appt = Item

    modDebugLog.DTrace MOD_NAME, "CalendarItems_ItemAdd '" & appt.subject & _
        "' responseStatus=" & appt.ResponseStatus

    BeginCalendarBusy appt
    On Error GoTo CleanupAdd

    ' Garde anti-sync MAPI (cf. IsExternalSync) : si l'item arrive par
    ' synchro Exchange (invitation deja acceptee ailleurs), on ne declenche
    ' PAS la popup Projet. Le filtre olResponseOrganized ci-dessous l'attrape
    ' deja dans le cas le plus courant, IsExternalSync est une securite
    ' supplementaire pour les rares cas ou MAPI ferait remonter un add avec
    ' un status non encore stabilise.
    ' v10.20 (F6-06) : plus de MarkProposed ici. La marque "ProjectAsk" n'est
    ' relue que par ItemChange, pour une reunion ACCEPTEE : sur un rendez-vous
    ' que j'organise, elle ne valait qu'un enregistrement (et un passage
    ' complet de la chaine calendrier) de trop a chaque creation.
    If Not IsExternalSync(appt) Then
        If appt.ResponseStatus = olResponseOrganized Then
            modProjectTag.CheckAssignment appt
        End If
    End If

    modMeetingColor.ApplyCategory appt
    modAntiOverlap.TryShift appt
    modAppointmentPrivacy.ApplyIfPersonal appt
    modMeetingToAppointment.DemoteIfEmptyMeeting appt

    EndCalendarBusy
    Exit Sub

CleanupAdd:
    modDebugLog.DError MOD_NAME, "ItemAdd handler err " & Err.Number & _
        " - " & Err.Description
    EndCalendarBusy
End Sub

Private Sub CalendarItems_ItemChange(ByVal Item As Object)
    If bTaggingBusy Then
        QueueCalendarReplay Item
        Exit Sub
    End If
    If bCalendarEventsSuspended Then Exit Sub
    If IsTechnicalEvent(Item) Then Exit Sub

    Dim appt As Outlook.AppointmentItem
    Set appt = Item

    modDebugLog.DTrace MOD_NAME, "CalendarItems_ItemChange '" & appt.subject & "'"

    BeginCalendarBusy appt
    On Error GoTo Cleanup1

    modProjectTag.CheckReTag appt

    If Not SubjectAlreadyTagged(appt.subject) Then
        If Not AlreadyProposed(appt) Then
            If appt.ResponseStatus = olResponseAccepted Then
                ' Garde anti-sync MAPI : un ItemChange provoque par la
                ' synchro Exchange d'une invitation deja acceptee sur iOS
                ' (ou autre client) ne doit PAS declencher la popup Projet.
                ' Cf. IsExternalSync. C'est ICI le vrai point d'entree
                ' du bug, le filtre olResponseAccepted laissait tout passer.
                If Not IsExternalSync(appt) Then
                    modProjectTag.CheckAssignment appt
                    MarkProposed appt
                End If
            End If
        End If
    End If

    modMeetingColor.ApplyCategory appt
    modAntiOverlap.TryShift appt
    modAppointmentPrivacy.ApplyIfPersonal appt
    modMeetingToAppointment.DemoteIfEmptyMeeting appt

Cleanup1:
    If Err.Number <> 0 Then
        modDebugLog.DError MOD_NAME, "ItemChange handler err " & Err.Number & _
            " - " & Err.Description
    End If
    EndCalendarBusy
End Sub

'-----------------------------------------------------------------------------
' BeginCalendarBusy / EndCalendarBusy (v10.20, F6-05) - encadrent le
' traitement d'un rendez-vous. Begin : garde bTaggingBusy et rendez-vous en
' cours (ses propres enregistrements ne sont pas mis en file). End : garde
' levee et, si des evenements ont ete mis en file pendant ce temps, rejeu
' arme (+CAL_REPLAY_DELAY_SEC).
'-----------------------------------------------------------------------------
Private Sub BeginCalendarBusy(ByVal appt As Object)
    bTaggingBusy = True
    mCalCurrentEID = ""
    On Error Resume Next
    mCalCurrentEID = CStr(appt.entryID)
End Sub

Private Sub EndCalendarBusy()
    On Error Resume Next
    bTaggingBusy = False
    mCalCurrentEID = ""
    If Not mCalPending Is Nothing Then
        If mCalPending.count > 0 Then modDefer.ArmUnique "CalReplay", CAL_REPLAY_DELAY_SEC
    End If
End Sub

'-----------------------------------------------------------------------------
' QueueCalendarReplay (v10.20, F6-05) - met en file (sans doublon) l'EntryID
' d'un rendez-vous dont l'evenement arrive pendant le traitement d'un autre
' (question d'etiquetage ouverte). Silencieux.
'-----------------------------------------------------------------------------
Private Sub QueueCalendarReplay(ByVal Item As Object)
    On Error Resume Next
    If Not TypeOf Item Is Outlook.AppointmentItem Then Exit Sub
    Dim eid As String
    eid = ""
    eid = CStr(Item.entryID)
    If Len(eid) = 0 Then Exit Sub
    If eid = mCalCurrentEID Then Exit Sub         ' le rendez-vous en cours de traitement
    If mCalPending Is Nothing Then Set mCalPending = CreateObject("Scripting.Dictionary")
    If mCalPending.Exists(eid) Then Exit Sub
    If mCalPending.count >= CAL_REPLAY_MAX Then
        modDebugLog.DError MOD_NAME, "Rejeu calendrier : file pleine (" & CAL_REPLAY_MAX & "), evenement ignore"
        Exit Sub
    End If
    mCalPending.Add eid, True
    modDebugLog.DDebug MOD_NAME, "Evenement calendrier mis en file pendant l'etiquetage (" & mCalPending.count & " en file)"
End Sub

'=============================================================================
' CalendarReplayDeferred (v10.20, F6-05) - cible modDefer (cle "CalReplay").
' Rejoue les evenements calendrier mis en file pendant la question
' d'etiquetage. Pour chaque rendez-vous encore lisible et non technique :
' meme chaine qu'un ItemChange (re-etiquetage, couleur, decalage, prive,
' conversion d'une reunion sans invite), SANS la question d'affectation d'un
' projet (le rattrapage ne l'ouvre pas sur un rendez-vous que l'utilisateur
' n'a pas sous les yeux ; CheckReTag peut, comme a l'ItemChange, proposer
' de restaurer une etiquette perdue). Traitement encore en cours : nouvel
' essai plus tard.
'=============================================================================
Public Sub CalendarReplayDeferred()
    On Error Resume Next
    If mCalPending Is Nothing Then Exit Sub
    If mCalPending.count = 0 Then Exit Sub
    If bTaggingBusy Or bCalendarEventsSuspended Then
        modDefer.ArmUnique "CalReplay", CAL_REPLAY_DELAY_SEC
        Exit Sub
    End If

    Dim eids As Variant, i As Long, n As Long, nDone As Long
    Dim itm As Object
    eids = mCalPending.keys
    n = mCalPending.count
    mCalPending.RemoveAll
    nDone = 0
    For i = 0 To n - 1
        Set itm = Nothing
        Set itm = Application.Session.GetItemFromID(CStr(eids(i)))
        Err.Clear
        If Not itm Is Nothing Then
            If Not IsTechnicalEvent(itm) Then
                ReplayCalendarItem itm
                nDone = nDone + 1
            End If
        End If
    Next i
    modDebugLog.DInfo MOD_NAME, "Rejeu calendrier : " & nDone & " rendez-vous traite(s) sur " & n & _
        " mis en file pendant l'etiquetage"
End Sub

Private Sub ReplayCalendarItem(ByVal appt As Outlook.AppointmentItem)
    On Error GoTo ReplayErr
    BeginCalendarBusy appt
    modProjectTag.CheckReTag appt
    modMeetingColor.ApplyCategory appt
    modAntiOverlap.TryShift appt
    modAppointmentPrivacy.ApplyIfPersonal appt
    modMeetingToAppointment.DemoteIfEmptyMeeting appt
    EndCalendarBusy
    Exit Sub
ReplayErr:
    modDebugLog.DError MOD_NAME, "Rejeu calendrier err " & Err.Number & " - " & Err.Description
    EndCalendarBusy
End Sub

Private Sub Application_ItemSend(ByVal Item As Object, Cancel As Boolean)
    On Error Resume Next

    ' v6 : point de passage watchdog (auto-reparation des liaisons mortes).
    ' Rebinder ICI sauve le mail en cours : mSentItems est rebranche AVANT
    ' que l'item n'arrive dans les Envoyes.
    CheckBindingsWatchdog

    If TypeOf Item Is Outlook.MeetingItem Then
        Dim mi As Outlook.MeetingItem
        Set mi = Item

        Dim appt As Outlook.AppointmentItem
        Set appt = mi.GetAssociatedAppointment(False)

        If Not appt Is Nothing Then
            If Not IsTechnicalEvent(appt) Then
                ' v10.20 - F6-19 (choix A) / F1-05 : l'etiquette projet
                ' interne (code, dossier, partenaire, opportunite) ne part
                ' jamais chez un invite externe. Seules les DEMANDES de
                ' reunion recopient l'objet du rendez-vous ; une annulation,
                ' une reponse ou un transfert garde l'objet prefixe par
                ' Outlook ("Annule : ", "Accepte : ", "TR : "). Le rendez-vous
                ' de l'organisateur garde son etiquette (seul l'objet du
                ' message envoye change).
                Dim isRequest As Boolean, hasExternal As Boolean
                isRequest = (LCase$(Left$(mi.MessageClass, 28)) = "ipm.schedule.meeting.request")
                hasExternal = True                       ' prudence si le test echoue
                hasExternal = modSmtp.HasExternalRecipient(mi)
                If isRequest Then
                    If hasExternal Then
                        mi.subject = modProjectTag.StripProjectTag(appt.subject)
                    Else
                        mi.subject = appt.subject
                    End If
                ElseIf hasExternal Then
                    mi.subject = modProjectTag.StripProjectTag(mi.subject)
                End If
                If hasExternal Then
                    modDebugLog.DDebug MOD_NAME, "ItemSend reunion : etiquette projet retiree de l'objet (destinataire externe)"
                End If
            End If
        End If
    ElseIf TypeOf Item Is Outlook.mailItem Then
        ' v10.3 (2026-06-17) : epingler le moniteur de la fenetre d'envoi pour
        ' le toast "reponse rangee" a venir. Ici, la fenetre de reponse
        ' (Inspector) ou la fenetre courrier (volet de lecture) est ENCORE au
        ' premier plan, donc GetForegroundWindow est fiable. Au moment ou le
        ' toast part reellement (mSentItems_ItemAdd -> TryMoveSentItem), cette
        ' fenetre s'est fermee et le premier plan est instable. Best-effort.
        modNotify.RememberToastMonitorFromForeground

        ' v10.19 : mail de CAMPAGNE (brouillon genere ou envoi automatique
        ' de modMailCampaign) -> aucune detection "attente de reponse",
        ' aucun suivi Urgent, aucune relance personnalisee : un envoi en lot
        ' ne doit ouvrir aucun pop-up. Tout se joue a l'arrivee dans
        ' Elements envoyes (mSentItems_ItemAdd -> HandleSentItem).
        ' v10.20 (F5-09) : CampaignOnItemSend note un envoi fait depuis un
        ' compte SECONDAIRE (ses Elements envoyes ne sont pas surveilles) et
        ' arme le releve differe de ce compte (+30 s, cle "CampaignSent").
        ' Compte principal : rien. Aucune lecture Excel dans l'evenement.
        If modMailCampaign.IsCampaignItem(Item) Then
            modDebugLog.DTrace MOD_NAME, "ItemSend : mail de campagne, detections sautees"
            modMailCampaign.CampaignOnItemSend Item
            On Error GoTo 0
            Exit Sub
        End If

        ' v11 (2026-07-13) : niveau URGENT -> suivi automatique. Le bandeau
        ' Urgent porte la date limite ; on la lit, on synchronise les deux
        ' bandeaux dessus, et on PROGRAMME l'ecriture du suivi apres l'envoi
        ' (differe "UrgentFlush" : pas de COM Excel synchrone dans ItemSend).
        ' Date non renseignee -> garde-fou (annuler / envoyer sans suivi).
        Dim urgentHandled As Boolean
        urgentHandled = False
        If modMailConfidentiality.HasUrgentBanner(Item) Then
            Dim urgDue As Date
            urgDue = modMailConfidentiality.UrgentBannerDate(Item)
            If urgDue = 0 Then
                ' v10.9 : via modUi.Ask. Boite ouverte depuis
                ' Application_ItemSend, donc pendant la fermeture de la
                ' fenetre de redaction : premier plan instable, et un
                ' "Non" jamais vu annulerait un envoi sans que
                ' l'utilisateur comprenne pourquoi.
                If modUi.Ask(T("Le bandeau Urgent n'a pas de date (jour non renseign~e).", _
                               "The Urgent banner has no date (day not filled).") & vbCrLf & _
                             T("Envoyer sans suivi malgr~e tout ?", _
                               "Send without tracking anyway?"), _
                             T("Suivi Mails", "Mail Tracking"), _
                             vbQuestion + vbYesNo + vbDefaultButton2) = vbNo Then
                    Cancel = True
                    On Error GoTo 0
                    Exit Sub
                End If
                ' Oui : envoi sans suivi.
                urgentHandled = True
            Else
                modMailConfidentiality.FinalizeUrgentBanner Item
                modMailResponseTracker.EnqueueUrgentTracking Item, urgDue
                modDefer.Arm "UrgentFlush", 2
                urgentHandled = True
            End If
        End If

        ' v7 : relance reellement envoyee. Le brouillon de relance porte le
        ' tag "FollowupConvID" (prepare depuis 11/2026 par le bouton Relancer
        ' du classeur Suivi des mails, modMailTracker v1.4 ; avant : mode
        ' Personnaliser de modMailResponseTracker, retire). Si present, on
        ' programme la mise a jour de la ligne de suivi APRES l'envoi
        ' (differe : pas de COM Excel synchrone dans ItemSend). Tant que le
        ' brouillon n'est pas envoye, rien.
        Dim isFollowup As Boolean
        isFollowup = False
        If Not urgentHandled Then
            Dim fuProp As Object
            Set fuProp = Item.UserProperties.Find("FollowupConvID")
            If Not fuProp Is Nothing Then
                Dim fuCID As String
                fuCID = CStr(fuProp.Value)
                If Len(fuCID) > 0 Then
                    modDefer.Arm "FollowupSent", 2, fuCID
                    isFollowup = True
                End If
            End If
        End If

        ' v3.1 (2026-05-28) : detection "attente de reponse" (systeme expert).
        ' LECTURE SEULE ici : EvaluateAndStore ne fait que calculer un score
        ' et, si suspicion, memoriser le mail + armer un differe +2 s. Le
        ' pop-up est affiche PLUS TARD (apres le tri QuickMove et le
        ' deplacement Envoye->dossier de modReplyInPlace), pour ne JAMAIS
        ' s'intercaler dans la chaine critique d'envoi.
        ' v10.20 (F4-03, F1-04) : seulement pour un mail ni Urgent (deja suivi
        ' d'office, ou envoye "sans suivi" par choix) ni relance d'un suivi :
        ' sinon la fenetre "Ce mail semble attendre une reponse" proposait un
        ' suivi deja acquis (seconde ligne pour le meme fil).
        If Not urgentHandled And Not isFollowup Then modReplySuspicion.EvaluateAndStore Item
    End If

    On Error GoTo 0
End Sub

' WATCHDOG BINDINGS (v6)
' Les evenements Application_* de CE module sont cables par l'hote VBA et
' continuent de tomber apres un reset d'etat (Ctrl+S VBE) ; les WithEvents
' DECLARES (mSentItems, CalendarItems, g_AppEvents...), eux, meurent.
' Chaque envoi / reception / rappel devient donc un point de controle :
' si une des 3 liaisons jamais legitimement mortes est tombee, on
' rebranche tout en silence (ReinitSilent recharge AUSSI la config :
' couvre les globales g_* remises a zero) et on previent par un toast.
' Les watchers de fenetres a role (courrier/calendrier) et InlineResponse sont
' exclus du DECLENCHEUR (ils peuvent etre legitimement absents sans fenetre
' active) mais sont reconstruits au passage par le rebind.
'=============================================================================
Private Sub CheckBindingsWatchdog()
    On Error Resume Next
    If bWatchdogBusy Then Exit Sub

    Dim dead As Boolean
    dead = False
    If mSentItems Is Nothing Then dead = True
    If CalendarItems Is Nothing Then dead = True
    If Not modNewMeetingDefault.InspectorEventsBound() Then dead = True
    If Not dead Then Exit Sub

    bWatchdogBusy = True
    modDebugLog.DError MOD_NAME, "Watchdog : liaisons mortes detectees -> ReinitSilent"
    modDevTools.ReinitSilent
    modNotify.Notify T("Liaisons ~ev~enements r~epar~ees automatiquement", _
                       "Event bindings repaired automatically")
    bWatchdogBusy = False
End Sub

'=============================================================================
' QUICKMOVE + REPLY-IN-PLACE
'-----------------------------------------------------------------------------
' v2 (2026-05-11) : prompt asynchrone + garde de reentrance.
'
' Probleme corrige :
'   mSentItems_ItemAdd affichait une MsgBox modale directement depuis le
'   handler d'evenement. Si plusieurs mails sont envoyes en rafale (ou si
'   l'utilisateur tarde a repondre), chaque ItemAdd empile une MsgBox derriere
'   la precedente. Outlook se retrouve avec N MsgBox cachees, le thread UI
'   bloque, et plus aucun clic n'arrive a destination.
'
' Solution (v2) :
'   1. ItemAdd ne fait plus QUE l'auto-classement silencieux (match unique).
'      Si pas de match unique, l'EntryID part dans une file FIFO et le
'      handler rend la main immediatement.
'   2. Un differe modDefer +1 s declenchait ProcessNextSentPrompt, qui
'      depilait le premier EntryID et affichait la fenetre de classement (a
'      l'epoque une MsgBox avec vbMsgBoxSetForeground ; depuis v10.8 une
'      InputBox via modUi.AskText, sans ce drapeau : cf. en-tete v10.9).
'   3. mQuickMovePromptBusy garantit qu'une seule fenetre est ouverte a la
'      fois.
'
' v10.20 (F3-06, choix A) : plus de fenetre automatique apres l'envoi. La
'   file des envoyes non classes est videe a la demande par FileSentPending
'   (commande "Classer les envoyes en attente") ; un toast annonce chaque
'   mail mis en file. ProcessNextSentPrompt et la cle modDefer "SentPrompt"
'   sont retires.
'=============================================================================
Private Sub mSentItems_ItemAdd(ByVal Item As Object)
    On Error Resume Next

    If Not TypeOf Item Is Outlook.mailItem Then Exit Sub

    ' v10.19 : mail de CAMPAGNE -> modMailCampaign le range (dossier de la
    ' campagne, du code de l'objet, ou du mail d'origine pour une relance)
    ' et ecrit la feuille en differe. JAMAIS propose au classement : 150
    ' envois ouvraient 150 fenetres en file avec v11.
    If modMailCampaign.HandleSentItem(Item) Then Exit Sub

    ' 0. Suspicion attente reponse : capturer l'EntryID du mail envoye AVANT
    '    tout deplacement (ReplyInPlace deplace hors des Envoyes). Sans effet
    '    si le mail n'a pas ete juge suspect a l'ItemSend. Le pop-up de suivi
    '    est affiche en differe +2s par modReplySuspicion (les deplacements
    '    par modQuickMove.MoveItemTo lui signalent le nouvel EntryID).
    modReplySuspicion.CaptureSentItem Item

    ' 1. Reply-in-place : range automatiquement la reponse dans le dossier
    '    du mail parent (v10 : post-it memorise au brouillon, sinon repli
    '    sur le dossier du fil via l'index de conversations). Silencieux.
    If modReplyInPlace.TryMoveSentItem(Item) Then Exit Sub

    ' 2. Auto-classement silencieux si match unique sur le prefill
    '    (codes DOSS / contrat / opportunite trouves dans sujet+corps).
    '    IMPORTANT v3 (2026-05-27) : utiliser ExtractPrefillSilent et non
    '    ExtractPrefill (aucune fenetre depuis ce handler ItemAdd).
    '    v10.20 (F3-12, F3-07) : correspondance BORNEE sur le dernier segment
    '    du chemin (modQuickMove.FindCodeMatches : "1500" ne designe plus un
    '    dossier "... 01500 ...", un dossier de projet a sous-dossiers ne sort
    '    plus en 1 + n lignes) ; repli sur le chemin entier pour un code NON
    '    numerique seulement. Deplacement par MoveItemTo (F10-06, F3-08).
    If Not modQuickMove.IsIndexLoaded() Then modQuickMove.SmartLoad

    Dim prefill As String
    prefill = modQuickMove.ExtractPrefillSilent(Item)

    If Len(prefill) > 0 Then
        Dim matchIndexes() As Long
        Dim matchCount As Long
        matchCount = modQuickMove.FindCodeMatches(prefill, matchIndexes)
        If matchCount = 0 And (prefill Like "*[!0-9]*") Then
            matchCount = modQuickMove.FindMatches(prefill, matchIndexes)
        End If

        If matchCount = 1 Then
            Dim targetFolder As Object
            Set targetFolder = modQuickMove.IndexFolderObject(matchIndexes(1))
            If Not targetFolder Is Nothing Then
                Dim movedToName As String, movedSubj As String
                movedToName = targetFolder.Name
                movedSubj = CStr(Item.subject)       ' lu AVANT le deplacement
                If modQuickMove.MoveItemTo(Item, targetFolder) Then
                    ' v8 : toast sur CE deplacement (classement AUTOMATIQUE :
                    ' garde, F10-12).
                    modNotify.NotifyFiled movedToName
                    modDebugLog.DInfo MOD_NAME, "ItemAdd auto-classe '" & _
                        modDebugLog.LogClip(movedSubj) & "' -> " & _
                        modQuickMove.IndexFolderPath(matchIndexes(1))
                    Exit Sub
                End If
                ' Deplacement refuse (erreur au journal) : le mail rejoint la file.
            End If
        End If
    End If

    ' 3. Pas de match unique : file d'attente. v10.20 (F3-06, choix A) : plus
    '    de fenetre automatique 1 s apres l'envoi (3 a 10 par jour, refermee
    '    sans classer deux fois sur trois, parfois ouverte cachee pendant des
    '    heures en bloquant Outlook). Un toast annonce le mail ; la commande
    '    "Classer les envoyes en attente" (FileSentPending) ouvre la fenetre
    '    quand l'utilisateur le decide.
    If mPendingSentEntryIDs Is Nothing Then Set mPendingSentEntryIDs = New Collection

    Dim eid As String
    eid = Item.entryID
    If Len(eid) = 0 Then Exit Sub

    ' Eviter doublons dans la file
    Dim k As Variant, dejaPresent As Boolean
    dejaPresent = False
    For Each k In mPendingSentEntryIDs
        If CStr(k) = eid Then dejaPresent = True : Exit For
    Next k
    If dejaPresent Then Exit Sub
    mPendingSentEntryIDs.Add eid

    modDebugLog.DInfo MOD_NAME, "Envoye non classe mis en attente (" & mPendingSentEntryIDs.count & " en attente)"
    modUi.Toast T("Mail envoy~e non class~e (", "Sent mail not filed (") & mPendingSentEntryIDs.count & _
                T(" en attente)", " pending)") & vbLf & _
                T("Bouton Classer les envoy~es en attente pour le ranger.", _
                  "Use the File pending sent mails button to file it."), uiInfo, True

    On Error GoTo 0
End Sub

'=============================================================================
' FileSentPending (v10.20, F3-06 choix A) - commande "Classer les envoyes en
' attente" (ruban : modQuickMove.QuickMoveSentPending). Ouvre la fenetre de
' classement pour chaque envoye de la file, dans l'ordre d'envoi :
'   - un mail deja range ou supprime entre-temps est retire sans question ;
'   - vide = laisser dans Envoyes (retire de la file) ; Annuler = arreter la
'     serie (le mail courant et les suivants restent en file).
' Encadre par modDefer.BeginHeavy / EndHeavy : aucun traitement lourd differe
' ne demarre pendant la fenetre (meme regle que l'ancienne cle "SentPrompt").
' File de session : perdue au redemarrage d'Outlook (les mails restent dans
' Elements envoyes, classables a la main).
'=============================================================================
Public Sub FileSentPending()
    On Error Resume Next
    If mQuickMovePromptBusy Then Exit Sub        ' fenetre deja ouverte
    If mPendingSentEntryIDs Is Nothing Then Set mPendingSentEntryIDs = New Collection
    PurgeSentQueue
    If mPendingSentEntryIDs.count = 0 Then
        modUi.Toast T("Aucun mail envoy~e en attente de classement.", _
                      "No sent mail waiting to be filed."), uiInfo
        Exit Sub
    End If

    If Not modQuickMove.IsIndexLoaded() Then modQuickMove.SmartLoad

    Dim eid As String, itm As Object
    Dim outcome As Long, pos As Long, guard As Long
    Dim nFiled As Long, nLeft As Long
    pos = 0
    guard = mPendingSentEntryIDs.count + 50      ' filet : jamais de boucle sans fin
    mQuickMovePromptBusy = True
    modDefer.BeginHeavy
    Do While mPendingSentEntryIDs.count > 0 And guard > 0
        guard = guard - 1
        eid = CStr(mPendingSentEntryIDs.Item(1))
        Set itm = Nothing
        Set itm = SentQueueItem(eid)
        If itm Is Nothing Then
            RemoveFromSentQueue eid               ' range ou supprime entre-temps
        Else
            pos = pos + 1
            outcome = SENT_LEFT
            outcome = PromptForSentItem(itm, " (" & pos & "/" & (pos + mPendingSentEntryIDs.count - 1) & ")")
            If outcome = SENT_STOP Then Exit Do
            RemoveFromSentQueue eid
            If outcome = SENT_FILED Then nFiled = nFiled + 1 Else nLeft = nLeft + 1
        End If
        Err.Clear
    Loop
    modDefer.EndHeavy
    mQuickMovePromptBusy = False
    modDebugLog.DInfo MOD_NAME, "Classer les envoyes en attente : " & nFiled & " classe(s), " & _
        nLeft & " laisse(s) dans Envoyes, " & mPendingSentEntryIDs.count & " encore en attente"
End Sub

'-----------------------------------------------------------------------------
' SentQueueItem (v10.20) - mail de la file encore dans Elements envoyes (par
' defaut), ou Nothing (introuvable, supprime ou deja range ailleurs).
'-----------------------------------------------------------------------------
Private Function SentQueueItem(ByVal eid As String) As Object
    On Error Resume Next
    Set SentQueueItem = Nothing
    Dim itm As Object, sentEid As String, parentEid As String
    Set itm = Nothing
    Set itm = Application.Session.GetItemFromID(eid)
    If itm Is Nothing Then Exit Function
    sentEid = ""
    parentEid = ""
    sentEid = CStr(Application.Session.GetDefaultFolder(olFolderSentMail).entryID)
    parentEid = CStr(itm.Parent.entryID)
    If Len(sentEid) > 0 And Len(parentEid) > 0 Then
        If Not Application.Session.CompareEntryIDs(sentEid, parentEid) Then Exit Function
    End If
    Set SentQueueItem = itm
End Function

' File purgee des mails deja ranges ou supprimes (v10.20).
Private Sub PurgeSentQueue()
    On Error Resume Next
    Dim i As Long
    For i = mPendingSentEntryIDs.count To 1 Step -1
        If SentQueueItem(CStr(mPendingSentEntryIDs.Item(i))) Is Nothing Then mPendingSentEntryIDs.Remove i
    Next i
End Sub

' Retire un EntryID de la file (premiere occurrence ; v10.20).
Private Sub RemoveFromSentQueue(ByVal eid As String)
    On Error Resume Next
    Dim i As Long
    For i = 1 To mPendingSentEntryIDs.count
        If CStr(mPendingSentEntryIDs.Item(i)) = eid Then
            mPendingSentEntryIDs.Remove i
            Exit Sub
        End If
    Next i
End Sub

'=============================================================================
' PromptForSentItem (v10.8 / v10.20) - FENETRE UNIQUE de classement d'un mail
' envoye, appelee par FileSentPending pour chaque mail de la file.
'
' Une seule InputBox, reutilisee en boucle ; son TEXTE porte l'etat du tour :
'   - liste numerotee des dossiers trouves (v10.20 : recherche lancee
'     d'office sur le code detecte au premier tour ; liste commune
'     modQuickMove.FormatMatchList, bornee a la place restante, F3-21) ;
'   - liste numerotee des candidats detectes dans le corps (collecte
'     silencieuse via modQuickMove.CollectBodyCandidates) ;
'   - v10.20 (F3-06) : 3 dossiers recents, choisis par r1, r2, r3 ;
'   - mention "Aucun dossier pour X" et commande "+" (creer le dossier).
' Saisies :
'   - VIDE = laisser dans Envoyes (SENT_LEFT) ; ANNULER = arreter la serie
'     (SENT_STOP, v10.20 : AskTextEx distingue les deux) ;
'   - liste de DOSSIERS affichee + 1-2 chiffres -> deplacement (v10.20,
'     F3-05 : liste refaite si l'index a ete renumerote pendant la saisie) ;
'   - liste de CANDIDATS affichee + 1 chiffre -> ce code devient le texte de
'     recherche ;
'   - tout autre texte (ex. un code a 4 chiffres) = recherche : dernier
'     segment du chemin d'abord, chemin entier si rien (v10.20, F3-07) ; un
'     seul dossier trouve -> deplacement.
' Deplacement par FileSentTo (MoveItemTo, recents, toast selon
' ToastOnManualFiling). Issue au journal en INFO (F10-13).
'=============================================================================
Private Function PromptForSentItem(ByVal Item As Object, ByVal posText As String) As Long
    On Error Resume Next
    PromptForSentItem = SENT_LEFT

    Const MAX_CAND As Long = 9         ' candidats du corps affiches (1 chiffre)
    Const MAX_LIST As Long = 20        ' dossiers affiches au plus
    Const PROMPT_MAX As Long = 1000    ' une boite de saisie est coupee vers 1 024 caracteres

    ' v10.6 - FIX conserve : le sujet reste HORS de T() (le decodeur
    ' d'accents A() alterait un sujet reel contenant "~e"/"~a").
    Dim subjPart As String
    subjPart = ShortText(CStr(Item.subject), 70)

    ' Prefill silencieux : sujet, puis noms de PJ, puis candidat UNIQUE corps.
    Dim searchText As String
    searchText = modQuickMove.ExtractPrefillSilent(Item)

    ' Candidats multiples du corps : collectes SANS fenetre, listes une seule
    ' fois, numerotes, dans la fenetre unique.
    Dim cand() As String, candCtx() As String, nCand As Long, candShown As Long
    nCand = 0
    candShown = 0
    If Len(searchText) = 0 Then
        nCand = modQuickMove.CollectBodyCandidates(Item, cand, candCtx)
    End If

    Dim header As String              ' message d'etat du tour precedent
    Dim fIdx() As Long, fCnt As Long  ' resultats dossiers affiches
    Dim mIdx() As Long, mCnt As Long
    Dim rIdx() As Long, rCnt As Long  ' v10.20 : dossiers recents affiches (r1..r3)
    Dim shown As Long                 ' v10.20 : lignes de dossiers REELLEMENT affichees
    Dim budget As Long                ' v10.20 : place restante pour la liste de dossiers
    Dim i As Long, n As Long
    Dim choice As String, prompt As String, tail As String, block As String, ln As String
    Dim lastZero As String            ' v10.11 : derniere recherche a zero resultat
    Dim lastSearch As String          ' v10.20 (F3-05) : recherche de la liste affichee
    Dim newIdx As Long                ' v10.11 : index du dossier cree (commande "+")
    Dim gen0 As Long                  ' v10.20 (F3-05) : generation de l'index a l'affichage
    Dim cancelled As Boolean
    Dim boxTitle As String

    boxTitle = T("QuickMove - Mail envoy~e", "QuickMove - Sent mail") & posText
    fCnt = 0
    shown = 0
    header = ""

    ' v10.20 (F3-06) : code detecte -> recherche lancee d'office (avant : il
    ' etait seulement pre-rempli, il fallait valider pour voir la liste).
    ' Meme un dossier unique est PROPOSE en liste : le classement automatique
    ' a deja juge ce code insuffisant pour classer sans question.
    If Len(searchText) > 0 Then
        mCnt = SearchSentFolders(searchText, mIdx)
        If mCnt = 0 Then
            header = NoFolderHeader(searchText)
            lastZero = searchText
        Else
            fCnt = mCnt
            fIdx = mIdx
            lastSearch = searchText
            header = ListHeader(mCnt, searchText)
            searchText = ""
        End If
    End If

    Do
        ' --- Construction du texte de la fenetre unique ---
        prompt = T("Vide = laisser dans Envoy~es ; Annuler = arr~xter la s~erie.", _
                   "Empty = leave in Sent Items; Cancel = stop.") & vbCrLf
        prompt = prompt & T("Sujet : ", "Subject: ") & subjPart & vbCrLf
        If Len(header) > 0 Then prompt = prompt & vbCrLf & header

        rCnt = 0
        If fCnt > 0 Then
            tail = vbCrLf & T("NUMERO pour d~eplacer, ou TEXTE pour affiner :", _
                              "NUMBER to move, or TEXT to refine:")
            ' Resultats de la derniere recherche : liste commune, bornee a la
            ' place restante (F3-21) ; shown = lignes reellement affichees.
            shown = 0
            budget = PROMPT_MAX - Len(prompt) - Len(tail) - 40
            If budget < 150 Then budget = 150
            prompt = prompt & modQuickMove.FormatMatchList(fIdx, fCnt, MAX_LIST, shown, budget)
            If fCnt > shown Then
                prompt = prompt & T("... et ", "... and ") & (fCnt - shown) & T(" autres", " more") & vbCrLf
            End If
        Else
            If nCand > 0 Then
                tail = vbCrLf & T("NUMERO de la liste, rN = r~ecent, ou nom du dossier :", _
                                  "NUMBER from the list, rN = recent, or folder name:")
            Else
                tail = vbCrLf & T("Nom du dossier cible, ou rN = r~ecent :", _
                                  "Target folder name, or rN = recent:")
            End If
            ' Candidats du corps (une seule liste), bornes a la place restante.
            If nCand > 0 Then
                block = T("Num~eros d~etect~es dans le corps :", "Numbers detected in the body:") & vbCrLf
                candShown = 0
                For i = 1 To modQuickMove.MinVal(nCand, MAX_CAND)
                    ln = "[" & i & "] " & cand(i) & "   (..." & ShortText(candCtx(i), 40) & "...)" & vbCrLf
                    If Len(prompt) + Len(block) + Len(ln) + Len(tail) > PROMPT_MAX Then Exit For
                    block = block & ln
                    candShown = i
                Next i
                If nCand > candShown Then
                    block = block & T("... et ", "... and ") & (nCand - candShown) & T(" autres", " more") & vbCrLf
                End If
                prompt = prompt & block
            End If
            ' v10.20 (F3-06) : dossiers recents (r1..r3), s'il reste la place.
            block = RecentBlock(rIdx, rCnt)
            If rCnt > 0 Then
                If Len(prompt) + Len(block) + Len(tail) <= PROMPT_MAX Then
                    prompt = prompt & block
                Else
                    rCnt = 0
                End If
            End If
            ' Consigne finale selon ce qui est REELLEMENT affiche (jamais plus
            ' longue que celle qui a servi au calcul de la place).
            If nCand > 0 And rCnt = 0 Then
                tail = vbCrLf & T("NUMERO de la liste, ou nom du dossier cible :", _
                                  "NUMBER from the list, or target folder name:")
            ElseIf nCand = 0 And rCnt = 0 Then
                tail = vbCrLf & T("Tapez le nom du dossier cible :", "Type the target folder name:")
            End If
        End If
        prompt = prompt & tail

        ' v10.9 : modUi.AskText(Ex) remonte Outlook au premier plan AVANT
        ' d'afficher (InputBox n'a aucun drapeau de premier plan) ; si
        ' Windows refuse, clignotement de la barre des taches + toast.
        ' v10.20 : AskTextEx distingue Annuler (arreter la serie) d'un champ
        ' vide valide (laisser ce mail dans Envoyes). Valeurs par defaut =
        ' arret : un appel qui echouerait ne bouclerait jamais.
        gen0 = modQuickMove.IndexGeneration()
        cancelled = True
        choice = ""
        ' MSGLOOP-OK : texte borne a PROMPT_MAX (listes coupees a la place restante)
        choice = modUi.AskTextEx(prompt, boxTitle, cancelled, searchText)
        If cancelled Then
            modDebugLog.DInfo MOD_NAME, "SentPrompt : serie arretee (Annuler), mail garde en attente"
            PromptForSentItem = SENT_STOP
            Exit Function
        End If
        choice = Trim(choice)
        If Len(choice) = 0 Then
            modDebugLog.DInfo MOD_NAME, "SentPrompt : laisse dans Envoyes"
            PromptForSentItem = SENT_LEFT
            Exit Function
        End If

        ' v10.11 : "+" apres un tour "Aucun dossier" = creer le dossier
        ' manquant (modQuickMove.CreateFolderInteractive : code + nom +
        ' partenaire, dossier Outlook + index + ligne Suivi de temps),
        ' puis classer CE mail dedans.
        If choice = "+" And Len(lastZero) > 0 Then
            newIdx = 0
            If modQuickMove.CreateFolderInteractive(lastZero, newIdx) Then
                If FileSentTo(Item, newIdx, "dossier cree") Then PromptForSentItem = SENT_FILED
                Exit Function
            End If
            ' Creation annulee ou en echec : on repart sur la recherche.
            header = ""
            searchText = lastZero
            GoTo NextTurn
        End If

        ' v10.20 (F3-06) : rN = dossier recent affiche.
        If rCnt > 0 And (LCase$(choice) Like "r#") Then
            n = CLng(Mid$(choice, 2))
            If n >= 1 And n <= rCnt Then
                If modQuickMove.IndexGeneration() <> gen0 Then
                    header = StaleListHeader()
                    searchText = ""
                    GoTo NextTurn
                End If
                If FileSentTo(Item, rIdx(n), "recent") Then PromptForSentItem = SENT_FILED
                Exit Function
            End If
        End If

        ' --- Interpretation d'une saisie numerique courte ---
        If IsNumeric(choice) Then
            n = 0
            n = CLng(choice)
            If fCnt > 0 And Len(choice) <= 2 Then
                ' Liste de dossiers affichee : 1-2 chiffres = choix direct.
                If n >= 1 And n <= shown Then
                    ' v10.20 (F3-05) : index renumerote pendant la saisie
                    ' (reindexation, re-scan) : les numeros affiches ne
                    ' designent plus les memes dossiers -> liste refaite.
                    If modQuickMove.IndexGeneration() <> gen0 Then
                        fCnt = SearchSentFolders(lastSearch, fIdx)
                        header = StaleListHeader()
                        If fCnt = 0 Then
                            header = header & NoFolderHeader(lastSearch)
                            lastZero = lastSearch
                        End If
                        searchText = ""
                        GoTo NextTurn
                    End If
                    If FileSentTo(Item, fIdx(n), "liste") Then PromptForSentItem = SENT_FILED
                    Exit Function
                End If
                header = T("Num~ero invalide.", "Invalid number.") & vbCrLf
                searchText = ""
                GoTo NextTurn
            ElseIf fCnt = 0 And nCand > 0 And Len(choice) = 1 _
                   And n >= 1 And n <= candShown Then
                ' Liste de candidats affichee : 1 chiffre = ce code devient
                ' le texte de recherche (les codes a 4 chiffres passent en
                ' recherche normale ci-dessous).
                choice = cand(n)
            End If
        End If

        ' --- Recherche de dossiers ---
        searchText = choice
        mCnt = SearchSentFolders(searchText, mIdx)

        If mCnt = 1 Then
            If FileSentTo(Item, mIdx(1), "recherche") Then PromptForSentItem = SENT_FILED
            Exit Function
        ElseIf mCnt = 0 Then
            ' Integre a la fenetre du tour suivant (plus de MsgBox separee).
            ' v10.11 : "+" propose la creation du dossier manquant.
            header = NoFolderHeader(searchText)
            lastZero = searchText
            fCnt = 0
        Else
            fCnt = mCnt
            fIdx = mIdx
            lastSearch = searchText
            header = ListHeader(mCnt, searchText)
            nCand = 0                 ' on masque les candidats du corps
            searchText = ""           ' saisie suivante = numero ou nouveau texte
        End If
NextTurn:
    Loop
End Function

'-----------------------------------------------------------------------------
' Aides de PromptForSentItem (v10.20).
' SearchSentFolders : recherche tapee (ou code detecte) - dernier segment du
'   chemin d'abord (un dossier de projet ne ressort plus avec tous ses
'   sous-dossiers), chemin entier si rien (F3-07) ; resultats regroupes par
'   boite, comme QuickMove (la numerotation suit ce nouvel ordre).
'-----------------------------------------------------------------------------
Private Function SearchSentFolders(ByVal txt As String, ByRef idx() As Long) As Long
    On Error Resume Next
    Dim n As Long
    n = 0
    n = modQuickMove.FindMatchesLeaf(txt, idx)
    If n = 0 Then n = modQuickMove.FindMatches(txt, idx)
    If n > 1 Then modQuickMove.GroupMatchesByStore idx, n
    SearchSentFolders = n
End Function

Private Function ListHeader(ByVal n As Long, ByVal txt As String) As String
    ListHeader = n & T(" dossier(s) pour ", " folder(s) for ") & """" & ShortText(txt, 40) & """ :" & vbCrLf
End Function

Private Function NoFolderHeader(ByVal txt As String) As String
    NoFolderHeader = T("Aucun dossier pour ", "No folder for ") & """" & ShortText(txt, 40) & """." & vbCrLf & _
                     T("+ = cr~eer ce dossier", "+ = create this folder") & vbCrLf
End Function

Private Function StaleListHeader() As String
    StaleListHeader = T("(Liste mise ~a jour pendant la saisie : choisissez ~a nouveau.)", _
                        "(List updated while typing: choose again.)") & vbCrLf
End Function

'-----------------------------------------------------------------------------
' RecentBlock (v10.20, F3-06) - texte "Recents :" + jusqu'a SENT_RECENTS_MAX
' dossiers recents de QuickMove (modQuickMove.mRecentFolders, index valides
' seulement), numerotes r1, r2, r3 ; rIdx / rCnt : index affiches.
'-----------------------------------------------------------------------------
Private Function RecentBlock(ByRef rIdx() As Long, ByRef rCnt As Long) As String
    On Error Resume Next
    Dim i As Long, idx As Long, s As String
    rCnt = 0
    s = ""
    ReDim rIdx(1 To SENT_RECENTS_MAX)
    For i = 1 To modQuickMove.mRecentCount
        If rCnt >= SENT_RECENTS_MAX Then Exit For
        idx = modQuickMove.mRecentFolders(i)
        If idx > 0 And idx <= modQuickMove.IndexCount() Then
            rCnt = rCnt + 1
            rIdx(rCnt) = idx
            s = s & "[r" & rCnt & "] " & ShortText(modQuickMove.DisplayPath(idx), SENT_PATH_MAX, True) & vbCrLf
        End If
    Next i
    If rCnt > 0 Then s = T("R~ecents :", "Recent:") & vbCrLf & s
    RecentBlock = s
End Function

' Texte raccourci a maxLen caracteres ("..." en fin, ou en debut si keepEnd).
Private Function ShortText(ByVal s As String, ByVal maxLen As Long, _
                           Optional ByVal keepEnd As Boolean = False) As String
    If Len(s) <= maxLen Or maxLen < 4 Then
        ShortText = s
    ElseIf keepEnd Then
        ShortText = "..." & Right$(s, maxLen - 3)
    Else
        ShortText = Left$(s, maxLen - 3) & "..."
    End If
End Function

'-----------------------------------------------------------------------------
' FileSentTo (v10.20) - classe un envoye dans le dossier d'index idx :
' deplacement par modQuickMove.MoveItemTo (element renvoye garde : suivi
' "attente de reponse" et annulation QuickMoveUndo, F10-06 / F3-08), dossier
' ajoute aux recents, toast "Range dans" seulement si ToastOnManualFiling=1
' (classement manuel, F10-12). Echec : toast d'avertissement, le mail reste
' dans Elements envoyes (avant : toast "classe" meme quand Move echouait).
'-----------------------------------------------------------------------------
Private Function FileSentTo(ByVal Item As Object, ByVal idx As Long, ByVal how As String) As Boolean
    On Error Resume Next
    FileSentTo = False
    Dim fld As Object, fldName As String
    Set fld = Nothing
    Set fld = modQuickMove.IndexFolderObject(idx)
    If fld Is Nothing Then
        modDebugLog.DError MOD_NAME, "SentPrompt : dossier introuvable (index " & idx & ")"
    Else
        fldName = CStr(fld.Name)
        If modQuickMove.MoveItemTo(Item, fld) Then
            FileSentTo = True
            modQuickMove.AddToRecent idx
            If modConfig.ToastOnManualFiling() Then modNotify.NotifyFiled fldName
            modDebugLog.DInfo MOD_NAME, "SentPrompt : classe (" & how & ") -> " & modQuickMove.IndexFolderPath(idx)
            Exit Function
        End If
    End If
    modUi.Toast T("Classement impossible : le mail reste dans Envoy~es (d~etail dans le journal).", _
                  "Filing failed: the mail stays in Sent Items (details in the log)."), uiWarn
End Function

'=============================================================================
' v10.16 - CANAL DE COMMANDES Excel -> Outlook
' BindCommandChannel : cree au besoin le dossier de notes
' Notes\ClassicOutlook-Toolkit et surveille ses elements (ajout ET
' modification : le classeur reutilise une seule note). Appele par
' Application_Startup et RebindEvents. Silencieux ; un echec est journalise.
' HandleToolkitCommand : lit la note "TOOLKIT|<commande>|<horodatage>[|args]" ;
' ignore une commande deja traitee (v10.20 : dernier horodatage traite garde
' dans le registre, Toolkit\CmdChannel\LastStamp) ou trop ancienne (v10.20 :
' INFO + toast si elle a moins de 15 min) ; une note deja acquittee par
' modMailResponseTracker ("ACK|...") est ignoree ; REFRESH -> meme
' actualisation differee que RefreshMailTracking (classeur retrouve par son
' nom) ; CAMPAIGN_xxx -> modMailCampaign.CampaignRemote (v10.19, le reste de
' la note = arguments "classeur|feuille|extra..."). Aucune fenetre, aucun
' traitement lourd dans l'evenement.
'=============================================================================
Private Sub BindCommandChannel()
    On Error Resume Next
    Set mCmdItems = Nothing
    Dim notesFld As Outlook.folder
    Set notesFld = Nothing
    Set notesFld = Application.Session.GetDefaultFolder(olFolderNotes)
    If notesFld Is Nothing Then
        modDebugLog.DError MOD_NAME, "Canal de commandes : dossier Notes introuvable (" & Err.Number & ")"
        Err.Clear
        Exit Sub
    End If
    Dim fld As Outlook.folder
    Set fld = Nothing
    Set fld = notesFld.Folders(CMD_FOLDER_NAME)
    If fld Is Nothing Then
        Err.Clear
        Set fld = notesFld.Folders.Add(CMD_FOLDER_NAME, olFolderNotes)
    End If
    If fld Is Nothing Then
        modDebugLog.DError MOD_NAME, "Canal de commandes : creation du dossier impossible (" & Err.Number & ")"
        Err.Clear
        Exit Sub
    End If
    Set mCmdItems = fld.items
    modDebugLog.DInfo MOD_NAME, "Canal de commandes pret (" & fld.FolderPath & ")"
    Err.Clear
End Sub

Private Sub HandleToolkitCommand(ByVal Item As Object)
    On Error Resume Next
    Dim txt As String
    txt = ""
    txt = CStr(Item.body)
    txt = Replace(txt, vbCr, vbLf)
    If InStr(1, txt, vbLf) > 0 Then txt = Left$(txt, InStr(1, txt, vbLf) - 1)
    txt = Trim$(txt)

    Dim parts() As String
    parts = Split(txt, "|")
    If UBound(parts) < 2 Then Exit Sub
    If parts(0) <> "TOOLKIT" Then Exit Sub

    Dim stamp As String
    stamp = parts(2)
    If Len(stamp) < 14 Then Exit Sub

    ' v10.20 (F1-08) : derniere commande traitee MEMORISEE dans le registre
    ' (avant : en memoire seulement, perdue au redemarrage d'Outlook ou a la
    ' remise a zero du projet VBA -> une commande ecrite moins de 2 min
    ' avant un redemarrage etait executee une seconde fois par l'echo de
    ' synchronisation). Les horodatages "aaaammjjhhnnss-mmm" se comparent
    ' comme des chaines : tout horodatage <= au dernier traite est ignore.
    If Len(mLastCmdStamp) = 0 Then mLastCmdStamp = GetSetting("Toolkit", "CmdChannel", "LastStamp", "")
    Err.Clear
    ' Valeur memorisee dans le futur (horloge decalee un jour) : ignoree,
    ' sinon plus aucune commande ne passerait.
    If StrComp(Left$(mLastCmdStamp, 8), Format$(Now + 1, "yyyymmdd"), vbBinaryCompare) > 0 Then mLastCmdStamp = ""
    If Len(mLastCmdStamp) > 0 Then
        If StrComp(stamp, mLastCmdStamp, vbBinaryCompare) <= 0 Then Exit Sub
    End If

    Dim sentAt As Date
    sentAt = 0
    sentAt = DateSerial(CInt(Mid$(stamp, 1, 4)), CInt(Mid$(stamp, 5, 2)), CInt(Mid$(stamp, 7, 2))) + _
             TimeSerial(CInt(Mid$(stamp, 9, 2)), CInt(Mid$(stamp, 11, 2)), CInt(Mid$(stamp, 13, 2)))
    If Err.Number <> 0 Or sentAt = 0 Then
        Err.Clear
        Exit Sub
    End If

    ' Commande marquee traitee (ou ecartee) des maintenant : ni l'echo de
    ' synchronisation ni la reecriture de la note ne la rejouent. Sauf
    ' horodatage dans le futur (horloge decalee) : il bloquerait les
    ' commandes suivantes, on l'ecarte sans le memoriser.
    If DateDiff("s", Now, sentAt) <= CMD_MAX_AGE_SEC Then
        mLastCmdStamp = stamp
        SaveSetting "Toolkit", "CmdChannel", "LastStamp", stamp
    End If
    Err.Clear

    Dim ageSec As Long
    ageSec = Abs(DateDiff("s", sentAt, Now))
    If ageSec > CMD_MAX_AGE_SEC Then
        ' v10.20 (F1-08) : INFO (avant : DEBUG, invisible) ; toast si la
        ' commande est encore recente (Outlook occupe quand le classeur l'a
        ' ecrite) : l'utilisateur sait qu'il doit la relancer.
        modDebugLog.DInfo MOD_NAME, "Canal de commandes : commande trop ancienne ignoree (" & _
            parts(1) & ", " & ageSec & " s)"
        If ageSec <= CMD_NOTIFY_MAX_SEC Then
            modUi.Toast T("Commande du classeur ignor~ee (trop ancienne) : relancez-la.", _
                          "Workbook command ignored (too old): run it again."), uiWarn
        End If
        Exit Sub
    End If

    ' v10.19 : arguments = tout ce qui suit l'horodatage (3e separateur).
    Dim cmd As String, args As String, i As Long
    cmd = UCase$(parts(1))
    args = ""
    For i = 3 To UBound(parts)
        If i > 3 Then args = args & "|"
        args = args & parts(i)
    Next i

    Select Case cmd
        Case "REFRESH"
            modDebugLog.DInfo MOD_NAME, "Canal de commandes : actualisation du suivi demandee par Excel"
            modMailResponseTracker.RequestDashboardRefresh Nothing
        Case "REPAIR"
            modDebugLog.DInfo MOD_NAME, "Canal de commandes : reparation du suivi demandee par Excel"
            modMailResponseTracker.RequestTrackerRepair
        Case Else
            If Left$(cmd, 9) = "CAMPAIGN_" Then
                modDebugLog.DInfo MOD_NAME, "Canal de commandes : " & cmd & " demandee par le classeur Campagnes"
                modMailCampaign.CampaignRemote cmd, args
            Else
                modDebugLog.DInfo MOD_NAME, "Canal de commandes : commande inconnue '" & parts(1) & "'"
            End If
    End Select
End Sub
