Public Sub Rbn_Action(control As Object)
    Dim act As String
    act = ""
    On Error Resume Next
    act = UCase$(CStr(control.Tag))
    On Error GoTo 0
    If Len(act) > 0 Then RunAction act
End Sub

' Libelles dynamiques : compteurs, et Archiver / Restaurer selon la feuille.
Public Sub Rbn_GetLabel(control As Object, ByRef returnedVal)
    On Error Resume Next
    Dim id As String
    id = ""
    id = CStr(control.id)
    EnsureSchema ThisWorkbook
    Dim n As Long
    Select Case id
        Case "lblValidate"
            n = CountStatus(TrackSheet(), "R" & ChrW$(233) & "pondu ?")
            returnedVal = n & A(" ~a valider")
        Case "lblLate"
            ' v1.4 (F4-08) : suivis en cours dont la date limite est passee
            ' (comme la colonne Echeance), et non le libelle de statut, fige
            ' depuis le dernier recalcul d'Outlook.
            n = CountLate(TrackSheet())
            returnedVal = n & " en retard"
        Case "lblChecked"
            n = CountMarks(ListSheet())
            If n > 1 Then
                returnedVal = n & A(" coch~ees")
            Else
                returnedVal = n & A(" coch~ee")
            End If
        Case "btnArchive", "btnArchiver"         ' v1.4 : nouvel et ancien identifiant
            If OnArchiveSheet() Then returnedVal = "Restaurer" Else returnedVal = "Archiver"
        Case Else
            returnedVal = ""
    End Select
End Sub

' v1.4 (F8-21) - Boutons grises hors de leur feuille (avant : une boite de
' message apres le clic) : Ouvrir le mail, Ouvrir le dossier et Archiver /
' Restaurer sur Suivi et Archives ; Relancer, Cocher les retards, Prolonger
' et Valider sur Suivi seulement ; Actualiser, Reparer et Aide toujours.
Public Sub Rbn_GetEnabled(control As Object, ByRef returnedVal)
    On Error Resume Next
    Dim act As String
    act = ""
    act = UCase$(CStr(control.Tag))
    Select Case act
        Case "", "REFRESH", "REPAIR", "HELP"
            returnedVal = True
        Case "OPEN", "OPEN_FOLDER", "ARCHIVE"
            returnedVal = (OnTrackSheet() Or OnArchiveSheet())
        Case Else                                ' FOLLOWUP, CHECK_LATE, EXTEND, VALIDATE
            returnedVal = OnTrackSheet()
    End Select
End Sub

Public Sub Rbn_GetImage(control As Object, ByRef returnedVal)
    On Error Resume Next
    If OnArchiveSheet() Then
        returnedVal = "SourceControlCheckOut"
    Else
        returnedVal = "SourceControlCheckIn"
    End If
End Sub

' Colonne coche : un clic coche/decoche la ligne (la cellule active passe
' sur l'objet), un clic sur l'en-tete decoche tout. Puis recalcul de la
' feuille : le surlignage de la ligne active (CELL("row")) suit la selection.
' v1.4 (F4-18) : seulement au CLIC DE SOURIS. Arriver au clavier sur la
' colonne A (touche Origine, fleche gauche) ne coche plus rien : la ligne
' cochee par accident etait ensuite prise par Prolonger, Valider ou
' Archiver a la place de la ligne active.
Public Sub OnSelectionChanged(ByVal sh As Object, ByVal Target As Range)
    If mBusy Then Exit Sub
    On Error GoTo Fin
    If TypeName(sh) <> "Worksheet" Then Exit Sub
    Dim ws As Worksheet
    Set ws = sh
    If ws.Name <> SH_TRACK And ws.Name <> SH_ARCHIVE Then Exit Sub
    EnsureSchema ws.Parent

    If COL_MARK >= 1 And COL_SUBJECT >= 1 Then
        If Target.CountLarge = 1 Then
            If Target.Column = COL_MARK And PrimaryButtonDown() Then
                Dim r As Long
                r = Target.Row
                mBusy = True
                Application.EnableEvents = False
                If r = 1 Then
                    ClearAllMarks ws
                ElseIf HasId(ws, r) Then
                    ToggleMark ws, r
                End If
                Dim tr As Long
                tr = r
                If tr < 2 Then tr = 2
                ws.Cells(tr, COL_SUBJECT).Select
                Application.EnableEvents = True
                mBusy = False
                RefreshRibbon
            End If
        End If
    End If
    ws.Calculate
    Exit Sub
Fin:
    Application.EnableEvents = True
    mBusy = False
End Sub

' v1.4 (F4-18) - Bouton principal de la souris enfonce en ce moment ? (Le
' changement de selection arrive pendant l'appui.) Boutons inverses
' (gaucher) pris en compte. En cas de doute : non.
Private Function PrimaryButtonDown() As Boolean
    On Error Resume Next
    PrimaryButtonDown = False
    Dim vk As Long
    vk = VK_LBUTTON
    If GetSystemMetrics(SM_SWAPBUTTON) <> 0 Then vk = VK_RBUTTON
    PrimaryButtonDown = ((GetAsyncKeyState(vk) And &H8000) <> 0)
End Function

' Edition du delai (feuille Suivi) -> recalcul instantane de la date limite
' et du statut, sans rouvrir le tableau de bord.
Public Sub OnSheetChanged(ByVal sh As Object, ByVal Target As Range)
    On Error GoTo Fin
    If TypeName(sh) <> "Worksheet" Then Exit Sub
    If sh.Name <> SH_TRACK Then Exit Sub
    Dim ws As Worksheet
    Set ws = sh
    EnsureSchema ws.Parent
    If COL_DELAY < 1 Or COL_ID < 1 Then Exit Sub
    Dim isect As Range
    Set isect = Application.Intersect(Target, ws.Columns(COL_DELAY))
    If isect Is Nothing Then Exit Sub
    Dim lastRow As Long
    lastRow = LastDataRow(ws)
    If lastRow < 2 Then Exit Sub
    Set isect = Application.Intersect(isect, ws.Range(ws.Cells(2, COL_DELAY), ws.Cells(lastRow, COL_DELAY)))
    If isect Is Nothing Then Exit Sub
    Application.EnableEvents = False
    Dim cell As Range
    For Each cell In isect.Cells
        If HasId(ws, cell.Row) Then RecomputeRow ws, cell.Row
    Next cell
Fin:
    Application.EnableEvents = True
End Sub

' Appele par Outlook (Application.Run) a la fin d'une actualisation
' demandee par le bouton Actualiser.
' v1.4 (F4-19) : une action en attente (mail qui n'etait plus joignable)
' reprend 2 s plus tard, une fois Outlook libere (jamais pendant son appel).
Public Sub AfterOutlookRefresh()
    On Error Resume Next
    mSchemaLoaded = False
    EnsureSchema ThisWorkbook
    EnsureSetup TrackSheet()
    EnsureSetup ArchiveSheet()
    RefreshRibbon
    mCmdBody = ""                        ' Outlook a repondu : plus d'accuse a attendre
    CancelAckCheck
    SetStatus A("Suivi actualis~e ~a ") & Format$(Now, "hh:nn") & "."
    ScheduleStatusClear 20
    If Len(mPendingAct) > 0 Then
        CancelReplay
        mReplayAt = Now + TimeSerial(0, 0, REPLAY_DELAY_SEC)
        Application.OnTime mReplayAt, "'" & ThisWorkbook.Name & "'!ReplayPendingAction"
    End If
End Sub

    End If

'=============================================================================
Private Sub RunAction(ByVal act As String)
    On Error GoTo Fin
    If act = "REFRESH" Then
        DoRefresh
        Exit Sub
    End If
    If act = "REPAIR" Then
        DoRepair
        Exit Sub
    End If
    If act = "HELP" Then                 ' v1.4 (F8-20) : bouton Aide
        DoHelp
        Exit Sub
    End If

    Dim sh As Object
    Set sh = Nothing
    On Error Resume Next
    Set sh = ActiveSheet
    On Error GoTo Fin
    If sh Is Nothing Then Exit Sub
    If TypeName(sh) <> "Worksheet" Then
        ShowInfo A("S~electionnez une ligne de la feuille ") & SH_TRACK & "."
        Exit Sub
    End If
    If Not sh.Parent Is ThisWorkbook Then Exit Sub
    Dim ws As Worksheet
    Set ws = sh
    EnsureSchema ws.Parent

    Dim onArchive As Boolean
    If ws.Name = SH_TRACK Then
        onArchive = False
    ElseIf ws.Name = SH_ARCHIVE Then
        onArchive = True
    Else
        ShowInfo A("Les actions du suivi portent sur les feuilles ") & SH_TRACK & " et " & SH_ARCHIVE & "."
        Exit Sub
    End If

    ' Garde-fou : les indices lus dans _Schema doivent correspondre a la
    ' feuille. Sinon relecture forcee, puis refus si l'ecart persiste.
    If Not LayoutMatches(ws) Then
        mSchemaLoaded = False
        EnsureSchema ws.Parent
        If Not LayoutMatches(ws) Then
            MsgBox A("Les colonnes de la feuille ne correspondent pas ~a la feuille _Schema : ") & _
                   A("aucune action n'a ~et~e faite.") & vbCrLf & vbCrLf & _
                   A("Ouvrez le tableau de bord depuis Outlook, puis recommencez."), _
                   vbExclamation, APP_TITLE
            Exit Sub
        End If
    End If

    Dim rowList As Collection
    Select Case act
        Case "OPEN", "OPEN_FOLDER"
            Dim r As Long
            r = ActiveCell.Row
            If r < 2 Or Not HasId(ws, r) Then
                ShowInfo A("S~electionnez d'abord une ligne de suivi (une cellule de la ligne suffit).")
                Exit Sub
            End If
            If act = "OPEN" Then DoOpen ws, r
            If act = "OPEN_FOLDER" Then DoOpenFolder ws, r
        Case "FOLLOWUP", "CHECK_LATE"
            If onArchive Then
                ShowInfo A("Ligne archiv~ee : restaurez-la d'abord pour la relancer.")
                Exit Sub
            End If
            If act = "CHECK_LATE" Then
                DoCheckLate ws
                Exit Sub
            End If
            ' v1.4 (F4-16) : lignes cochees, sinon ligne active (comme
            ' Prolonger, Valider et Archiver ; avant : ligne active seule).
            Set rowList = TargetRows(ws)
            If rowList.Count = 0 Then Exit Sub
            DoFollowUpRows ws, rowList
        Case "EXTEND", "VALIDATE", "ARCHIVE"
            If onArchive And act <> "ARCHIVE" Then
                ShowInfo A("Ligne archiv~ee : seuls Ouvrir le mail, Ouvrir le dossier et Restaurer sont disponibles ici.")
                Exit Sub
            End If
            Set rowList = TargetRows(ws)
            If rowList.Count = 0 Then Exit Sub
            If act = "EXTEND" Then DoExtend ws, rowList
            If act = "VALIDATE" Then DoValidate ws, rowList
            If act = "ARCHIVE" Then
                If onArchive Then DoRestore ws, rowList Else DoArchive ws, rowList
            End If
    End Select
    Exit Sub
Fin:
    MsgBox A("Action impossible : ") & Err.Number & " - " & Err.Description, vbExclamation, APP_TITLE
End Sub

' Lignes visees : les lignes cochees s'il y en a, sinon la ligne active.
' Vide (avec message) si rien d'exploitable.
Private Function TargetRows(ByVal ws As Worksheet) As Collection
    Dim res As Collection
    Set res = CheckedRows(ws)
    If res.Count = 0 Then
        Dim r As Long
        r = ActiveCell.Row
        If r >= 2 Then
            If HasId(ws, r) Then res.Add r
        End If
    End If
    If res.Count = 0 Then

' v1.4 (F4-01) : derniere ligne dont la colonne ID est remplie, lignes

' derniere ligne VISIBLE : avec la feuille Archives (ou Suivi) filtree,

' est lue d'un bloc jusqu'au bas de la zone utilisee (une lecture de
' plage voit aussi les lignes masquees). Repli : ancienne methode.
Private Function LastDataRow(ByVal ws As Worksheet) As Long
    On Error GoTo Fallback
    LastDataRow = 1
    If COL_ID < 1 Then Exit Function
    Dim usedLast As Long, v As Variant, r As Long
    usedLast = ws.UsedRange.Row + ws.UsedRange.Rows.Count - 1
    If usedLast < 2 Then Exit Function
    v = ws.Range(ws.Cells(1, COL_ID), ws.Cells(usedLast, COL_ID)).Value
    If Not IsArray(v) Then Exit Function

        If IsError(v(r, 1)) Then
            LastDataRow = r
            Exit Function
        End If
        If Len(CStr(v(r, 1))) > 0 Then
            LastDataRow = r
            Exit Function
        End If
    Next r
    Exit Function
Fallback:
    On Error Resume Next
    LastDataRow = 1
    LastDataRow = ws.Cells(ws.Rows.Count, COL_ID).End(xlUp).Row
End Function

' Surlignage jaune de la ligne active (sauf la cellule Statut, qui garde sa
' couleur). Mise en forme conditionnelle sur =ROW()=CELL("row") : rien n'est
' ecrit dans le fichier ; OnSelectionChanged recalcule la feuille a chaque
' changement de selection. Les anciennes regles de ce module sont retirees
' avant pose (les indices de colonnes ont pu changer).
Private Sub EnsureHighlightRule(ByVal ws As Worksheet)
    On Error Resume Next
    If COL_ID < 1 Or COL_STATUS < 1 Then Exit Sub
    Dim lastVis As Long
    lastVis = COL_NBFU
    If lastVis < COL_SUBJECT Then lastVis = COL_SUBJECT

    Dim i As Long, f As String
    For i = ws.Cells.FormatConditions.Count To 1 Step -1
        f = ""
        f = CStr(ws.Cells.FormatConditions(i).Formula1)
        If InStr(1, f, "CELL(", vbTextCompare) > 0 Then ws.Cells.FormatConditions(i).Delete
    Next i

    Dim formula As String
    formula = "=AND(ROW()=CELL(""row""),$" & ColLetter(ws, COL_ID) & "2<>"""")"
    If COL_STATUS > 1 Then
        AddHighlight ws.Range(ws.Cells(2, 1), ws.Cells(ws.Rows.Count, COL_STATUS - 1)), formula
    End If
    If lastVis > COL_STATUS Then
        AddHighlight ws.Range(ws.Cells(2, COL_STATUS + 1), ws.Cells(ws.Rows.Count, lastVis)), formula
    End If
End Sub

Private Sub AddHighlight(ByVal rng As Range, ByVal formula As String)
    On Error Resume Next
    Dim fc As Object
    Set fc = rng.FormatConditions.Add(Type:=xlExpression, Formula1:=formula)
    If fc Is Nothing Then Exit Sub
    fc.Interior.Color = RGB(255, 244, 171)
    fc.StopIfTrue = False
End Sub

Private Function ColLetter(ByVal ws As Worksheet, ByVal col As Long) As String

End Function

'=============================================================================
' OUVRIR LE DOSSIER : la fenetre principale d'Outlook navigue vers le
' dossier qui contient le mail suivi, mail selectionne s'il est visible dans
' la vue. Sans fenetre principale ouverte, nouvelle fenetre.
'=============================================================================
Private Sub DoOpenFolder(ByVal ws As Worksheet, ByVal r As Long)
    On Error GoTo Fin
    Dim ol As Object
    Set ol = GetOutlook()
    If ol Is Nothing Then Exit Sub

    Dim itm As Object
    Set itm = ResolveMail(ol, ws, r)
    If itm Is Nothing Then
        MailNotFound "OPEN_FOLDER", ws, r   ' v1.4 (F4-19)
        Exit Sub
    End If

    Dim fld As Object
    Set fld = itm.Parent
    SyncFolderCell ws, r, itm

    Dim olExp As Object
    Set olExp = Nothing
    On Error Resume Next
    Set olExp = ol.ActiveExplorer
    On Error GoTo Fin
    If olExp Is Nothing Then
        fld.Display
        On Error Resume Next
        Set olExp = ol.ActiveExplorer
        On Error GoTo Fin
    Else
        Set olExp.CurrentFolder = fld
    End If
    If olExp Is Nothing Then Exit Sub

    On Error Resume Next
    Dim selectable As Boolean, winState As Long
    selectable = False
    selectable = olExp.IsItemSelectableInView(itm)
    If selectable Then
        olExp.ClearSelection
        olExp.AddToSelection itm
    End If
    winState = -1
    winState = olExp.WindowState
    If winState = 1 Then olExp.WindowState = 0             ' reduite -> agrandie
    olExp.Activate
    AppActivate CStr(olExp.Caption)
    Exit Sub
Fin:
    MsgBox A("Ouverture du dossier impossible : ") & Err.Number & " - " & Err.Description, _
           vbExclamation, A("Ouvrir le dossier")
End Sub

'=============================================================================
' RELANCER : brouillon "Repondre a tous" pre-rempli, affiche pour relecture.
' Les pieces jointes du mail suivi (le mail envoye par soi dont la date est
' la plus proche de la ligne) sont remises d'office : une reponse a tous les
' retire. Texte en francais ou en anglais selon la langue de ce mail. La
' ligne n'est mise a jour qu'a l'ENVOI reel (Outlook : Application_ItemSend
' -> RegisterFollowupSent, grace a la propriete FollowupConvID).
' v1.4 (F4-16, choix A) : seul moteur de relance, sur les lignes COCHEES
' (sinon la ligne active). Un brouillon par ligne : les FOLLOWUP_SHOWN
' premiers affiches, les suivants enregistres dans Brouillons (avec la meme
' propriete : leur envoi depuis Brouillons met aussi la ligne a jour),
' FOLLOWUP_MAX par clic au plus. Lignes "Repondu", "Repondu ?" et "Traite"
' ignorees (F4-09 : jamais de relance a qui a deja repondu). Mail
' introuvable (deplace) : Outlook est prie d'actualiser et la relance de
' ces lignes reprend seule ensuite (F4-19). Compte rendu dans la barre
' d'etat ; fenetre seulement si rien n'a pu etre prepare ou en cas d'erreur.
'=============================================================================
Private Sub DoFollowUpRows(ByVal ws As Worksheet, ByVal rowList As Collection)
    On Error GoTo Fin
    Dim ol As Object
    Set ol = GetOutlook()
    If ol Is Nothing Then Exit Sub

    Dim v As Variant, r As Long, st As String, res As Long
    Dim nShown As Long, nSaved As Long, nSkipped As Long, nOver As Long, nFail As Long
    Dim missing As Collection, done As Collection
    Set missing = New Collection
    Set done = New Collection
    mLastFollowErr = ""
    For Each v In rowList
        r = CLng(v)
        st = Trim$(CellText(ws.Cells(r, COL_STATUS).Value))
        If IsTerminalStatus(st) Then
            nSkipped = nSkipped + 1
        ElseIf nShown + nSaved >= FOLLOWUP_MAX Then
            nOver = nOver + 1
        Else
            res = PrepareFollowUp(ol, ws, r, (nShown < FOLLOWUP_SHOWN))
            Select Case res
                Case 1
                    nShown = nShown + 1
                    done.Add r
                Case 2
                    nSaved = nSaved + 1
                    done.Add r
                Case 0
                    missing.Add r
                Case Else
                    nFail = nFail + 1
            End Select
        End If
    Next v

    If done.Count > 0 Then
        Application.EnableEvents = False
        ClearMarks ws, done
        Application.EnableEvents = True
        RefreshRibbon
    End If

    Dim queued As Boolean
    queued = False
    If missing.Count > 0 Then queued = QueueRetry("FOLLOWUP", ws, missing)

    If done.Count = 0 And missing.Count = 0 And nFail = 0 Then
        If nSkipped > 0 Then
            ShowInfo A("Aucune relance pr~epar~ee : ") & nSkipped & _
                     A(" ligne(s) d~ej~a r~epondue(s), ~a valider ou trait~ee(s).")
        End If
        Exit Sub
    End If
    If missing.Count > 0 And Not queued And done.Count = 0 And nFail = 0 Then
        ShowMissingMail
        Exit Sub
    End If

    Dim msg As String
    msg = A("Relances : ") & nShown & A(" affich~ee(s)")
    If nSaved > 0 Then msg = msg & ", " & nSaved & A(" dans Brouillons")
    If nSkipped > 0 Then msg = msg & ", " & nSkipped & A(" ignor~ee(s) (r~eponse re~cue ou trait~ee)")
    If nOver > 0 Then msg = msg & ", " & nOver & A(" au-del~a de ") & FOLLOWUP_MAX & A(" (cliquez de nouveau)")
    If missing.Count > 0 Then
        If queued Then
            msg = msg & ", " & missing.Count & A(" mail(s) d~eplac~e(s) : recherche dans Outlook, reprise ensuite")
        Else
            msg = msg & ", " & missing.Count & A(" mail(s) introuvable(s)")
        End If
    End If
    If nFail > 0 Then msg = msg & ", " & nFail & A(" en erreur")
    SetStatus msg & "."
    ScheduleStatusClear STATUS_TIMEOUT_SEC
    If nFail > 0 Then
        MsgBox A("Relance impossible pour ") & nFail & A(" ligne(s) : ") & mLastFollowErr, vbExclamation, "Relancer"
    End If
    Exit Sub
Fin:
    Application.EnableEvents = True
    MsgBox A("Relance impossible : ") & Err.Number & " - " & Err.Description, vbExclamation, "Relancer"
End Sub

' Un brouillon de relance pour la ligne r. Retour : 1 = affiche, 2 =
' enregistre dans Brouillons, 0 = mail de la ligne introuvable, -1 = erreur
' (mLastFollowErr).
Private Function PrepareFollowUp(ByVal ol As Object, ByVal ws As Worksheet, ByVal r As Long, _
                                 ByVal showIt As Boolean) As Long
    Dim tmpDir As String
    tmpDir = ""
    PrepareFollowUp = -1
    On Error GoTo Fin

    Dim src As Object
    Set src = ResolveMail(ol, ws, r)
    If src Is Nothing Then
        PrepareFollowUp = 0
        Exit Function
    End If
    SyncFolderCell ws, r, src

    Dim orig As Object
    Set orig = FindTrackedSentMail(ol, src, RowSentDate(ws, r))

    ' v1.4 (F4-09) : "Repondre a tous" sur LE mail suivi (envoye par soi)
    ' quand il est retrouve : ses destinataires sont ceux du suivi. La
    ' reference d'une ligne peut etre un autre message du fil (apres une
    ' reparation d'ancrage cote Outlook) : repondre a celui-la ecrirait a
    ' son expediteur et a ses destinataires, pas a ceux du mail suivi.
    Dim base As Object
    If orig Is Nothing Then Set base = src Else Set base = orig

    Dim lang As String
    lang = DetectLanguage(base)

    Dim attPaths As Collection
    Set attPaths = New Collection
    If Not orig Is Nothing Then
        tmpDir = NewTempDir()
        If Len(tmpDir) > 0 Then CopyRealAttachments orig, tmpDir, attPaths
    End If

    Dim fu As Object
    Set fu = base.ReplyAll
    fu.htmlBody = InsertAfterBody(fu.htmlBody, BuildRelanceHtml(attPaths.Count, lang))
    Dim v As Variant
    For Each v In attPaths
        fu.Attachments.Add CStr(v)
    Next v

    On Error Resume Next
    Dim up As Object
    Set up = fu.UserProperties.Find(FOLLOWUP_PROP)
    If up Is Nothing Then Set up = fu.UserProperties.Add(FOLLOWUP_PROP, 1)   ' olText
    up.Value = CStr(ws.Cells(r, COL_CONV).Value)
    On Error GoTo Fin

    If showIt Then
        fu.Display
        PrepareFollowUp = 1
    Else
        fu.Save
        PrepareFollowUp = 2
    End If
    On Error Resume Next
    If showIt Then ol.ActiveWindow.Activate
    RemoveTempDir tmpDir
    Exit Function
Fin:
    mLastFollowErr = Err.Number & " - " & Err.Description
    On Error Resume Next
    RemoveTempDir tmpDir
    PrepareFollowUp = -1
End Function

'=============================================================================
' COCHER LES RETARDS (v1.4 - F4-16) : coche les suivis en cours dont la date
' limite est passee (lendemain de la date limite, comme la colonne
' Echeance), decoche les autres ; Relancer prepare ensuite leurs brouillons.
' Une affectation pour toute la colonne.
'=============================================================================
Private Sub DoCheckLate(ByVal ws As Worksheet)
    On Error GoTo Fin
    If COL_MARK < 1 Or COL_ID < 1 Or COL_STATUS < 1 Or COL_DUE < 1 Then Exit Sub
    Dim lastRow As Long
    lastRow = LastDataRow(ws)
    If lastRow < 2 Then Exit Sub
    Dim ids As Variant, sts As Variant, dues As Variant, marks() As Variant
    Dim i As Long, n As Long, nLate As Long
    n = lastRow - 1
    ids = ReadCol(ws, COL_ID, lastRow)
    sts = ReadCol(ws, COL_STATUS, lastRow)
    dues = ReadCol(ws, COL_DUE, lastRow)
    ReDim marks(1 To n, 1 To 1)
    For i = 1 To n
        marks(i, 1) = ""
        If Len(CellText(ids(i, 1))) > 0 Then
            If IsLate(CellText(sts(i, 1)), dues(i, 1)) Then
                marks(i, 1) = ChrW$(MARK_GLYPH)
                nLate = nLate + 1
            End If
        End If
    Next i
    Application.EnableEvents = False
    ws.Range(ws.Cells(2, COL_MARK), ws.Cells(lastRow, COL_MARK)).Value = marks
    Application.EnableEvents = True
    RefreshRibbon
    If nLate = 0 Then
        SetStatus A("Aucun suivi en retard.")
    Else
        SetStatus nLate & A(" suivi(s) en retard coch~e(s) : Relancer pr~epare leurs brouillons.")
    End If
    ScheduleStatusClear 20
    Exit Sub
Fin:
    Application.EnableEvents = True
    MsgBox A("Cocher les retards impossible : ") & Err.Number & " - " & Err.Description, vbExclamation, APP_TITLE
End Sub

'=============================================================================
' PROLONGER (une ou plusieurs lignes) : la nouvelle date limite se compte a
' partir d'AUJOURD'HUI (pas de la date d'envoi). La date limite etant
' DERIVEE du delai (t0 + delai, t0 = derniere relance sinon date d'envoi),
' on ajuste le DELAI pour que la prolongation survive au recalcul d'Outlook.
' Une date peut etre saisie directement. Prolonger acquitte les reponses
' deja recues (comme ExtendRow cote Outlook).
'=============================================================================
Private Sub DoExtend(ByVal ws As Worksheet, ByVal rowList As Collection)
    On Error GoTo Fin
    Dim head As String
    If rowList.Count = 1 Then
        Dim curDue As Date
        curDue = 0
        On Error Resume Next
        curDue = CDate(ws.Cells(CLng(rowList(1)), COL_DUE).Value)
        On Error GoTo Fin
        head = A("Date limite actuelle : ") & DueText(curDue)
    Else
        head = rowList.Count & A(" lignes s~electionn~ees.")
    End If

    Dim prompt As String
    prompt = head & vbCrLf & vbCrLf & _
             A("La nouvelle date limite se compte ~a partir d'AUJOURD'HUI (") & _
             Format$(Date, "dd/mm/yyyy") & ")," & vbCrLf & _
             A("pas ~a partir de la date d'envoi du mail.") & vbCrLf & _
             A("Exemple : 7 -> ") & Format$(Date + 7, "dd/mm/yyyy") & vbCrLf & vbCrLf & _
             A("Nombre de jours, ou une date (jj/mm/aaaa) :")

    Dim s As String
    s = InputBox(prompt, A("Prolonger le d~elai de r~eponse"), "7")
    If Len(Trim$(s)) = 0 Then Exit Sub

    Dim newDue As Date
    If Not ParseExtendInput(s, newDue) Then
        MsgBox A("Saisie non reconnue : indiquez un nombre de jours (ex. 7) ou une date future (ex. ") & _
               Format$(Date + 7, "dd/mm/yyyy") & ").", vbExclamation, A("Prolonger le d~elai de r~eponse")
        Exit Sub
    End If

    Application.EnableEvents = False
    Dim v As Variant, r As Long, t0 As Date
    For Each v In rowList
        r = CLng(v)
        t0 = RowFloor(ws, r)
        ws.Cells(r, COL_DELAY).Value = DateDiff("d", Int(t0), newDue)
        If COL_ACK > 0 And COL_RESPONSE > 0 Then
            ws.Cells(r, COL_ACK).Value = ws.Cells(r, COL_RESPONSE).Value
        End If
        RecomputeRow ws, r
    Next v
    ClearMarks ws, rowList
    Application.EnableEvents = True
    RefreshRibbon
    Exit Sub
Fin:
    Application.EnableEvents = True
    MsgBox A("Prolongation impossible : ") & Err.Number & " - " & Err.Description, _
           vbExclamation, "Prolonger"
End Sub

'=============================================================================
' VALIDER LA REPONSE (une ou plusieurs lignes) : lignes "Repondu ?" (reponse
' recue, pas encore confirmee) -> acquittement (AckCount = nb de reponses)
' et statut "Repondu". Cote Outlook, une reponse SUPPLEMENTAIRE re-signalera
' la ligne. Comparaison sur le statut ACCENTUE (la feuille v6 cherchait
' "Repondu ?" sans accent : la validation etait toujours refusee).
'=============================================================================
Private Sub DoValidate(ByVal ws As Worksheet, ByVal rowList As Collection)
    On Error GoTo Fin
    Dim pending As String
    pending = "R" & ChrW$(233) & "pondu ?"

    Dim v As Variant, r As Long, st As String
    Dim nDone As Long, nSkipped As Long, firstSkipped As String
    Application.EnableEvents = False
    For Each v In rowList
        r = CLng(v)
        st = Trim$(CStr(ws.Cells(r, COL_STATUS).Value))
        If StrComp(st, pending, vbTextCompare) = 0 Then
            ws.Cells(r, COL_ACK).Value = ws.Cells(r, COL_RESPONSE).Value
            ws.Cells(r, COL_STATUS).Value = "R" & ChrW$(233) & "pondu"
            StyleStatus ws, r, "R" & ChrW$(233) & "pondu"
            EnsureDeadlineFormula ws, r
            nDone = nDone + 1
        Else
            nSkipped = nSkipped + 1
            If Len(firstSkipped) = 0 Then firstSkipped = st
        End If
    Next v
    ClearMarks ws, rowList
    Application.EnableEvents = True
    RefreshRibbon

    If nDone = 0 Then
        MsgBox A("Rien ~a valider : le statut est ") & firstSkipped & "." & vbCrLf & vbCrLf & _
               A("Valider la r~eponse ne concerne que les lignes au statut R~epondu ? : ") & _
               A("une r~eponse re~cue, pas encore confirm~ee."), vbInformation, A("Valider la r~eponse")
    ElseIf nSkipped > 0 Then
        SetStatus nDone & A(" r~eponse(s) valid~ee(s), ") & nSkipped & A(" ligne(s) ignor~ee(s) (statut diff~erent de R~epondu ?).")
        ScheduleStatusClear 15
    End If
    Exit Sub
Fin:
    Application.EnableEvents = True
    MsgBox A("Validation impossible : ") & Err.Description, vbExclamation, A("Valider la r~eponse")
End Sub

' ARCHIVER (une ou plusieurs lignes) : les lignes quittent le suivi courant
' pour la feuille "Archives" (creee a la premiere utilisation). Colonnes
' techniques conservees : Ouvrir le mail / Ouvrir le dossier / Restaurer
' restent possibles depuis les archives. Cote Outlook, FindRowByConv ne
' cherche que dans Suivi : un nouveau mail sur une conversation archivee
' cree une ligne NEUVE (voulu).
' v1.4 : plus de confirmation (F4-19 : l'action se defait par Restaurer),
' compte rendu dans la barre d'etat ; ligne d'origine supprimee SEULEMENT
' si sa copie est bien arrivee (F4-13) ; ligne de destination calculee sous
' les lignes masquees par un filtre (F4-01).
'=============================================================================
Private Sub DoArchive(ByVal ws As Worksheet, ByVal rowList As Collection)
    On Error GoTo Fin
    Dim wsA As Worksheet
    Set wsA = EnsureArchiveSheet(ws)
    If wsA Is Nothing Then Exit Sub

    Application.EnableEvents = False
    Application.ScreenUpdating = False
    Dim i As Long, r As Long, destRow As Long, nDone As Long, nFail As Long
    For i = rowList.Count To 1 Step -1
        r = CLng(rowList(i))
        destRow = LastDataRow(wsA) + 1

        If CopyRowTo(ws, r, wsA, destRow) Then
            If COL_ARCHIVED_AT > 0 Then wsA.Cells(destRow, COL_ARCHIVED_AT).Value = Format$(Now, "dd/mm/yyyy hh:nn")
            ws.Rows(r).Delete
            nDone = nDone + 1

            nFail = nFail + 1

    Next i
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    RefreshRibbon
    If nDone > 0 Then
        SetStatus nDone & A(" suivi(s) archiv~e(s) - Restaurer depuis la feuille ") & SH_ARCHIVE & "."
        ScheduleStatusClear 20
    End If
    If nFail > 0 Then
        MsgBox nFail & A(" ligne(s) non archiv~ee(s) : la copie vers la feuille ") & SH_ARCHIVE & _
               A(" a ~echou~e, elles restent dans ") & SH_TRACK & ".", vbExclamation, "Archiver"

Fin:

    Application.EnableEvents = True
    MsgBox A("Archivage impossible : ") & Err.Description, vbExclamation, "Archiver"
End Sub

' Copie une ligne : valeurs et formats de date, coche effacee, couleur du
' statut ; pas de copie des mises en forme conditionnelles.
' v1.4 (F4-13) : SANS presse-papiers (avant : Copy / PasteSpecial sous On
' Error Resume Next, sans resultat rendu : un collage rate - presse-papiers
' pris par une autre application - laissait la destination vide et
' l'appelant supprimait quand meme la ligne d'origine ; le contenu copie
' par l'utilisateur etait en outre perdu a chaque archivage). Les textes
' gardent leur nature de texte (prefixe ', comme le collage de valeurs :
' un objet "12/03" n'est pas transforme en date). True seulement si la
' ligne est bien arrivee (meme identifiant de suivi).
Private Function CopyRowTo(ByVal src As Worksheet, ByVal r As Long, ByVal dst As Worksheet, _
                           ByVal destRow As Long) As Boolean
    CopyRowTo = False
    On Error GoTo Fin
    If COL_ID < 1 Then Exit Function
    Dim n As Long, c As Long
    n = LastKnownCol()
    Dim vals As Variant
    vals = src.Range(src.Cells(r, 1), src.Cells(r, n)).Value
    For c = 1 To n
        If VarType(vals(1, c)) = vbString Then
            If Len(vals(1, c)) > 0 Then vals(1, c) = "'" & vals(1, c)
        End If
    Next c
    dst.Range(dst.Cells(destRow, 1), dst.Cells(destRow, n)).Value = vals
    CopyNumberFormat src, r, dst, destRow, COL_SENT
    CopyNumberFormat src, r, dst, destRow, COL_DUE
    CopyNumberFormat src, r, dst, destRow, COL_LASTFU
    If COL_MARK > 0 Then dst.Cells(destRow, COL_MARK).ClearContents
    StyleStatus dst, destRow, Trim$(CellText(dst.Cells(destRow, COL_STATUS).Value))

    Dim idS As String, idD As String
    idS = CellText(src.Cells(r, COL_ID).Value)
    idD = CellText(dst.Cells(destRow, COL_ID).Value)
    CopyRowTo = (Len(idS) > 0 And idS = idD)
    Exit Function
Fin:
    CopyRowTo = False
End Function

' Derniere colonne utile d'une ligne de suivi (colonne "Archive le" comprise).
Private Function LastKnownCol() As Long
    Dim m As Long
    m = COL_ARCHIVED_AT
    If COL_ACK > m Then m = COL_ACK
    If COL_ENTRY > m Then m = COL_ENTRY
    If COL_CONV > m Then m = COL_CONV
    If COL_ID > m Then m = COL_ID
    If COL_NBFU > m Then m = COL_NBFU
    LastKnownCol = m
End Function

Private Sub CopyNumberFormat(ByVal src As Worksheet, ByVal r As Long, ByVal dst As Worksheet, _
                             ByVal destRow As Long, ByVal col As Long)
    On Error Resume Next
    If col < 1 Then Exit Sub
    dst.Cells(destRow, col).NumberFormat = src.Cells(r, col).NumberFormat
End Sub

' Copie ratee : la ligne de destination (vide avant la copie) est videe.
Private Sub ClearRowSafe(ByVal ws As Worksheet, ByVal r As Long)
    On Error Resume Next
    If r < 2 Then Exit Sub
    ws.Range(ws.Cells(r, 1), ws.Cells(r, LastKnownCol())).ClearContents
End Sub

'=============================================================================
' EnsureArchiveSheet - cree la feuille "Archives" a la premiere utilisation,
' avec la meme ligne d'en-tete que le suivi et les memes colonnes masquees.
' v1.4 : en-tete copie SANS presse-papiers (Copy Destination : valeurs et
' formats, donc le bandeau de la charte commune, F8-22).
'=============================================================================
Private Function EnsureArchiveSheet(ByVal ws As Worksheet) As Worksheet
    On Error Resume Next
    Dim wb As Workbook
    Set wb = ws.Parent
    Dim wsA As Worksheet
    Set wsA = Nothing
    Set wsA = wb.Worksheets(SH_ARCHIVE)
    If Not wsA Is Nothing Then
        Set EnsureArchiveSheet = wsA
        Exit Function
    End If

    Err.Clear
    Set wsA = wb.Worksheets.Add(After:=ws)
    If wsA Is Nothing Then Exit Function
    wsA.Name = SH_ARCHIVE

    ws.Rows(1).Copy Destination:=wsA.Rows(1)
    wsA.Rows(1).RowHeight = ws.Rows(1).RowHeight
    If COL_ARCHIVED_AT > 0 Then

        wsA.Cells(1, COL_ARCHIVED_AT).Font.Color = ws.Cells(1, COL_SUBJECT).Font.Color

        wsA.Columns(c).ColumnWidth = ws.Columns(c).ColumnWidth
        wsA.Columns(c).Hidden = ws.Columns(c).Hidden
        wsA.Columns(c).WrapText = ws.Columns(c).WrapText
    Next c
    EnsureSetup wsA

    wsA.Rows(1).Font.Bold = True
    wsA.Activate
    wsA.Cells(2, 1).Select
    ActiveWindow.FreezePanes = True
    ws.Activate

    Set EnsureArchiveSheet = wsA
End Function

Private Sub DoRefresh()

' v1.4 - Demande d'actualisation a Outlook (appel direct, sinon note de
' commande + accuse attendu). False si elle n'a pas pu partir (message si
' showErrors).
Private Function RequestOutlookRefresh(ByVal ol As Object, ByVal showErrors As Boolean) As Boolean
    RequestOutlookRefresh = False
    Dim errNo As Long
    On Error Resume Next
    Err.Clear
    ol.RefreshMailTracking ThisWorkbook
    errNo = Err.Number
    Err.Clear
    On Error GoTo 0
    If errNo = 0 Then
        RequestOutlookRefresh = True
        Exit Function
    End If

    Dim why As String
    If Not SendOutlookCommand(ol, "REFRESH", why) Then
        If showErrors Then
            MsgBox A("Outlook n'a pas re~cu la demande d'actualisation : ") & why & vbCrLf & vbCrLf & _
                   A("En attendant : bouton Suivi Mails du ruban Outlook."), vbExclamation, "Actualiser"
        End If
        Exit Function
    End If
    ScheduleAckCheck
    RequestOutlookRefresh = True
End Function

' v1.2 - Ecrit une commande pour Outlook dans la note unique du dossier
' Notes\ClassicOutlook-Toolkit (creee si le dossier est vide). Outlook
' (ThisOutlookSession v10.16) surveille ce dossier et execute la commande.
' False + raison dans why si impossible.
' v1.4 (F4-19) : texte envoye memorise pour le controle de l'accuse.
Private Function SendOutlookCommand(ByVal ol As Object, ByVal cmd As String, ByRef why As String) As Boolean
    SendOutlookCommand = False
    why = ""
    On Error Resume Next

    Dim notesFld As Object
    Set notesFld = Nothing
    Set notesFld = ol.Session.GetDefaultFolder(OL_FOLDER_NOTES)
    If notesFld Is Nothing Then
        why = A("dossier Notes d'Outlook inaccessible.")
        Exit Function
    End If

    Dim fld As Object
    Set fld = Nothing
    Set fld = notesFld.Folders(CMD_FOLDER_NAME)
    If fld Is Nothing Then
        why = A("le dossier de commandes n'existe pas encore. Collez ThisOutlookSession v10.16 dans Outlook, ") & _
              A("puis fermez et rouvrez Outlook.")
        Exit Function
    End If

    Dim itm As Object
    Set itm = Nothing
    Set itm = fld.Items.GetFirst
    If itm Is Nothing Then
        Err.Clear
        Set itm = fld.Items.Add(OL_NOTE_ITEM)
    End If
    If itm Is Nothing Then
        why = A("cr~eation de la note de commande impossible.")
        Exit Function
    End If

    Dim body As String
    body = "TOOLKIT|" & cmd & "|" & Format$(Now, "yyyymmddhhnnss") & "-" & _
           Format$(Int(Timer * 1000) Mod 1000, "000")
    Err.Clear
    itm.Body = body
    itm.Save
    If Err.Number <> 0 Then
        why = A("~ecriture de la note de commande impossible (erreur ") & Err.Number & ")."
        Err.Clear
        Exit Function
    End If
    mCmdBody = body
    mCmdName = cmd
    mAckTries = 0
    SendOutlookCommand = True
End Function

'=============================================================================
' ACCUSE DE RECEPTION (v1.4 - F4-19). Outlook (modMailResponseTracker v6.7)
' reecrit la note "TOOLKIT|..." en "ACK|horodatage|commande|etat" des qu'il
' prend la demande (etat BUSY : une actualisation tournait deja). Controle
' ACK_FIRST_SEC s apres l'envoi, puis encore ACK_SECOND_SEC s plus tard
' (Outlook occupe) ; note toujours inchangee : Outlook n'ecoute plus le
' canal de commandes, et l'utilisateur le sait (avant : "Actualisation en
' cours..." pendant 2 min, puis rien).
'=============================================================================
Private Sub ScheduleAckCheck()
    On Error Resume Next
    CancelAckCheck
    mAckAt = Now + TimeSerial(0, 0, ACK_FIRST_SEC)
    Application.OnTime mAckAt, "'" & ThisWorkbook.Name & "'!CheckCommandAck"
End Sub

Private Sub CancelAckCheck()
    On Error Resume Next
    If mAckAt <> 0 Then Application.OnTime mAckAt, "'" & ThisWorkbook.Name & "'!CheckCommandAck", , False
    mAckAt = 0
End Sub

Public Sub CheckCommandAck()
    On Error Resume Next
    mAckAt = 0
    If Len(mCmdBody) = 0 Then Exit Sub
    Dim ol As Object
    Set ol = Nothing
    Set ol = GetObject(, "Outlook.Application")
    If ol Is Nothing Then Exit Sub                    ' Outlook ferme entre-temps : rien a dire

    Dim itm As Object, body As String
    Set itm = CommandNote(ol)
    If itm Is Nothing Then Exit Sub
    body = ""
    body = Trim$(Replace(Replace(CStr(itm.Body), vbCr, ""), vbLf, ""))
    If Left$(body, 4) = "ACK|" Then
        mCmdBody = ""
        If InStr(1, body, "|BUSY", vbTextCompare) > 0 Then
            SetStatus A("Outlook ~etait d~ej~a en train de traiter le suivi : r~esultat dans un instant.")
            ScheduleStatusClear STATUS_TIMEOUT_SEC
        End If
        Exit Sub
    End If
    If body <> mCmdBody Then                          ' autre commande depuis : rien a conclure
        mCmdBody = ""
        Exit Sub
    End If

    mAckTries = mAckTries + 1
    If mAckTries < 2 Then
        mAckAt = Now + TimeSerial(0, 0, ACK_SECOND_SEC)
        Application.OnTime mAckAt, "'" & ThisWorkbook.Name & "'!CheckCommandAck"
        Exit Sub
    End If

' REPARER (v1.3) : lignes dont l'objet ne correspond plus au mail ouvert, au

' AfterOutlookRefresh. Confirmation : l'action reecrit des colonnes.

              vbQuestion + vbYesNo, A("R~eparer")) <> vbYes Then Exit Sub

    Set ol = GetOutlook()
    If ol Is Nothing Then Exit Sub

               A("En attendant : macro RepairTrackerRows dans Outlook (Alt+F8)."), vbExclamation, A("R~eparer")
        Exit Sub
    End If
    ScheduleAckCheck
    SetStatus A("R~eparation en cours dans Outlook...")
    ScheduleStatusClear STATUS_TIMEOUT_SEC
End Sub

'=============================================================================
' RECALCUL (date limite derivee du delai + statut binaire ; echeance par
' formule)
'=============================================================================
Private Sub RecomputeRow(ByVal ws As Worksheet, ByVal r As Long)
    On Error Resume Next
    Dim delayDays As Long
    delayDays = 0
    delayDays = CLng(ws.Cells(r, COL_DELAY).Value)

    Dim t0 As Date
    t0 = RowFloor(ws, r)
    Dim due As Date
    due = DateAdd("d", delayDays, t0)
    ws.Cells(r, COL_DUE).Value = due
    ws.Cells(r, COL_DUE).NumberFormat = "dd/mm/yyyy"
    EnsureDeadlineFormula ws, r

    ' Statuts terminaux : on ne touche pas (Outlook gere "Repondu"/"Traite").
    Dim st As String
    st = CStr(ws.Cells(r, COL_STATUS).Value)
    If st = "R" & ChrW$(233) & "pondu" Then Exit Sub
    If st = "Trait" & ChrW$(233) Then Exit Sub

    Dim fc As Long
    fc = 0
    fc = CLng(ws.Cells(r, COL_NBFU).Value)

    ' v1.4 (F4-15) : en retard le LENDEMAIN de la date limite (jours
    ' entiers, comme la colonne Echeance) ; avant : des l'heure d'envoi +
    ' delai, et un suivi urgent du jour etait en retard des sa creation.
    Dim ns As String
    If Date > Int(CDbl(due)) Then
        ns = "En retard"
    ElseIf fc > 0 Then
        ns = "Relanc" & ChrW$(233)
    Else
        ns = "En attente"
    End If
    ws.Cells(r, COL_STATUS).Value = ns
    StyleStatus ws, r, ns
End Sub

' t0 d'une ligne : derniere relance si presente, sinon date d'envoi.
Private Function RowFloor(ByVal ws As Worksheet, ByVal r As Long) As Date
    On Error Resume Next
    Dim lf As Date, sd As Date
    lf = 0: sd = 0
    sd = CDate(ws.Cells(r, COL_SENT).Value)
    If Len(CStr(ws.Cells(r, COL_LASTFU).Value)) > 0 Then lf = CDate(ws.Cells(r, COL_LASTFU).Value)
    If lf > sd Then RowFloor = lf Else RowFloor = sd
End Function

Private Function RowSentDate(ByVal ws As Worksheet, ByVal r As Long) As Date
    On Error Resume Next
    RowSentDate = 0
    RowSentDate = CDate(ws.Cells(r, COL_SENT).Value)
End Function

' Echeance = formule (J-x / Aujourd'hui / Retard x j, vide pour Repondu,
' Repondu ? et Traite), ecrite si la cellule n'en a pas. Meme formule que
' modMailResponseTracker.DeadlineFormulaR1C1 (Outlook).
Private Sub EnsureDeadlineFormula(ByVal ws As Worksheet, ByVal r As Long)
    On Error Resume Next
    If COL_DEADLINE < 1 Or COL_STATUS < 1 Or COL_DUE < 1 Then Exit Sub
    Dim c As Range
    Set c = ws.Cells(r, COL_DEADLINE)
    Dim hasF As Boolean
    hasF = False
    hasF = CBool(c.HasFormula)
    If hasF Then Exit Sub
    Dim st As String, du As String
    st = "RC" & COL_STATUS
    du = "RC" & COL_DUE
    c.FormulaR1C1 = "=IF(OR(" & st & "=""R" & ChrW$(233) & "pondu""," & st & "=""R" & ChrW$(233) & "pondu ?""," & _
        st & "=""Trait" & ChrW$(233) & """," & du & "=""""),""""," & _
        "IFERROR(IF(INT(" & du & ")-TODAY()>0,""J-""&(INT(" & du & ")-TODAY())," & _
        "IF(INT(" & du & ")-TODAY()=0,""Aujourd'hui"",""Retard ""&(TODAY()-INT(" & du & "))&"" j"")),""""))"
End Sub

Private Sub StyleStatus(ByVal ws As Worksheet, ByVal r As Long, ByVal status As String)
    On Error Resume Next
    If COL_STATUS < 1 Then Exit Sub
    Dim c As Range
    Set c = ws.Cells(r, COL_STATUS)
    c.Font.Bold = True
    Select Case status
        Case "R" & ChrW$(233) & "pondu"
            c.Interior.Color = RGB(198, 239, 206): c.Font.Color = RGB(0, 97, 0)
        Case "R" & ChrW$(233) & "pondu ?"
            c.Interior.Color = RGB(255, 217, 179): c.Font.Color = RGB(197, 90, 17)
        Case "En retard"
            c.Interior.Color = RGB(255, 174, 174): c.Font.Color = RGB(156, 0, 0)
        Case "Relanc" & ChrW$(233)
            c.Interior.Color = RGB(197, 217, 241): c.Font.Color = RGB(31, 73, 125)
        Case "Trait" & ChrW$(233)
            c.Interior.Color = RGB(217, 217, 217): c.Font.Color = RGB(64, 64, 64)
        Case Else
            c.Interior.Color = RGB(230, 230, 230): c.Font.Color = RGB(89, 89, 89)
    End Select
End Sub

' MAIL DEPLACE (v1.4 - F4-19). L'identifiant du mail d'une ligne n'est plus

    Else

' Memorise l'action et ses lignes (par identifiant de suivi : le tri
' d'Outlook deplace les lignes), puis demande l'actualisation. False si
' deja en reprise (pas de boucle), ligne sans identifiant ou demande
' impossible.
Private Function QueueRetry(ByVal act As String, ByVal ws As Worksheet, ByVal rowSet As Collection) As Boolean
    QueueRetry = False
    On Error GoTo Fin
    If mReplaying Then Exit Function
    Dim ids As String, v As Variant, idv As String
    ids = ";"
    For Each v In rowSet
        idv = CellText(ws.Cells(CLng(v), COL_ID).Value)
        If Len(idv) = 0 Then Exit Function
        ids = ids & idv & ";"
    Next v
    Dim ol As Object
    Set ol = Nothing
    On Error Resume Next
    Set ol = GetObject(, "Outlook.Application")
    On Error GoTo Fin
    If ol Is Nothing Then Exit Function
    If Not RequestOutlookRefresh(ol, False) Then Exit Function
    mPendingAct = act
    mPendingIds = ids
    mPendingSheet = ws.Name
    mPendingAt = Now
    SetStatus A("Mail d~eplac~e : Outlook le recherche, l'action reprendra seule ~a la fin de l'actualisation...")
    ScheduleStatusClear STATUS_TIMEOUT_SEC
    QueueRetry = True
    Exit Function
Fin:
    QueueRetry = False
End Function

' Reprise de l'action en attente (rendez-vous pose par AfterOutlookRefresh).
' Demande de plus de 10 min : oubliee.
Public Sub ReplayPendingAction()
    On Error GoTo Fin
    mReplayAt = 0
    If Len(mPendingAct) = 0 Then Exit Sub
    Dim act As String, ids As String, shName As String, askedAt As Date
    act = mPendingAct
    ids = mPendingIds
    shName = mPendingSheet
    askedAt = mPendingAt
    mPendingAct = ""
    mPendingIds = ""
    mPendingSheet = ""
    If Now - askedAt > TimeSerial(0, 10, 0) Then Exit Sub

    Dim ws As Worksheet
    Set ws = Nothing
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(shName)
    On Error GoTo Fin
    If ws Is Nothing Then Exit Sub
    EnsureSchema ThisWorkbook
    If Not LayoutMatches(ws) Then Exit Sub
    Dim rowSet As Collection
    Set rowSet = RowsByIds(ws, ids)
    If rowSet.Count = 0 Then Exit Sub

    mReplaying = True
    Select Case act
        Case "OPEN"
            DoOpen ws, CLng(rowSet(1))
        Case "OPEN_FOLDER"
            DoOpenFolder ws, CLng(rowSet(1))
        Case "FOLLOWUP"
            DoFollowUpRows ws, rowSet
    End Select
    mReplaying = False
    Exit Sub
Fin:
    mReplaying = False
End Sub

' Lignes (ordre croissant) dont l'identifiant de suivi figure dans ids
' (";ID1;ID2;").
Private Function RowsByIds(ByVal ws As Worksheet, ByVal ids As String) As Collection
    Dim res As Collection
    Set res = New Collection
    Set RowsByIds = res
    On Error Resume Next
    If COL_ID < 1 Then Exit Function
    Dim lastRow As Long, v As Variant, i As Long, idv As String
    lastRow = LastDataRow(ws)
    If lastRow < 2 Then Exit Function
    v = ReadCol(ws, COL_ID, lastRow)
    For i = 1 To lastRow - 1
        idv = CellText(v(i, 1))
        If Len(idv) > 0 Then
            If InStr(1, ids, ";" & idv & ";", vbBinaryCompare) > 0 Then res.Add i + 1
        End If
    Next i
End Function

'=============================================================================
' FICHIER DES ECHEANCES (v1.4 - F4-07) : apres chaque enregistrement du
' classeur (ThisWorkbook.Workbook_AfterSave), une ligne par suivi de la
' feuille Suivi qui n'est ni "Repondu" ni "Traite" : "date limite
' aaaa-mm-jj <TAB> statut <TAB> objet", dans <dossier de l'outil>\Sorties\
' suivi-echeances.txt (dossier de l'outil lu dans le registre, ou Outlook
' le memorise). Meme format que modMailResponseTracker.WriteDueCache
' (Outlook) : le rappel du matin d'Outlook le lit sans ouvrir Excel, et
' voit ainsi les Prolonger / Valider / Archiver faits ici. Silencieux.
'=============================================================================
Public Sub WriteDueCache()
    Dim f As Integer
    f = 0
    On Error GoTo Fin
    Dim path As String
    path = DueCachePath()
    If Len(path) = 0 Then Exit Sub
    EnsureSchema ThisWorkbook
    Dim ws As Worksheet
    Set ws = TrackSheet()
    If ws Is Nothing Then Exit Sub
    If Not LayoutMatches(ws) Then Exit Sub
    If COL_ID < 1 Or COL_STATUS < 1 Or COL_DUE < 1 Or COL_SUBJECT < 1 Then Exit Sub

    Dim lastRow As Long, i As Long
    Dim txt As String, st As String, dueV As Date
    txt = "# ClassicOutlook-Toolkit suivi-echeances v1 " & Format$(Now, "yyyy-mm-dd hh:nn:ss") & vbCrLf
    lastRow = LastDataRow(ws)
    If lastRow >= 2 Then
        Dim ids As Variant, sts As Variant, dues As Variant, subs As Variant
        ids = ReadCol(ws, COL_ID, lastRow)
        sts = ReadCol(ws, COL_STATUS, lastRow)
        dues = ReadCol(ws, COL_DUE, lastRow)
        subs = ReadCol(ws, COL_SUBJECT, lastRow)
        For i = 1 To lastRow - 1
            If Len(CellText(ids(i, 1))) > 0 Then
                st = Trim$(CellText(sts(i, 1)))
                If st <> "R" & ChrW$(233) & "pondu" And st <> "Trait" & ChrW$(233) Then
                    dueV = CellDate(dues(i, 1))
                    If dueV > 0 Then txt = txt & Format$(dueV, "yyyy-mm-dd")
                    txt = txt & vbTab & st & vbTab & OneLine(CellText(subs(i, 1))) & vbCrLf
                End If
            End If
        Next i
    End If
    f = FreeFile
    Open path For Output As #f
    Print #f, txt;
    Close #f
    f = 0
    Exit Sub
Fin:
    On Error Resume Next
    If f > 0 Then Close #f
End Sub

' <dossier de l'outil>\Sorties\suivi-echeances.txt ("" si le dossier de
' l'outil n'est pas memorise dans le registre, ou sans dossier Sorties :
' ancien emplacement a plat, Outlook ecrit alors seul le fichier).
Private Function DueCachePath() As String
    On Error Resume Next
    DueCachePath = ""
    Dim d As String
    d = Trim$(GetSetting("ClassicOutlookToolkit", "Paths", "ToolkitDir", ""))
    If Len(d) = 0 Then Exit Function
    If Right$(d, 1) = "\" Then d = Left$(d, Len(d) - 1)
    If Len(Dir(d & "\Sorties", vbDirectory)) = 0 Then Exit Function
    DueCachePath = d & "\Sorties\" & DUE_CACHE_FILE
End Function

' Texte sur une ligne, sans tabulation, 120 caracteres au plus.
Private Function OneLine(ByVal s As String) As String
    s = Replace(Replace(Replace(s, vbTab, " "), vbCr, " "), vbLf, " ")
    If Len(s) > 120 Then s = Left$(s, 120)
    OneLine = s
End Function

'=============================================================================
' Retrouve, dans le fil du mail de reference, LE mail suivi : envoye par
' soi, date d'envoi la plus proche de la "Date envoi" de la ligne. La
' reference d'une ligne peut etre n'importe quel mail du fil (apres une
' reparation d'ancrage cote Outlook) : sans ce tri, on risquerait de joindre
' les pieces jointes d'une REPONSE recue. Nothing si rien d'envoye par soi.
'=============================================================================
Private Function FindTrackedSentMail(ByVal ol As Object, ByVal src As Object, _
                                     ByVal sentDate As Date) As Object
    On Error Resume Next
    Set FindTrackedSentMail = Nothing

    Dim myAddr As String, myName As String
    myAddr = ""
    myName = ""
    myAddr = LCase$(CStr(ol.Session.CurrentUser.AddressEntry.Address))
    myName = CStr(ol.Session.CurrentUser.Name)

    Dim best As Object
    Set best = Nothing
    Dim bestGap As Double
    bestGap = 1E+30

    Dim conv As Object
    Set conv = Nothing
    Set conv = src.GetConversation
    If Not conv Is Nothing Then
        Dim storeID As String
        storeID = ""
        storeID = CStr(src.Parent.storeID)
        Dim tbl As Object
        Set tbl = conv.GetTable
        Dim guard As Long, rw As Object, itm As Object, gap As Double, cls As Long
        guard = 0
        Do Until tbl.EndOfTable
            guard = guard + 1
            If guard > 500 Then Exit Do
            Set rw = Nothing
            Set rw = tbl.GetNextRow
            If rw Is Nothing Then Exit Do
            Set itm = Nothing
            Set itm = ol.Session.GetItemFromID(CStr(rw.Item("EntryID")), storeID)
            If Not itm Is Nothing Then
                cls = 0
                cls = itm.Class
                If cls = OL_MAIL_CLASS Then
                    If IsFromMe(itm, myAddr, myName) Then
                        gap = 1E+30
                        gap = Abs(CDbl(itm.SentOn) - CDbl(sentDate))
                        If gap < bestGap Then
                            bestGap = gap
                            Set best = itm
                        End If
                    End If
                End If
            End If
        Loop
    End If

    If best Is Nothing Then
        If IsFromMe(src, myAddr, myName) Then Set best = src
    End If
    Set FindTrackedSentMail = best
End Function

Private Function IsFromMe(ByVal itm As Object, ByVal myAddr As String, ByVal myName As String) As Boolean
    On Error Resume Next
    IsFromMe = False
    Dim senderAddr As String
    senderAddr = ""
    senderAddr = LCase$(CStr(itm.SenderEmailAddress))
    If Len(myAddr) > 0 And Len(senderAddr) > 0 Then
        If senderAddr = myAddr Then
            IsFromMe = True
            Exit Function
        End If
    End If
    Dim senderNm As String
    senderNm = ""
    senderNm = CStr(itm.SenderName)
    If Len(myName) > 0 And Len(senderNm) > 0 Then
        If StrComp(senderNm, myName, vbTextCompare) = 0 Then IsFromMe = True
    End If
End Function

'=============================================================================
' Langue du mail suivi ("FR" ou "EN") : mots typiques comptes dans le texte
' propre du mail (avant l'historique cite). Francais en cas de doute.
'=============================================================================
Private Function DetectLanguage(ByVal itm As Object) As String
    On Error Resume Next
    DetectLanguage = "FR"
    Dim txt As String
    txt = ""
    txt = CStr(itm.Body)
    If Len(txt) = 0 Then Exit Function
    txt = OwnTextPart(txt)

    ' Lettres gardees (accents compris), tout le reste devient un espace.
    Dim out As String, i As Long, ch As String, code As Long
    out = Space$(Len(txt))
    For i = 1 To Len(txt)
        ch = LCase$(Mid$(txt, i, 1))
        code = AscW(ch)
        If (code >= 97 And code <= 122) Or code > 127 Then
            Mid$(out, i, 1) = ch
        End If
    Next i

    Dim words() As String, w As Variant, nFr As Long, nEn As Long
    words = Split(out, " ")
    For Each w In words
        If Len(w) > 0 Then
            If InStr(1, FR_WORDS, " " & w & " ", vbBinaryCompare) > 0 Then nFr = nFr + 1
            If InStr(1, EN_WORDS, " " & w & " ", vbBinaryCompare) > 0 Then nEn = nEn + 1
        End If
    Next w
    If nEn > nFr Then DetectLanguage = "EN"
End Function

' Texte du mail avant l'historique cite (reponse en haut), 3000 caracteres max.
Private Function OwnTextPart(ByVal txt As String) As String
    Dim marks As Variant, i As Long, p As Long, cut As Long
    marks = Array(vbLf & "De :", vbLf & "From:", "-----Message", "-----Original", " a " & ChrW$(233) & "crit :", " wrote:")
    cut = Len(txt) + 1
    For i = LBound(marks) To UBound(marks)
        p = InStr(1, txt, CStr(marks(i)), vbTextCompare)
        If p > 0 And p < cut Then cut = p
    Next i
    If cut > 3000 Then cut = 3000
    OwnTextPart = Left$(txt, cut - 1)
End Function

'=============================================================================
' Copie dans runDir les VRAIES pieces jointes d'un mail. Regles identiques a
' modThreadAttachments.Classify (Outlook) : fichiers et mails joints
' seulement ; ecartes : pieces masquees, images referencees dans le corps
' (cid:), images de signature (imageNNN / Outlook-xxx, < 30 Ko), liens
' cloud, fichiers bloques par Outlook, smime.p7s / smime.p7m, winmail.dat,
' .emz / .wmz. Un sous-dossier par fichier : le nom d'origine est garde.
'=============================================================================
Private Sub CopyRealAttachments(ByVal itm As Object, ByVal runDir As String, ByVal attPaths As Collection)
    On Error Resume Next
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    If fso Is Nothing Then Exit Sub
    Dim atts As Object, n As Long
    Set atts = Nothing
    Set atts = itm.Attachments
    If atts Is Nothing Then Exit Sub
    n = 0
    n = atts.Count

    Dim html As String, htmlLoaded As Boolean
    Dim j As Long, k As Long, att As Object, nm As String, subDir As String, p As String
    For j = 1 To n
        Set att = Nothing
        Set att = atts.Item(j)
        If Not att Is Nothing Then
            If IsRealAttachment(att, itm, html, htmlLoaded) Then
                nm = SafeFileName(AttFileName(att))
                If Len(nm) > 0 Then
                    k = k + 1
                    subDir = runDir & "\" & k
                    fso.CreateFolder subDir
                    p = subDir & "\" & nm
                    Err.Clear
                    att.SaveAsFile p
                    If Err.Number = 0 Then attPaths.Add p
                    Err.Clear
                End If
            End If
        End If
    Next j
End Sub

Private Function IsRealAttachment(ByVal att As Object, ByVal itm As Object, _
                                  ByRef html As String, ByRef htmlLoaded As Boolean) As Boolean
    IsRealAttachment = False
    On Error Resume Next

    Dim ty As Long
    ty = -1
    ty = CLng(att.Type)
    If ty <> 1 And ty <> 5 Then Exit Function              ' olByValue, olEmbeddeditem

    Dim pa As Object
    Set pa = Nothing
    Set pa = att.PropertyAccessor

    Dim meth As Long
    meth = 0
    If Not pa Is Nothing Then meth = CLng(pa.GetProperty(PR_ATTACH_METHOD))
    If meth = 7 Then Exit Function                         ' lien cloud

    Dim hiddenFlag As Boolean
    hiddenFlag = False
    If Not pa Is Nothing Then hiddenFlag = CBool(pa.GetProperty(PR_ATTACH_HIDDEN))
    If hiddenFlag Then Exit Function

    Dim bl As Long
    bl = 0
    bl = CLng(att.BlockLevel)
    If bl <> 0 Then Exit Function                          ' bloque par Outlook

    Dim nm As String
    nm = AttFileName(att)
    If Len(nm) = 0 Then Exit Function
    If IsTechnicalName(nm) Then Exit Function

    Dim cid As String
    cid = ""
    If Not pa Is Nothing Then cid = Trim$(CStr(pa.GetProperty(PR_ATTACH_CONTENT_ID)))
    cid = Replace(Replace(cid, "<", ""), ">", "")
    If Len(cid) > 0 Then
        If Not htmlLoaded Then
            html = ""
            html = CStr(itm.htmlBody)
            htmlLoaded = True
        End If
        If Len(html) > 0 Then
            If InStr(1, html, "cid:" & cid, vbTextCompare) > 0 Then Exit Function
        End If
    End If

    Dim sz As Long
    sz = 0
    sz = CLng(att.Size)
    If IsSignatureImage(nm, sz) Then Exit Function

    IsRealAttachment = True
End Function

Private Function AttFileName(ByVal att As Object) As String
    On Error Resume Next
    Dim nm As String, ty As Long
    nm = ""
    nm = Trim$(CStr(att.Filename))
    If Len(nm) = 0 Then nm = Trim$(CStr(att.DisplayName))
    ty = 0
    ty = CLng(att.Type)
    If ty = 5 Then                                         ' mail joint
        If Len(nm) = 0 Then nm = "message"
        If StrComp(Right$(nm, 4), ".msg", vbTextCompare) <> 0 Then nm = nm & ".msg"
    End If
    AttFileName = nm
End Function

Private Function IsTechnicalName(ByVal nm As String) As Boolean
    Dim s As String
    IsTechnicalName = False
    s = LCase$(Trim$(nm))
    Select Case s
        Case "smime.p7s", "smime.p7m", "winmail.dat"
            IsTechnicalName = True
            Exit Function
    End Select
    If Right$(s, 4) = ".emz" Or Right$(s, 4) = ".wmz" Then IsTechnicalName = True
End Function

Private Function IsSignatureImage(ByVal nm As String, ByVal sz As Long) As Boolean
    IsSignatureImage = False
    If sz <= 0 Or sz >= SIG_IMAGE_MAX Then Exit Function
    Dim s As String, p As Long, ext As String, stem As String
    s = LCase$(Trim$(nm))
    p = InStrRev(s, ".")
    If p <= 1 Then Exit Function
    ext = Mid$(s, p + 1)
    Select Case ext
        Case "png", "jpg", "jpeg", "gif", "bmp"
        Case Else
            Exit Function
    End Select
    stem = Left$(s, p - 1)
    If Left$(stem, 5) = "image" And Len(stem) > 5 Then
        IsSignatureImage = IsAllDigits(Mid$(stem, 6))
    ElseIf Left$(stem, 8) = "outlook-" Then
        IsSignatureImage = True
    End If
End Function

Private Function IsAllDigits(ByVal s As String) As Boolean
    IsAllDigits = False
    If Len(s) = 0 Then Exit Function
    Dim i As Long, ch As String
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        If ch < "0" Or ch > "9" Then Exit Function
    Next i
    IsAllDigits = True
End Function

' Dossier temporaire de la relance : %TEMP%\SuiviMails-PJ\<horodatage>. Les
' restes de plus d'un jour (session interrompue) sont purges au passage.
Private Function NewTempDir() As String
    On Error Resume Next
    NewTempDir = ""
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    If fso Is Nothing Then Exit Function
    Dim tmpRoot As String
    tmpRoot = Environ$("TEMP")
    If Len(tmpRoot) = 0 Then tmpRoot = Environ$("TMP")
    If Len(tmpRoot) = 0 Then Exit Function
    tmpRoot = tmpRoot & "\" & TEMP_SUBDIR

    Dim ok As Boolean
    ok = False
    ok = fso.FolderExists(tmpRoot)
    If Not ok Then fso.CreateFolder tmpRoot

    Dim old As Collection, sf As Object, age As Double, v As Variant
    Set old = New Collection
    For Each sf In fso.GetFolder(tmpRoot).SubFolders
        age = 0
        age = Now - sf.DateCreated
        If age > 1 Then old.Add CStr(sf.Path)
    Next sf
    For Each v In old
        fso.DeleteFolder CStr(v), True
    Next v

    Dim d As String
    d = tmpRoot & "\" & Format$(Now, "yyyymmdd-hhnnss")
    fso.CreateFolder d
    ok = False
    ok = fso.FolderExists(d)
    If ok Then NewTempDir = d
End Function

' Supprime le dossier temporaire d'une relance (seulement sous TEMP_SUBDIR).
Private Sub RemoveTempDir(ByVal d As String)
    On Error Resume Next
    If Len(d) = 0 Then Exit Sub
    If InStr(1, d, "\" & TEMP_SUBDIR & "\", vbTextCompare) = 0 Then Exit Sub
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    fso.DeleteFolder d, True
End Sub

Private Function SafeFileName(ByVal s As String) As String
    Dim Bad As Variant, i As Long
    Bad = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    For i = LBound(Bad) To UBound(Bad)
        s = Replace(s, CStr(Bad(i)), "_")
    Next i
    s = Trim$(s)
    If Len(s) > 150 Then s = Right$(s, 150)
    SafeFileName = s
End Function

'=============================================================================
' CORPS DE RELANCE (FR ou EN) + mention "generee automatiquement" (le
' brouillon est envoye a la main, d'ou "generee" et non "envoyee"). Avec des
' pieces jointes, une phrase les annonce.
'=============================================================================
Private Function BuildRelanceHtml(ByVal nAttach As Long, ByVal lang As String) As String
    Dim s As String, mention As String
    If lang = "EN" Then
        s = "Hello," & vbCrLf & _
            "I am following up on the message below, which is still awaiting a reply."
        If nAttach = 1 Then
            s = s & " Please find attached the document concerned."
        ElseIf nAttach > 1 Then
            s = s & " Please find attached the documents concerned."
        End If
        s = s & " Have you had a chance to review it, or would you need any further information from me?" & vbCrLf & _
            "I remain at your disposal and thank you in advance." & vbCrLf & _
            "Best regards,"
        mention = "Follow-up generated automatically."
    Else
        s = A("Bonjour,") & vbCrLf & _
            A("Je me permets de revenir vers vous au sujet de ce message, rest~e sans retour ~a ce jour.")
        If nAttach = 1 Then
            s = s & A(" Vous trouverez ci-joint le document concern~e.")
        ElseIf nAttach > 1 Then
            s = s & A(" Vous trouverez ci-joint les documents concern~es.")
        End If
        s = s & A(" Auriez-vous pu en prendre connaissance, ou avez-vous besoin d'un compl~ement de ma part ?") & _
            vbCrLf & A("Je reste ~a votre disposition et vous remercie par avance.") & vbCrLf & _
            A("Bien cordialement,")
        mention = A("Relance g~en~er~ee automatiquement.")
    End If

    Dim html As String
    html = "<div>" & Replace(EncodeHtml(s), vbCrLf, "<br>") & "</div>"
    html = html & "<div style=""color:#888888;font-size:9pt;font-style:italic;margin-top:2px"">" & _
                  EncodeHtml(mention) & "</div>"
    BuildRelanceHtml = html
End Function

Private Function InsertAfterBody(ByVal htmlBody As String, ByVal fragment As String) As String
    Dim posBody As Long, posEnd As Long
    posBody = InStr(1, htmlBody, "<body", vbTextCompare)
    If posBody > 0 Then
        posEnd = InStr(posBody, htmlBody, ">")
        If posEnd > 0 Then
            InsertAfterBody = Left$(htmlBody, posEnd) & fragment & _
                              StripLeadingBlanks(Mid$(htmlBody, posEnd + 1))
            Exit Function
        End If
    End If
    InsertAfterBody = fragment & StripLeadingBlanks(htmlBody)
End Function

Private Function StripLeadingBlanks(ByVal s As String) As String
    Dim pats() As String
    pats = Split("<p class=MsoNormal><o:p>&nbsp;</o:p></p>|" & _
                 "<p class=MsoNormal><o:p></o:p></p>|" & _
                 "<p class=MsoNormal>&nbsp;</p>|<p class=MsoNormal></p>|" & _
                 "<o:p>&nbsp;</o:p>|<o:p></o:p>|" & _
                 "<div>&nbsp;</div>|<div><br></div>|<div><br/></div>|<div></div>|" & _
                 "<p>&nbsp;</p>|<p></p>|&nbsp;|<br>|<br/>|<br />", "|")
    Dim changed As Boolean
    Do
        changed = False
        s = LTrimWS(s)
        Dim i As Long
        For i = LBound(pats) To UBound(pats)
            Do While Len(s) >= Len(pats(i)) And _
                     StrComp(Left$(s, Len(pats(i))), pats(i), vbTextCompare) = 0
                s = Mid$(s, Len(pats(i)) + 1)
                s = LTrimWS(s)
                changed = True
            Loop
        Next i
    Loop While changed
    StripLeadingBlanks = s
End Function

Private Function LTrimWS(ByVal s As String) As String
    Do While Len(s) > 0
        Dim ch As String
        ch = Left$(s, 1)
        If ch = " " Or ch = vbCr Or ch = vbLf Or ch = vbTab Then
            s = Mid$(s, 2)
        Else
            Exit Do
        End If
    Loop
    LTrimWS = s
End Function

Private Function EncodeHtml(ByVal s As String) As String
    s = Replace(s, "&", "&amp;")
    s = Replace(s, "<", "&lt;")
    s = Replace(s, ">", "&gt;")
    EncodeHtml = s
End Function

' Saisie "Prolonger" : nombre de jours a compter d'aujourd'hui, ou date
' (jj/mm/aaaa, jj/mm). False si illisible ou pas dans le futur.
Private Function ParseExtendInput(ByVal s As String, ByRef newDue As Date) As Boolean
    On Error GoTo Bad
    ParseExtendInput = False
    s = Trim$(s)
    If Len(s) = 0 Then Exit Function
    If InStr(1, s, "/") > 0 Or InStr(1, s, ".") > 0 Or InStr(1, s, "-") > 0 Then
        If Not IsDate(s) Then Exit Function
        newDue = DateValue(s)
    Else
        If Not IsNumeric(s) Then Exit Function
        Dim n As Long
        n = CLng(Val(s))
        If n <= 0 Then Exit Function
        newDue = Date + n
    End If
    If newDue <= Date Then Exit Function
    ParseExtendInput = True
    Exit Function
Bad:
    ParseExtendInput = False
End Function

' "29/07/2026 (depassee de 68 j)" / "(dans 12 j)" / "(aujourd'hui)".
Private Function DueText(ByVal d As Date) As String
    If d = 0 Then
        DueText = "-"
        Exit Function
    End If
    Dim n As Long
    n = DateDiff("d", Date, Int(d))
    DueText = Format$(d, "dd/mm/yyyy")
    If n > 0 Then
        DueText = DueText & " (dans " & n & " j)"
    ElseIf n = 0 Then
        DueText = DueText & " (aujourd'hui)"
    Else
        DueText = DueText & A(" (d~epass~ee de ") & (-n) & " j)"
    End If
End Function

' Objets des lignes visees, pour une confirmation (MAX_LISTED premiers).
Private Function SubjectList(ByVal ws As Worksheet, ByVal rowList As Collection) As String
    On Error Resume Next
    Dim s As String, i As Long, subj As String
    For i = 1 To rowList.Count
        If i > MAX_LISTED Then
            s = s & "  ... et " & (rowList.Count - MAX_LISTED) & " autre(s)" & vbCrLf
            Exit For
        End If
        subj = ""
        subj = CStr(ws.Cells(CLng(rowList(i)), COL_SUBJECT).Value)
        If Len(subj) > 70 Then subj = Left$(subj, 70) & "..."
        s = s & "  - " & subj & vbCrLf
    Next i
    SubjectList = s
End Function

'=============================================================================
' A : decode les marqueurs d'accent ASCII (table de modConfig.A() cote
' Outlook) : ~e e aigu, ~f e grave, ~a a grave, ~E E aigu, ~i i circ,
' ~o o circ, ~x e circ, ~c c cedille, ~u u circ, ~U u grave, ~A a circ,
' ~y i trema, ~I I circ. Marqueur inconnu : "~" conserve tel quel.
'=============================================================================
Private Function A(ByVal s As String) As String
    Dim out As String, i As Long, n As Long, ch As String
    n = Len(s)
    i = 1
    Do While i <= n
        ch = Mid$(s, i, 1)
        If ch = "~" And i < n Then
            Select Case Mid$(s, i + 1, 1)
                Case "e": out = out & ChrW$(233): i = i + 2
                Case "f": out = out & ChrW$(232): i = i + 2
                Case "a": out = out & ChrW$(224): i = i + 2
                Case "E": out = out & ChrW$(201): i = i + 2
                Case "i": out = out & ChrW$(238): i = i + 2
                Case "o": out = out & ChrW$(244): i = i + 2
                Case "x": out = out & ChrW$(234): i = i + 2
                Case "c": out = out & ChrW$(231): i = i + 2
                Case "u": out = out & ChrW$(251): i = i + 2
                Case "U": out = out & ChrW$(249): i = i + 2
                Case "A": out = out & ChrW$(226): i = i + 2
                Case "y": out = out & ChrW$(239): i = i + 2
                Case "I": out = out & ChrW$(206): i = i + 2
                Case Else
                    out = out & ch
                    i = i + 1
            End Select
        Else
            out = out & ch
            i = i + 1
        End If
    Loop
    A = out
End Function
