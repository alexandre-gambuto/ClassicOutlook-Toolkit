'=============================================================================
' modContactNormalize v1.5  (2026-08-20)
' Purpose (EN): harmonise contact cards across ALL contact folders of a
'   chosen store (or the current selection / open card): phone numbers to
'   E.164 (+33...), names to "Firstname LASTNAME" (particles lower case),
'   postal addresses (street in title case, city and country upper case),
'   e-mails lower case, text fields trimmed. Dry-run first, per-field report,
'   no UserForm.
'
' Harmonise les fiches contact de TOUS les dossiers de contacts du compte
' choisi (ou de la selection courante / de la fiche ouverte) :
'   - TELEPHONES en E.164 compact (+33612345678). 0X national -> +CC ;
'     00 -> + ; "+33 (0)6" -> +336 ; 9 chiffres sans le 0 -> +CC (signale
'     "suppose") ; numeros courts (<= 6 chiffres : 112, 3635) conserves ;
'     toute valeur contenant des lettres (poste, ext.) est laissee intacte
'     et signalee. Fax volontairement hors perimetre.
'   - NOMS : Prenom "Jean-Baptiste", Nom "de CHAISEMARTIN" (particules en
'     minuscules, liste config NameParticles). Une fiche dont le prenom OU
'     le nom est vide est reputee ENTITE (societe, service) et n'est pas
'     renommee. FileAs suit le motif deja en place ("Nom, Prenom" ou
'     "Prenom Nom") ; un FileAs personnalise n'est pas touche. FullName est
'     pose PUIS les composantes First/Middle/Last : resultat identique quel
'     que soit le sens de recomposition d'Outlook, verifie apres Save.
'   - ADRESSES : rue en casse de titre (1er mot alphabetique capitalise,
'     mots-outils en minuscules : de, la, du... ; sigles BP, CS, CEDEX en
'     majuscules ; jetons contenant des chiffres intacts), ville et pays en
'     MAJUSCULES (config), "92320 Chatillon" saisi dans le champ Ville ->
'     code postal + ville separes. Convention lisible, pas la norme postale
'     AFNOR (tout en capitales sans accents), volontairement.
'   - E-MAILS SMTP en minuscules ; champs texte (societe, fonction, service)
'     debarrasses des espaces parasites.
'   - CATEGORIES (v1.3) : NON traitees par l'harmonisation. Une purge est
'     proposee A LA FIN de ContactNormalizeAll si des fiches en portent
'     (confirmation TAPEE), ou lancable seule par ContactClearCategories.
'     Motif : Outlook sur le web PROJETTE les dossiers de contacts du client
'     lourd en categories. Une categorie stockee sur la fiche s'ajoute donc
'     a celle du dossier - deux etiquettes pour un seul rangement. Vider les
'     categories laisse le dossier faire foi.
'   - FETE (v1.1) : date de la fete du prenom (calendrier des saints usuel,
'     ~500 prenoms + equivalents italiens courants + config NameDayExtra)
'     posee dans la UserProperty "NameDay" au format MM-JJ. Prenom compose :
'     "Jean-X" / "Marie-X" se fetent au second prenom, les autres au premier
'     (Jean-Baptiste, Jean-Marie, Marie-Madeleine ont leur entree). Prenom
'     absent de la table = signale au rapport, rien n'est ecrit. Les entites
'     ne sont pas concernees. Lue ensuite par modContactBirthday.
'
' RAPPORT : <Bureau>\Macro VBA\Outlook\ContactNormalize-AAAAMMJJ-HHMMSS.txt
' (UTF-8) : une ligne par champ modifie (avant -> apres) + avertissements
' (numeros non normalisables). Ouverture proposee dans Notepad.
'
' FONCTION PUBLIQUE PARTAGEE (v1.5) :
'   PhoneFormatCanonical(raw) : formate un numero selon la convention du
'   toolkit (E.164 + PhoneOutputSpaced), "" si non reconnu. Source de verite
'   unique appelee par modContactFromMail (et tout futur module).
'
' SUBS PUBLIQUES (epinglables au QAT) :
'   ContactClearCategories     vide le champ Categories de toutes les fiches
'                              du compte choisi (Simulation / Appliquer)
'   ContactNormalizeAll        tout le compte choisi (Simulation / Appliquer)
'   ContactNormalizeSelected   la fiche ouverte ou la selection courante :
'                              c'est la SONDE - lancer sur 1-2 fiches avant
'                              le balayage complet
'
' A LANCER AVANT modContactBirthday (le sujet des evenements reprend
' Prenom / NOM tels qu'ils sont dans la fiche).
'
' Dependances : modConfig (Cfg/CfgB, T, A, MacroVBADir), modDebugLog,
'   modMsgBoxCustom.Confirm3.
'
' Cles config.ini (section CONTACTS - HARMONISATION) : NormalizeNames,
'   NormalizePhones, NormalizeAddresses, NormalizeEmails, NormalizeNameDay,
'   NameDayExtra, ClearCategoriesPrompt,
'   PhoneDefaultCountryCode, PhoneOutputSpaced, NameParticles,
'   ContactFileAsFormat, AddressLowercaseWords, AddressUppercaseWords,
'   AddressCityUppercase, AddressCountryUppercase, AddressDefaultCountry,
'   ContactSkipFolders (partagee avec modContactBirthday).
'=============================================================================

'=============================================================================
' ContactNormalizeAll  (epinglable au QAT)
' Compte -> Simulation / Appliquer -> tous les dossiers de contacts.
'=============================================================================
Public Sub ContactNormalizeAll()
    Dim errNo As Long, errDesc As String
    On Error GoTo ErrH
    LoadSettings
    ResetCounters

    Dim st As Outlook.Store
    Set st = PickStore()
    If st Is Nothing Then Exit Sub

    Dim choice As VbMsgBoxResult
    choice = AskMode(T("Compte : ", "Account: ") & st.DisplayName)
    If choice = vbCancel Then Exit Sub
    Dim dryRun As Boolean
    dryRun = (choice = vbYes)

    modDebugLog.DInfo MOD_NAME, "NormalizeAll start store=" & st.DisplayName & " dryRun=" & dryRun
    ReportHeader dryRun, "Compte : " & st.DisplayName
    ProcessFolderTree st.GetRootFolder, dryRun
    WriteReport
    modDebugLog.DInfo MOD_NAME, "NormalizeAll end scanned=" & mScanned & " changed=" & mChanged & _
        " fields=" & mFieldChanges & " errors=" & mErrors & " cats=" & mCatFound & " dryRun=" & dryRun
    ShowSummary dryRun
    If mClearCatPrompt And mCatFound > 0 Then OfferClearCategories st
    Exit Sub
ErrH:
    errNo = Err.Number
    errDesc = Err.Description
    On Error Resume Next
    modDebugLog.DError MOD_NAME, "ContactNormalizeAll err " & errNo & " - " & errDesc
    ReportLine "!! ERREUR " & errNo & " - " & errDesc
    WriteReport
    MsgBox T("Erreur : ", "Error: ") & errNo & " - " & errDesc, vbExclamation, UiTitle()
End Sub

'=============================================================================
' ContactNormalizeSelected  (epinglable au QAT) - la SONDE.
' Fiche ouverte (inspecteur) sinon selection de l'explorateur.
'=============================================================================
Public Sub ContactNormalizeSelected()
    Dim errNo As Long, errDesc As String
    On Error GoTo ErrH
    LoadSettings
    ResetCounters

    Dim picked As Collection
    Set picked = New Collection
    Dim insp As Outlook.Inspector
    On Error Resume Next
    Set insp = Application.ActiveInspector
    On Error GoTo ErrH
    If Not insp Is Nothing Then
        If TypeName(insp.CurrentItem) = "ContactItem" Then picked.Add insp.CurrentItem
    End If
    If picked.Count = 0 Then
        Dim sel As Outlook.Selection, idx As Long
        On Error Resume Next
        Set sel = Application.ActiveExplorer.Selection
        On Error GoTo ErrH
        If Not sel Is Nothing Then
            For idx = 1 To sel.Count
                If TypeName(sel.Item(idx)) = "ContactItem" Then picked.Add sel.Item(idx)
            Next idx
        End If
    End If
    If picked.Count = 0 Then
        MsgBox T("Aucune fiche contact ouverte ni s~electionn~ee.", "No contact card open or selected."), _
               vbInformation, UiTitle()
        Exit Sub
    End If

    Dim choice As VbMsgBoxResult
    choice = AskMode(picked.Count & T(" fiche(s) s~electionn~ee(s)", " selected card(s)"))
    If choice = vbCancel Then Exit Sub
    Dim dryRun As Boolean
    dryRun = (choice = vbYes)

    modDebugLog.DInfo MOD_NAME, "NormalizeSelected start n=" & picked.Count & " dryRun=" & dryRun
    ReportHeader dryRun, "Selection : " & picked.Count & " fiche(s)"
    Dim itm As Object
    For Each itm In picked
        NormalizeContact itm, "(selection)", dryRun
    Next itm
    WriteReport
    ShowSummary dryRun
    Exit Sub
ErrH:
    errNo = Err.Number
    errDesc = Err.Description
    On Error Resume Next
    modDebugLog.DError MOD_NAME, "ContactNormalizeSelected err " & errNo & " - " & errDesc
    ReportLine "!! ERREUR " & errNo & " - " & errDesc
    WriteReport
    MsgBox T("Erreur : ", "Error: ") & errNo & " - " & errDesc, vbExclamation, UiTitle()
End Sub

' Proposition en fin d'harmonisation (mCatFound fiches concernees).
Private Sub OfferClearCategories(ByVal st As Outlook.Store)
    Dim msg As String
    msg = mCatFound & T(" fiche(s) portent encore une cat~egorie." & vbCrLf & vbCrLf & _
          "Outlook sur le web projette d~ej~a tes DOSSIERS de contacts en cat~egories : " & _
          "une cat~egorie stock~ee sur la fiche fait donc double emploi." & vbCrLf & vbCrLf & _
          "Vider les cat~egories maintenant ?", _
          " card(s) still carry a category." & vbCrLf & vbCrLf & _
          "Outlook on the web already projects your contact FOLDERS as categories: " & _
          "a category stored on the card is therefore redundant." & vbCrLf & vbCrLf & _
          "Clear the categories now?")
    If MsgBox(msg, vbYesNo + vbQuestion + vbDefaultButton2, UiTitle()) <> vbYes Then Exit Sub
    RunClearCategories st
End Sub

' Passe de purge : Simulation / Appliquer, confirmation TAPEE avant ecriture.
Private Sub RunClearCategories(ByVal st As Outlook.Store)
    Dim choice As VbMsgBoxResult
    choice = AskMode(T("Purge des cat~egories - compte : ", "Category purge - account: ") & st.DisplayName)
    If choice = vbCancel Then Exit Sub
    Dim dryRun As Boolean
    dryRun = (choice = vbYes)
    If Not dryRun Then
        Dim word As String, typed As String
        word = UCase$(T("VIDER", "CLEAR"))
        typed = InputBox(T("Cette action retire la cat~egorie de TOUTES les fiches du compte." & vbCrLf & _
                           "Les dossiers ne sont pas touch~es." & vbCrLf & vbCrLf & "Tape ", _
                           "This removes the category from EVERY card of the account." & vbCrLf & _
                           "Folders are not affected." & vbCrLf & vbCrLf & "Type ") & word & _
                         T(" pour confirmer :", " to confirm:"), UiTitle())
        If UCase$(Trim$(typed)) <> word Then Exit Sub
    End If
    ResetCounters
    mClearMode = True
    modDebugLog.DInfo MOD_NAME, "ClearCategories start store=" & st.DisplayName & " dryRun=" & dryRun
    ReportLine "modContactNormalize " & MOD_VERSION & " - " & Format$(Now, "yyyy-mm-dd hh:nn:ss")
    ReportLine IIf(dryRun, "MODE : SIMULATION (aucune modification)", "MODE : APPLIQUE")
    ReportLine "PURGE DES CATEGORIES - compte : " & st.DisplayName
    ReportLine String$(78, "-")
    ProcessFolderTree st.GetRootFolder, dryRun
    mClearMode = False
    WriteReport
    modDebugLog.DInfo MOD_NAME, "ClearCategories end found=" & mCatFound & " cleared=" & mCatCleared & _
        " errors=" & mErrors & " dryRun=" & dryRun
    MsgBox IIf(dryRun, T("SIMULATION - rien n'a ~et~e ~ecrit.", "DRY RUN - nothing was written."), _
                       T("Termin~e.", "Done.")) & vbCrLf & vbCrLf & _
           T("Fiches parcourues : ", "Cards scanned: ") & mScanned & vbCrLf & _
           T("Fiches avec cat~egorie : ", "Cards with a category: ") & mCatFound & vbCrLf & _
           T("Cat~egories vid~ees : ", "Categories cleared: ") & mCatCleared & vbCrLf & _
           T("Erreurs : ", "Errors: ") & mErrors & vbCrLf & vbCrLf & _
           T("Rapport : ", "Report: ") & mReportPath, vbInformation, UiTitle()
End Sub

Private Sub LoadSettings()
    modConfig.EnsureLoaded
    mDoNames = modConfig.CfgB("NormalizeNames", True)
    mDoPhones = modConfig.CfgB("NormalizePhones", True)
    mDoAddresses = modConfig.CfgB("NormalizeAddresses", True)
    mDoEmails = modConfig.CfgB("NormalizeEmails", True)
    mCountryCode = DigitsOnly(modConfig.Cfg("PhoneDefaultCountryCode", "33"))
    If Len(mCountryCode) = 0 Then mCountryCode = "33"
    mSpaced = modConfig.CfgB("PhoneOutputSpaced", False)
    mParticles = "|" & LCase$(modConfig.Cfg("NameParticles", DEFAULT_PARTICLES)) & "|"
    mLowerWords = "|" & StripAccents(LCase$(modConfig.Cfg("AddressLowercaseWords", DEFAULT_LOWER_WORDS))) & "|"
    mUpperWords = "|" & StripAccents(LCase$(modConfig.Cfg("AddressUppercaseWords", DEFAULT_UPPER_WORDS))) & "|"
    mCityUpper = modConfig.CfgB("AddressCityUppercase", True)
    mCountryUpper = modConfig.CfgB("AddressCountryUppercase", True)
    mDefaultCountry = Trim$(modConfig.Cfg("AddressDefaultCountry", ""))
    mFileAsFormat = UCase$(Trim$(modConfig.Cfg("ContactFileAsFormat", "LastFirst")))
    mSkipFolders = modConfig.Cfg("ContactSkipFolders", DEFAULT_SKIP)
    mDoNameDay = modConfig.CfgB("NormalizeNameDay", True)
    mClearCatPrompt = modConfig.CfgB("ClearCategoriesPrompt", True)
    LoadNameDays modConfig.Cfg("NameDayExtra", "")
End Sub

' Table des fetes -> Dictionary ; NameDayExtra (prenom=MM-JJ|...) surcharge.
Private Sub LoadNameDays(ByVal extra As String)
    Set mNameDays = CreateObject("Scripting.Dictionary")
    Dim parts() As String, idx As Long, kv() As String, k As String, v As String
    parts = Split(NameDayTable() & extra, "|")
    For idx = LBound(parts) To UBound(parts)
        If InStr(parts(idx), "=") > 0 Then
            kv = Split(parts(idx), "=")
            k = StripAccents(LCase$(Trim$(kv(0))))
            v = Trim$(kv(1))
            If Len(k) > 0 And v Like "##-##" Then mNameDays.Item(k) = v
        End If
    Next idx
End Sub

'=============================================================================
' Choix du compte (store). Defaut = store du dossier courant.
'=============================================================================
Private Function PickStore() As Outlook.Store
    On Error GoTo ErrH
    Dim ns As Outlook.NameSpace
    Set ns = Application.Session
    Dim cur As Outlook.Store
    On Error Resume Next
    Set cur = Application.ActiveExplorer.CurrentFolder.Store
    On Error GoTo ErrH
    Dim listTxt As String, idx As Long, defIdx As Long
    For idx = 1 To ns.Stores.Count
        listTxt = listTxt & idx & " - " & ns.Stores(idx).DisplayName & vbCrLf
        If Not cur Is Nothing Then
            If ns.Stores(idx).StoreID = cur.StoreID Then defIdx = idx
        End If
    Next idx
    If defIdx = 0 Then defIdx = 1
    Dim ans As String
    ans = Trim$(InputBox(T("Compte dont les contacts sont ~a harmoniser (num~ero) :", _
                           "Account whose contacts will be harmonised (number):") & _
                         vbCrLf & vbCrLf & listTxt, UiTitle(), CStr(defIdx)))
    If Len(ans) = 0 Then Exit Function
    If Not IsNumeric(ans) Then GoTo BadChoice
    Dim num As Long
    num = CLng(ans)
    If num < 1 Or num > ns.Stores.Count Then GoTo BadChoice
    Set PickStore = ns.Stores(num)
    Exit Function
BadChoice:
    MsgBox T("Choix invalide.", "Invalid choice."), vbExclamation, UiTitle()
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "PickStore err " & Err.Number & " - " & Err.Description
End Function

'=============================================================================
' Parcours recursif : chaque dossier de type Contact (non systeme, non
' cache) est traite.
'=============================================================================
Private Sub ProcessFolderTree(ByVal fld As Outlook.MAPIFolder, ByVal dryRun As Boolean)
    On Error Resume Next
    Dim isContacts As Boolean
    isContacts = (fld.DefaultItemType = olContactItem)
    If Err.Number <> 0 Then isContacts = False: Err.Clear
    If isContacts Then
        If IsSkippedFolder(fld) Then
            modDebugLog.DDebug MOD_NAME, "Dossier ignore : " & fld.FolderPath
        Else
            ProcessContacts fld, dryRun
        End If
    End If
    Dim child As Outlook.MAPIFolder
    For Each child In fld.Folders
        ProcessFolderTree child, dryRun
    Next child
End Sub

Private Function IsSkippedFolder(ByVal fld As Outlook.MAPIFolder) As Boolean
    ' Pack v3 (C3) : delegation modContactCommon (helper partage).
    IsSkippedFolder = modContactCommon.IsSkippedFolder(fld, mSkipFolders)
End Function

' Dossier situe sous les Elements supprimes de son compte (corbeille) ?
Private Function IsInDeletedItems(ByVal fld As Outlook.MAPIFolder) As Boolean
    ' Pack v3 (C3) : delegation modContactCommon (helper partage).
    IsInDeletedItems = modContactCommon.IsInDeletedItems(fld)
End Function


Private Sub ProcessContacts(ByVal fld As Outlook.MAPIFolder, ByVal dryRun As Boolean)
    On Error GoTo ErrH
    mFolders = mFolders + 1
    Dim before As Long, beforeC As Long
    before = mScanned
    beforeC = mChanged
    ReportLine "=== Dossier " & fld.FolderPath & " ==="
    Dim itm As Object, cnt As Long
    For Each itm In fld.Items
        cnt = cnt + 1
        If TypeName(itm) = "ContactItem" Then NormalizeContact itm, fld.Name, dryRun
        If (cnt Mod 20) = 0 Then DoEvents
    Next itm
    ReportLine "    -> " & (mScanned - before) & " contacts, " & (mChanged - beforeC) & " a modifier"
    Exit Sub
ErrH:
    mErrors = mErrors + 1
    modDebugLog.DError MOD_NAME, "ProcessContacts " & fld.FolderPath & " err " & Err.Number & " - " & Err.Description
    ReportLine "!! Dossier " & fld.FolderPath & " : erreur " & Err.Number & " - " & Err.Description
End Sub

'=============================================================================
' Traitement d'une fiche : chaque bloc calcule les valeurs cibles, les
' compare, ecrit la ligne de rapport et (hors simulation) affecte le champ.
' Un seul Save par fiche, puis controles de recomposition Outlook.
'=============================================================================
Private Sub NormalizeContact(ByVal ct As Outlook.ContactItem, ByVal fldName As String, ByVal dryRun As Boolean)
    Dim errNo As Long, errDesc As String
    Dim who As String
    On Error GoTo ErrH
    mScanned = mScanned + 1
    ResetScratch

    who = SafeProp(ct, "FullName")
    If Len(who) = 0 Then who = SafeProp(ct, "FileAs")
    If Len(who) = 0 Then who = SafeProp(ct, "CompanyName")

    Dim lines As String, changes As Long
    Dim cats As String
    cats = Trim$(SafeProp(ct, "Categories"))
    If Len(cats) > 0 Then mCatFound = mCatFound + 1
    If mClearMode Then
        If Len(cats) = 0 Then Exit Sub
        ReportLine "[" & IIf(dryRun, "SIMUL ", "MODIF ") & "] " & who & "   (" & fldName & ")"
        ReportLine "   " & T("Cat~egories", "Categories") & " : " & cats & "  ->  (vide)"
        mChanged = mChanged + 1
        mFieldChanges = mFieldChanges + 1
        mCatCleared = mCatCleared + 1
        If Not dryRun Then
            ct.Categories = ""
            If Not SafeSave(ct) Then
                mErrors = mErrors + 1
                ReportLine "   !! Save en echec"
            End If
        End If
        Exit Sub
    End If
    If mDoNames Then changes = changes + DoNames(ct, lines, dryRun)
    If mDoPhones Then changes = changes + DoPhones(ct, lines, dryRun)
    If mDoAddresses Then changes = changes + DoAddresses(ct, lines, dryRun)
    If mDoEmails Then changes = changes + DoEmails(ct, lines, dryRun)
    changes = changes + DoTextFields(ct, lines, dryRun)
    If mDoNameDay Then changes = changes + DoNameDay(ct, lines, dryRun)

    If Len(lines) = 0 Then Exit Sub
    ReportLine "[" & IIf(changes > 0, IIf(dryRun, "SIMUL ", "MODIF "), "NOTE  ") & "] " & who & "   (" & fldName & ")"
    mReport = mReport & lines
    If changes = 0 Then Exit Sub
    mChanged = mChanged + 1
    mFieldChanges = mFieldChanges + changes
    If dryRun Then Exit Sub

    If Not SafeSave(ct) Then
        mErrors = mErrors + 1
        ReportLine "   !! Save en echec, fiche non modifiee"
        Exit Sub
    End If
    PostSaveChecks ct
    Exit Sub
ErrH:
    errNo = Err.Number
    errDesc = Err.Description
    mErrors = mErrors + 1
    modDebugLog.DError MOD_NAME, "NormalizeContact err " & errNo & " - " & errDesc & " (" & who & ")"
    ReportLine "[ERREUR] " & who & " : " & errNo & " - " & errDesc
End Sub

Private Function DoNames(ByVal ct As Outlook.ContactItem, ByRef lines As String, ByVal dryRun As Boolean) As Long
    On Error GoTo ErrH
    Dim first As String, middle As String, last As String, civ As String, suffix As String
    first = CollapseSpaces(Trim$(SafeProp(ct, "FirstName")))
    middle = CollapseSpaces(Trim$(SafeProp(ct, "MiddleName")))
    last = CollapseSpaces(Trim$(SafeProp(ct, "LastName")))
    civ = Trim$(SafeProp(ct, "Title"))
    suffix = Trim$(SafeProp(ct, "Suffix"))
    If Len(first) = 0 Or Len(last) = 0 Then
        ' Entite (societe, service, numero...) : jamais renommee.
        mEntities = mEntities + 1
        Exit Function
    End If

    Dim nFirst As String, nMiddle As String, nLast As String
    nFirst = CapName(first)
    nMiddle = CapName(middle)
    nLast = UpperSurname(last)

    Dim oldFull As String, newFull As String
    oldFull = SafeProp(ct, "FullName")
    newFull = JoinWords(civ, nFirst, nMiddle, nLast, suffix)

    Dim oldFA As String, newFA As String
    oldFA = SafeProp(ct, "FileAs")
    newFA = RebuildFileAs(oldFA, first, middle, last, nFirst, nMiddle, nLast)

    Dim cnt As Long
    If StrComp(SafeProp(ct, "FirstName"), nFirst, vbBinaryCompare) <> 0 Then
        cnt = cnt + 1
        AddLine lines, T("Pr~enom", "First name"), SafeProp(ct, "FirstName"), nFirst
    End If
    If StrComp(SafeProp(ct, "MiddleName"), nMiddle, vbBinaryCompare) <> 0 Then
        cnt = cnt + 1
        AddLine lines, T("Deuxi~fme pr~enom", "Middle name"), SafeProp(ct, "MiddleName"), nMiddle
    End If
    If StrComp(SafeProp(ct, "LastName"), nLast, vbBinaryCompare) <> 0 Then
        cnt = cnt + 1
        AddLine lines, T("Nom", "Last name"), SafeProp(ct, "LastName"), nLast
    End If
    If StrComp(oldFA, newFA, vbBinaryCompare) <> 0 Then
        cnt = cnt + 1
        AddLine lines, "FileAs", oldFA, newFA
    End If
    If cnt = 0 Then Exit Function

    mExpFull = newFull: mNewFirst = nFirst: mNewMiddle = nMiddle: mNewLast = nLast
    If dryRun Then
        DoNames = cnt
        Exit Function
    End If

    ' FullName d'abord (Outlook le re-decoupe), puis les composantes qui
    ' font foi, puis FileAs. Civilite / suffixe reposes a l'identique.
    ct.FullName = newFull
    ct.Title = civ
    ct.FirstName = nFirst
    ct.MiddleName = nMiddle
    ct.LastName = nLast
    ct.Suffix = suffix
    ct.FileAs = newFA

    ' Noms d'affichage des e-mails ("Ancien Nom (adresse)") suivent.
    Dim idx As Long, dn As String
    If Len(oldFull) > 0 And StrComp(oldFull, newFull, vbBinaryCompare) <> 0 Then
        For idx = 1 To 3
            dn = SafeProp(ct, "Email" & idx & "DisplayName")
            If InStr(1, dn, oldFull, vbBinaryCompare) > 0 Then
                CallByName ct, "Email" & idx & "DisplayName", VbLet, Replace(dn, oldFull, newFull)
            End If
        Next idx
    End If
    DoNames = cnt
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "DoNames err " & Err.Number & " - " & Err.Description
    AddNote lines, "!! noms : erreur " & Err.Number & " - " & Err.Description
    DoNames = 0
End Function

' Prenom(s) : chaque mot capitalise, tirets et apostrophes respectes
' ("jean-baptiste" -> "Jean-Baptiste", "d'angelo" -> "D'Angelo").
Private Function CapName(ByVal s As String) As String
    Dim words() As String, idx As Long
    s = CollapseSpaces(Trim$(s))
    If Len(s) = 0 Then Exit Function
    words = Split(s, " ")
    For idx = LBound(words) To UBound(words)
        words(idx) = CapHyphenated(words(idx), False)
    Next idx
    CapName = Join(words, " ")
End Function

' Nom : MAJUSCULES, particules en minuscules ("de la fontaine" ->
' "de la FONTAINE", "d'estaing" -> "d'ESTAING"). Un nom mono-mot n'est
' jamais traite comme particule.
Private Function UpperSurname(ByVal s As String) As String
    Dim words() As String, idx As Long, w As String, base As String
    Dim pos As Long, pre As String, apos As String, rest As String
    s = CollapseSpaces(Trim$(s))
    If Len(s) = 0 Then Exit Function
    words = Split(s, " ")
    For idx = LBound(words) To UBound(words)
        w = words(idx)
        base = StripAccents(LCase$(w))
        If UBound(words) > LBound(words) And IsParticle(base) Then
            words(idx) = LCase$(w)
        Else
            pos = ApostrophePos(w)
            If pos > 1 And pos < Len(w) Then
                pre = Left$(w, pos - 1)
                apos = Mid$(w, pos, 1)
                rest = Mid$(w, pos + 1)
                If IsParticle(StripAccents(LCase$(pre))) Then
                    words(idx) = LCase$(pre) & apos & UCase$(rest)
                Else
                    words(idx) = UCase$(w)
                End If
            Else
                words(idx) = UCase$(w)
            End If
        End If
    Next idx
    UpperSurname = Join(words, " ")
End Function

Private Function IsParticle(ByVal base As String) As Boolean
    IsParticle = (InStr(1, mParticles, "|" & base & "|", vbBinaryCompare) > 0)
End Function

' Reconstruit FileAs selon le motif deja en place ; un FileAs personnalise
' (aucun motif reconnu) est laisse tel quel.
Private Function RebuildFileAs(ByVal oldFA As String, ByVal oFirst As String, ByVal oMiddle As String, _
                               ByVal oLast As String, ByVal nFirst As String, ByVal nMiddle As String, _
                               ByVal nLast As String) As String
    Dim oFM As String, nFM As String
    oFM = JoinWords(oFirst, oMiddle, "", "", "")
    nFM = JoinWords(nFirst, nMiddle, "", "", "")
    RebuildFileAs = oldFA
    If Len(Trim$(oldFA)) = 0 Then
        If mFileAsFormat = "FIRSTLAST" Then
            RebuildFileAs = nFM & " " & nLast
        Else
            RebuildFileAs = nLast & ", " & nFM
        End If
    ElseIf StrComp(oldFA, oLast & ", " & oFM, vbTextCompare) = 0 Then
        RebuildFileAs = nLast & ", " & nFM
    ElseIf StrComp(oldFA, oLast & "," & oFM, vbTextCompare) = 0 Then
        RebuildFileAs = nLast & ", " & nFM
    ElseIf StrComp(oldFA, oFM & " " & oLast, vbTextCompare) = 0 Then
        RebuildFileAs = nFM & " " & nLast
    ElseIf StrComp(oldFA, oLast & " " & oFM, vbTextCompare) = 0 Then
        RebuildFileAs = nLast & " " & nFM
    ElseIf StrComp(oldFA, oLast & ", " & oFirst, vbTextCompare) = 0 Then
        RebuildFileAs = nLast & ", " & nFirst
    ElseIf StrComp(oldFA, oFirst & " " & oLast, vbTextCompare) = 0 Then
        RebuildFileAs = nFirst & " " & nLast
    End If
End Function

' E.164 : "+" & indicatif & numero national sans le 0. Statuts en tete.
Private Function NormalizePhone(ByVal raw As String, ByRef status As Long) As String
    Dim s As String, idx As Long, ch As String, digits As String
    Dim hasPlus As Boolean, hasLetter As Boolean, cc As String, result As String
    NormalizePhone = raw
    status = PH_SAME
    s = Trim$(raw)
    If Len(s) = 0 Then Exit Function

    For idx = 1 To Len(s)
        ch = Mid$(s, idx, 1)
        Select Case ch
            Case "0" To "9"
                digits = digits & ch
            Case "+"
                If Len(digits) = 0 And Not hasPlus Then
                    hasPlus = True
                Else
                    hasLetter = True
                End If
            Case " ", ".", "-", "(", ")", "/", vbTab, ChrW$(160)
                ' separateurs ignores
            Case Else
                hasLetter = True
        End Select
    Next idx
    If hasLetter Then
        status = PH_LETTERS
        Exit Function
    End If
    cc = mCountryCode

    If hasPlus Or (Left$(digits, 2) = "00" And Len(digits) >= 10) Then
        If Not hasPlus Then digits = Mid$(digits, 3)
        ' "+33 (0)6 12..." -> retirer le 0 national residuel
        If Left$(digits, Len(cc)) = cc Then
            If Mid$(digits, Len(cc) + 1, 1) = "0" And Len(digits) = Len(cc) + 10 Then
                digits = cc & Mid$(digits, Len(cc) + 2)
            End If
        End If
        If Len(digits) < 8 Or Len(digits) > 15 Then
            status = PH_INVALID
            Exit Function
        End If
        result = "+" & digits
        status = PH_CHANGED
    Else
        If Len(digits) <= 6 Then
            status = PH_SHORT
            Exit Function
        End If
        If Len(digits) = 10 And Left$(digits, 1) = "0" Then
            result = "+" & cc & Mid$(digits, 2)
            status = PH_CHANGED
        ElseIf Len(digits) = 9 And Left$(digits, 1) <> "0" And cc = "33" Then
            result = "+" & cc & digits
            status = PH_ASSUMED
        Else
            status = PH_INVALID
            Exit Function
        End If
    End If

    If mSpaced Then result = SpaceE164(result)
    If StrComp(result, raw, vbBinaryCompare) = 0 Then status = PH_SAME
    NormalizePhone = result
End Function

' "+33612345678" -> "+33 6 12 34 56 78" (francais uniquement, sinon intact).
Private Function SpaceE164(ByVal e164 As String) As String
    SpaceE164 = e164
    If Left$(e164, 3) = "+33" And Len(e164) = 12 Then
        SpaceE164 = "+33 " & Mid$(e164, 4, 1) & " " & Mid$(e164, 5, 2) & " " & Mid$(e164, 7, 2) & _
                    " " & Mid$(e164, 9, 2) & " " & Mid$(e164, 11, 2)
    End If
End Function

'-----------------------------------------------------------------------------
' PhoneFormatCanonical (v1.5) - PUBLIQUE, sans etat : formate un numero brut
' selon la convention du toolkit (E.164, espace si PhoneOutputSpaced=1).
' Renvoie "" si le numero n'est pas reconnu (l'appelant garde alors sa
' valeur ou l'ecarte). Recharge les 2 cles concernees a chaque appel : sur
' d'etre a jour meme apres un Ctrl+S (globales remises a zero).
'-----------------------------------------------------------------------------
Public Function PhoneFormatCanonical(ByVal raw As String) As String
    On Error Resume Next
    mCountryCode = DigitsOnly(modConfig.Cfg("PhoneDefaultCountryCode", "33"))
    If Len(mCountryCode) = 0 Then mCountryCode = "33"
    mSpaced = modConfig.CfgB("PhoneOutputSpaced", False)
    Dim status As Long, res As String
    res = NormalizePhone(raw, status)
    Select Case status
        Case PH_SAME, PH_CHANGED, PH_ASSUMED
            PhoneFormatCanonical = res
        Case Else
            PhoneFormatCanonical = ""
    End Select
    Err.Clear
End Function

Private Function DoAddresses(ByVal ct As Outlook.ContactItem, ByRef lines As String, ByVal dryRun As Boolean) As Long
    On Error GoTo ErrH
    Dim groups() As String, gi As Long, grp As String, cnt As Long
    Dim street As String, city As String, postal As String, country As String
    Dim nStreet As String, nCity As String, nPostal As String, nCountry As String
    groups = Split(ADDR_GROUPS, "|")
    For gi = LBound(groups) To UBound(groups)
        grp = groups(gi)
        street = SafeProp(ct, grp & "AddressStreet")
        city = SafeProp(ct, grp & "AddressCity")
        postal = SafeProp(ct, grp & "AddressPostalCode")
        country = SafeProp(ct, grp & "AddressCountry")
        If Len(Trim$(street & city & postal & country)) > 0 Then
            nStreet = NormalizeStreet(street)
            nCity = CollapseSpaces(Trim$(city))
            nPostal = CollapseSpaces(Trim$(postal))
            nCountry = CollapseSpaces(Trim$(country))
            ' "92320 Chatillon" saisi dans Ville, code postal vide -> separer
            If Len(nPostal) = 0 And nCity Like "##### *" Then
                nPostal = Left$(nCity, 5)
                nCity = Trim$(Mid$(nCity, 6))
            End If
            If mCityUpper Then nCity = UCase$(nCity)
            If mCountryUpper Then nCountry = UCase$(nCountry)
            If Len(nCountry) = 0 And Len(mDefaultCountry) > 0 And Len(nCity) > 0 And nPostal Like "#####" Then
                nCountry = IIf(mCountryUpper, UCase$(mDefaultCountry), mDefaultCountry)
            End If
            cnt = cnt + ApplyField(ct, grp & "AddressStreet", street, nStreet, GroupLabel(grp) & T(" - rue", " - street"), lines, dryRun)
            cnt = cnt + ApplyField(ct, grp & "AddressPostalCode", postal, nPostal, GroupLabel(grp) & T(" - code postal", " - postal code"), lines, dryRun)
            cnt = cnt + ApplyField(ct, grp & "AddressCity", city, nCity, GroupLabel(grp) & T(" - ville", " - city"), lines, dryRun)
            cnt = cnt + ApplyField(ct, grp & "AddressCountry", country, nCountry, GroupLabel(grp) & T(" - pays", " - country"), lines, dryRun)
            If StrComp(city, nCity, vbBinaryCompare) <> 0 Or StrComp(street, nStreet, vbBinaryCompare) <> 0 _
               Or StrComp(postal, nPostal, vbBinaryCompare) <> 0 Or StrComp(country, nCountry, vbBinaryCompare) <> 0 Then
                mExpStreet(gi) = nStreet: mExpCity(gi) = nCity
                mExpPostal(gi) = nPostal: mExpCountry(gi) = nCountry
            End If
        End If
    Next gi
    DoAddresses = cnt
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "DoAddresses err " & Err.Number & " - " & Err.Description
    AddNote lines, "!! adresses : erreur " & Err.Number & " - " & Err.Description
    DoAddresses = cnt
End Function

' Rue : lignes multiples respectees ; par ligne, 1er mot alphabetique
' capitalise, mots-outils en minuscules, sigles en majuscules, jetons a
' chiffres intacts.
Private Function NormalizeStreet(ByVal s As String) As String
    Dim arr() As String, li As Long
    s = Replace(s, vbCrLf, vbLf)
    s = Replace(s, vbCr, vbLf)
    arr = Split(s, vbLf)
    For li = LBound(arr) To UBound(arr)
        arr(li) = NormalizeStreetLine(CollapseSpaces(Trim$(arr(li))))
    Next li
    NormalizeStreet = Join(arr, vbCrLf)
End Function

Private Function NormalizeStreetLine(ByVal ln As String) As String
    Dim words() As String, wi As Long, w As String, tail As String, base As String
    Dim firstAlphaDone As Boolean
    If Len(ln) = 0 Then Exit Function
    words = Split(ln, " ")
    For wi = LBound(words) To UBound(words)
        w = words(wi)
        tail = ""
        Do While Len(w) > 0
            If Right$(w, 1) = "," Or Right$(w, 1) = ";" Then
                tail = Right$(w, 1) & tail
                w = Left$(w, Len(w) - 1)
            Else
                Exit Do
            End If
        Loop
        If Len(w) > 0 Then
            base = StripAccents(LCase$(w))
            If HasDigit(w) Then
                ' 12B, 3eme, 1er : intact
            ElseIf IsUpperWord(base) Then
                w = UCase$(w)
            ElseIf Not firstAlphaDone Then
                w = CapHyphenated(w, True)
                w = UCase$(Left$(w, 1)) & Mid$(w, 2)
                firstAlphaDone = True
            ElseIf IsLowerWord(base) Then
                w = LCase$(w)
            Else
                w = CapHyphenated(w, True)
            End If
        End If
        words(wi) = w & tail
    Next wi
    NormalizeStreetLine = Join(words, " ")
End Function

Private Function IsLowerWord(ByVal base As String) As Boolean
    IsLowerWord = (InStr(1, mLowerWords, "|" & base & "|", vbBinaryCompare) > 0)
End Function

Private Function IsUpperWord(ByVal base As String) As Boolean
    IsUpperWord = (InStr(1, mUpperWords, "|" & base & "|", vbBinaryCompare) > 0)
End Function

'-----------------------------------------------------------------------------
' FETE DU PRENOM -> UserProperty NameDay ("MM-JJ")
'-----------------------------------------------------------------------------
Private Function DoNameDay(ByVal ct As Outlook.ContactItem, ByRef lines As String, ByVal dryRun As Boolean) As Long
    On Error GoTo ErrH
    Dim first As String, last As String
    first = Trim$(SafeProp(ct, "FirstName"))
    last = Trim$(SafeProp(ct, "LastName"))
    If Len(first) = 0 Or Len(last) = 0 Then Exit Function      ' entite : pas de fete
    Dim mmdd As String
    mmdd = NameDayFor(first)
    If Len(mmdd) = 0 Then
        mNameDayUnknown = mNameDayUnknown + 1
        AddNote lines, T("f" & ChrW$(234) & "te : pr~enom absent du calendrier : ", "name day: first name not in the table: ") & first
        Exit Function
    End If
    Dim up As Outlook.UserProperty, cur As String
    Set up = ct.UserProperties.Find(UP_NAMEDAY)
    If Not up Is Nothing Then cur = Trim$(CStr(up.Value))
    If cur = mmdd Then Exit Function
    AddLine lines, T("F" & ChrW$(234) & "te", "Name day"), cur, Mid$(mmdd, 4, 2) & "/" & Left$(mmdd, 2)
    mNameDaySet = mNameDaySet + 1
    If Not dryRun Then
        If up Is Nothing Then Set up = ct.UserProperties.Add(UP_NAMEDAY, olText, False)
        up.Value = mmdd
    End If
    DoNameDay = 1
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "DoNameDay err " & Err.Number & " - " & Err.Description
    AddNote lines, "!! fete : erreur " & Err.Number & " - " & Err.Description
End Function

' Prenom -> "MM-JJ" ou "". Compose : entree exacte, sinon second prenom si le
' premier est Jean / Marie, sinon premier prenom.
Private Function NameDayFor(ByVal firstName As String) As String
    Dim k As String, parts() As String
    k = StripAccents(LCase$(Trim$(firstName)))
    k = Replace(k, " ", "-")
    If mNameDays.Exists(k) Then
        NameDayFor = mNameDays.Item(k)
        Exit Function
    End If
    If InStr(k, "-") > 0 Then
        parts = Split(k, "-")
        If (parts(0) = "jean" Or parts(0) = "marie") And UBound(parts) >= 1 Then
            If mNameDays.Exists(parts(1)) Then
                NameDayFor = mNameDays.Item(parts(1))
                Exit Function
            End If
        End If
        If mNameDays.Exists(parts(0)) Then NameDayFor = mNameDays.Item(parts(0))
    End If
End Function

' Table des fetes : prenom (minuscules, sans accent) -> MM-JJ.
' Calendrier des saints usuel (type calendrier de La Poste). Completable
' sans toucher au code via config NameDayExtra=prenom=MM-JJ|prenom=MM-JJ.

'-----------------------------------------------------------------------------
' Controles apres Save : Outlook peut recomposer FullName et les blocs
' d'adresse composites a sa maniere. On relit, et on force si besoin.
'-----------------------------------------------------------------------------
Private Sub PostSaveChecks(ByVal ct As Outlook.ContactItem)
    On Error Resume Next
    Dim needSave As Boolean
    If Len(mExpFull) > 0 Then
        If StrComp(SafeProp(ct, "FullName"), mExpFull, vbBinaryCompare) <> 0 Then
            ct.FullName = mExpFull
            ct.FirstName = mNewFirst
            ct.MiddleName = mNewMiddle
            ct.LastName = mNewLast
            needSave = True
            ReportLine "   ~ FullName force apres Save (Outlook l'avait recompose : " & SafeProp(ct, "FullName") & ")"
            modDebugLog.DInfo MOD_NAME, "FullName force : " & mExpFull
        End If
    End If
    Dim groups() As String, gi As Long, comp As String, comp2 As String
    groups = Split(ADDR_GROUPS, "|")
    For gi = LBound(groups) To UBound(groups)
        If Len(mExpCity(gi)) > 0 Or Len(mExpStreet(gi)) > 0 Then
            comp = SafeProp(ct, groups(gi) & "Address")
            If (Len(mExpCity(gi)) > 0 And InStr(1, comp, mExpCity(gi), vbBinaryCompare) = 0) _
               Or (Len(mExpStreet(gi)) > 0 And InStr(1, comp, mExpStreet(gi), vbBinaryCompare) = 0) Then
                comp2 = mExpStreet(gi)
                If Len(Trim$(mExpPostal(gi) & " " & mExpCity(gi))) > 0 Then
                    comp2 = comp2 & IIf(Len(comp2) > 0, vbCrLf, "") & Trim$(mExpPostal(gi) & " " & mExpCity(gi))
                End If
                If Len(mExpCountry(gi)) > 0 Then comp2 = comp2 & vbCrLf & mExpCountry(gi)
                CallByName ct, groups(gi) & "Address", VbLet, comp2
                CallByName ct, groups(gi) & "AddressStreet", VbLet, mExpStreet(gi)
                CallByName ct, groups(gi) & "AddressPostalCode", VbLet, mExpPostal(gi)
                CallByName ct, groups(gi) & "AddressCity", VbLet, mExpCity(gi)
                CallByName ct, groups(gi) & "AddressCountry", VbLet, mExpCountry(gi)
                needSave = True
                ReportLine "   ~ bloc adresse " & groups(gi) & " reconstruit apres Save"
                modDebugLog.DInfo MOD_NAME, "Adresse composite reconstruite (" & groups(gi) & ")"
            End If
        End If
    Next gi
    If needSave Then
        If Not SafeSave(ct) Then ReportLine "   !! Save de controle en echec"
    End If
End Sub

' Compare, ecrit la ligne de rapport et (hors simulation) affecte via
' CallByName. Renvoie 1 si le champ change, 0 sinon.
Private Function ApplyField(ByVal ct As Outlook.ContactItem, ByVal propName As String, _
                            ByVal oldV As String, ByVal newV As String, ByVal lbl As String, _
                            ByRef lines As String, ByVal dryRun As Boolean) As Long
    If StrComp(oldV, newV, vbBinaryCompare) = 0 Then Exit Function
    AddLine lines, lbl, oldV, newV
    If Not dryRun Then CallByName ct, propName, VbLet, newV
    ApplyField = 1
End Function

Private Function SafeProp(ByVal ct As Object, ByVal propName As String) As String
    On Error Resume Next
    SafeProp = ""
    SafeProp = CStr(CallByName(ct, propName, VbGet))
    Err.Clear
End Function

Private Sub AddLine(ByRef lines As String, ByVal lbl As String, ByVal oldV As String, ByVal newV As String)
    lines = lines & "   " & lbl & " : " & Flat(oldV) & "  ->  " & Flat(newV) & vbCrLf
End Sub

Private Sub AddNote(ByRef lines As String, ByVal txt As String)
    lines = lines & "   " & Flat(txt) & vbCrLf
End Sub

Private Function Flat(ByVal s As String) As String
    Flat = Replace(Replace(s, vbCrLf, " / "), vbLf, " / ")
End Function

Private Function JoinWords(ByVal p1 As String, ByVal p2 As String, ByVal p3 As String, _
                           ByVal p4 As String, ByVal p5 As String) As String
    JoinWords = CollapseSpaces(Trim$(p1 & " " & p2 & " " & p3 & " " & p4 & " " & p5))
End Function

' Capitalise un mot en respectant tirets et apostrophes. applyLower :
' segments qui sont des mots-outils d'adresse laisses en minuscules
' ("bourg-la-reine" -> "Bourg-la-Reine", "l'eglise" -> "l'Eglise").
Private Function CapHyphenated(ByVal w As String, ByVal applyLower As Boolean) As String
    Dim segs() As String, si As Long
    segs = Split(w, "-")
    For si = LBound(segs) To UBound(segs)
        If applyLower And IsLowerWord(StripAccents(LCase$(segs(si)))) Then
            segs(si) = LCase$(segs(si))
        Else
            segs(si) = CapApostrophe(segs(si), applyLower)
        End If
    Next si
    CapHyphenated = Join(segs, "-")
End Function

Private Function CapApostrophe(ByVal seg As String, ByVal applyLower As Boolean) As String
    Dim pos As Long, pre As String, apos As String, rest As String
    pos = ApostrophePos(seg)
    If pos > 1 And pos < Len(seg) Then
        pre = Left$(seg, pos - 1)
        apos = Mid$(seg, pos, 1)
        rest = Mid$(seg, pos + 1)
        If applyLower And IsLowerWord(StripAccents(LCase$(pre))) Then
            CapApostrophe = LCase$(pre) & apos & CapSimple(rest)
        Else
            CapApostrophe = CapSimple(pre) & apos & CapSimple(rest)
        End If
    Else
        CapApostrophe = CapSimple(seg)
    End If
End Function

Private Function CapSimple(ByVal s As String) As String
    If Len(s) = 0 Then Exit Function
    CapSimple = UCase$(Left$(s, 1)) & LCase$(Mid$(s, 2))
End Function

' Position de la premiere apostrophe (droite ou typographique), 0 si aucune.
Private Function ApostrophePos(ByVal s As String) As Long
    Dim p1 As Long, p2 As Long
    p1 = InStr(s, "'")
    p2 = InStr(s, ChrW$(8217))
    If p1 = 0 Then
        ApostrophePos = p2
    ElseIf p2 = 0 Then
        ApostrophePos = p1
    ElseIf p1 < p2 Then
        ApostrophePos = p1
    Else
        ApostrophePos = p2
    End If
End Function

' Save avec 3 essais (Exchange Online : erreurs transitoires).
Private Function SafeSave(ByVal itm As Object) As Boolean
    Dim tries As Long
    On Error Resume Next
    For tries = 1 To 3
        Err.Clear
        itm.Save
        If Err.Number = 0 Then
            SafeSave = True
            Exit Function
        End If
        modDebugLog.DDebug MOD_NAME, "Save essai " & tries & " err " & Err.Number & " - " & Err.Description
        Sleep 300 * tries
    Next tries
    SafeSave = False
End Function

Private Sub ReportLine(ByVal s As String)
    mReport = mReport & s & vbCrLf
End Sub

Private Sub WriteReport()
    On Error Resume Next
    Dim dirPath As String
    dirPath = modConfig.MacroVBADir()
    If Dir(dirPath, vbDirectory) = "" Then MkDir dirPath
    mReportPath = dirPath & "\ContactNormalize-" & Format$(Now, "yyyymmdd-hhnnss") & ".txt"
    Dim stm As Object
    Set stm = CreateObject("ADODB.Stream")
    stm.Type = 2
    stm.Charset = "utf-8"
    stm.Open
    stm.WriteText mReport
    stm.SaveToFile mReportPath, 2
    stm.Close
    If Err.Number <> 0 Then
        modDebugLog.DError MOD_NAME, "WriteReport err " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
End Sub
