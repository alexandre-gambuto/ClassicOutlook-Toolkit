'=============================================================================
' Maintenance  (POINT D'ENTREE - a epingler au QAT)
'=============================================================================
Public Sub Maintenance()
    On Error GoTo ErrH
    Dim choice As String
    Do
        choice = modUi.AskText( _
            "=== " & UCase$(TITLE) & " ===" & vbCrLf & vbCrLf & _
            T("1 - Logs", "1 - Logs") & vbCrLf & _
            T("2 - Sauvegardes", "2 - Backups") & vbCrLf & _
            T("3 - Configuration", "3 - Configuration") & vbCrLf & _
            T("4 - Diagnostic", "4 - Diagnostics") & vbCrLf & _
            T("5 - Bindings & cache", "5 - Bindings & cache") & vbCrLf & vbCrLf & _
            T("0 - Quitter", "0 - Quit") & vbCrLf & vbCrLf & _
            T("Num~ero :", "Number:"), TITLE)

        If Len(choice) = 0 Then Exit Sub          ' Annuler ou vide -> sortie
        Select Case Trim$(choice)
            Case "0": Exit Sub
            Case "1": MenuLogs
            Case "2": MenuBackups
            Case "3": MenuConfiguration
            Case "4": MenuDiagnostic
            Case "5": MenuBindingsCache
            Case Else: MsgBox T("Choix invalide.", "Invalid choice."), vbExclamation, TITLE
        End Select
    Loop
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "Maintenance err " & Err.Number & " - " & Err.Description
    MsgBox T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, vbExclamation, TITLE
End Sub

'=============================================================================
' MaintCycleLogLevel  (epinglable au QAT)
' Fait tourner le niveau de log : OFF -> INFO -> DEBUG -> TRACE -> OFF.
'=============================================================================
Public Sub MaintCycleLogLevel()
    Dim n As Long
    n = modDebugLog.GlobalLevel
    n = n + 1
    If n > modDebugLog.LOG_TRACE Then n = modDebugLog.LOG_ERROR
    modDebugLog.SetGlobalLevel n
    ' MsgBox utile quand l'action est lancee en bouton direct (hors menu) :
    MsgBox T("Niveau log -> ", "Log level -> ") & LogLevelText() & vbCrLf & vbCrLf & _
           LogLevelHelp(), vbInformation, TITLE
End Sub

' Aide affichee a chaque changement de niveau : rappelle ce que chaque
' niveau ecrit dans debug.log (du moins au plus verbeux). Marque le niveau
' courant d'une fleche.
Private Function LogLevelHelp() As String
    Dim cur As Long
    cur = modDebugLog.GlobalLevel
    LogLevelHelp = _
        Marker(cur, modDebugLog.LOG_ERROR) & T("OFF   : erreurs uniquement (defaut)", "OFF   : errors only (default)") & vbCrLf & _
        Marker(cur, modDebugLog.LOG_INFO) & T("INFO  : + actions reussies (RDV decale, mail range...)", "INFO  : + successful actions (appt shifted, mail filed...)") & vbCrLf & _
        Marker(cur, modDebugLog.LOG_DEBUG) & T("DEBUG : + decisions et valeurs intermediaires", "DEBUG : + decisions and intermediate values") & vbCrLf & _
        Marker(cur, modDebugLog.LOG_TRACE) & T("TRACE : + entree/sortie de fonctions (tout, verbeux)", "TRACE : + function entry/exit (everything, verbose)")
End Function

Private Function Marker(ByVal cur As Long, ByVal lvl As Long) As String
    If cur = lvl Then
        Marker = ">> "
    Else
        Marker = "   "
    End If
End Function

Private Function LogLevelText() As String
    Select Case modDebugLog.GlobalLevel
        Case modDebugLog.LOG_ERROR: LogLevelText = T("OFF (erreurs seules)", "OFF (errors only)")
        Case modDebugLog.LOG_INFO:  LogLevelText = "INFO"
        Case modDebugLog.LOG_DEBUG: LogLevelText = "DEBUG"
        Case modDebugLog.LOG_TRACE: LogLevelText = "TRACE"
        Case Else: LogLevelText = "LVL " & modDebugLog.GlobalLevel
    End Select
End Function

'=============================================================================
' MaintFilterModules  (epinglable au QAT)
' Pseudo-UI rechargee : liste numerotee de tous les modules connus avec
' statut ACTIF/off. Taper un numero -> toggle. 0 / vide -> quitter en
' gardant la config courante.
'=============================================================================
Public Sub MaintFilterModules()
    Dim mods() As String, n As Long
    n = SplitModules(mods)
    If n = 0 Then
        MsgBox T("Aucun module connu (v~erifier KnownModules dans config.ini).", "No known module (check KnownModules in config.ini)."), _
               vbExclamation, TITLE
        Exit Sub
    End If

    Dim choice As String, idx As Long
    Do
        ' Construire l'affichage avec statut
        Dim disp As String, i As Long
        Dim filterStr As String
        filterStr = CurrentFilter()
        disp = T("=== FILTRE MODULES ===", "=== MODULE FILTER ===") & vbCrLf & _
               T("(ACTIF = trace dans le log)", "(ACTIF = traced in the log)") & vbCrLf & vbCrLf
        If Len(filterStr) = 0 Then
            disp = disp & T("Filtre courant : (tous actifs)", "Current filter: (all active)") & vbCrLf & vbCrLf
        Else
            disp = disp & T("Filtre courant : ", "Current filter: ") & filterStr & vbCrLf & vbCrLf
        End If
        For i = 0 To n - 1
            disp = disp & Format$(i + 1, "00") & " - " & _
                   Left$(mods(i) & String(14, " "), 14) & " [" & _
                   IIf(IsActive(mods(i), filterStr), T("ACTIF", "ON   "), "  off") & "]" & vbCrLf
        Next i
        disp = disp & vbCrLf & T("Numero pour basculer, 0 = quitter (garde la config) :", "Number to toggle, 0 = quit (keep config):")

        choice = modUi.AskText(disp, TITLE)
        Select Case Trim$(choice)
            Case "", "0": Exit Sub
            Case Else
                If IsNumeric(choice) Then
                    idx = CLng(choice)
                    If idx >= 1 And idx <= n Then
                        ToggleModule mods, n, mods(idx - 1)
                    Else
                        MsgBox T("Num~ero hors plage.", "Number out of range."), vbExclamation, TITLE
                    End If
                Else
                    MsgBox T("Entree invalide.", "Invalid entry."), vbExclamation, TITLE
                End If
        End Select
    Loop
End Sub

Private Function LogFilterText() As String
    Dim f As String
    f = CurrentFilter()
    If Len(f) = 0 Then
        LogFilterText = T("(tous actifs)", "(all active)")
    Else
        LogFilterText = f
    End If
End Function

' Le filtre courant n'est pas lisible directement depuis modDebugLog
' (champ prive). On en garde une copie locale synchronisee a chaque
' SetEnabledModules emis ici. Comme la seule UI qui touche au filtre est
' ce module, la copie reste fiable au sein d'une session.
' (Declaration de mFilterLocal en tete de module.)

Private Function CurrentFilter() As String
    CurrentFilter = mFilterLocal
End Function

Private Function IsActive(ByVal moduleName As String, ByVal filterStr As String) As Boolean
    If Len(filterStr) = 0 Then
        IsActive = True            ' filtre vide -> tous actifs
    Else
        IsActive = (InStr(1, "|" & filterStr & "|", "|" & moduleName & "|", vbTextCompare) > 0)
    End If
End Function

' Bascule un module. Logique :
'   - filtre vide (tous actifs) + toggle X  -> filtre = tous SAUF X
'   - filtre non vide + X present           -> retirer X
'   - filtre non vide + X absent            -> ajouter X
'   - si apres ajout le filtre contient TOUS les modules -> revenir a vide
'     (etat "tous actifs" canonique, plus propre dans le log)
Private Sub ToggleModule(ByRef mods() As String, ByVal n As Long, ByVal target As String)
    Dim filterStr As String
    filterStr = CurrentFilter()

    If Len(filterStr) = 0 Then
        ' Tous actifs -> on desactive target : filtre = tous les autres
        Dim s As String, i As Long
        For i = 0 To n - 1
            If StrComp(mods(i), target, vbTextCompare) <> 0 Then
                s = s & IIf(Len(s) > 0, "|", "") & mods(i)
            End If
        Next i
        ApplyFilter s
        Exit Sub
    End If

    If InStr(1, "|" & filterStr & "|", "|" & target & "|", vbTextCompare) > 0 Then
        ' Present -> retirer
        ApplyFilter RemoveFromPipe(filterStr, target)
    Else
        ' Absent -> ajouter
        Dim nf As String
        nf = filterStr & "|" & target
        ' Si on vient de tout reactiver, revenir a "vide" (canonique)
        If CountPipe(nf) >= n Then
            ApplyFilter ""
        Else
            ApplyFilter nf
        End If
    End If
End Sub

Private Sub ApplyFilter(ByVal filterStr As String)
    mFilterLocal = filterStr
    modDebugLog.SetEnabledModules filterStr
End Sub

Private Function RemoveFromPipe(ByVal pipe As String, ByVal item As String) As String
    Dim parts() As String, i As Long, s As String
    parts = Split(pipe, "|")
    For i = LBound(parts) To UBound(parts)
        If Len(parts(i)) > 0 And StrComp(parts(i), item, vbTextCompare) <> 0 Then
            s = s & IIf(Len(s) > 0, "|", "") & parts(i)
        End If
    Next i
    RemoveFromPipe = s
End Function

Private Function CountPipe(ByVal pipe As String) As Long
    If Len(pipe) = 0 Then
        CountPipe = 0
    Else
        CountPipe = UBound(Split(pipe, "|")) + 1
    End If
End Function

Private Function SplitModules(ByRef mods() As String) As Long
    Dim raw As String
    raw = modConfig.Cfg("KnownModules", MODULES_DEFAULT)
    If Len(Trim$(raw)) = 0 Then raw = MODULES_DEFAULT
    mods = Split(raw, "|")
    SplitModules = UBound(mods) - LBound(mods) + 1
End Function

'=============================================================================
' MaintOpenConfig  (epinglable au QAT)
' Ouvre config.ini dans Notepad. Propose de creer un modele si absent.
'=============================================================================
Public Sub MaintOpenConfig()
    On Error GoTo ErrH
    Dim p As String
    p = modConfig.ConfigPath()
    If Dir(p) = "" Then
        ' Creer un fichier modele si absent, pour edition immediate.
        If MsgBox(T("Aucun config.ini.", "No config.ini.") & vbCrLf & _
                  T("Cr~eer un mod~fle commente maintenant ?", "Create a commented template now?"), _
                  vbQuestion + vbYesNo, TITLE) = vbYes Then
            CreateConfigTemplate p
        Else
            Exit Sub
        End If
    End If
    Shell "notepad.exe """ & p & """", vbNormalFocus
    Exit Sub
ErrH:
    MsgBox T("Erreur ouverture config : ", "Config open error: ") & Err.Number & " - " & Err.Description, _
           vbExclamation, TITLE
End Sub

'=============================================================================
' MaintOpenVBAFolder  (epinglable au QAT)
' Ouvre <Bureau>\Macro VBA\Outlook\VBA dans l'Explorateur. Propose de
' creer le dossier s'il n'existe pas.
'=============================================================================
Public Sub MaintOpenVBAFolder()
    On Error GoTo ErrH
    Dim p As String
    p = modConfig.VBADir()
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(p) Then
        If MsgBox(T("Le dossier VBA n'existe pas :", "The VBA folder does not exist:") & vbCrLf & p & vbCrLf & vbCrLf & _
                  T("Le cr~eer ?", "Create it?"), vbQuestion + vbYesNo, TITLE) = vbYes Then
            CreateFolderRec p
        Else
            Exit Sub
        End If
    End If

' Teste l'ecriture reelle : cree puis supprime un fichier sonde.
Private Function TestWrite(ByVal dirPath As String) As String
    On Error GoTo ErrH
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(dirPath) Then
        TestWrite = T("KO (dossier absent)", "KO (folder missing)")
        Exit Function
    End If
    Dim probe As String
    probe = dirPath & "\__maint_probe_" & Format$(Now, "hhnnss") & ".tmp"
    Dim f As Integer
    f = FreeFile
    Open probe For Output As #f
    Print #f, "probe"
    Close #f
    If Dir(probe) <> "" Then
        Kill probe
        TestWrite = T("OK (ecriture/suppression)", "OK (write/delete)")
    Else
        TestWrite = T("KO (ecriture echouee)", "KO (write failed)")
    End If
    Exit Function
ErrH:

End Function

' procedure a partir d'une chaine. Une ligne par module instrumente.

' ModVersion, (2) ajouter sa Case ici, (3) verifier qu'il figure dans
' MODULES_DEFAULT. Rien d'autre.
'

'=============================================================================
' MaintModuleVersions (v3) - versions REELLEMENT chargees dans le VBE.
' Lues dans la constante de chaque module, pas dans config.ini : un module
' non recolle apres une livraison affiche donc son ANCIENNE version, ce qui
' est exactement le signal recherche.
'=============================================================================
Public Sub MaintModuleVersions()
    On Error GoTo ErrH

    Dim mods() As String, n As Long, i As Long
    n = SplitModules(mods)

    Dim v As String
    Dim known As String, unknown As String
    Dim nbKnown As Long, nbUnknown As Long

    For i = 0 To n - 1
        v = ModuleVersionOf(mods(i))
        If Len(v) > 0 Then
            known = known & "  " & Left$(mods(i) & String(16, " "), 16) & _
                    " " & v & vbCrLf
            nbKnown = nbKnown + 1
        Else
            If Len(unknown) > 0 Then unknown = unknown & ", "
            unknown = unknown & mods(i)
            nbUnknown = nbUnknown + 1
        End If
    Next i

    Dim r As String
    r = "=== " & T("VERSIONS CHARG~EES DANS LE VBE", "VERSIONS LOADED IN THE VBE") & " ===" & vbCrLf & _
        T("(lues dans le code lui-m~xme, pas dans config.ini)", _
          "(read from the code itself, not from config.ini)") & vbCrLf & vbCrLf & _
        known & vbCrLf & _
        T("Non instrument~es (", "Not instrumented (") & nbUnknown & _
        T(") :", "):") & vbCrLf & _
        "  " & unknown & vbCrLf & vbCrLf & _
        nbKnown & "/" & n & " " & T("module(s) instrument~e(s).", "module(s) instrumented.")

    modDebugLog.DInfo MOD_NAME, "MaintModuleVersions : " & nbKnown & "/" & n & " instrumentes"
    modUi.Inform r, TITLE
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "MaintModuleVersions err " & Err.Number & " - " & Err.Description
    MsgBox T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, _
           vbExclamation, TITLE
End Sub

'=============================================================================
' AssertBindings  (health-check des WithEvents - v2, 2026-06-10)
' Verifie que les bindings evenementiels critiques sont vivants :
'   1. SentItems ItemAdd        (ThisOutlookSession.mSentItems)
'   2. CalendarItems Add/Change (ThisOutlookSession.CalendarItems)
'   3. Explorer FolderSwitch    (ThisOutlookSession.olMainExplorer)
'   4. Inspectors NewInspector  (modNewMeetingDefault.g_AppEvents ->
'                                clsAppEvents.m_Inspectors)
'   5. Explorer InlineResponse  (clsAppEvents.m_Explorer)
' Un WithEvents remis a Nothing par un Ctrl+S VBE = plus AUCUN evenement,
' silencieusement : c'est la 1re cause historique des regressions.
'
' silentIfOk:=True  : log seul si tout est OK ; MsgBox d'alerte sinon
'                     (mode demarrage / fin de Reinit).
' silentIfOk:=False : rapport systematique ; OK -> MsgBoxC auto-fermee 5 s,
'                     KO -> MsgBox persistante + rappel Reinit (mode menu).
' Retour : True si les 5 bindings sont vivants.
'
' NB : les libelles de bindings sont des chaines FONCTIONNELLES de
' diagnostic (noms d'evenements Outlook), volontairement hors T().
' NB : le binding 5 (InlineResponse) peut etre legitimement KO si aucune
' fenetre Outlook n'existait a l'initialisation de clsAppEvents
' (ActiveExplorer = Nothing) ; un Reinit le repare.
' Epinglable au QAT.
'=============================================================================
Public Function AssertBindings(Optional ByVal silentIfOk As Boolean = False) As Boolean
    On Error Resume Next

    ' Pack v3 : 6e binding = watchers InboxAlert (compte comme vivant si la
    ' fonction est desactivee par config, pour ne pas alerter a tort).
    Const NB_BINDINGS As Long = 6
    Dim names(1 To NB_BINDINGS) As String
    Dim alive(1 To NB_BINDINGS) As Boolean
    Dim i As Long

    names(1) = T("SentItems ItemAdd (QuickMove / ReplyInPlace)", "SentItems ItemAdd (QuickMove / ReplyInPlace)")
    alive(1) = ThisOutlookSession.SentItemsEventsBound

    names(2) = T("CalendarItems ItemAdd/ItemChange (tag / couleur / anti-chev.)", "CalendarItems ItemAdd/ItemChange (tag / colour / anti-overlap)")
    alive(2) = ThisOutlookSession.CalendarEventsBound

    names(3) = T("Explorer FolderSwitch (calendrier fenetre separee)", "Explorer FolderSwitch (calendar in separate window)")
    alive(3) = ThisOutlookSession.MainExplorerEventsBound

    names(4) = T("Inspectors NewInspector (duree defaut / ReplyInPlace / Maximize)", "Inspectors NewInspector (default duration / ReplyInPlace / Maximize)")
    alive(4) = modNewMeetingDefault.InspectorEventsBound

    names(5) = T("Explorer InlineResponse (ReplyInPlace volet de lecture)", "Explorer InlineResponse (ReplyInPlace reading pane)")
    alive(5) = modNewMeetingDefault.InlineExplorerBound

    names(6) = T("Inbox ItemAdd (alertes nouveaux mails)", "Inbox ItemAdd (new mail alerts)")
    If modConfig.InboxAlertEnabled() Then
        alive(6) = modInboxAlert.InboxAlertBound()
    Else
        alive(6) = True
    End If

    Dim allOk As Boolean, okCount As Long, deadList As String
    allOk = True
    For i = 1 To NB_BINDINGS
        If alive(i) Then
            okCount = okCount + 1
            modDebugLog.DDebug MOD_NAME, "Binding OK   : " & names(i)
        Else
            allOk = False
            deadList = deadList & "  - " & names(i) & vbCrLf
            modDebugLog.DError MOD_NAME, "Binding MORT : " & names(i)
        End If
    Next i

    AssertBindings = allOk
    modDebugLog.DInfo MOD_NAME, "AssertBindings : " & okCount & "/" & NB_BINDINGS & " OK"

    If allOk Then
        If Not silentIfOk Then
            ' Rapport positif auto-ferme (n'exige aucun clic).
            modMsgBoxCustom.MsgBoxC _
                T("Bindings WithEvents : 6/6 actifs.", "WithEvents bindings: 6/6 alive."), _
                vbInformation + vbOKOnly, TITLE, "", "", "", 5000, 1
        End If
    Else
        MsgBox T("BINDINGS EVENEMENTIELS MORTS :", "DEAD EVENT BINDINGS:") & vbCrLf & vbCrLf & _
               deadList & vbCrLf & _
               T("Les automatismes associ~es ne tournent plus.", "The related automations are no longer running.") & vbCrLf & _
               T("Correctif : Maintenance > Bindings & cache > Reinit.", "Fix: Maintenance > Bindings & cache > Reinit."), _
               vbExclamation, TITLE
    End If
End Function

Private Sub CreateFolderRec(ByVal path As String)
    On Error Resume Next
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    If fso.FolderExists(path) Then Exit Sub
    Dim parent As String
    parent = fso.GetParentFolderName(path)
    If Len(parent) > 0 And Not fso.FolderExists(parent) Then CreateFolderRec parent
    If Not fso.FolderExists(path) Then fso.CreateFolder path
End Sub
