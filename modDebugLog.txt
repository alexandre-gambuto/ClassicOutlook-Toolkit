Option Explicit

' =====================================================================
' modDebugLog v2.2
' Purpose (EN): centralized file logger shared by all modules. 4 levels
'   (ERROR/INFO/DEBUG/TRACE), per-module filtering, automatic size rotation.
'
' Logger fichier centralise pour tous les modules du projet.
'
' v2.2 (2026-11-16) : audit 11/2026 - F10-03, F10-13, F1-10 :
'   (a) le niveau du journal se relit tout seul dans config.ini a la
'   premiere ligne apres une remise a zero du projet VBA (modification ou
'   enregistrement dans l'editeur). Avant, il retombait a ERREUR jusqu'au
'   prochain LoadConfig et les lignes INFO etaient perdues : 3 fermetures
'   propres sur 4 n'avaient pas leur "SESSION END" ;
'   (b) rotation sur deux generations (debug.log.old puis debug.log.old2) :
'   l'historique precedent n'est plus ecrase a chaque rotation ;
'   (c) LastLineTime() : heure de la derniere ligne du journal (diagnostic
'   de la fermeture precedente au demarrage) ; LogClip() : texte tronque
'   pour le journal (objets de mails) ;
'   (d) en-tete remis d'accord avec le code (niveau par defaut, chemin).
'
' v2.1 (2026-07-05) : (a) un seul Open rate ne coupe plus le log pour
'   toute la session (le fichier vit sous le Bureau OneDrive : un verrou
'   de synchronisation est souvent TRANSITOIRE) -> suspension apres 3
'   echecs consecutifs seulement, re-sondage automatique apres 5 min.
'   (b) perf : verification du dossier 1x/session et test de rotation
'   1x/50 ecritures (3 acces disque par ligne auparavant).
'   (c) commentaire d'echo VBE aligne sur le comportement reel.
'
' 4 niveaux :
'   ERROR (0) : erreurs/exceptions
'   INFO  (1) : evenements metier (RDV decale, mail range, categorie posee)
'   DEBUG (2) : details pour diagnostic (valeurs intermediaires)
'   TRACE (3) : trace fine (entree/sortie de fonctions)
'
' Controle via 2 mecanismes combines :
'   1. Niveau global (variable m_GlobalLevel) : seuls les messages de
'      niveau <= ce niveau sortent (ERROR toujours).
'   2. Filtre par module (m_EnabledModules, "ModA|ModB") : si vide, tous
'      les modules ; sinon seulement ceux listes (SetEnabledModules).
'
' Niveau par defaut : DefaultLogLevel de config.ini (defaut 1 = INFO),
' applique par LoadConfig et, depuis v2.2, relu tout seul a la premiere
' ligne ecrite apres une remise a zero du projet VBA. A chaud (fenetre
' Execution) : LogOff / LogInfoMode / LogDebugMode / LogTraceMode, ou
' SetGlobalLevel n (jusqu'au prochain demarrage).
'
' Ecrit dans : <outil>\Sorties\debug.log (modConfig.MacroVBADir()).
' Rotation a LOG_MAX_SIZE octets : debug.log -> debug.log.old ->
' debug.log.old2 (v2.2 : deux generations conservees).
'
' API :
'   DLog(moduleName, level, msg)    - log unitaire
'   DError(module, msg)             - raccourci ERROR
'   DInfo(module, msg)              - raccourci INFO
'   DDebug(module, msg)             - raccourci DEBUG
'   DTrace(module, msg)             - raccourci TRACE
'   DLogSection(title)              - separateur visuel
'   OpenLog                         - ouvre debug.log dans Notepad
'   ClearLog                        - remet le fichier a zero
'   LogFilePath                     - chemin complet
'   SetGlobalLevel(level)           - change le niveau a chaud
'   LastLineTime()                  - heure de la derniere ligne (v2.2)
'   LogClip(texte, longueur)        - texte tronque pour le journal (v2.2)
' =====================================================================

' === Configuration par defaut ===
' Passer a LOG_DEBUG ou LOG_TRACE pour une session de diagnostic.
Private m_GlobalLevel As Long   ' initialise au premier appel
' v2.2 (F10-03) : niveau deja pris dans config.ini (ou pose a la main) ?
' Remis a False par une remise a zero du projet VBA, comme m_GlobalLevel.
Private m_LevelInit As Boolean

' Niveaux (constantes publiques pour que les modules les utilisent)
Public Const LOG_ERROR As Long = 0
Public Const LOG_INFO As Long = 1
Public Const LOG_DEBUG As Long = 2
Public Const LOG_TRACE As Long = 3

' Rotation : taille max avant rotation (2 Mo)
Private Const LOG_MAX_SIZE As Long = 2097152

Private Const LOG_FILENAME As String = "debug.log"
Private Const LOG_FILENAME_OLD As String = "debug.log.old"
Private Const LOG_FILENAME_OLD2 As String = "debug.log.old2"   ' v2.2
' Pack v3 (D7) : le sous-dossier n'est plus duplique ici ; le chemin vient
' de modConfig.MacroVBADir() (source unique du dossier des sorties).
Private Const MOD_VERSION As String = "v2.2 - 2026-11-16"

' v2.2 : LastLineTime lit au plus ce nombre d'octets en fin de fichier.
Private Const TAIL_BYTES As Long = 8192

' Liste des modules actifs (vide = tous). Exemple : "AntiChev|NewMeeting"
Private m_EnabledModules As String

' Anti-spam d'erreurs disque (v2.1). AVANT : un seul Open rate desactivait
' le log pour TOUTE la session. APRES : suspension apres LOG_FAIL_MAX
' echecs CONSECUTIFS, puis re-sondage automatique apres LOG_RETRY_MINUTES.
Private m_LogDisabled As Boolean
Private m_OpenFailStreak As Long
Private m_RetryAt As Date
Private Const LOG_FAIL_MAX As Long = 3
Private Const LOG_RETRY_MINUTES As Long = 5

' Perf (v2.1) : Dir/MkDir/FileLen a chaque ligne = 3 acces disque par
' message. Dossier verifie une fois par session ; rotation testee toutes
' les ROTATION_CHECK_EVERY ecritures (depassement max ~50 lignes).
Private m_DirEnsured As Boolean
Private m_WritesSinceCheck As Long
Private Const ROTATION_CHECK_EVERY As Long = 50

' ---------------------------------------------------------------------
' Accesseurs de configuration
' ---------------------------------------------------------------------
Public Property Get GlobalLevel() As Long
    GlobalLevel = m_GlobalLevel
End Property

Public Sub SetGlobalLevel(ByVal level As Long)
    m_GlobalLevel = level
    m_LevelInit = True     ' v2.2 : un niveau pose explicitement prime
    DLog "modDebugLog", LOG_INFO, "GlobalLevel -> " & LevelName(level)
End Sub

Public Sub SetEnabledModules(ByVal list As String)
    ' list = "ModA|ModB" pour filtrer, "" pour tout autoriser
    m_EnabledModules = list
    DLog "modDebugLog", LOG_INFO, "EnabledModules -> " & _
         IIf(Len(list) = 0, "(tous)", list)
End Sub

' ---------------------------------------------------------------------
' API principale
' ---------------------------------------------------------------------
Public Sub DLog(ByVal moduleName As String, ByVal level As Long, _
                ByVal msg As String)
    On Error Resume Next
    If m_LogDisabled Then
        If Now < m_RetryAt Then Exit Sub
        ' v2.1 : re-sondage apres suspension (verrou OneDrive transitoire).
        m_LogDisabled = False
        m_OpenFailStreak = 0
        m_DirEnsured = False
    End If

    ' v2.2 (F10-03) : premiere ligne depuis le chargement du projet VBA (ou
    ' depuis une remise a zero par l'editeur) : relire le niveau dans
    ' config.ini AVANT de filtrer. Sans cela m_GlobalLevel vaut 0 (ERREUR)
    ' et toute ligne INFO est jetee jusqu'au prochain LoadConfig (perte de
    ' "SESSION END", de "Reinit lance"...). Le drapeau est pose AVANT
    ' l'appel : une ligne ecrite pendant la lecture de config.ini ne relance
    ' pas la lecture (pas de boucle).
    If Not m_LevelInit Then
        m_LevelInit = True
        m_GlobalLevel = modConfig.ConfiguredLogLevel()
    End If

    ' Filtre par niveau
    If level > m_GlobalLevel Then Exit Sub
    
    ' Filtre par module (si liste non vide)
    If Len(m_EnabledModules) > 0 Then
        If InStr(1, "|" & m_EnabledModules & "|", "|" & moduleName & "|", _
                 vbTextCompare) = 0 Then
            Exit Sub
        End If
    End If
    
    Dim path As String
    path = LogFilePath()
    
    If Not m_DirEnsured Then
        EnsureLogDir
        m_DirEnsured = True
    End If
    m_WritesSinceCheck = m_WritesSinceCheck - 1
    If m_WritesSinceCheck <= 0 Then
        m_WritesSinceCheck = ROTATION_CHECK_EVERY
        CheckRotation path
    End If
    
    Dim f As Integer
    f = FreeFile
    Err.Clear
    Open path For Append As #f
    If Err.Number <> 0 Then
        ' v2.1 : suspension apres LOG_FAIL_MAX echecs consecutifs seulement.
        m_OpenFailStreak = m_OpenFailStreak + 1
        If m_OpenFailStreak >= LOG_FAIL_MAX Then
            m_LogDisabled = True
            m_RetryAt = DateAdd("n", LOG_RETRY_MINUTES, Now)
            Debug.Print "[modDebugLog] Log suspendu " & LOG_RETRY_MINUTES & _
                        " min (Open KO x" & m_OpenFailStreak & ") : " & path
        End If
        Err.Clear
        Exit Sub
    End If
    m_OpenFailStreak = 0
    Print #f, Format(Now, "yyyy-mm-dd hh:nn:ss") & " | " & _
              LevelShort(level) & " | " & Left(moduleName & "               ", 15) & " | " & msg
    Close #f
    
    ' Echo vers la fenetre Execution du VBE pour ERROR/INFO/DEBUG. TRACE
    ' est volontairement EXCLU (trop bavard pour le buffer limite de la
    ' fenetre). (v2.1 : commentaire aligne sur le comportement reel.)
    If level <= LOG_DEBUG Then
        Debug.Print "[" & moduleName & "/" & LevelShort(level) & "] " & msg
    End If
End Sub

Public Sub DError(ByVal moduleName As String, ByVal msg As String)
    DLog moduleName, LOG_ERROR, msg
End Sub

Public Sub DInfo(ByVal moduleName As String, ByVal msg As String)
    DLog moduleName, LOG_INFO, msg
End Sub

Public Sub DDebug(ByVal moduleName As String, ByVal msg As String)
    DLog moduleName, LOG_DEBUG, msg
End Sub

Public Sub DTrace(ByVal moduleName As String, ByVal msg As String)
    DLog moduleName, LOG_TRACE, msg
End Sub

Public Sub DLogSection(ByVal title As String)
    DLog "system", LOG_INFO, String(60, "=")
    DLog "system", LOG_INFO, title
    DLog "system", LOG_INFO, String(60, "=")
End Sub

' ---------------------------------------------------------------------
' Utilitaires
' ---------------------------------------------------------------------
Public Function ModVersion() As String
    ModVersion = MOD_VERSION
End Function

Public Function LogFilePath() As String
    LogFilePath = modConfig.MacroVBADir() & "\" & LOG_FILENAME
End Function

Private Function LogFilePathOld() As String
    LogFilePathOld = modConfig.MacroVBADir() & "\" & LOG_FILENAME_OLD
End Function

Public Sub OpenLog()
    Dim path As String
    path = LogFilePath()
    If Dir(path) = "" Then
        MsgBox T("Aucun fichier log pour l'instant : ", "No log file yet: ") & path, _
               vbInformation, "modDebugLog"
        Exit Sub
    End If
    Shell "notepad.exe """ & path & """", vbNormalFocus
End Sub

Public Sub ClearLog()
    On Error GoTo ErrH
    Dim path As String
    path = LogFilePath()
    EnsureLogDir
    Dim f As Integer
    f = FreeFile
    Open path For Output As #f
    Close #f
    m_LogDisabled = False
    m_OpenFailStreak = 0
    m_DirEnsured = False
    MsgBox T("Log vide : ", "Log cleared: ") & path, vbInformation, "modDebugLog"
    Exit Sub
ErrH:
    MsgBox T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, _
           vbExclamation, "modDebugLog"
End Sub

' ---------------------------------------------------------------------
' Internes
' ---------------------------------------------------------------------
Private Sub EnsureLogDir()
    On Error Resume Next
    ' Pack v3 : creation segment par segment du chemin MacroVBADir (parent
    ' puis sous-dossier), sans dupliquer les noms de dossiers.
    Dim subDir As String, baseDir As String, p As Long
    subDir = modConfig.MacroVBADir()
    p = InStrRev(subDir, "\")
    If p > 0 Then
        baseDir = Left$(subDir, p - 1)
        If Dir(baseDir, vbDirectory) = "" Then MkDir baseDir
    End If
    If Dir(subDir, vbDirectory) = "" Then MkDir subDir
    On Error GoTo 0
End Sub

' v2.2 (F10-13) : deux generations. debug.log -> debug.log.old ->
' debug.log.old2 (la plus ancienne est supprimee). Si le decalage de .old
' echoue (fichier tenu ouvert), .old est remplace comme avant. Taille lue
' par FileLen (erreur = fichier absent) : plus de Dir(), qui remettait a
' zero la boucle Dir$ d'un appelant toutes les 50 lignes.
Private Sub CheckRotation(ByVal path As String)
    On Error Resume Next
    Dim size As Long
    size = -1
    size = FileLen(path)
    If size < LOG_MAX_SIZE Then Exit Sub        ' absent (-1) ou sous le seuil

    Dim oldPath As String, old2Path As String, oldLen As Long
    oldPath = LogFilePathOld()
    old2Path = modConfig.MacroVBADir() & "\" & LOG_FILENAME_OLD2
    Err.Clear
    Kill old2Path
    Err.Clear
    Name oldPath As old2Path
    Err.Clear
    oldLen = -1
    oldLen = FileLen(oldPath)
    If oldLen >= 0 Then
        Err.Clear
        Kill oldPath                            ' decalage impossible : ancien .old perdu
    End If
    Err.Clear
    Name path As oldPath
    On Error GoTo 0
End Sub

'---------------------------------------------------------------------
' LastLineTime (v2.2 - audit 11/2026, F10-03 / F10-14)
' Heure de la derniere ligne horodatee de debug.log (0 si fichier absent
' ou illisible). A appeler AU DEMARRAGE, AVANT toute ligne de la nouvelle
' session (donc avant modConfig.LoadConfig) : elle donne alors l'heure de
' la fin de la session precedente, a comparer a la marque de fermeture
' propre posee par Application_Quit. Lecture des TAIL_BYTES derniers
' octets seulement ; ne journalise rien.
'---------------------------------------------------------------------
Public Function LastLineTime() As Date
    On Error Resume Next
    LastLineTime = 0
    Dim path As String, n As Long, startPos As Long
    path = LogFilePath()
    n = -1
    n = FileLen(path)
    If Err.Number <> 0 Or n <= 0 Then Exit Function

    startPos = n - TAIL_BYTES + 1
    If startPos < 1 Then startPos = 1
    Dim f As Integer, chunk As String
    f = FreeFile
    Err.Clear
    Open path For Binary Access Read Shared As #f
    If Err.Number <> 0 Then Exit Function
    chunk = String$(n - startPos + 1, vbNullChar)
    Get #f, startPos, chunk
    Close #f
    If Err.Number <> 0 Then Exit Function

    Dim lines() As String, i As Long, s As String, d As Date
    lines = Split(Replace(chunk, vbCr, ""), vbLf)
    For i = UBound(lines) To LBound(lines) Step -1
        s = lines(i)
        ' Format ecrit par DLog : "yyyy-mm-dd hh:nn:ss | ..."
        If Len(s) >= 19 Then
            If Mid$(s, 5, 1) = "-" And Mid$(s, 8, 1) = "-" And Mid$(s, 11, 1) = " " And _
               Mid$(s, 14, 1) = ":" And Mid$(s, 17, 1) = ":" And IsNumeric(Left$(s, 4)) Then
                d = 0
                d = DateSerial(CInt(Left$(s, 4)), CInt(Mid$(s, 6, 2)), CInt(Mid$(s, 9, 2))) + _
                    TimeSerial(CInt(Mid$(s, 12, 2)), CInt(Mid$(s, 15, 2)), CInt(Mid$(s, 18, 2)))
                If d > 0 Then
                    LastLineTime = d
                    Exit Function
                End If
            End If
        End If
    Next i
End Function

'---------------------------------------------------------------------
' LogClip (v2.2 - audit 11/2026, F10-13) : texte raccourci pour le
' journal (objet de mail, nom de fichier...) : maxLen caracteres au plus,
' "..." si coupe, retours a la ligne remplaces par des espaces. Le journal
' vit dans un dossier synchronise : n'y ecrire que le necessaire.
'---------------------------------------------------------------------
Public Function LogClip(ByVal s As String, Optional ByVal maxLen As Long = 20) As String
    s = Replace(Replace(s, vbCr, " "), vbLf, " ")
    If maxLen < 1 Then maxLen = 1
    If Len(s) > maxLen Then
        LogClip = Left$(s, maxLen) & "..."
    Else
        LogClip = s
    End If
End Function

Private Function LevelName(ByVal level As Long) As String
    Select Case level
        Case LOG_ERROR: LevelName = "ERROR"
        Case LOG_INFO: LevelName = "INFO"
        Case LOG_DEBUG: LevelName = "DEBUG"
        Case LOG_TRACE: LevelName = "TRACE"
        Case Else: LevelName = "LVL" & level
    End Select
End Function

Private Function LevelShort(ByVal level As Long) As String
    Select Case level
        Case LOG_ERROR: LevelShort = "ERR "
        Case LOG_INFO: LevelShort = "INFO"
        Case LOG_DEBUG: LevelShort = "DBG "
        Case LOG_TRACE: LevelShort = "TRC "
        Case Else: LevelShort = "????"
    End Select
End Function
