'=============================================================================
' DesktopPath
' Renvoie le chemin complet du Bureau (avec redirection OneDrive suivie).
' Valeur mise en cache apres premier appel.
'=============================================================================
Public Function DesktopPath() As String
    If Len(mDesktopPath) > 0 Then
        DesktopPath = mDesktopPath
        Exit Function
    End If

    Dim p As String
    p = ResolveDesktopPath()

    ' Normalisation : retirer un eventuel separateur final
    If Right(p, 1) = "\" Then p = Left(p, Len(p) - 1)

    mDesktopPath = p
    DesktopPath = mDesktopPath
End Function

'=============================================================================
' TimeTrackingPath
' Chemin complet de "Suivi de temps.xlsm".
' Surchargeable par la cle INI "TimeTrackingPath" (chemin absolu complet).
'=============================================================================
Public Function TimeTrackingPath() As String
    Dim o As String
    o = Cfg("TimeTrackingPath", "")
    If Len(o) > 0 Then
        TimeTrackingPath = o
    Else
        EnsureLoaded
        TimeTrackingPath = DesktopPath() & "\" & g_TimeTrackingFilename
    End If
End Function

'=============================================================================
' Cfg
' Renvoie la valeur string d'une cle de config.ini.
' Charge le fichier au premier appel (lazy). defaultVal si cle absente.
'=============================================================================
Public Function Cfg(ByVal key As String, ByVal defaultVal As String) As String
    On Error Resume Next
    EnsureConfigLoaded
    If mCfg Is Nothing Then
        Cfg = defaultVal
        Exit Function
    End If
    If mCfg.Exists(LCase$(key)) Then
        Cfg = mCfg(LCase$(key))
    Else
        Cfg = defaultVal
    End If
End Function

'=============================================================================
' CfgL
' Valeur Long d'une cle. defaultVal si absente ou non numerique.
'=============================================================================
Public Function CfgL(ByVal key As String, ByVal defaultVal As Long) As Long
    Dim s As String
    s = Cfg(key, "")
    If Len(s) = 0 Then
        CfgL = defaultVal
    ElseIf IsNumeric(s) Then
        CfgL = CLng(s)
    Else
        CfgL = defaultVal
    End If
End Function

'=============================================================================
' CfgB
' Valeur Boolean d'une cle. 1 / true / vrai / oui / yes = True.
'=============================================================================
Public Function CfgB(ByVal key As String, ByVal defaultVal As Boolean) As Boolean
    Dim s As String
    s = LCase$(Trim$(Cfg(key, "")))
    If Len(s) = 0 Then
        CfgB = defaultVal
        Exit Function
    End If
    Select Case s
        Case "1", "true", "vrai", "oui", "yes", "y", "o"
            CfgB = True
        Case "0", "false", "faux", "non", "no", "n"
            CfgB = False
        Case Else
            CfgB = defaultVal
    End Select
End Function

'=============================================================================
' ReloadConfig
' Force le rechargement complet : relit config.ini depuis le disque PUIS
' re-peuple les variables globales g_*. C'est ce qu'appelle le bouton
' "Recharger config" du menu Maintenance.
'=============================================================================
Public Sub ReloadConfig()
    mCfgLoaded = False
    Set mCfg = Nothing
    EnsureConfigLoaded
    mGlobalsLoaded = False
    LoadConfig
End Sub

'=============================================================================
' EnsureConfigLoaded (interne)
' Charge config.ini une seule fois. Cree mCfg meme si le fichier est absent
' (dictionnaire vide) pour que les appels suivants soient stables.
'=============================================================================
Private Sub EnsureConfigLoaded()
    On Error GoTo ErrH
    If mCfgLoaded And Not (mCfg Is Nothing) Then Exit Sub

    Set mCfg = CreateObject("Scripting.Dictionary")
    mCfg.CompareMode = 1   ' TextCompare (insensible a la casse, mais on stocke deja en LCase)
    mCfgLoaded = False

    Dim path As String
    path = ConfigPath()
    If Dir(path) = "" Then
        ' Pas de fichier : dictionnaire vide, defauts utilises partout.
        Exit Sub
    End If

    ' Lecture du fichier ENTIER en un bloc, puis decoupage tolerant aux
    ' terminateurs de ligne (CRLF Windows, CR Mac classique, LF Unix).
    ' Motif : Line Input #f casse uniquement sur CR/CRLF. Un config.ini
    ' reenregistre en LF seul (editeur Unix) est avale comme une unique
    ' ligne geante, commencant par ';' (en-tete commentaire), donc 0 cle
    ' lue. Ce bug etait silencieux : les defauts codes masquaient l'absence
    ' de chargement. (v3.4)
    Dim f As Integer, contenu As String
    Dim eqPos As Long, k As String, v As String
    f = FreeFile
    Open path For Input As #f
    If LOF(f) > 0 Then
        contenu = Input$(LOF(f), f)
    End If
    Close #f

    ' Normaliser tous les terminateurs vers vbLf avant Split.
    contenu = Replace(contenu, vbCrLf, vbLf)
    contenu = Replace(contenu, vbCr, vbLf)

    Dim lignes() As String
    lignes = Split(contenu, vbLf)

    Dim i As Long, line As String
    For i = LBound(lignes) To UBound(lignes)
        line = Trim$(lignes(i))
        If Len(line) = 0 Then GoTo NextLine
        If Left$(line, 1) = ";" Or Left$(line, 1) = "#" Then GoTo NextLine
        If Left$(line, 1) = "[" Then GoTo NextLine   ' section ignoree (parse plat)
        eqPos = InStr(line, "=")
        If eqPos < 2 Then GoTo NextLine               ' pas de cle valide
        k = LCase$(Trim$(Left$(line, eqPos - 1)))
        v = Trim$(Mid$(line, eqPos + 1))
        If Len(k) > 0 Then mCfg(k) = v                ' derniere occurrence gagne
NextLine:
    Next i

    mCfgLoaded = True
    modDebugLog.DInfo MOD_NAME, "config.ini charge (" & mCfg.count & " cles)"
    Exit Sub

ErrH:
    modDebugLog.DError MOD_NAME, "EnsureConfigLoaded err " & Err.Number & _
        " - " & Err.Description
    ' On garde un dictionnaire vide : defauts partout, pas de plantage.
    If mCfg Is Nothing Then Set mCfg = CreateObject("Scripting.Dictionary")
End Sub

'=============================================================================
' Resolution interne du Bureau (logique calquee sur modOneNote.GetDesktopPath)
'=============================================================================
Private Function ResolveDesktopPath() As String
    ' Priorite 1 : WScript.Shell.SpecialFolders("Desktop")
    ' -> suit la redirection OneDrive (Bureau / Desktop selon la langue Windows)
    On Error Resume Next
    Dim sH As Object
    Set sH = CreateObject("WScript.Shell")
    Dim p As String
    p = sH.SpecialFolders("Desktop")
    If Err.Number = 0 And Len(p) > 0 Then
        Dim fso As Object
        Set fso = CreateObject("Scripting.FileSystemObject")
        If fso.FolderExists(p) Then
            On Error GoTo 0
            ResolveDesktopPath = p
            Exit Function
        End If
    End If
    Err.Clear

    ' Fallbacks : emplacements connus. La variable OneDriveCommercial est
    ' generique (independante du nom de l'organisation) : elle pointe la
    ' racine OneDrive Entreprise quel que soit le tenant. Puis OneDrive
    ' perso, puis Bureau/Desktop local.
    Dim candidats As Variant
    candidats = Array( _
        Environ("OneDriveCommercial") & "\Bureau", _
        Environ("OneDriveCommercial") & "\Desktop", _
        Environ("OneDrive") & "\Bureau", _
        Environ("OneDrive") & "\Desktop", _
        Environ("USERPROFILE") & "\OneDrive\Bureau", _
        Environ("USERPROFILE") & "\OneDrive\Desktop", _
        Environ("USERPROFILE") & "\Bureau", _
        Environ("USERPROFILE") & "\Desktop")

    Dim fso2 As Object
    Set fso2 = CreateObject("Scripting.FileSystemObject")
    Dim candidat As Variant
    For Each candidat In candidats
        If Len(candidat) > 0 Then
            If fso2.FolderExists(CStr(candidat)) Then
                ResolveDesktopPath = CStr(candidat)
                On Error GoTo 0
                Exit Function
            End If
        End If
    Next candidat

    ' Dernier recours
    ResolveDesktopPath = Environ("USERPROFILE")
    On Error GoTo 0
End Function

'=============================================================================
' IsOpportunity
' True si la chaine correspond a un code opportunite, pour l'un des types
' configures (g_OpportunityTypes, ex. "OP"). Reproduit le test
' x Like "20##_OP_####" mais pilote par config (plusieurs types via |).
' (## = annee 2000-2099 ; #### = numero de g_OpportunityNumberLength chiffres.)
'=============================================================================
Public Function IsOpportunity(ByVal s As String) As Boolean
    On Error Resume Next
    EnsureLoaded
    Dim types() As String, i As Long, pat As String
    types = Split(g_OpportunityTypes, "|")
    For i = LBound(types) To UBound(types)
        If Len(Trim$(types(i))) > 0 Then
            pat = "20##_" & Trim$(types(i)) & "_" & _
                  String(g_OpportunityNumberLength, "#")
            If s Like pat Then
                IsOpportunity = True
                Exit Function
            End If
        End If
    Next i
End Function

'=============================================================================
' ScanOpportunityCode (pack v3 - A3)
' Renvoie le 1er code opportunite trouve dans txt ("20YY_<type>_<N chiffres>"),
' sinon "". Remplace les fenetres glissantes de 12 caracteres codees en dur
' dans modProjectTag (ScanOpportunity / IsValidProjectTag) : la largeur est
' DERIVEE de la config (types et longueur du numero), type par type.
'=============================================================================
Public Function ScanOpportunityCode(ByVal txt As String) As String
    On Error Resume Next
    EnsureLoaded
    ScanOpportunityCode = ""
    Dim types() As String, i As Long, pat As String, win As Long, p As Long
    types = Split(g_OpportunityTypes, "|")
    For i = LBound(types) To UBound(types)
        If Len(Trim$(types(i))) > 0 Then
            pat = "20##_" & Trim$(types(i)) & "_" & String(g_OpportunityNumberLength, "#")
            win = Len(pat)
            For p = 1 To Len(txt) - (win - 1)
                If Mid$(txt, p, win) Like pat Then
                    ScanOpportunityCode = Mid$(txt, p, win)
                    Exit Function
                End If
            Next p
        End If
    Next i
End Function

'=============================================================================
' ContainsOpportunityToken
' True si la chaine contient un token de type opportunite entoure de "_"
' (ex. "_OP_"), pour l'un des types configures. Reproduit le test
' historique InStr(s, "_OP_") > 0 (plusieurs types possibles via |).
'=============================================================================
Public Function ContainsOpportunityToken(ByVal s As String) As Boolean
    On Error Resume Next
    EnsureLoaded
    Dim types() As String, i As Long
    types = Split(g_OpportunityTypes, "|")
    For i = LBound(types) To UBound(types)
        If Len(Trim$(types(i))) > 0 Then
            If InStr(s, "_" & Trim$(types(i)) & "_") > 0 Then

                Exit Function
            End If
        End If
    Next i
End Function

'=============================================================================
' EscapeRegex (interne)
' Echappe les caracteres speciaux regex dans une chaine litterale, pour
' qu'un prefixe metier (g_ProjectPrefix, g_ContractYearPrefix) puisse etre
' insere dans un pattern sans casser la regex. Cas usuels (DOSS, 202) : aucun
' caractere special, renvoi inchange.
'=============================================================================
Private Function EscapeRegex(ByVal s As String) As String
    Dim i As Long, ch As String, out As String
    Const SPECIAUX As String = "\.^$*+?()[]{}|"
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        If InStr(SPECIAUX, ch) > 0 Then out = out & "\"
        out = out & ch
    Next i
    EscapeRegex = out
End Function

'=============================================================================
' A (helper accents)
' Source de verite unique pour les chaines UI accentuees. Decode des
' marqueurs ASCII inline "~X" en caracteres accentues, ce qui permet de
' garder le source 100% ASCII (immunise au cycle export .txt / import VBE,
' ou les accents litteraux se corrompent) tout en affichant correctement.
'
' A utiliser pour TOUTE chaine visible (MsgBox / InputBox / titres /
' StatusBar). NE PAS utiliser pour les chaines fonctionnelles (categories
' Outlook comparees, DASL, cles config, chemins, noms de fichiers) ni pour
' les logs (DInfo/DError restent ASCII).
'
' Table des marqueurs (reconstruite a partir des usages du projet) :
'   ~e -> e accent aigu        (ChrW 233)
'   ~f -> e accent grave       (ChrW 232)
'   ~a -> a accent grave       (ChrW 224)
'   ~E -> E majuscule aigu     (ChrW 201)
'   ~i -> i accent circonflexe (ChrW 238)
'   ~o -> o accent circonflexe (ChrW 244)
'   ~x -> e accent circonflexe (ChrW 234)   (pack v3)
'   ~c -> c cedille             (ChrW 231)   (pack v3)
'   ~u -> u accent circonflexe (ChrW 251)   (pack v3)
'   ~U -> u accent grave       (ChrW 249)   (pack v3)
'   ~A -> a accent circonflexe (ChrW 226)   (pack v3)
'   ~y -> i trema              (ChrW 239)   (pack v3)
'   ~I -> I accent circonflexe (ChrW 206)   (pack v3 : "BO~ITES" du projet analytics)
' Un "~" non suivi d'une lettre connue est laisse tel quel.
' PACK v3 : avant, "fen~etre" / "m~eme" / "~etre" etaient ecrits avec ~e
' (e aigu) faute de marqueur e-circonflexe : 25 libelles affichaient
' "fenetre" avec un accent aigu. Marqueur ~x ajoute, occurrences corrigees.
'=============================================================================
Public Function LangIsEN() As Boolean
    EnsureLoaded
    LangIsEN = (g_Language = "EN")
End Function

Public Function A(ByVal s As String) As String
    Dim out As String, i As Long, n As Long, ch As String, nx As String
    n = Len(s)
    i = 1
    Do While i <= n
        ch = Mid$(s, i, 1)
        If ch = "~" And i < n Then
            nx = Mid$(s, i + 1, 1)
            Select Case nx
                Case "e": out = out & ChrW$(233): i = i + 2
                Case "f": out = out & ChrW$(232): i = i + 2
                Case "a": out = out & ChrW$(224): i = i + 2
                Case "E": out = out & ChrW$(201): i = i + 2
                Case "i": out = out & ChrW$(238): i = i + 2
                Case "o": out = out & ChrW$(244): i = i + 2
                Case "x": out = out & ChrW$(234): i = i + 2
                Case "c": out = out & ChrW$(231): i = i + 2
                Case "u": out = out & ChrW$(251): i = i + 2
                Case "U": out = out & ChrW$(249): i = i + 2
                Case "A": out = out & ChrW$(226): i = i + 2
                Case "y": out = out & ChrW$(239): i = i + 2
                Case "I": out = out & ChrW$(206): i = i + 2
                Case Else
                    ' Marqueur inconnu : on garde le ~ tel quel.
                    out = out & ch
                    i = i + 1
            End Select
        Else
            out = out & ch
            i = i + 1
        End If
    Loop
    A = out
End Function

'=============================================================================
' T (helper langue UI)
' Choisit la chaine a afficher selon g_Language ("FR" par defaut, ou "EN").
'   T(fr, en) -> en si g_Language="EN" ET en non vide, sinon fr.
' Le repli FR (en="") garantit que le commutateur ne casse jamais l'UI, meme
' a moitie traduite. Les deux chaines passent par A() : le FR decode ses
' marqueurs d'accents (~e, ~f...), l'EN (nativement ASCII) est inchange.
'
' A utiliser pour TOUTE chaine visible (MsgBox / InputBox / titres /
' StatusBar), exactement comme A(). NE PAS utiliser pour les chaines
' fonctionnelles (categories Outlook comparees, DASL, cles config, chemins,
' noms de fichiers) ni pour les logs (DInfo/DError restent ASCII, FR).
'
' Exemple :
'   MsgBox T("Dossier d~ej~a rang~e.", "Already filed."), vbInformation
'=============================================================================
Public Function T(ByVal fr As String, ByVal en As String) As String
    EnsureLoaded
    If g_Language = "EN" And Len(en) > 0 Then
        T = A(en)
    Else
        T = A(fr)
    End If
End Function

'=============================================================================
' ContainsProjectCode
' True si le texte contient <prefixe><N chiffres> (ex. "DOSS1234").
' Pilote par config (prefixe + longueur). Insensible a la casse.
'=============================================================================
Public Function ContainsProjectCode(ByVal s As String) As Boolean
    On Error Resume Next
    EnsureLoaded
    ' Pack v3 : aligne sur ScanProjectCode (EXACTEMENT N chiffres). L'ancien
    ' Like "*PFX####*" acceptait aussi PFX12345 (5 chiffres).
    ContainsProjectCode = (Len(ScanProjectCode(s)) = g_ProjectCodeLength)
End Function

'=============================================================================
' ScanProjectCode
' Cherche <prefixe> suivi d'EXACTEMENT g_ProjectCodeLength chiffres et
' renvoie ces chiffres (sinon ""). Pilote par config (prefixe + longueur).
'=============================================================================
Public Function ScanProjectCode(ByVal txt As String) As String
    On Error Resume Next
    EnsureLoaded
    ScanProjectCode = ""
    Dim pfx As String: pfx = UCase$(g_ProjectPrefix)
    Dim lp As Long: lp = Len(pfx)
    Dim up As String: up = UCase$(txt)
    Dim pos As Long
    pos = InStr(1, up, pfx)
    Do While pos > 0
        Dim j As Long: j = pos + lp
        Dim num As String: num = ""
        Do While j <= Len(txt)
            Dim ch As String: ch = Mid$(txt, j, 1)
            If ch >= "0" And ch <= "9" Then
                num = num & ch
                j = j + 1
            Else
                Exit Do
            End If
        Loop
        If Len(num) = g_ProjectCodeLength Then
            ScanProjectCode = num
            Exit Function
        End If
        pos = InStr(pos + 1, up, pfx)
    Loop
End Function

'=============================================================================
' FindContractCode
' Renvoie le 1er code contrat trouve (fenetre glissante "20##_<N chiffres>"),
' sinon "". Pilote par config (prefixe annee + longueur du numero).
' (Annee 20## = 2000-2099 ; un code opportunite "20XX_OP_..." ne matche pas,
'  les lettres O/T ne pouvant tenir une position chiffre #.)
'=============================================================================
Public Function FindContractCode(ByVal txt As String) As String
    On Error Resume Next
    EnsureLoaded
    FindContractCode = ""
    Dim win As Long
    win = 4 + 1 + g_ContractNumberLength   ' "20" + 2 chiffres + "_" + N chiffres
    ' Pack v3 : le separateur accepte _ OU - comme le regex g_ContractPattern
    ' (avant : "_" seul ici, "[_\-]" dans le regex -> detections divergentes).
    Dim pat As String
    pat = "20##[_-]" & String(g_ContractNumberLength, "#")
    Dim i As Long
    For i = 1 To Len(txt) - (win - 1)
        Dim x As String: x = Mid$(txt, i, win)
        If x Like pat Then
            FindContractCode = x
            Exit Function
        End If
    Next i
End Function

'=============================================================================
' ContainsContractCode : True si FindContractCode renvoie un resultat.
'=============================================================================
Public Function ContainsContractCode(ByVal s As String) As Boolean
    ContainsContractCode = (Len(FindContractCode(s)) > 0)
End Function

End Function
