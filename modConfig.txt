'=============================================================================
' DesktopPath
' Renvoie le chemin complet du Bureau (avec redirection OneDrive suivie).
' Valeur mise en cache apres premier appel.
'=============================================================================
Public Function DesktopPath() As String
    If Len(mDesktopPath) > 0 Then
        DesktopPath = mDesktopPath
        Exit Function
    End If

    Dim p As String
    p = ResolveDesktopPath()

    ' Normalisation : retirer un eventuel separateur final
    If Right(p, 1) = "\" Then p = Left(p, Len(p) - 1)

    mDesktopPath = p
    DesktopPath = mDesktopPath
End Function

' TimeTrackingPath
' Chemin complet de "Suivi de temps.xlsm" : <outil>\Classeurs (v3.17 ;
' ancien emplacement : racine du Bureau).
' Surchargeable par la cle INI "TimeTrackingPath" (chemin absolu complet).
'=============================================================================
Public Function TimeTrackingPath() As String
    Dim o As String
    o = Cfg("TimeTrackingPath", "")
    If Len(o) > 0 Then
        TimeTrackingPath = o
    Else
        EnsureLoaded
        TimeTrackingPath = WorkbooksDir() & "\" & g_TimeTrackingFilename
    End If
End Function

'=============================================================================
' ToolkitDir (v3.17)
' Dossier de l'outil : config.ini a sa racine, Classeurs\, Sorties\,
' Ressources\, Code\. Resolu une fois par session (cache), dans l'ordre :
'   1. dossier memorise (registre) s'il contient config.ini ;
'   2. dossier "ClassicOutlook-Toolkit" contenant config.ini, a la racine
'      puis un niveau sous la racine de : OneDrive Entreprise, OneDrive
'      perso, OneDrive, Bureau (ex. <OneDrive perso>\Atelier VBA\
'      ClassicOutlook-Toolkit). Memorise des qu'il est trouve ;
'   3. dossier memorise sans config.ini (choisi a la main via
'      SetToolkitFolder, premiere installation) ;
'   4. ancien emplacement <Bureau>\Macro VBA\Outlook s'il existe (avec ou
'      sans config.ini) : utilise tel quel (tout a plat), jamais memorise ;
'   5. a defaut : <Bureau>\ClassicOutlook-Toolkit (premiere installation,
'      cf. modQuickStart).
' Ne lit pas config.ini et ne journalise pas : appele par ConfigPath() et,
' via MacroVBADir(), par le logger lui-meme.
'=============================================================================
Public Function ToolkitDir() As String
    If Len(mToolkitDir) = 0 Then ResolveToolkitDir
    ToolkitDir = mToolkitDir
End Function

'=============================================================================
' SetToolkitFolder (v3.17) - macro (Alt+F8, ou fenetre Execution du VBE)
' Designe a la main le dossier de l'outil (celui qui contient config.ini),
' si la recherche automatique ne le trouve pas ou pour en changer. Le
' chemin est memorise (registre HKCU) puis la configuration est relue.
' Champ vide = oubli du dossier memorise (retour a la recherche auto).
'=============================================================================
Public Sub SetToolkitFolder()
    On Error GoTo ErrH
    Dim ttl As String, ans As String, p As String
    ttl = T("Dossier de l'outil", "Toolkit folder")
    ans = InputBox(T("Dossier de l'outil, celui qui contient config.ini (chemin copi~e depuis l'Explorateur).", _
                     "Toolkit folder, the one holding config.ini (path copied from Explorer).") & vbCrLf & vbCrLf & _
                   T("Vide = recherche automatique.", "Empty = automatic search."), ttl, ToolkitDir())
    If StrPtr(ans) = 0 Then Exit Sub                 ' Annuler
    p = NormDir(Replace(ans, """", ""))
    If Len(p) > 0 Then
        If Not FolderOk(p) Then
            MsgBox T("Dossier introuvable :", "Folder not found:") & vbCrLf & p, vbExclamation, ttl
            Exit Sub
        End If
        If Not HasConfig(p) Then
            If MsgBox(T("Ce dossier ne contient pas config.ini. Le retenir quand m~xme ?", _
                        "This folder holds no config.ini. Keep it anyway?"), _
                      vbQuestion + vbYesNo, ttl) <> vbYes Then Exit Sub
        End If
    End If
    ' L'ancien emplacement n'est jamais memorise (il est reconnu tout seul) :
    ' memorise, il masquerait le nouveau dossier a chaque demarrage.
    Dim legacyPick As Boolean
    legacyPick = IsLegacyPath(p)
    If legacyPick Then
        SaveSetting REG_APP_TK, REG_SECTION_PATHS, REG_KEY_TOOLKITDIR, ""
    Else
        SaveSetting REG_APP_TK, REG_SECTION_PATHS, REG_KEY_TOOLKITDIR, p
    End If
    InvalidateDesktopCache
    If Len(p) > 0 Then
        ' Choix explicite : pris tel quel pour cette session (meme sans
        ' config.ini, que le guide de demarrage ou Maintenance > Ouvrir
        ' config creera ici).
        mToolkitDir = p
        mToolkitLegacy = legacyPick
    End If
    ' Dossier Sorties cree tout de suite : le journal (modDebugLog) ne le
    ' recree pas en cours de session.
    On Error Resume Next
    If FolderOk(ToolkitDir()) Then
        If Not FolderOk(MacroVBADir()) Then MkDir MacroVBADir()
    End If
    On Error GoTo ErrH
    ReloadConfig
    modDebugLog.DInfo MOD_NAME, "SetToolkitFolder : " & ToolkitDir()
    MsgBox T("Dossier de l'outil :", "Toolkit folder:") & vbCrLf & ToolkitDir() & vbCrLf & _
           IIf(legacyPick, T("(ancien emplacement : utilis~e tel quel, non m~emoris~e)", _
                             "(former location: used as is, not remembered)") & vbCrLf, "") & vbCrLf & _
           T("Red~emarrer Outlook pour que tous les modules le prennent en compte.", _
             "Restart Outlook so that every module picks it up."), vbInformation, ttl
    Exit Sub
ErrH:
    MsgBox T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, vbExclamation, _
           T("Dossier de l'outil", "Toolkit folder")
End Sub

'=============================================================================
' ResolveToolkitDir (interne, v3.17) - voir ToolkitDir.
'=============================================================================
Private Sub ResolveToolkitDir()
    On Error Resume Next
    Dim reg As String, p As String
    mToolkitLegacy = False

    ' 1. Dossier memorise, s'il contient config.ini. L'ancien emplacement
    '    n'est jamais pris d'ici (il passerait devant le nouveau dossier).
    reg = NormDir(GetSetting(REG_APP_TK, REG_SECTION_PATHS, REG_KEY_TOOLKITDIR, ""))
    If IsLegacyPath(reg) Then reg = ""
    If HasConfig(reg) Then
        mToolkitDir = reg
        Exit Sub
    End If

    ' 2. Recherche : OneDrive Entreprise / perso / Bureau, racine puis un niveau
    p = FindToolkitDir()
    If Len(p) > 0 Then
        mToolkitDir = p
        If StrComp(p, reg, vbTextCompare) <> 0 Then
            SaveSetting REG_APP_TK, REG_SECTION_PATHS, REG_KEY_TOOLKITDIR, p
        End If
        Exit Sub
    End If

    ' 3. Dossier memorise sans config.ini (choisi a la main)
    If FolderOk(reg) Then
        mToolkitDir = reg
        Exit Sub
    End If

    ' 4. Ancien emplacement : utilise tel quel, jamais memorise. Reconnu
    '    meme sans config.ini (installation qui tourne sur les defauts :
    '    debug.log y est cree depuis toujours).
    p = DesktopPath() & "\" & LEGACY_SUBDIR
    If FolderOk(p) Then
        mToolkitDir = p
        mToolkitLegacy = True
        Exit Sub
    End If

    ' 5. Premiere installation
    mToolkitDir = DesktopPath() & "\" & TOOLKIT_FOLDER
    If Len(mToolkitDir) <= Len(TOOLKIT_FOLDER) + 1 Then
        mToolkitDir = Environ$("USERPROFILE") & "\" & TOOLKIT_FOLDER
    End If
End Sub

'=============================================================================
' FindToolkitDir (interne, v3.17)
' Premier dossier "ClassicOutlook-Toolkit" contenant config.ini, a la
' racine de chaque base, puis un niveau sous la racine. "" si aucun.
'=============================================================================
Private Function FindToolkitDir() As String
    On Error Resume Next
    FindToolkitDir = ""
    Dim bases(1 To 4) As String, i As Long, j As Long, p As String
    bases(1) = NormDir(Environ$("OneDriveCommercial"))
    bases(2) = NormDir(Environ$("OneDriveConsumer"))
    bases(3) = NormDir(Environ$("OneDrive"))
    bases(4) = NormDir(DesktopPath())
    ' Doublons neutralises (OneDrive = l'un des deux autres en general)
    For i = 2 To 4
        For j = 1 To i - 1
            If StrComp(bases(i), bases(j), vbTextCompare) = 0 Then bases(i) = ""
        Next j
    Next i
    ' Racine de chaque base d'abord
    For i = 1 To 4
        If Len(bases(i)) > 0 Then
            If HasConfig(bases(i) & "\" & TOOLKIT_FOLDER) Then
                FindToolkitDir = bases(i) & "\" & TOOLKIT_FOLDER
                Exit Function
            End If
        End If
    Next i
    ' Puis un niveau sous la racine
    For i = 1 To 4
        If Len(bases(i)) > 0 Then
            p = ScanOneLevel(bases(i))
            If Len(p) > 0 Then
                FindToolkitDir = p
                Exit Function
            End If
        End If
    Next i
End Function

' Sous-dossiers directs de base : premier <base>\<x>\ClassicOutlook-Toolkit
' contenant config.ini. Erreur d'acces : recherche abandonnee pour cette base.
Private Function ScanOneLevel(ByVal base As String) As String
    On Error GoTo Fin
    ScanOneLevel = ""
    Dim fso As Object, sf As Object, p As String
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(base) Then Exit Function
    For Each sf In fso.GetFolder(base).SubFolders
        p = base & "\" & CStr(sf.Name) & "\" & TOOLKIT_FOLDER
        If HasConfig(p) Then
            ScanOneLevel = p
            Exit Function
        End If
    Next sf
Fin:
End Function

' True si le dossier p contient config.ini. FSO plutot que Dir$ : l'etat de
' Dir$ est partage avec l'appelant (boucle Dir$ en cours).
Private Function HasConfig(ByVal p As String) As Boolean
    On Error Resume Next
    Dim ok As Boolean
    ok = False
    If Len(p) > 0 Then ok = CreateObject("Scripting.FileSystemObject").FileExists(p & "\" & CONFIG_FILENAME)
    HasConfig = ok
End Function

' True si le dossier p existe (erreur = non).
Private Function FolderOk(ByVal p As String) As Boolean
    On Error Resume Next
    Dim ok As Boolean
    ok = False
    If Len(p) > 0 Then ok = CreateObject("Scripting.FileSystemObject").FolderExists(p)
    FolderOk = ok
End Function

' True si p est l'ancien emplacement <Bureau>\Macro VBA\Outlook (choisi a
' la main ou memorise) : disposition a plat, comme avant v3.17.
Private Function IsLegacyPath(ByVal p As String) As Boolean
    IsLegacyPath = False
    If Len(p) = 0 Then Exit Function
    IsLegacyPath = (StrComp(NormDir(p), DesktopPath() & "\" & LEGACY_SUBDIR, vbTextCompare) = 0)
End Function

' Chemin de dossier normalise : espaces et separateur final retires.
Private Function NormDir(ByVal p As String) As String
    p = Trim$(p)
    Do While Len(p) > 3 And Right$(p, 1) = "\"
        p = Left$(p, Len(p) - 1)
    Loop
    NormDir = p
End Function

'=============================================================================
' Cfg
' Renvoie la valeur string d'une cle de config.ini.
' Charge le fichier au premier appel (lazy). defaultVal si cle absente.
' v3.17 : jetons %TOOLKIT% / %VARIABLE% remplaces (ExpandTokens).
'=============================================================================
Public Function Cfg(ByVal key As String, ByVal defaultVal As String) As String
    On Error Resume Next
    EnsureConfigLoaded
    If mCfg Is Nothing Then
        Cfg = defaultVal
        Exit Function
    End If
    If mCfg.Exists(LCase$(key)) Then
        Cfg = ExpandTokens(CStr(mCfg(LCase$(key))))   ' v3.17 : %TOOLKIT%, %VARIABLE%
    Else
        Cfg = defaultVal
    End If
End Function

'=============================================================================
' ExpandTokens (interne, v3.17)
' Jetons dans une valeur de config.ini : %TOOLKIT% = dossier de l'outil,
' %NOM% = variable d'environnement Windows (ex. %OneDriveConsumer%,
' %USERPROFILE%). Jeton inconnu ou vide : laisse tel quel. Valeur sans "%" :
' renvoyee sans traitement.
'=============================================================================
Private Function ExpandTokens(ByVal s As String) As String
    ExpandTokens = s
    If InStr(1, s, "%") = 0 Then Exit Function
    Dim res As String, pos As Long, p1 As Long, p2 As Long, v As String
    pos = 1
    Do
        p1 = InStr(pos, s, "%")
        If p1 = 0 Then Exit Do
        p2 = InStr(p1 + 1, s, "%")
        If p2 = 0 Then Exit Do
        v = TokenValue(Mid$(s, p1 + 1, p2 - p1 - 1))
        If Len(v) > 0 Then
            res = res & Mid$(s, pos, p1 - pos) & v
            pos = p2 + 1
        Else
            res = res & Mid$(s, pos, p1 - pos + 1)
            pos = p1 + 1
        End If
    Loop
    ExpandTokens = res & Mid$(s, pos)
End Function

' Valeur d'un jeton (nom sans les %). "" si nom invalide ou inconnu.
Private Function TokenValue(ByVal nm As String) As String
    On Error Resume Next
    TokenValue = ""
    If Len(nm) = 0 Or Len(nm) > 40 Then Exit Function
    If nm Like "*[!A-Za-z0-9_]*" Then Exit Function
    If UCase$(nm) = "TOOLKIT" Then
        TokenValue = ToolkitDir()
    Else
        TokenValue = Environ$(nm)
    End If
End Function

'=============================================================================
' CfgL
' Valeur Long d'une cle. defaultVal si absente ou non numerique.
' v3.18 (F2-08) : une valeur hors des limites d'un Long (ex. un zero de
' trop) ou non numerique prend defaultVal et est signalee UNE fois au
' journal, au lieu de lever l'erreur 6 qui arretait LoadConfig et laissait
' a zero tous les reglages lus apres elle.
'=============================================================================
Public Function CfgL(ByVal key As String, ByVal defaultVal As Long) As Long
    Dim s As String
    s = Cfg(key, "")
    CfgL = defaultVal
    If Len(s) = 0 Then Exit Function
    If Not IsNumeric(s) Then
        WarnBadValue key, s, CStr(defaultVal)
        Exit Function
    End If
    On Error GoTo Bad
    CfgL = CLng(s)
    Exit Function
Bad:
    CfgL = defaultVal
    WarnBadValue key, s, CStr(defaultVal)
End Function

'=============================================================================
' CfgB
' Valeur Boolean d'une cle. 1 / true / vrai / oui / yes = True.
' v3.18 (F2-08) : valeur non reconnue -> defaut, signalee une fois.
'=============================================================================
Public Function CfgB(ByVal key As String, ByVal defaultVal As Boolean) As Boolean
    Dim s As String
    s = LCase$(Trim$(Cfg(key, "")))
    If Len(s) = 0 Then
        CfgB = defaultVal
        Exit Function
    End If
    Select Case s
        Case "1", "true", "vrai", "oui", "yes", "y", "o"
            CfgB = True
        Case "0", "false", "faux", "non", "no", "n"
            CfgB = False
        Case Else
            CfgB = defaultVal
            WarnBadValue key, s, IIf(defaultVal, "1", "0")
    End Select
End Function

'=============================================================================
' WarnBadValue (interne, v3.18 - F2-08)
' Signale au journal (ERREUR, une seule fois par cle et par chargement de
' config.ini) une valeur illisible et la valeur par defaut retenue a la
' place. Le nom de la cle est ecrit tel quel : c'est lui qu'il faut
' chercher dans config.ini.
'=============================================================================
Private Sub WarnBadValue(ByVal key As String, ByVal rawValue As String, ByVal usedValue As String)
    On Error Resume Next
    If mBadKeys Is Nothing Then
        Set mBadKeys = CreateObject("Scripting.Dictionary")
        mBadKeys.CompareMode = 1
    End If
    If mBadKeys.Exists(key) Then Exit Sub
    mBadKeys(key) = True
    modDebugLog.DError MOD_NAME, "config.ini : cle " & key & " = '" & Left$(rawValue, 40) & _
        "' illisible ou hors limites, valeur par defaut utilisee (" & usedValue & ")"
End Sub

' IsPersonalCategoryName (pack v3) : True si le nom de categorie donne est la
' categorie "personnel" configuree (ou sa variante courte). Comparaison
' insensible a la casse, sur un nom DEJA isole (pas la liste complete).
'=============================================================================
Public Function IsPersonalCategoryName(ByVal catName As String) As Boolean
    Dim c As String
    c = Trim$(catName)
    If Len(c) = 0 Then Exit Function
    If StrComp(c, PersonalCategory(), vbTextCompare) = 0 Then
        IsPersonalCategoryName = True
    ElseIf Len(PersonalCategoryAlt()) > 0 Then
        If StrComp(c, PersonalCategoryAlt(), vbTextCompare) = 0 Then IsPersonalCategoryName = True
    End If
End Function

' ContainsPersonalCategory (pack v3) : True si la liste de categories d'un
' item (separateurs , ; |) contient la categorie "personnel" configuree.
' Remplace les tests InStr(cats, "Personnel") disperses dans 4 modules.
'=============================================================================
Public Function ContainsPersonalCategory(ByVal cats As String) As Boolean
    If Len(cats) = 0 Then Exit Function
    Dim tmp As String
    tmp = Replace(Replace(cats, ";", ","), "|", ",")
    Dim parts() As String, i As Long
    parts = Split(tmp, ",")
    For i = LBound(parts) To UBound(parts)
        If IsPersonalCategoryName(parts(i)) Then
            ContainsPersonalCategory = True
            Exit Function
        End If
    Next i
End Function

'=============================================================================
' EnsureConfigLoaded (interne)
' Charge config.ini une seule fois. Cree mCfg meme si le fichier est absent
' (dictionnaire vide) pour que les appels suivants soient stables.
' v3.18 (F2-08) : un seul essai par chargement, fichier absent compris
' (mCfgTried ; avant, chaque Cfg() refaisait la lecture et un Dir() sur le
' chemin OneDrive tant que config.ini manquait). ReloadConfig relance un
' essai. Existence testee par FSO (HasConfig) : Dir() remettait a zero la
' boucle Dir$ en cours chez l'appelant.
'=============================================================================
Private Sub EnsureConfigLoaded()
    On Error GoTo ErrH
    If mCfgTried And Not (mCfg Is Nothing) Then Exit Sub
    mCfgTried = True

    Set mCfg = CreateObject("Scripting.Dictionary")
    mCfg.CompareMode = 1   ' TextCompare (insensible a la casse, mais on stocke deja en LCase)
    mCfgLoaded = False

    Dim path As String
    path = ConfigPath()
    If Not HasConfig(ToolkitDir()) Then
        ' Pas de fichier : dictionnaire vide, defauts utilises partout.
        Exit Sub
    End If

    ' Lecture du fichier ENTIER en un bloc, puis decoupage tolerant aux
    ' terminateurs de ligne (CRLF Windows, CR Mac classique, LF Unix).
    ' Motif : Line Input #f casse uniquement sur CR/CRLF. Un config.ini
    ' reenregistre en LF seul (editeur Unix) est avale comme une unique
    ' ligne geante, commencant par ';' (en-tete commentaire), donc 0 cle
    ' lue. Ce bug etait silencieux : les defauts codes masquaient l'absence
    ' de chargement. (v3.4)
    Dim f As Integer, contenu As String
    Dim eqPos As Long, k As String, v As String
    f = FreeFile
    Open path For Input As #f
    If LOF(f) > 0 Then
        contenu = Input$(LOF(f), f)
    End If
    Close #f

    ' Normaliser tous les terminateurs vers vbLf avant Split.
    contenu = Replace(contenu, vbCrLf, vbLf)
    contenu = Replace(contenu, vbCr, vbLf)

    Dim lignes() As String
    lignes = Split(contenu, vbLf)

    Dim i As Long, line As String
    For i = LBound(lignes) To UBound(lignes)
        line = Trim$(lignes(i))
        If Len(line) = 0 Then GoTo NextLine
        If Left$(line, 1) = ";" Or Left$(line, 1) = "#" Then GoTo NextLine
        If Left$(line, 1) = "[" Then GoTo NextLine   ' section ignoree (parse plat)
        eqPos = InStr(line, "=")
        If eqPos < 2 Then GoTo NextLine               ' pas de cle valide
        k = LCase$(Trim$(Left$(line, eqPos - 1)))
        v = Trim$(Mid$(line, eqPos + 1))
        If Len(k) > 0 Then mCfg(k) = v                ' derniere occurrence gagne
NextLine:
    Next i

    mCfgLoaded = True
    modDebugLog.DInfo MOD_NAME, "config.ini charge (" & mCfg.count & " cles)"
    Exit Sub

ErrH:
    modDebugLog.DError MOD_NAME, "EnsureConfigLoaded err " & Err.Number & _
        " - " & Err.Description
    ' On garde un dictionnaire vide : defauts partout, pas de plantage.
    If mCfg Is Nothing Then Set mCfg = CreateObject("Scripting.Dictionary")
    CloseQuietly f
    ' v3.18 : erreur de LECTURE (verrou OneDrive passager...) : nouvel essai
    ' au prochain appel, comme avant. Seul le fichier absent est definitif.
    mCfgTried = False
End Sub

' Ferme un numero de fichier sans jamais lever d'erreur (appelable depuis un
' gestionnaire d'erreur actif). 0 = rien a fermer.
Private Sub CloseQuietly(ByVal fileNum As Integer)
    On Error Resume Next
    If fileNum <> 0 Then Close #fileNum
End Sub

'=============================================================================
' Resolution interne du Bureau (logique calquee sur modOneNote.GetDesktopPath)
'=============================================================================
Private Function ResolveDesktopPath() As String
    ' Priorite 1 : WScript.Shell.SpecialFolders("Desktop")
    ' -> suit la redirection OneDrive (Bureau / Desktop selon la langue Windows)
    On Error Resume Next
    Dim sH As Object
    Set sH = CreateObject("WScript.Shell")
    Dim p As String
    p = sH.SpecialFolders("Desktop")
    If Err.Number = 0 And Len(p) > 0 Then
        Dim fso As Object
        Set fso = CreateObject("Scripting.FileSystemObject")
        If fso.FolderExists(p) Then
            On Error GoTo 0
            ResolveDesktopPath = p
            Exit Function
        End If
    End If
    Err.Clear

    ' Fallbacks : emplacements connus. La variable OneDriveCommercial est
    ' generique (independante du nom de l'organisation) : elle pointe la
    ' racine OneDrive Entreprise quel que soit le tenant. Puis OneDrive
    ' perso, puis Bureau/Desktop local.
    Dim candidats As Variant
    candidats = Array( _
        Environ("OneDriveCommercial") & "\Bureau", _
        Environ("OneDriveCommercial") & "\Desktop", _
        Environ("OneDrive") & "\Bureau", _
        Environ("OneDrive") & "\Desktop", _
        Environ("USERPROFILE") & "\OneDrive\Bureau", _
        Environ("USERPROFILE") & "\OneDrive\Desktop", _
        Environ("USERPROFILE") & "\Bureau", _
        Environ("USERPROFILE") & "\Desktop")

    Dim fso2 As Object
    Set fso2 = CreateObject("Scripting.FileSystemObject")
    Dim candidat As Variant
    For Each candidat In candidats
        If Len(candidat) > 0 Then
            If fso2.FolderExists(CStr(candidat)) Then
                ResolveDesktopPath = CStr(candidat)
                On Error GoTo 0
                Exit Function
            End If
        End If
    Next candidat

    ' Dernier recours
    ResolveDesktopPath = Environ("USERPROFILE")
    On Error GoTo 0

'=============================================================================
' IsOpportunity
' True si la chaine correspond a un code opportunite, pour l'un des types
' configures (g_OpportunityTypes, ex. "OP"). Reproduit le test
' x Like "20##_OP_####" mais pilote par config (plusieurs types via |).
' (## = annee 2000-2099 ; #### = numero de g_OpportunityNumberLength chiffres.)
'=============================================================================
Public Function IsOpportunity(ByVal s As String) As Boolean
    On Error Resume Next
    EnsureLoaded
    Dim types() As String, i As Long, pat As String
    types = Split(g_OpportunityTypes, "|")
    For i = LBound(types) To UBound(types)
        If Len(Trim$(types(i))) > 0 Then
            pat = "20##_" & Trim$(types(i)) & "_" & _
                  String(g_OpportunityNumberLength, "#")
            If s Like pat Then
                IsOpportunity = True
                Exit Function
            End If
        End If
    Next i
End Function

' Renvoie le 1er code opportunite trouve dans txt ("20YY_<type>_<N chiffres>"),
' sinon "". Remplace les fenetres glissantes de 12 caracteres codees en dur
' dans modProjectTag (ScanOpportunity / IsValidProjectTag) : la largeur est
' DERIVEE de la config (types et longueur du numero), type par type.
'=============================================================================
Public Function ScanOpportunityCode(ByVal txt As String) As String
    On Error Resume Next
    EnsureLoaded
    ScanOpportunityCode = ""
    Dim types() As String, i As Long, pat As String, win As Long, p As Long
    types = Split(g_OpportunityTypes, "|")

'=============================================================================
' EscapeRegex (interne)
' Echappe les caracteres speciaux regex dans une chaine litterale, pour
' qu'un prefixe metier (g_ProjectPrefix, g_ContractYearPrefix) puisse etre
' insere dans un pattern sans casser la regex. Cas usuels (DOSS, 202) : aucun
' caractere special, renvoi inchange.
'=============================================================================
Private Function EscapeRegex(ByVal s As String) As String
    Dim i As Long, ch As String, out As String
    Const SPECIAUX As String = "\.^$*+?()[]{}|"
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        If InStr(SPECIAUX, ch) > 0 Then out = out & "\"
        out = out & ch
    Next i
    EscapeRegex = out
End Function

'=============================================================================
' A (helper accents)
' Source de verite unique pour les chaines UI accentuees. Decode des
' marqueurs ASCII inline "~X" en caracteres accentues, ce qui permet de
' garder le source 100% ASCII (immunise au cycle export .txt / import VBE,
' ou les accents litteraux se corrompent) tout en affichant correctement.
'
' A utiliser pour TOUTE chaine visible (MsgBox / InputBox / titres /
' StatusBar). NE PAS utiliser pour les chaines fonctionnelles (categories
' Outlook comparees, DASL, cles config, chemins, noms de fichiers) ni pour
' les logs (DInfo/DError restent ASCII).
'
' Table des marqueurs (reconstruite a partir des usages du projet) :
'   ~e -> e accent aigu        (ChrW 233)
'   ~f -> e accent grave       (ChrW 232)
'   ~a -> a accent grave       (ChrW 224)
'   ~E -> E majuscule aigu     (ChrW 201)
'   ~i -> i accent circonflexe (ChrW 238)
'   ~o -> o accent circonflexe (ChrW 244)
'   ~x -> e accent circonflexe (ChrW 234)   (pack v3)
'   ~c -> c cedille             (ChrW 231)   (pack v3)
'   ~u -> u accent circonflexe (ChrW 251)   (pack v3)
'   ~U -> u accent grave       (ChrW 249)   (pack v3)
'   ~A -> a accent circonflexe (ChrW 226)   (pack v3)
'   ~y -> i trema              (ChrW 239)   (pack v3)
'   ~I -> I accent circonflexe (ChrW 206)   (pack v3 : "BO~ITES" du projet analytics)
' Un "~" non suivi d'une lettre connue est laisse tel quel.
' PACK v3 : avant, "fen~etre" / "m~eme" / "~etre" etaient ecrits avec ~e
' (e aigu) faute de marqueur e-circonflexe : 25 libelles affichaient
' "fenetre" avec un accent aigu. Marqueur ~x ajoute, occurrences corrigees.
'=============================================================================
Public Function LangIsEN() As Boolean
    EnsureLoaded
    LangIsEN = (g_Language = "EN")
End Function

Public Function A(ByVal s As String) As String
    Dim out As String, i As Long, n As Long, ch As String, nx As String
    n = Len(s)
    i = 1
    Do While i <= n
        ch = Mid$(s, i, 1)
        If ch = "~" And i < n Then
            nx = Mid$(s, i + 1, 1)
            Select Case nx
                Case "e": out = out & ChrW$(233): i = i + 2
                Case "f": out = out & ChrW$(232): i = i + 2
                Case "a": out = out & ChrW$(224): i = i + 2
                Case "E": out = out & ChrW$(201): i = i + 2
                Case "i": out = out & ChrW$(238): i = i + 2
                Case "o": out = out & ChrW$(244): i = i + 2
                Case "x": out = out & ChrW$(234): i = i + 2
                Case "c": out = out & ChrW$(231): i = i + 2
                Case "u": out = out & ChrW$(251): i = i + 2
                Case "U": out = out & ChrW$(249): i = i + 2
                Case "A": out = out & ChrW$(226): i = i + 2
                Case "y": out = out & ChrW$(239): i = i + 2
                Case "I": out = out & ChrW$(206): i = i + 2
                Case Else
                    ' Marqueur inconnu : on garde le ~ tel quel.
                    out = out & ch
                    i = i + 1
            End Select
        Else
            out = out & ch
            i = i + 1
        End If
    Loop
    A = out
End Function

'=============================================================================
' T (helper langue UI)
' Choisit la chaine a afficher selon g_Language ("FR" par defaut, ou "EN").
'   T(fr, en) -> en si g_Language="EN" ET en non vide, sinon fr.
' Le repli FR (en="") garantit que le commutateur ne casse jamais l'UI, meme
' a moitie traduite. Les deux chaines passent par A() : le FR decode ses
' marqueurs d'accents (~e, ~f...), l'EN (nativement ASCII) est inchange.
'
' A utiliser pour TOUTE chaine visible (MsgBox / InputBox / titres /
' StatusBar), exactement comme A(). NE PAS utiliser pour les chaines
' fonctionnelles (categories Outlook comparees, DASL, cles config, chemins,
' noms de fichiers) ni pour les logs (DInfo/DError restent ASCII, FR).
'
' Exemple :
'   MsgBox T("Dossier d~ej~a rang~e.", "Already filed."), vbInformation
'=============================================================================
Public Function T(ByVal fr As String, ByVal en As String) As String
    EnsureLoaded
    If g_Language = "EN" And Len(en) > 0 Then
        T = A(en)
    Else
        T = A(fr)
    End If
End Function

' ContainsProjectCode
' True si le texte contient <prefixe><N chiffres> (ex. "DOSS1234").
' Pilote par config (prefixe + longueur). Insensible a la casse.
'=============================================================================
Public Function ContainsProjectCode(ByVal s As String) As Boolean
    On Error Resume Next
    EnsureLoaded
    ' Pack v3 : aligne sur ScanProjectCode (EXACTEMENT N chiffres). L'ancien
    ' Like "*PFX####*" acceptait aussi PFX12345 (5 chiffres).
    ContainsProjectCode = (Len(ScanProjectCode(s)) = g_ProjectCodeLength)
End Function

'=============================================================================
' ScanProjectCode
' Cherche <prefixe> suivi d'EXACTEMENT g_ProjectCodeLength chiffres et
' renvoie ces chiffres (sinon ""). Pilote par config (prefixe + longueur).
'=============================================================================
Public Function ScanProjectCode(ByVal txt As String) As String
    On Error Resume Next
    EnsureLoaded
    ScanProjectCode = ""
    Dim pfx As String: pfx = UCase$(g_ProjectPrefix)
    Dim lp As Long: lp = Len(pfx)
    Dim up As String: up = UCase$(txt)
    Dim pos As Long
    pos = InStr(1, up, pfx)
    Do While pos > 0
        Dim j As Long: j = pos + lp
        Dim num As String: num = ""
        Do While j <= Len(txt)
            Dim ch As String: ch = Mid$(txt, j, 1)
            If ch >= "0" And ch <= "9" Then
                num = num & ch
                j = j + 1
            Else
                Exit Do
            End If
        Loop
        If Len(num) = g_ProjectCodeLength Then
            ScanProjectCode = num
            Exit Function
        End If
        pos = InStr(pos + 1, up, pfx)
    Loop
End Function

'=============================================================================
' FindContractCode
' Renvoie le 1er code contrat trouve (fenetre glissante "20##_<N chiffres>"),
' sinon "". Pilote par config (prefixe annee + longueur du numero).
' (Annee 20## = 2000-2099 ; un code opportunite "20XX_OP_..." ne matche pas,
'  les lettres O/T ne pouvant tenir une position chiffre #.)
'=============================================================================
Public Function FindContractCode(ByVal txt As String) As String
    On Error Resume Next
    EnsureLoaded
    FindContractCode = ""
    Dim win As Long
    win = 4 + 1 + g_ContractNumberLength   ' "20" + 2 chiffres + "_" + N chiffres
    ' Pack v3 : le separateur accepte _ OU - comme le regex g_ContractPattern
    ' (avant : "_" seul ici, "[_\-]" dans le regex -> detections divergentes).
    Dim pat As String
    pat = "20##[_-]" & String(g_ContractNumberLength, "#")
    Dim i As Long
    For i = 1 To Len(txt) - (win - 1)
        Dim x As String: x = Mid$(txt, i, win)
        If x Like pat Then
            FindContractCode = x
            Exit Function
        End If
    Next i
End Function

'=============================================================================
' ContainsContractCode : True si FindContractCode renvoie un resultat.
'=============================================================================
Public Function ContainsContractCode(ByVal s As String) As Boolean
    ContainsContractCode = (Len(FindContractCode(s)) > 0)
End Function
