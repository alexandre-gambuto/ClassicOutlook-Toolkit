' v12 : ouvre le classeur des campagnes (Suivi des Campagnes.xlsm, chemin
' config.ini) dans l'Excel de l'utilisateur. A epingler au ruban / a la
' barre d'acces rapide d'Outlook.
Public Sub CampaignOpenWorkbook()
    On Error GoTo ErrH
    Dim wbPath As String
    wbPath = modConfig.CampaignWorkbookPath()
    If Len(Dir$(wbPath)) = 0 Then
        modUi.Inform T("Classeur des campagnes introuvable :", "Campaigns workbook not found:") & vbCrLf & _
                     wbPath & vbCrLf & vbCrLf & _
                     T("Copie le classeur des campagnes ~a cet emplacement (ou renseigne CampaignWorkbookPath dans config.ini).", _
                       "Copy the campaigns workbook there (or set CampaignWorkbookPath in config.ini)."), _
                     T("Campagnes", "Campaigns"), vbExclamation
        Exit Sub
    End If

    Dim xlApp As Object, wb As Object, wasOpen As Boolean, createdByUs As Boolean
    If Not modExcelBridge.OpenWorkbook(wbPath, False, xlApp, wb, wasOpen, createdByUs) Then
        modUi.Fail T("Ouverture du classeur impossible (voir debug.log).", "Cannot open the workbook (see debug.log)."), T("Campagnes", "Campaigns")
        Exit Sub
    End If
    On Error Resume Next
    xlApp.Visible = True
    wb.Windows(1).Visible = True
    wb.Activate
    AppActivate xlApp.Caption
    On Error GoTo ErrH
    modDebugLog.DInfo MOD_NAME, "Classeur Campagnes ouvert : " & wbPath
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "CampaignOpenWorkbook " & Err.Number & " - " & Err.Description
    modUi.Fail T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, T("Campagnes", "Campaigns")
End Sub


' v12.1 (F5-15) : cible du bouton "Aide" de l'onglet Emaileur (Projet.CampaignHelp,
' macro absente jusqu'ici) : ouvre le classeur des campagnes sur sa feuille
' "Aide" (le mode d'emploi complet).
Public Sub CampaignHelp()
    On Error Resume Next
    CampaignOpenWorkbook
    Dim xlApp As Object, wb As Object, ws As Object
    Set xlApp = RunningExcel()
    If xlApp Is Nothing Then Exit Sub
    Set wb = modExcelBridge.FindOpenWorkbookByRef(xlApp, modConfig.CampaignWorkbookPath())
    If wb Is Nothing Then Exit Sub
    Set ws = wb.Worksheets("Aide")
    If ws Is Nothing Then
        modDebugLog.DInfo MOD_NAME, "CampaignHelp : feuille Aide absente du classeur"
        Exit Sub
    End If
    wb.Activate
    ws.Activate
End Sub


' Ancien format : feuille de 9 en-tetes pre-formates (ligne 1) dans un
' nouveau classeur ou un classeur ouvert. Conserve pour les usages hors
' du classeur des campagnes ; le classeur dedie cree ses feuilles lui-meme.
Public Sub CampaignCreateSheet()
    On Error GoTo ErrH
    Dim xlApp As Object
    Set xlApp = RunningExcel()

    Dim wb As Object, ws As Object, isNew As Boolean
    If xlApp Is Nothing Then
        Set xlApp = CreateObject("Excel.Application")
        xlApp.Visible = True
        isNew = True
    Else
        xlApp.Visible = True
        Dim choice As Long
        choice = ChooseTargetWorkbook(xlApp, wb)   ' 1=nouveau ; 2=wb defini ; 0=annule
        If choice = 0 Then Exit Sub
        isNew = (choice = 1)
    End If

    If isNew Then
        Set wb = xlApp.Workbooks.Add
        Set ws = wb.Worksheets(1)
    Else
        Set ws = wb.Worksheets.Add
    End If

    On Error Resume Next
    ws.Name = UniqueSheetName(wb, "Campagne email")
    On Error GoTo ErrH

    BuildLegacySheet xlApp, ws

    modDebugLog.DInfo MOD_NAME, "Feuille (ancien format) creee : " & ws.Name
    modUi.Inform T("Feuille d'emailing cr~e~ee (ancien format, en-t~xtes en ligne 1).", "Mailing sheet created (legacy format, headers on row 1).") & vbCrLf & _
                 T("Remplis les lignes, garde le classeur ouvert et actif, puis lance la g~en~eration.", "Fill the rows, keep the workbook open and active, then run the generation."), _
                 T("Campagnes", "Campaigns")
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "CampaignCreateSheet " & Err.Number & " - " & Err.Description
    modUi.Fail T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, T("Campagnes", "Campaigns")
End Sub


' Tester la LIGNE COURANTE de la feuille active : 1 brouillon ouvert.
Public Sub CampaignGenerateRow()
    On Error GoTo ErrH
    Dim ctx As CampaignCtx
    If Not LoadActiveContext(ctx) Then Exit Sub

    Dim rw As Long
    On Error Resume Next
    rw = ctx.XlApp.ActiveCell.Row
    On Error GoTo ErrH
    If rw <= ctx.HeaderRow Then
        modUi.Inform T("Dans Excel, place le curseur sur une ligne de donn~ees de la table, puis relance.", _
                       "In Excel, put the cursor on a data row of the table, then run again."), _
                     T("Campagnes", "Campaigns"), vbExclamation
        Exit Sub
    End If
    TestRow ctx, rw
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "CampaignGenerateRow " & Err.Number & " - " & Err.Description
    modUi.Fail T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, T("Campagnes", "Campaigns")
End Sub


' Tous les brouillons de la feuille active (lignes sans statut).
Public Sub CampaignGenerateBatch()
    On Error GoTo ErrH
    Dim ctx As CampaignCtx
    If Not LoadActiveContext(ctx) Then Exit Sub
    If SendBusyOn(ctx) Then
        modUi.Inform T("Un envoi est en cours sur cette feuille : attends la fin ou utilise Arr~xter.", _
                       "A send is running on this sheet: wait for it to finish or use Stop."), _
                     T("Campagnes", "Campaigns"), vbExclamation
        Exit Sub
    End If

    If modUi.Ask(T("Feuille : '", "Sheet: '") & ctx.SheetName & "'" & vbCrLf & _
                 T("Classeur : ", "Workbook: ") & ctx.WbName & vbCrLf & _
                 T("Compte d'envoi : ", "Sending account: ") & ctx.AccountLabel & vbCrLf & vbCrLf & _
                 T("G~en~erer les brouillons des lignes sans statut ?", "Generate the drafts of the rows without status?"), _

' Efface les statuts (et le controle) des lignes NON envoyees de la feuille
' active, apres confirmation. v12.1 (F5-12) : meme regle que le bouton du
' classeur (Oui = lignes non envoyees) ; une ligne "Envoye..." garde son
' statut, sinon le prochain Envoyer lui renverrait le mail.
Public Sub CampaignResetStatuses()
    On Error GoTo ErrH
    Dim ctx As CampaignCtx
    If Not LoadActiveContext(ctx) Then Exit Sub
    If SendBusyOn(ctx) Then
        modUi.Inform T("Un envoi est en cours sur cette feuille : attends la fin ou utilise Arr~xter.", _
                       "A send is running on this sheet: wait for it to finish or use Stop."), _
                     T("Campagnes", "Campaigns"), vbExclamation
        Exit Sub
    End If

    Dim lastRow As Long
    lastRow = LastDataRow(ctx)
    If lastRow <= ctx.HeaderRow Then
        modUi.Inform T("Aucun statut ~a effacer.", "No status to clear."), T("Campagnes", "Campaigns")
        Exit Sub
    End If

    Dim stB As Variant, i As Long, st As String, nReset As Long, nSent As Long
    stB = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_STATUS), lastRow, ctx.Map(R_STATUS))
    For i = 1 To UBound(stB, 1)
        st = SafeText(stB(i, 1))
        If Len(st) > 0 Then
            If Left$(st, 6) = "Envoye" Then
                nSent = nSent + 1
            Else
                nReset = nReset + 1
            End If
        End If
    Next i
    If nReset = 0 Then
        modUi.Inform T("Aucune ligne non envoy~ee ~a r~einitialiser (brouillon ou ~echec).", "No unsent row to reset (draft or failure).") & _
                     IIf(nSent > 0, vbCrLf & nSent & T(" ligne(s) d~ej~a envoy~ee(s) : statut conserv~e.", " row(s) already sent: status kept."), ""), _
                     T("Campagnes", "Campaigns")
        Exit Sub
    End If

    If modUi.Ask(T("Effacer le Statut (et le Contr~ole) des ", "Clear the Status (and Check) of the ") & nReset & _
                 T(" ligne(s) NON envoy~ee(s) (brouillons, ~echecs) ?", " row(s) NOT sent (drafts, failures)?") & vbCrLf & _
                 T("Elles redeviennent envoyables. Les lignes d~ej~a envoy~ees (", "They become sendable again. Rows already sent (") & nSent & _
                 T(") ne sont pas touch~ees.", ") are not touched."), _
                 T("Campagnes - r~einitialisation", "Campaigns - reset"), vbExclamation + vbYesNo, False) <> vbYes Then Exit Sub

    Dim rw As Long, n As Long
    For i = 1 To UBound(stB, 1)
        st = SafeText(stB(i, 1))
        If Len(st) > 0 And Left$(st, 6) <> "Envoye" Then
            rw = ctx.HeaderRow + i
            ctx.Ws.Cells(rw, ctx.Map(R_STATUS)).ClearContents
            If HasRole(ctx, R_CHECK) Then ctx.Ws.Cells(rw, ctx.Map(R_CHECK)).ClearContents
            n = n + 1
        End If
    Next i
    modDebugLog.DInfo MOD_NAME, "Statuts effaces (" & ctx.SheetName & ") : " & n & " ligne(s) non envoyee(s), " & nSent & " envoyee(s) conservee(s)"
    modUi.Toast n & T(" statut(s) effac~e(s) (lignes envoy~ees conserv~ees).", " status(es) cleared (sent rows kept)."), uiSuccess
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "CampaignResetStatuses " & Err.Number & " - " & Err.Description
    modUi.Fail T("Erreur : ", "Error: ") & Err.Number & " - " & Err.Description, T("Campagnes", "Campaigns")
End Sub

'=============================================================================
' COMMANDES DISTANTES (classeur des campagnes -> canal de commandes ->
' ThisOutlookSession -> CampaignRemote -> modDefer "CampaignCmd" ->
' CampaignRunPending). Aucune fenetre bloquante : resultats dans la feuille,
' toasts, puis rappel du classeur.
'=============================================================================

' Appele par ThisOutlookSession.HandleToolkitCommand (contexte evenement :
' on ne fait que memoriser et armer le differe).
'   cmd     : "CAMPAIGN_CHECK" ... (cf. RunRemoteCommand)
'   argText : "classeur|feuille|extra1|extra2" (classeur = nom ou chemin)
Public Sub CampaignRemote(ByVal cmd As String, ByVal argText As String)
    On Error Resume Next
    If gPendingCmds Is Nothing Then Set gPendingCmds = New Collection
    gPendingCmds.Add cmd & vbTab & argText
    modDefer.ArmUnique DEFER_CMD_KEY, 1
    modDebugLog.DInfo MOD_NAME, "Commande recue du classeur : " & cmd
End Sub


' Cible modDefer "CampaignCmd" : execute les commandes en attente.
Public Sub CampaignRunPending()
    On Error Resume Next
    If gPendingCmds Is Nothing Then Exit Sub
    Dim packed As String, p As Long
    Do While gPendingCmds.Count > 0
        packed = CStr(gPendingCmds.Item(1))
        gPendingCmds.Remove 1
        p = InStr(packed, vbTab)
        If p > 0 Then RunRemoteCommand Left$(packed, p - 1), Mid$(packed, p + 1)
    Loop
End Sub


Private Sub RunRemoteCommand(ByVal cmd As String, ByVal argText As String)
    On Error GoTo ErrH
    Dim args() As String
    args = Split(argText, "|")
    If UBound(args) < 1 Then
        modDebugLog.DError MOD_NAME, "Commande " & cmd & " : arguments incomplets (" & argText & ")"
        Exit Sub
    End If
    Dim wbRef As String, sheet As String, extra1 As String, extra2 As String
    wbRef = Trim$(args(0))
    sheet = Trim$(args(1))
    If UBound(args) >= 2 Then extra1 = Trim$(args(2))
    If UBound(args) >= 3 Then extra2 = Trim$(args(3))

    Dim cmdU As String
    cmdU = UCase$(cmd)
    If cmdU = "CAMPAIGN_STOP" Then
        CampaignStop
        Exit Sub
    End If

    Dim ctx As CampaignCtx
    Dim wasOpen As Boolean, createdByUs As Boolean
    If Not AcquireContext(wbRef, sheet, ctx, wasOpen, createdByUs) Then
        modUi.Toast T("Campagne : classeur ou feuille introuvable", "Campaign: workbook or sheet not found") & vbLf & sheet, uiError, True
        Exit Sub
    End If

    ' v12.1 (F5-01) : un envoi cadence en cours sur CETTE feuille relit ses
    ' lignes une a une ; ni brouillons, ni second envoi, ni relances tant
    ' qu'il tourne (doublons, lignes deplacees).
    If cmdU = "CAMPAIGN_DRAFTS" Or cmdU = "CAMPAIGN_SEND" Or cmdU = "CAMPAIGN_FOLLOWUP" Then
        If SendBusyOn(ctx) Then
            modDebugLog.DInfo MOD_NAME, "Commande " & cmd & " refusee : envoi en cours sur la feuille"
            modUi.Toast T("Campagne : un envoi est en cours sur cette feuille. Commande ignor~ee (Arr~xter d'abord).", _
                          "Campaign: a send is running on this sheet. Command ignored (Stop first)."), uiWarn, True
            ReleaseContext ctx, wasOpen, createdByUs, False
            Exit Sub
        End If
    End If

    Dim keepOpen As Boolean
    keepOpen = False
    Select Case cmdU
        Case "CAMPAIGN_CHECK"
            RunCheck ctx, False
        Case "CAMPAIGN_TEST"
            TestRow ctx, CLng(Val(extra1))
        Case "CAMPAIGN_DRAFTS"
            GenerateDrafts ctx, False
        Case "CAMPAIGN_SEND"
            keepOpen = StartSend(ctx, KIND_MAIN, "", wasOpen, createdByUs)
        Case "CAMPAIGN_REFRESH"
            RefreshReplies ctx, False
        Case "CAMPAIGN_FOLLOWUP"
            ' extra1 = DRAFTS | SEND ; extra2 = "12;15;20" (lignes cochees) ou vide
            If UCase$(extra1) = "SEND" Then
                keepOpen = StartSend(ctx, KIND_FU, extra2, wasOpen, createdByUs)
            Else
                FollowUp ctx, "DRAFTS", extra2, False
            End If
        Case "CAMPAIGN_CLEANDRAFTS"
            Dim removed As Long
            removed = PurgeCampaignDrafts(ctx.WbName, ctx.SheetName)
            modUi.Toast removed & T(" brouillon(s) de cette campagne supprim~e(s).", " draft(s) of this campaign deleted."), uiSuccess
        Case Else
            modDebugLog.DError MOD_NAME, "Commande inconnue : " & cmd
    End Select

    If Not keepOpen Then ReleaseContext ctx, wasOpen, createdByUs, True
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "RunRemoteCommand " & cmd & " : " & Err.Number & " - " & Err.Description
    On Error Resume Next
    modUi.Toast T("Campagne : erreur ", "Campaign: error ") & Err.Number & " (" & cmd & ")", uiError, True
End Sub

' Instance Excel en cours (validee, fantome ecarte) ou Nothing.
Private Function RunningExcel() As Object
    On Error Resume Next
    Set RunningExcel = Nothing
    Set RunningExcel = GetObject(, "Excel.Application")
    If Not RunningExcel Is Nothing Then
        If Not modCommon.ExcelAlive(RunningExcel) Then
            Set RunningExcel = Nothing
            modDebugLog.DInfo MOD_NAME, "RunningExcel : instance fantome ignoree"
        End If
    End If
End Function


' Contexte de la FEUILLE ACTIVE de l'Excel ouvert (ancien menu). Sans Excel
' ou sans classeur : selecteur de fichier. quiet=True : aucun message.
Private Function LoadActiveContext(ByRef ctx As CampaignCtx, Optional ByVal quiet As Boolean = False) As Boolean
    On Error GoTo ErrH
    LoadActiveContext = False
    Dim xlApp As Object, created As Boolean
    Set xlApp = RunningExcel()
    If xlApp Is Nothing Then
        If quiet Then Exit Function
        Set xlApp = CreateObject("Excel.Application")
        created = True
    End If
    xlApp.Visible = True

    If xlApp.Workbooks.Count = 0 Then
        If quiet Then Exit Function
        Dim f As Variant
        f = xlApp.GetOpenFilename("Classeurs Excel (*.xls; *.xlsx; *.xlsm),*.xls;*.xlsx;*.xlsm", , _
                                  T("Choisir le classeur de campagne", "Choose the campaign workbook"))
        If VarType(f) = vbBoolean Then
            If created Then xlApp.Quit
            Exit Function
        End If
        xlApp.Workbooks.Open CStr(f)
        xlApp.Visible = True
    End If

    Dim ws As Object
    Set ws = xlApp.ActiveSheet
    If ws Is Nothing Then Exit Function

    ' Feuilles anciennes : boite deduite du dossier Outlook courant (v11).
    ResolveLegacyTarget

    If Not LoadContext(ctx, xlApp, ws.Parent, ws, Not quiet) Then
        If Not quiet Then
            modUi.Inform T("La feuille active n'est pas une feuille de campagne : aucune colonne destinataire / email reconnue.", _
                           "The active sheet is not a campaign sheet: no recipient / email column recognized."), _
                         T("Campagnes", "Campaigns"), vbExclamation
        End If
        Exit Function
    End If
    LoadActiveContext = True
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "LoadActiveContext " & Err.Number & " - " & Err.Description
    LoadActiveContext = False
End Function


' Contexte d'une feuille designee par le classeur (commande distante).
' wbRef : NOM du classeur (prefere : un chemin OneDrive peut etre une URL)
' ou chemin complet. Classeur ouvert chez l'utilisateur -> wasOpen=True ;
' sinon ouverture en silence depuis le chemin config.ini.
Private Function AcquireContext(ByVal wbRef As String, ByVal sheet As String, ByRef ctx As CampaignCtx, _
                                ByRef wasOpen As Boolean, ByRef createdByUs As Boolean) As Boolean
    On Error GoTo ErrH
    AcquireContext = False
    wasOpen = False
    createdByUs = False

    Dim xlApp As Object, wb As Object
    Set xlApp = RunningExcel()
    ' v12.1 : regle commune du pont (chemin complet, sinon nom de fichier :
    ' un chemin OneDrive peut se presenter en https).
    If Not xlApp Is Nothing Then Set wb = modExcelBridge.FindOpenWorkbookByRef(xlApp, wbRef)
    If Not wb Is Nothing Then
        wasOpen = True
    Else
        Dim wbPath As String
        If InStr(wbRef, "\") > 0 And LCase$(Left$(wbRef, 4)) <> "http" Then
            wbPath = wbRef
        Else
            wbPath = modConfig.CampaignWorkbookPath()
        End If
        Set xlApp = Nothing
        If Not modExcelBridge.OpenWorkbook(wbPath, False, xlApp, wb, wasOpen, createdByUs) Then Exit Function
    End If

    Dim ws As Object
    On Error Resume Next
    Set ws = wb.Worksheets(sheet)
    On Error GoTo ErrH
    If ws Is Nothing Then
        modDebugLog.DError MOD_NAME, "Feuille introuvable : " & sheet
        If Not wasOpen Then modExcelBridge.CloseWorkbook xlApp, wb, wasOpen, createdByUs, False
        Exit Function
    End If

    If Not LoadContext(ctx, xlApp, wb, ws, False) Then
        modDebugLog.DError MOD_NAME, "Feuille sans colonne destinataire : " & sheet
        If Not wasOpen Then modExcelBridge.CloseWorkbook xlApp, wb, wasOpen, createdByUs, False
        Exit Function
    End If
    AcquireContext = True
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "AcquireContext " & Err.Number & " - " & Err.Description
    AcquireContext = False
End Function


' v12.1 (F5-01) : un envoi cadence vise-t-il deja cette feuille ? Aucune
' commande qui cree des brouillons ou reecrit les statuts ne doit alors
' tourner sur la meme feuille.
Private Function SendBusyOn(ByRef ctx As CampaignCtx) As Boolean
    On Error Resume Next
    SendBusyOn = False
    If Not gSendActive Then Exit Function
    If StrComp(gSendCtx.SheetName, ctx.SheetName, vbTextCompare) <> 0 Then Exit Function
    SendBusyOn = (StrComp(gSendCtx.WbName, ctx.WbName, vbTextCompare) = 0)
End Function

' Charge le contexte : ligne d'en-tete, roles, champs, parametres, compte.
' interactive=True : colonnes obligatoires manquantes demandees (ancien menu).
Private Function LoadContext(ByRef ctx As CampaignCtx, ByVal xlApp As Object, ByVal wb As Object, _
                             ByVal ws As Object, ByVal interactive As Boolean) As Boolean
    On Error GoTo ErrH
    LoadContext = False
    ctx.Loaded = False
    Set ctx.XlApp = xlApp
    Set ctx.Wb = wb
    Set ctx.Ws = ws
    ctx.WbPath = CStr(wb.FullName)
    ctx.WbName = CStr(wb.Name)
    ctx.SheetName = CStr(ws.Name)
    ctx.AccountError = ""

    ctx.HeaderRow = FindHeaderRow(ws)
    If ctx.HeaderRow = 0 Then
        If Not interactive Then Exit Function
        ctx.HeaderRow = 1   ' ancien comportement : en-tetes supposes en ligne 1, colonnes demandees
    End If

    Set ctx.Map = ResolveColumns(ws, ctx.HeaderRow, interactive, ctx.Fields)
    If ctx.Map Is Nothing Then Exit Function

    Set ctx.Params = ReadParams(ws, ctx.HeaderRow)
    ctx.Legacy = (ctx.Params.Count = 0)

    EnsureColumns ctx
    ResolveAccount ctx
    ctx.Loaded = True
    LoadContext = True
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "LoadContext " & Err.Number & " - " & Err.Description
    LoadContext = False
End Function


' Premiere ligne (1..HEADER_SCAN_ROWS) portant une colonne destinataire /
' email. 0 si aucune. Lecture en UN bloc COM. Une ligne du bloc de
' parametres (libelle connu en A, valeur en B, rien d'autre) n'est jamais
' prise pour l'en-tete, meme si sa valeur commence par "Email...".
Private Function FindHeaderRow(ByVal ws As Object) As Long
    On Error Resume Next
    FindHeaderRow = 0
    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, 1, 1, HEADER_SCAN_ROWS, HEADER_SCAN_COLS)
    Dim r As Long, c As Long, filled As Long, isParamRow As Boolean
    For r = LBound(blk, 1) To UBound(blk, 1)
        filled = 0
        For c = LBound(blk, 2) To UBound(blk, 2)
            If Len(Trim$(CStr(blk(r, c)))) > 0 Then filled = filled + 1
        Next c
        isParamRow = (filled <= 2 And Len(ParamKeyFor(NormalizeHeader(CStr(blk(r, 1))))) > 0)
        If Not isParamRow Then
            For c = LBound(blk, 2) To UBound(blk, 2)
                If RoleFromHeader(NormalizeHeader(CStr(blk(r, c)))) = R_TO Then
                    FindHeaderRow = r
                    Exit Function
                End If
            Next c
        End If
    Next r
End Function


' Dictionnaire role -> colonne, et champs de fusion (en-tete normalise ->
' colonne, roles compris). Colonnes obligatoires absentes : demandees si
' interactive, sinon Nothing.
Private Function ResolveColumns(ByVal ws As Object, ByVal headerRow As Long, ByVal interactive As Boolean, _
                                ByRef fields As Object) As Object
    On Error GoTo ErrH
    Dim map As Object
    Set map = CreateObject("Scripting.Dictionary")
    Set fields = CreateObject("Scripting.Dictionary")

    Dim lastCol As Long, c As Long, label As String, nrm As String, role As String
    lastCol = ws.Cells(headerRow, ws.Columns.Count).End(XL_TOLEFT).Column
    If lastCol < 1 Then lastCol = 1

    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, headerRow, 1, headerRow, lastCol)
    For c = 1 To lastCol
        label = CStr(blk(1, c))
        If Len(Trim$(label)) > 0 Then
            nrm = NormalizeHeader(label)
            If Not fields.Exists(nrm) Then fields.Add nrm, c
            role = RoleFromHeader(nrm)
            If Len(role) > 0 Then
                If Not map.Exists(role) Then map.Add role, c
            End If
        End If
    Next c

    If Not map.Exists(R_TO) Then
        If interactive Then map(R_TO) = AskColumn(ws, "Destinataires (A) / Email")
        If Not map.Exists(R_TO) Then Set ResolveColumns = Nothing: Exit Function
        If map(R_TO) = 0 Then Set ResolveColumns = Nothing: Exit Function
    End If
    If Not map.Exists(R_BODY) Then
        If interactive Then map(R_BODY) = AskColumn(ws, "Message (HTML)")
        If map.Exists(R_BODY) Then
            If map(R_BODY) = 0 Then Set ResolveColumns = Nothing: Exit Function
        Else
            Set ResolveColumns = Nothing: Exit Function
        End If
    End If
    Set ResolveColumns = map
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "ResolveColumns " & Err.Number & " - " & Err.Description
    Set ResolveColumns = Nothing
End Function


' minuscules, sans accents, lettres et chiffres seulement.
Private Function NormalizeHeader(ByVal s As String) As String
    Dim x As String
    x = LCase$(Trim$(s))
    x = Replace(x, Chr$(233), "e"): x = Replace(x, Chr$(232), "e")
    x = Replace(x, Chr$(234), "e"): x = Replace(x, Chr$(235), "e")
    x = Replace(x, Chr$(224), "a"): x = Replace(x, Chr$(226), "a")
    x = Replace(x, Chr$(238), "i"): x = Replace(x, Chr$(239), "i")
    x = Replace(x, Chr$(244), "o"): x = Replace(x, Chr$(249), "u")
    x = Replace(x, Chr$(251), "u"): x = Replace(x, Chr$(231), "c")
    Dim i As Long, ch As String, out As String
    For i = 1 To Len(x)
        ch = Mid$(x, i, 1)
        If (ch >= "a" And ch <= "z") Or (ch >= "0" And ch <= "9") Then out = out & ch
    Next i
    NormalizeHeader = out
End Function


' Role d'un en-tete normalise. v12.1 (F5-07) : noms EXACTS seulement (plus
' aucun fragment : "Src Statut", "Statut client", "Objet du marketing" ou
' "Logiciel Microsoft" ne prennent plus la place d'une colonne de
' pilotage) ; anciens en-tetes connus listes (feuille 9 colonnes :
' "Destinataires (A)", "Copies (Cc)", "Caches (Cci)", "Message (HTML)"...).
' Seule la colonne destinataire garde des prefixes (email..., courriel...,
' adresse...). COPIE IDENTIQUE dans modCampaignBook (classeur).
Private Function RoleFromHeader(ByVal n As String) As String
    RoleFromHeader = ""
    If Len(n) = 0 Then Exit Function
    Select Case n
        Case "convid": RoleFromHeader = R_CONV
        Case "entryid": RoleFromHeader = R_ENTRY
        Case "senton": RoleFromHeader = R_SENTON
        Case "replyid": RoleFromHeader = R_REPLYID
        Case "controle", "check": RoleFromHeader = R_CHECK
        Case "statut", "status": RoleFromHeader = R_STATUS
        Case "suivi", "tracking": RoleFromHeader = R_TRACK
        Case "reponses", "replies": RoleFromHeader = R_REPLIES
        Case "dernierereponse", "lastreply": RoleFromHeader = R_LASTREPLY
        Case "relances", "followups": RoleFromHeader = R_FOLLOWUPS
        Case "derniererelance", "lastfollowup": RoleFromHeader = R_LASTFU
        Case "relancehtml", "followuphtml": RoleFromHeader = R_FUBODY
        Case "messagehtml", "html", "corpshtml", "bodyhtml", "htmlbody": RoleFromHeader = R_BODY
        Case "cc", "copie", "copies", "copiescc", "encopie": RoleFromHeader = R_CC
        Case "bcc", "cci", "cachescci", "cachecci", "copiecachee", "copiescachees": RoleFromHeader = R_BCC
        Case "objet", "subject": RoleFromHeader = R_SUBJECT
        Case "piecesjointes", "piecejointe", "attachments", "attachment": RoleFromHeader = R_ATTACH
        Case "modele", "modeleoft", "modeleoftmsg", "template": RoleFromHeader = R_TEMPLATE
        Case "destinataire", "destinataires", "destinatairea", "destinatairesa", "mail", "to"
            RoleFromHeader = R_TO
        Case Else
            If Left$(n, 5) = "email" Or Left$(n, 8) = "courriel" Or Left$(n, 7) = "adresse" Then RoleFromHeader = R_TO
    End Select
End Function


Private Function AskColumn(ByVal ws As Object, ByVal fieldName As String) As Long
    Dim s As String
    s = Trim$(modUi.AskText(T("Colonne introuvable pour : ", "Column not found for: ") & fieldName & vbCrLf & _
                            T("Saisir la lettre de colonne (ex : A) :", "Enter the column letter (e.g. A):"), _
                            T("Campagnes - colonne manquante", "Campaigns - missing column")))
    If Len(s) = 0 Then AskColumn = 0: Exit Function
    On Error Resume Next
    AskColumn = ws.Range(s & "1").Column
    On Error GoTo 0
End Function


' Bloc de parametres : lignes 1..headerRow-1, libelle en A, valeur en B.
' Dictionnaire vide = feuille ancienne.
Private Function ReadParams(ByVal ws As Object, ByVal headerRow As Long) As Object
    Dim d As Object
    Set d = CreateObject("Scripting.Dictionary")
    Set ReadParams = d
    On Error Resume Next
    If headerRow <= 1 Then Exit Function
    Dim blk As Variant
    blk = modExcelBridge.ReadBlock(ws, 1, 1, headerRow - 1, 2)
    Dim r As Long, key As String, nrm As String
    For r = LBound(blk, 1) To UBound(blk, 1)
        nrm = NormalizeHeader(CStr(blk(r, 1)))
        key = ParamKeyFor(nrm)
        If Len(key) > 0 Then
            If Not d.Exists(key) Then d.Add key, Trim$(CStr(blk(r, 2)))
        End If
    Next r
End Function


' Cle P_* d'un libelle normalise (ordre = priorite : "messagederelance"
' avant "message"). "" si le libelle n'est pas un parametre connu.
Private Function ParamKeyFor(ByVal nrm As String) As String
    ParamKeyFor = ""
    If Len(nrm) = 0 Then Exit Function
    If Left$(nrm, Len(P_FOLLOWUP)) = P_FOLLOWUP Or Left$(nrm, 7) = "relance" Then
        ParamKeyFor = P_FOLLOWUP
    ElseIf Left$(nrm, Len(P_MESSAGE)) = P_MESSAGE Then
        ParamKeyFor = P_MESSAGE
    ElseIf Left$(nrm, Len(P_ACCOUNT)) = P_ACCOUNT Or Left$(nrm, 7) = "account" Then
        ParamKeyFor = P_ACCOUNT
    ElseIf Left$(nrm, Len(P_TEMPLATE)) = P_TEMPLATE Or Left$(nrm, 8) = "template" Then
        ParamKeyFor = P_TEMPLATE
    ElseIf Left$(nrm, Len(P_SUBJECT)) = P_SUBJECT Or Left$(nrm, 7) = "subject" Then
        ParamKeyFor = P_SUBJECT
    ElseIf Left$(nrm, Len(P_ATTACH)) = P_ATTACH Or Left$(nrm, 6) = "attach" Then
        ParamKeyFor = P_ATTACH
    ElseIf Left$(nrm, Len(P_FOLDER)) = P_FOLDER Or Left$(nrm, 6) = "folder" Then
        ParamKeyFor = P_FOLDER
    ElseIf Left$(nrm, Len(P_CREATED)) = P_CREATED Or Left$(nrm, 7) = "created" Then
        ParamKeyFor = P_CREATED
    ElseIf Left$(nrm, Len(P_NAME)) = P_NAME Or Left$(nrm, 8) = "campaign" Then
        ParamKeyFor = P_NAME
    End If
End Function

' v12.1 (F5-02) : ecrit un texte puis le RELIT ; False si Excel a refuse
' l'ecriture ou si la valeur relue differe (rien n'est alors suppose ecrit).
Private Function PutCellChecked(ByRef ctx As CampaignCtx, ByVal rw As Long, ByVal role As String, ByVal v As String) As Boolean
    On Error Resume Next
    PutCellChecked = False
    If Not HasRole(ctx, role) Then Exit Function
    Dim back As String, e As Long
    Err.Clear
    ctx.Ws.Cells(rw, ctx.Map(role)).Value = v
    e = Err.Number
    Err.Clear
    If e <> 0 Then
        modDebugLog.DError MOD_NAME, "Ecriture refusee par Excel : ligne " & rw & " colonne " & role & " (err " & e & ")"
        Exit Function
    End If
    back = Chr$(1)
    back = CStr(ctx.Ws.Cells(rw, ctx.Map(role)).Value)
    e = Err.Number
    Err.Clear
    If e <> 0 Or back <> v Then
        modDebugLog.DError MOD_NAME, "Ecriture non confirmee : ligne " & rw & " colonne " & role & " (err " & e & ")"
        Exit Function
    End If
    PutCellChecked = True
End Function

' Remplace {{Champ}} par la valeur de la colonne dont l'en-tete normalise
' correspond. Un champ inconnu reste tel quel (signale par le pre-vol).
Private Function MergeFields(ByRef ctx As CampaignCtx, ByVal rw As Long, ByVal txt As String) As String
    On Error Resume Next
    MergeFields = txt
    If InStr(txt, "{{") = 0 Then Exit Function
    If ctx.Fields Is Nothing Then Exit Function
    Dim out As String, p As Long, q As Long, fld As String, key As String, v As String
    out = txt
    p = InStr(out, "{{")
    Do While p > 0
        q = InStr(p + 2, out, "}}")
        If q = 0 Then Exit Do
        fld = Mid$(out, p + 2, q - p - 2)
        key = NormalizeHeader(fld)
        If ctx.Fields.Exists(key) Then
            v = CStr(ctx.Ws.Cells(rw, ctx.Fields(key)).Value)
            out = Left$(out, p - 1) & v & Mid$(out, q + 2)
            p = InStr(p + Len(v), out, "{{")
        Else
            p = InStr(q + 2, out, "{{")
        End If
    Loop
    MergeFields = out
End Function


' Champs inconnus ({{x}} sans colonne) d'un texte, separes par ", ".
Private Function UnknownFields(ByRef ctx As CampaignCtx, ByVal txt As String) As String
    On Error Resume Next
    UnknownFields = ""
    If InStr(txt, "{{") = 0 Then Exit Function
    Dim p As Long, q As Long, fld As String, out As String
    p = InStr(txt, "{{")
    Do While p > 0
        q = InStr(p + 2, txt, "}}")
        If q = 0 Then Exit Do
        fld = Mid$(txt, p + 2, q - p - 2)
        If Not ctx.Fields.Exists(NormalizeHeader(fld)) Then
            If InStr(out, "{{" & fld & "}}") = 0 Then
                If Len(out) > 0 Then out = out & ", "
                out = out & "{{" & fld & "}}"
            End If
        End If
        p = InStr(q + 2, txt, "{{")
    Loop
    UnknownFields = out
End Function

' Boite deduite du dossier Outlook courant (feuilles anciennes, menu).

    On Error Resume Next

    Set gLegacyAccount = Nothing

' Compte d'envoi de la campagne : parametre, sinon CampaignDefaultAccount
' (config.ini), sinon boite du dossier courant (feuille ancienne seulement),
' sinon compte par defaut d'Outlook.
Private Sub ResolveAccount(ByRef ctx As CampaignCtx)
    On Error Resume Next
    Set ctx.Account = Nothing
    Set ctx.Store = Nothing
    ctx.AccountError = ""
    ctx.AccountLabel = T("(compte par d~efaut)", "(default account)")

    Dim wanted As String
    wanted = Trim$(ParamValue(ctx, P_ACCOUNT))
    If Len(wanted) = 0 Then wanted = Trim$(modConfig.CampaignDefaultAccount())

    If Len(wanted) = 0 Then
        If ctx.Legacy And Not gLegacyStore Is Nothing Then
            If Not modStoreContext.IsDefaultStore(gLegacyStore) Then
                Set ctx.Store = gLegacyStore
                Set ctx.Account = gLegacyAccount
                ctx.AccountLabel = gLegacyStore.DisplayName
                If ctx.Account Is Nothing Then
                    ctx.AccountLabel = ctx.AccountLabel & T(" (sans compte propre : envoi 'de la part de')", " (no own account: send 'on behalf of')")
                End If
            End If
        End If
        Exit Sub
    End If

    Dim wantedL As String
    wantedL = LCase$(wanted)
    Dim acc As Object
    For Each acc In Application.Session.Accounts
        If LCase$(CStr(acc.SmtpAddress)) = wantedL Or LCase$(CStr(acc.DisplayName)) = wantedL Then
            Set ctx.Account = acc
            Set ctx.Store = acc.DeliveryStore
            ctx.AccountLabel = CStr(acc.SmtpAddress)
            If Len(ctx.AccountLabel) = 0 Then ctx.AccountLabel = CStr(acc.DisplayName)
            Exit Sub
        End If
    Next acc

    Dim st As Object
    For Each st In Application.Session.Stores
        If LCase$(CStr(st.DisplayName)) = wantedL Then
            Set ctx.Store = st
            ctx.AccountLabel = CStr(st.DisplayName) & T(" (envoi 'de la part de')", " (send 'on behalf of')")
            Exit Sub
        End If
    Next st

    ctx.AccountError = wanted
    ctx.AccountLabel = T("INTROUVABLE : ", "NOT FOUND: ") & wanted
    modDebugLog.DError MOD_NAME, "Compte d'envoi introuvable : " & wanted
End Sub


' Applique le compte d'envoi au mail (SendUsingAccount, ou SentOnBehalfOfName
' pour une boite partagee sans compte propre - droits Exchange requis).
Private Sub ApplySendAccount(ByRef ctx As CampaignCtx, ByVal mail As Object)
    On Error Resume Next
    If Not ctx.Account Is Nothing Then
        Set mail.SendUsingAccount = ctx.Account
        modDebugLog.DDebug MOD_NAME, "SendUsingAccount = " & ctx.Account.DisplayName
    ElseIf Not ctx.Store Is Nothing Then
        If Not modStoreContext.IsDefaultStore(ctx.Store) Then
            mail.SentOnBehalfOfName = ctx.Store.DisplayName
            modDebugLog.DInfo MOD_NAME, "SentOnBehalfOfName = " & ctx.Store.DisplayName
        End If
    End If
End Sub


' Dossier Brouillons de la boite d'envoi (repli : Brouillons par defaut).
Private Function TargetDraftsFolder(ByRef ctx As CampaignCtx) As Object
    On Error Resume Next
    Set TargetDraftsFolder = Nothing
    If Not ctx.Store Is Nothing Then Set TargetDraftsFolder = modStoreContext.SysFolder(ctx.Store, OL_FOLDER_DRAFTS)

End Function

' Lignes candidates (1-based Excel) :
'   KIND_MAIN : destinataire renseigne ET statut vide ;
'   KIND_FU   : rowsArg ("12;15") si fourni, sinon lignes envoyees dont le
'               Suivi est "Sans reponse".
Private Function CandidateRows(ByRef ctx As CampaignCtx, ByVal kind As String, ByVal rowsArg As String, _
                               ByRef rows() As Long) As Long
    On Error Resume Next
    Dim n As Long, lastRow As Long, r As Long
    n = 0
    lastRow = LastDataRow(ctx)
    ReDim rows(1 To 1)
    If lastRow <= ctx.HeaderRow Then CandidateRows = 0: Exit Function

    If kind = KIND_FU And Len(Trim$(rowsArg)) > 0 Then
        Dim parts() As String, i As Long, v As Long
        parts = Split(rowsArg, ";")
        ReDim rows(1 To UBound(parts) + 1)
        For i = LBound(parts) To UBound(parts)
            v = CLng(Val(parts(i)))
            If v > ctx.HeaderRow And v <= lastRow Then
                n = n + 1
                rows(n) = v
            End If
        Next i
        CandidateRows = n
        Exit Function
    End If

    Dim toBlk As Variant, stBlk As Variant, trBlk As Variant
    toBlk = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_TO), lastRow, ctx.Map(R_TO))
    stBlk = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_STATUS), lastRow, ctx.Map(R_STATUS))
    If kind = KIND_FU Then
        If Not HasRole(ctx, R_TRACK) Then CandidateRows = 0: Exit Function
        trBlk = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_TRACK), lastRow, ctx.Map(R_TRACK))
    End If

    ReDim rows(1 To lastRow - ctx.HeaderRow)
    Dim st As String
    For r = 1 To UBound(toBlk, 1)
        If Len(Trim$(CStr(toBlk(r, 1)))) > 0 Then
            st = Trim$(CStr(stBlk(r, 1)))
            If kind = KIND_MAIN Then
                If Len(st) = 0 Then
                    n = n + 1
                    rows(n) = ctx.HeaderRow + r
                End If
            Else
                If Left$(st, 6) = "Envoye" Then
                    If Trim$(CStr(trBlk(r, 1))) = TrkNoReply() Then
                        n = n + 1
                        rows(n) = ctx.HeaderRow + r
                    End If
                End If
            End If
        End If
    Next r
    CandidateRows = n
End Function


' Pre-vol des lignes rows(1..nRows). Ecrit Controle ("OK" / anomalie) pour
' chacune. Renvoie le nombre de lignes en anomalie ; okCount = lignes OK.
Private Function Preflight(ByRef ctx As CampaignCtx, ByRef rows() As Long, ByVal nRows As Long, _
                           ByVal kind As String, ByRef okCount As Long) As Long
    On Error GoTo ErrH
    Dim problems As Long, i As Long, rw As Long, pb As String
    okCount = 0
    problems = 0
    If nRows = 0 Then Preflight = 0: Exit Function

    ' Anomalies de niveau campagne : elles bloquent toutes les lignes.
    Dim campPb As String
    campPb = CampaignLevelProblems(ctx, kind)

    Dim seen As Object
    Set seen = CreateObject("Scripting.Dictionary")

    Dim lastRow As Long
    lastRow = LastDataRow(ctx)
    Dim chk As Variant
    chk = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_CHECK), lastRow, ctx.Map(R_CHECK))

    Dim adr As String
    For i = 1 To nRows
        rw = rows(i)
        pb = campPb & ValidateRow(ctx, rw, kind)
        adr = LCase$(Trim$(CleanAddressList(Cell(ctx, rw, R_TO))))
        If Len(adr) > 0 And kind = KIND_MAIN Then
            If seen.Exists(adr) Then
                pb = pb & T("doublon de la ligne ", "duplicate of row ") & seen(adr) & " ; "
            Else
                seen(adr) = rw
            End If
        End If
        If Len(pb) = 0 Then
            okCount = okCount + 1
            chk(rw - ctx.HeaderRow, 1) = CHK_OK
        Else
            problems = problems + 1
            If Right$(pb, 3) = " ; " Then pb = Left$(pb, Len(pb) - 3)
            chk(rw - ctx.HeaderRow, 1) = pb
            modDebugLog.DInfo MOD_NAME, "Pre-vol ligne " & rw & " : " & pb
        End If
    Next i

    ctx.Ws.Range(ctx.Ws.Cells(ctx.HeaderRow + 1, ctx.Map(R_CHECK)), ctx.Ws.Cells(lastRow, ctx.Map(R_CHECK))).Value = chk
    Preflight = problems
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "Preflight " & Err.Number & " - " & Err.Description
    Preflight = nRows
End Function


' Anomalies de niveau campagne ("" si aucune) : compte introuvable, modele
' du bloc absent, pieces jointes du bloc absentes, champ inconnu dans
' l'objet du bloc, dossier de classement introuvable ou ambigu (v12.1).
' Les champs inconnus du message (colonne de la ligne) sont controles par
' ValidateRow ; le message de relance seulement s'il est vide.
Private Function CampaignLevelProblems(ByRef ctx As CampaignCtx, ByVal kind As String) As String
    On Error Resume Next
    Dim pb As String
    If Len(ctx.AccountError) > 0 Then
        pb = pb & T("compte d'envoi introuvable (", "sending account not found (") & ctx.AccountError & ") ; "
    End If
    If Not ctx.Legacy Then
        Dim tpl As String
        tpl = EffectiveTemplate(ctx, 0)
        If Len(tpl) > 0 Then
            If Len(Dir$(tpl)) = 0 Then pb = pb & T("mod~fle introuvable (", "template not found (") & tpl & ") ; "
        End If
        pb = pb & MissingAttachments(ctx, ParamValue(ctx, P_ATTACH))
        Dim unk As String
        unk = UnknownFields(ctx, ParamValue(ctx, P_SUBJECT))
        If Len(unk) > 0 Then pb = pb & T("champ inconnu dans l'objet : ", "unknown field in subject: ") & unk & " ; "
    End If

    ' v12.1 (F5-13) : le dossier de classement est resolu ICI, comme il le
    ' sera a l'arrivee de chaque mail dans Elements envoyes : une saisie
    ' qui ne designe pas exactement un dossier bloque avant l'envoi au lieu
    ' d'etre ignoree apres, dans le journal seulement.
    Dim fTxt As String, nFld As Long, fld As Object
    fTxt = Trim$(ParamValue(ctx, P_FOLDER))
    If Len(fTxt) > 0 Then
        Set fld = ResolveFilingFolder(fTxt, nFld)
        If nFld = 0 Then
            pb = pb & T("dossier de classement introuvable (", "filing folder not found (") & fTxt & ") ; "
        ElseIf nFld > 1 Then
            pb = pb & T("dossier de classement ambigu (", "ambiguous filing folder (") & fTxt & ") : " & nFld & _
                      T(" dossiers ; ", " folders; ")
        ElseIf fld Is Nothing Then
            pb = pb & T("dossier de classement inaccessible (", "filing folder not reachable (") & fTxt & ") ; "
        End If
    End If
    CampaignLevelProblems = pb
End Function


' Anomalies d'UNE ligne ("" si aucune).
Private Function ValidateRow(ByRef ctx As CampaignCtx, ByVal rw As Long, ByVal kind As String) As String
    On Error Resume Next
    Dim pb As String
    Dim dest As String
    dest = Trim$(Cell(ctx, rw, R_TO))
    If Len(dest) = 0 Then
        pb = pb & T("destinataire manquant ; ", "missing recipient; ")
    ElseIf Not PlausibleAddresses(dest) Then
        pb = pb & T("adresse A douteuse ; ", "suspicious To address; ")
    End If
    Dim cc As String
    cc = Trim$(Cell(ctx, rw, R_CC))
    If Len(cc) > 0 Then
        If Not PlausibleAddresses(cc) Then pb = pb & T("adresse Cc douteuse ; ", "suspicious Cc address; ")
    End If
    Dim bc As String
    bc = Trim$(Cell(ctx, rw, R_BCC))
    If Len(bc) > 0 Then
        If Not PlausibleAddresses(bc) Then pb = pb & T("adresse Cci douteuse ; ", "suspicious Bcc address; ")
    End If

    If Len(EffectiveSubject(ctx, rw)) = 0 Then pb = pb & T("objet vide ; ", "empty subject; ")
    Dim unk As String
    unk = UnknownFields(ctx, Cell(ctx, rw, R_SUBJECT))
    If Len(unk) > 0 Then pb = pb & T("champ inconnu : ", "unknown field: ") & unk & " ; "

    If kind = KIND_MAIN Then
        If Len(Trim$(Cell(ctx, rw, R_BODY))) = 0 Then
            pb = pb & T("message vide (G~en~erer) ; ", "empty message (Generate); ")
        Else
            unk = UnknownFields(ctx, Cell(ctx, rw, R_BODY))
            If Len(unk) > 0 Then pb = pb & T("champ inconnu : ", "unknown field: ") & unk & " ; "
        End If
    Else
        If Len(Trim$(Cell(ctx, rw, R_FUBODY))) = 0 Then
            pb = pb & T("relance vide (G~en~erer) ; ", "empty follow-up (Generate); ")
        End If
        If Len(Trim$(Cell(ctx, rw, R_ENTRY))) = 0 Then
            pb = pb & T("mail d'origine inconnu (Actualiser les r~eponses) ; ", "original mail unknown (Refresh replies); ")
        End If
    End If

    ' Modele facultatif : vide = mail neuf avec la signature par defaut.
    Dim tpl As String
    tpl = EffectiveTemplate(ctx, rw)
    If Len(tpl) > 0 Then
        If Len(Dir$(tpl)) = 0 Then
            pb = pb & T("mod~fle introuvable (", "template not found (") & tpl & ") ; "
        ElseIf LCase$(Right$(tpl, 4)) <> ".oft" And LCase$(Right$(tpl, 4)) <> ".msg" Then
            pb = pb & T("mod~fle : extension .oft ou .msg attendue ; ", "template: .oft or .msg expected; ")
        End If
    End If

    pb = pb & MissingAttachments(ctx, Cell(ctx, rw, R_ATTACH))
    ValidateRow = pb
End Function


' Pieces jointes absentes du disque dans une liste (";" ou retours a la ligne).
Private Function MissingAttachments(ByRef ctx As CampaignCtx, ByVal listText As String) As String
    On Error Resume Next
    Dim out As String, parts() As String, j As Long, p As String
    Dim pj As String
    pj = NormalizeAttachSep(listText)
    If Len(Trim$(pj)) = 0 Then Exit Function
    parts = Split(pj, ";")
    For j = LBound(parts) To UBound(parts)
        p = ResolveRelPath(ctx, CleanPath(parts(j)))
        If Len(p) > 0 Then
            If Len(Dir$(p)) = 0 Then out = out & T("PJ introuvable (", "attachment not found (") & p & ") ; "
        End If
    Next j
    MissingAttachments = out
End Function


Private Function PlausibleAddresses(ByVal listText As String) As Boolean
    Dim parts() As String, i As Long, one As String
    parts = Split(listText, ";")
    For i = LBound(parts) To UBound(parts)
        one = Trim$(parts(i))
        If Len(one) > 0 Then
            If Not OneAddressPlausible(one) Then PlausibleAddresses = False: Exit Function
        End If
    Next i
    PlausibleAddresses = True
End Function


Private Function OneAddressPlausible(ByVal adr As String) As Boolean
    Dim pos As Long
    pos = InStr(adr, "@")
    If pos < 2 Then Exit Function
    If InStr(adr, " ") > 0 Then Exit Function
    If InStr(pos + 1, adr, ".") = 0 Then Exit Function
    OneAddressPlausible = True
End Function


' Verification seule (classeur ou menu) : Controle rempli, synthese.
Private Sub RunCheck(ByRef ctx As CampaignCtx, ByVal interactive As Boolean)
    Dim rows() As Long, n As Long, ok As Long, pbs As Long
    n = CandidateRows(ctx, KIND_MAIN, "", rows)
    If n = 0 Then
        If interactive Then
            modUi.Inform T("Aucune ligne ~a v~erifier (destinataire vide ou statut d~ej~a renseign~e).", "No row to check (empty recipient or status already set)."), T("Campagnes", "Campaigns")
        Else
            modUi.Toast T("Aucune ligne ~a v~erifier.", "No row to check."), uiInfo
        End If
        Exit Sub
    End If
    ExcelStatus ctx, T("Campagne : v~erification de ", "Campaign: checking ") & n & T(" ligne(s)...", " row(s)...")
    pbs = Preflight(ctx, rows, n, KIND_MAIN, ok)
    ExcelStatus ctx, ""
    Dim msg As String
    msg = ok & T(" ligne(s) pr~xte(s) sur ", " row(s) ready out of ") & n & "."
    If pbs > 0 Then msg = msg & vbLf & pbs & T(" ~a corriger : voir la colonne Contr~ole.", " to fix: see the Check column.")
    If interactive Then
        modUi.Inform msg, T("Campagnes - v~erification", "Campaigns - check")
    Else
        modUi.Toast msg, IIf(pbs > 0, uiWarn, uiSuccess), True
    End If
    modDebugLog.DInfo MOD_NAME, "Verification " & ctx.SheetName & " : ok=" & ok & " anomalies=" & pbs
End Sub

' Mail d'une ligne depuis le modele : destinataires, objet, corps HTML de la
' colonne insere au-dessus de la signature, pieces jointes (bloc + ligne),
' marqueur de campagne, compte d'envoi. Nothing + gLastFailure si echec.
Private Function BuildMail(ByRef ctx As CampaignCtx, ByVal rw As Long) As Object
    On Error GoTo ErrH
    gLastFailure = ""
    gAttachFail = ""
    Set BuildMail = Nothing

    Dim mail As Object
    Set mail = OpenFromTemplate(EffectiveTemplate(ctx, rw))
    If mail Is Nothing Then
        gLastFailure = T("mod~fle non ouvert", "template not opened")
        Exit Function
    End If

    mail.To = CleanAddressList(Cell(ctx, rw, R_TO))
    If HasRole(ctx, R_CC) Then mail.CC = CleanAddressList(Cell(ctx, rw, R_CC))
    If HasRole(ctx, R_BCC) Then mail.BCC = CleanAddressList(Cell(ctx, rw, R_BCC))
    mail.Subject = EffectiveSubject(ctx, rw)
    mail.HTMLBody = InsertBody(mail.HTMLBody, MergeFields(ctx, rw, Cell(ctx, rw, R_BODY)))

    AddAttachments ctx, rw, mail
    ' v12.1 (F5-21) : une piece jointe (ou image du modele .msg) presente au
    ' pre-vol mais impossible a joindre maintenant (fichier verrouille, fichier
    ' OneDrive non telecharge hors connexion, chemin trop long) : la ligne
    ' passe en echec avec la raison, le mail ne part pas incomplet.
    If Len(gAttachFail) > 0 Then
        gLastFailure = T("PJ non jointe : ", "attachment not added: ") & gAttachFail
        modDebugLog.DError MOD_NAME, "BuildMail ligne " & rw & " : piece(s) jointe(s) non ajoutee(s), mail abandonne"
        mail.Close 1   ' olDiscard
        Exit Function
    End If
    SetPayload mail, ctx, rw, KIND_MAIN, ""
    ApplySendAccount ctx, mail
    Set BuildMail = mail
    Exit Function
ErrH:
    gLastFailure = "err " & Err.Number
    modDebugLog.DError MOD_NAME, "BuildMail ligne " & rw & " : " & Err.Number & " - " & Err.Description
    Set BuildMail = Nothing
End Function


' Relance d'une ligne : "Repondre a tous" sur le mail d'origine (fil
' conserve), texte Relance (HTML) en tete, pieces jointes remises.
Private Function BuildFollowUp(ByRef ctx As CampaignCtx, ByVal rw As Long) As Object
    On Error GoTo ErrH
    gLastFailure = ""
    gAttachFail = ""
    Set BuildFollowUp = Nothing

    Dim origEID As String
    origEID = Trim$(Cell(ctx, rw, R_ENTRY))
    Dim orig As Object
    On Error Resume Next
    Set orig = Application.Session.GetItemFromID(origEID)
    On Error GoTo ErrH
    If orig Is Nothing Then
        gLastFailure = T("mail d'origine introuvable", "original mail not found")
        Exit Function
    End If

    Dim fu As Object
    Set fu = orig.ReplyAll
    Dim adr As String
    adr = CleanAddressList(Cell(ctx, rw, R_TO))
    If Len(adr) > 0 Then
        If InStr(1, CStr(fu.To), Split(adr, ";")(0), vbTextCompare) = 0 Then fu.To = adr
    End If
    fu.HTMLBody = InsertBody(fu.HTMLBody, MergeFields(ctx, rw, Cell(ctx, rw, R_FUBODY)))

    AddAttachments ctx, rw, fu
    If Len(gAttachFail) > 0 Then      ' v12.1 (F5-21) : voir BuildMail
        gLastFailure = T("PJ non jointe : ", "attachment not added: ") & gAttachFail
        modDebugLog.DError MOD_NAME, "BuildFollowUp ligne " & rw & " : piece(s) jointe(s) non ajoutee(s), relance abandonnee"
        fu.Close 1   ' olDiscard
        Exit Function
    End If
    SetPayload fu, ctx, rw, KIND_FU, origEID
    ApplySendAccount ctx, fu
    Set BuildFollowUp = fu
    Exit Function
ErrH:
    gLastFailure = "err " & Err.Number
    modDebugLog.DError MOD_NAME, "BuildFollowUp ligne " & rw & " : " & Err.Number & " - " & Err.Description
    Set BuildFollowUp = Nothing
End Function


' Pieces jointes du bloc de parametres puis de la ligne. v12.1 (F5-21) : un
' fichier disparu ou impossible a joindre est note dans gAttachFail (le
' constructeur abandonne alors le mail).
Private Sub AddAttachments(ByRef ctx As CampaignCtx, ByVal rw As Long, ByVal mail As Object)
    On Error Resume Next
    AddAttachmentList ctx, mail, ParamValue(ctx, P_ATTACH)
    AddAttachmentList ctx, mail, Cell(ctx, rw, R_ATTACH)
End Sub


Private Sub AddAttachmentList(ByRef ctx As CampaignCtx, ByVal mail As Object, ByVal listText As String)
    On Error Resume Next
    Dim pj As String, parts() As String, j As Long, p As String, e As Long
    pj = NormalizeAttachSep(listText)
    If Len(Trim$(pj)) = 0 Then Exit Sub
    parts = Split(pj, ";")
    For j = LBound(parts) To UBound(parts)
        p = ResolveRelPath(ctx, CleanPath(parts(j)))
        If Len(p) > 0 Then
            If Len(Dir$(p)) > 0 Then
                Err.Clear
                mail.Attachments.Add p
                e = Err.Number
                Err.Clear
                If e <> 0 Then
                    modDebugLog.DError MOD_NAME, "PJ non jointe (err " & e & ") : " & LeafName(p)
                    NoteAttachFail LeafName(p)
                End If
            Else
                modDebugLog.DError MOD_NAME, "PJ introuvable : " & p
                NoteAttachFail LeafName(p)
            End If
        End If
    Next j
End Sub


' v12.1 (F5-21) : piece jointe manquante pour le mail en construction.
Private Sub NoteAttachFail(ByVal fileName As String)
    If Len(gAttachFail) > 0 Then gAttachFail = gAttachFail & ", "
    gAttachFail = gAttachFail & fileName
End Sub


' Marqueur invisible : purge ciblee + reference de la ligne Excel (suivi des
' envois manuels, classement, relances).
Private Sub SetPayload(ByVal mail As Object, ByRef ctx As CampaignCtx, ByVal rw As Long, _
                       ByVal kind As String, ByVal origEID As String)
    On Error Resume Next
    Dim up As Object
    Set up = mail.UserProperties.Find(UPROP_CAMPAIGN)
    If up Is Nothing Then Set up = mail.UserProperties.Add(UPROP_CAMPAIGN, 1)   ' 1 = olText
    up.Value = ctx.WbPath & vbTab & ctx.SheetName & vbTab & CStr(rw) & vbTab & _
               LCase$(Trim$(CleanAddressList(Cell(ctx, rw, R_TO)))) & vbTab & _
               EffectiveSubject(ctx, rw) & vbTab & kind & vbTab & _
               Trim$(ParamValue(ctx, P_FOLDER)) & vbTab & origEID
End Sub


' .oft -> CreateItemFromTemplate (vrai brouillon, source preservee).
' .msg -> corps lu puis mail NEUF (un .msg copie via OpenSharedItem.Copy
'         atterrit dans la Boite de reception) ; v12 : pieces jointes et
'         images incorporees (logos de signature) reportees, identifiant de
'         contenu conserve. Le .msg source n'est jamais altere.
' vide -> mail NEUF ; GetInspector (sans affichage) y pose la signature par
'         defaut d'Outlook, le corps de la ligne est ensuite insere au-dessus.
Private Function OpenFromTemplate(ByVal tplPath As String) As Object
    On Error GoTo ErrH
    Set OpenFromTemplate = Nothing
    Dim ext As String
    ext = LCase$(Right$(tplPath, 4))

    If Len(Trim$(tplPath)) = 0 Then
        Dim plain As Object, insp As Object
        Set plain = Application.CreateItem(0)   ' olMailItem
        On Error Resume Next
        Set insp = plain.GetInspector            ' insere la signature par defaut
        Set insp = Nothing
        On Error GoTo ErrH
        Set OpenFromTemplate = plain
    ElseIf ext = ".oft" Then
        Set OpenFromTemplate = Application.CreateItemFromTemplate(tplPath)
    ElseIf ext = ".msg" Then
        Dim src As Object, mail As Object
        Set src = Application.Session.OpenSharedItem(tplPath)
        Set mail = Application.CreateItem(0)   ' olMailItem
        mail.HTMLBody = src.HTMLBody
        CopyAttachments src, mail
        src.Close 1   ' olDiscard
        Set OpenFromTemplate = mail
    Else
        modDebugLog.DError MOD_NAME, "Extension modele non geree : " & tplPath
    End If
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "OpenFromTemplate " & Err.Number & " - " & Err.Description & " (" & tplPath & ")"
    Set OpenFromTemplate = Nothing
End Function


' Reporte les pieces jointes d'un .msg (visibles et incorporees) dans un
' mail neuf via un fichier temporaire ; identifiant de contenu (cid:) et
' drapeau "masque" reposes sur la copie pour que les images du corps
' continuent de s'afficher. v12.1 (F5-21) : chaque piece non reportee est
' notee dans gAttachFail (le mail est alors abandonne par BuildMail).
Private Sub CopyAttachments(ByVal src As Object, ByVal dst As Object)
    On Error Resume Next
    Dim n As Long
    n = src.Attachments.Count
    If n = 0 Then Exit Sub
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    Dim tmpDir As String
    tmpDir = Environ$("TEMP")
    If Len(tmpDir) = 0 Then tmpDir = Environ$("TMP")
    If Len(tmpDir) = 0 Then
        modDebugLog.DError MOD_NAME, "CopyAttachments : dossier temporaire introuvable, " & n & " piece(s) non reportee(s)"
        NoteAttachFail T("pi~fces du mod~fle", "template attachments")
        Exit Sub
    End If
    tmpDir = tmpDir & "\Campaign-" & Format$(Now, "yyyymmdd-hhnnss")
    fso.CreateFolder tmpDir

    Dim i As Long, att As Object, nm As String, p As String, cid As String, hid As Boolean, newAtt As Object
    For i = 1 To n
        Set att = Nothing
        Set att = src.Attachments.Item(i)
        If Not att Is Nothing Then
            nm = Trim$(CStr(att.FileName))
            If Len(nm) = 0 Then nm = Trim$(CStr(att.DisplayName))
            If Len(nm) = 0 Then nm = "piece" & i
            nm = modCommon.SanitizeFileName(nm)
            p = tmpDir & "\" & i & "-" & nm
            Err.Clear
            att.SaveAsFile p
            If Err.Number = 0 Then
                cid = ""
                cid = Trim$(CStr(att.PropertyAccessor.GetProperty(PR_ATTACH_CONTENT_ID)))
                hid = False
                hid = CBool(att.PropertyAccessor.GetProperty(PR_ATTACH_HIDDEN))
                Set newAtt = Nothing
                Set newAtt = dst.Attachments.Add(p, 1, , nm)   ' 1 = olByValue
                If Not newAtt Is Nothing Then
                    If Len(cid) > 0 Then newAtt.PropertyAccessor.SetProperty PR_ATTACH_CONTENT_ID, cid
                    If hid Then newAtt.PropertyAccessor.SetProperty PR_ATTACH_HIDDEN, True
                Else
                    ' v12.1 (F5-21) : piece ou image du modele non reportee.
                    modDebugLog.DError MOD_NAME, "CopyAttachments : Attachments.Add KO (" & nm & ") err " & Err.Number
                    Err.Clear
                    NoteAttachFail nm
                End If
            Else
                modDebugLog.DError MOD_NAME, "CopyAttachments : SaveAsFile KO (" & nm & ") err " & Err.Number
                Err.Clear
                NoteAttachFail nm
            End If
        End If
    Next i
    fso.DeleteFolder tmpDir, True
End Sub


' Insere le corps juste apres <body...> ; la signature du modele reste
' en dessous. Repli : prepend.
Private Function InsertBody(ByVal tplHtml As String, ByVal bodyHtml As String) As String
    Dim posBody As Long, posEnd As Long
    posBody = InStr(1, tplHtml, "<body", vbTextCompare)
    If posBody > 0 Then
        posEnd = InStr(posBody, tplHtml, ">")
        If posEnd > 0 Then
            InsertBody = Left$(tplHtml, posEnd) & bodyHtml & Mid$(tplHtml, posEnd + 1)
            Exit Function
        End If
    End If
    InsertBody = bodyHtml & tplHtml
End Function


' Adresses NON resolues ("" si tout est resolu). Un domaine errone ou un
' TLD tronque ne se resout pas ; c'est Outlook qui tranche.
Private Function UnresolvedAddresses(ByVal mail As Object) As String
    On Error Resume Next
    Dim recips As Object, r As Object, i As Long, ko As String
    Set recips = mail.Recipients
    recips.ResolveAll
    For i = 1 To recips.Count
        Set r = recips.Item(i)
        If Not r.Resolved Then
            If Len(ko) > 0 Then ko = ko & "; "
            ko = ko & r.Name
        End If
    Next i
    UnresolvedAddresses = ko
End Function

' Brouillons des lignes sans statut. skipProblems=False (classeur) : une
' anomalie bloque tout. True (ancien menu) : question, puis lignes OK seules.
Private Sub GenerateDrafts(ByRef ctx As CampaignCtx, ByVal askOnProblems As Boolean)
    On Error GoTo ErrH
    Dim rows() As Long, n As Long, ok As Long, pbs As Long
    n = CandidateRows(ctx, KIND_MAIN, "", rows)
    If n = 0 Then
        modUi.Toast T("Rien ~a g~en~erer (lignes d~ej~a trait~ees ou sans destinataire).", "Nothing to generate (rows already processed or without recipient)."), uiInfo
        Exit Sub
    End If

    ExcelStatus ctx, T("Campagne : v~erification...", "Campaign: checking...")
    pbs = Preflight(ctx, rows, n, KIND_MAIN, ok)
    If pbs > 0 Then
        If askOnProblems Then
            If modUi.Ask(ok & T(" ligne(s) pr~xte(s), ", " row(s) ready, ") & pbs & T(" en anomalie (colonne Contr~ole).", " with problems (Check column).") & vbCrLf & _
                         T("Continuer en sautant les lignes en anomalie ?", "Continue, skipping the rows with problems?"), _
                         T("Campagnes - pr~e-vol", "Campaigns - preflight"), vbQuestion + vbYesNo, False) <> vbYes Then
                ExcelStatus ctx, ""
                Exit Sub
            End If
        Else
            ExcelStatus ctx, ""
            modUi.Toast pbs & T(" ligne(s) ~a corriger (colonne Contr~ole). Aucun brouillon cr~e~e.", " row(s) to fix (Check column). No draft created."), uiWarn, True
            Exit Sub
        End If
    End If

    ' v12.1 (F5-06) : un brouillon deja present pour une de ces lignes (statut
    ' reinitialise depuis) ferait double emploi avec le nouveau : supprime.
    Dim purged As Long
    purged = PurgeCampaignDrafts(ctx.WbName, ctx.SheetName, RowKeys(ctx, rows, n), KIND_MAIN)
    If purged > 0 Then modDebugLog.DInfo MOD_NAME, "Brouillons : " & purged & " ancien(s) brouillon(s) des memes lignes supprime(s)"
    If IsSecondarySendStore(ctx) Then RegisterSweepStore ctx.Store   ' F5-09

    Dim drafts As Object
    Set drafts = TargetDraftsFolder(ctx)

    Dim i As Long, rw As Long, done As Long, failed As Long, mail As Object, ko As String
    For i = 1 To n
        rw = rows(i)
        If Cell(ctx, rw, R_CHECK) = CHK_OK Then
            ExcelStatus ctx, T("Campagne : brouillon ligne ", "Campaign: draft row ") & rw & " (" & (done + failed + 1) & "/" & ok & ")"
            Set mail = BuildMail(ctx, rw)
            If mail Is Nothing Then
                PutCell ctx, rw, R_STATUS, ST_FAIL & Format$(Now, "dd/mm hh:nn") & " - " & gLastFailure
                failed = failed + 1
            Else
                ko = UnresolvedAddresses(mail)
                If Len(ko) > 0 Then
                    mail.Close 1
                    PutCell ctx, rw, R_STATUS, ST_FAIL & Format$(Now, "dd/mm hh:nn") & T(" - adresse non r~esolue (", " - unresolved address (") & ko & ")"
                    PutCell ctx, rw, R_CHECK, T("adresse non r~esolue (", "unresolved address (") & ko & ")"
                    failed = failed + 1
                Else
                    SaveToDrafts mail, drafts
                    PutCell ctx, rw, R_STATUS, ST_DRAFT & Format$(Now, "dd/mm/yyyy hh:nn")
                    done = done + 1
                End If
            End If
        End If
        If (i Mod 10) = 0 Then DoEvents
    Next i
    ExcelStatus ctx, ""

    modDebugLog.DInfo MOD_NAME, "Brouillons " & ctx.SheetName & " : ok=" & done & " ko=" & failed
    modUi.Toast done & T(" brouillon(s) cr~e~e(s) dans Brouillons (", " draft(s) created in Drafts (") & ctx.AccountLabel & ")" & _
                 IIf(failed > 0, vbLf & failed & T(" ~echec(s) : voir Statut.", " failure(s): see Status."), ""), _
                 IIf(failed > 0, uiWarn, uiSuccess), True
    Exit Sub
ErrH:
    ExcelStatus ctx, ""
    modDebugLog.DError MOD_NAME, "GenerateDrafts " & Err.Number & " - " & Err.Description
    modUi.Toast T("Brouillons : erreur ", "Drafts: error ") & Err.Number, uiError, True
End Sub


' Save puis Move vers les Brouillons de la boite d'envoi (l'EntryID change,
' la UserProperty survit). Echec du Move : reste en boite par defaut.
Private Sub SaveToDrafts(ByVal mail As Object, ByVal drafts As Object)
    On Error Resume Next
    mail.Save
    If drafts Is Nothing Then Exit Sub
    If mail.Parent.EntryID = drafts.EntryID Then Exit Sub
    Dim moved As Object
    Set moved = mail.Move(drafts)
    If moved Is Nothing Then modDebugLog.DError MOD_NAME, "Move vers Brouillons de la boite d'envoi refuse"
End Sub

' Demarre l'envoi des lignes candidates (kind M : lignes sans statut ;
' kind F : relances). Renvoie True si l'envoi a demarre (le contexte Excel
' est alors conserve jusqu'a FinishSend).
Private Function StartSend(ByRef ctx As CampaignCtx, ByVal kind As String, ByVal rowsArg As String, _
                           ByVal wasOpen As Boolean, ByVal createdByUs As Boolean) As Boolean
    On Error GoTo ErrH
    StartSend = False
    If gSendActive Then
        modUi.Toast T("Un envoi est d~ej~a en cours.", "A send is already in progress."), uiWarn, True
        Exit Function
    End If

    Dim rows() As Long, n As Long, ok As Long, pbs As Long
    n = CandidateRows(ctx, kind, rowsArg, rows)
    If n = 0 Then
        modUi.Toast T("Aucune ligne ~a envoyer.", "No row to send."), uiInfo
        Exit Function
    End If

    Dim stepSec As Long, batchSize As Long, pauseSec As Long, cap As Long
    stepSec = modConfig.CampaignStepSec(): If stepSec <= 0 Then stepSec = 20
    batchSize = modConfig.CampaignBatchSize(): If batchSize <= 0 Then batchSize = 50
    pauseSec = modConfig.CampaignPauseSec(): If pauseSec < 0 Then pauseSec = 180
    cap = modConfig.CampaignCap(): If cap <= 0 Then cap = 200
    If n > cap Then
        modUi.Toast n & T(" envois demand~es : plafond de s~ecurit~e (", " sends requested: safety cap (") & cap & T(") d~epass~e. R~eduis le lot.", ") exceeded. Reduce the batch."), uiError, True
        Exit Function
    End If

    If Not OutlookOnline() Then
        modUi.Toast T("Outlook est HORS CONNEXION : les mails ne partiraient pas. Envoi annul~e.", "Outlook is OFFLINE: mail would not go out. Send cancelled."), uiError, True
        Exit Function
    End If

    ExcelStatus ctx, T("Campagne : v~erification...", "Campaign: checking...")
    pbs = Preflight(ctx, rows, n, kind, ok)
    If pbs > 0 Then
        ExcelStatus ctx, ""
        modUi.Toast pbs & T(" ligne(s) ~a corriger (colonne Contr~ole). Rien n'est envoy~e.", " row(s) to fix (Check column). Nothing sent."), uiWarn, True
        Exit Function
    End If

    ' Brouillons deja generes pour les lignes de CET envoi (meme genre) : ils
    ' feraient double emploi avec l'envoi direct. v12.1 (F5-05 / F5-06) :
    ' purge limitee a ces lignes ; les brouillons des autres lignes (statut
    ' "Brouillon cree le", qui ne partent pas ici) restent intacts.
    Dim keys As Object, purged As Long
    Set keys = RowKeys(ctx, rows, n)
    purged = PurgeCampaignDrafts(ctx.WbName, ctx.SheetName, keys, kind)
    If purged > 0 Then modDebugLog.DInfo MOD_NAME, "Envoi : " & purged & " brouillon(s) des lignes envoyees purge(s)."

    gSendActive = True
    gSendCtx = ctx
    gSendWasOpen = wasOpen
    gSendCreatedByUs = createdByUs
    gSendKind = kind
    gSendOk = 0
    gSendKo = 0
    gSendSkip = 0
    gSendClaim = ""
    gSendClaimRow = 0
    gSendBusyTries = 0
    gSendAbortReason = ""
    gSendStepSec = stepSec
    gSendBatchSize = batchSize
    gSendPauseSec = pauseSec
    ' v12.1 (F5-09) : compte secondaire -> Elements envoyes releves.
    gSendSecondary = IsSecondarySendStore(ctx)
    If gSendSecondary Then RegisterSweepStore ctx.Store
    ' v12.1 (F5-01) : chaque position garde la cle de sa ligne (destinataire
    ' + objet) pour la retrouver si la table est triee pendant l'envoi.
    ReDim gSendRows(1 To n)
    ReDim gSendKeys(1 To n)
    ReDim gSendSubj(1 To n)
    Dim i As Long
    For i = 1 To n
        gSendRows(i) = rows(i)
        gSendKeys(i) = LCase$(Trim$(CleanAddressList(Cell(ctx, rows(i), R_TO))))
        gSendSubj(i) = EffectiveSubject(ctx, rows(i))
    Next i
    gSendTotal = n
    gSendPos = 1

    modDebugLog.DInfo MOD_NAME, "Envoi demarre (" & kind & ") : " & n & " mail(s), cadence " & stepSec & "s, feuille " & ctx.SheetName
    ExcelStatus ctx, T("Campagne : envoi de ", "Campaign: sending ") & n & " mail(s)..."
    modUi.Toast T("Envoi lanc~e : ", "Send started: ") & n & T(" mail(s), cadence ", " mail(s), throttle ") & stepSec & " s" & vbLf & _
                 T("Outlook reste utilisable. Arr~xter = interrompre.", "Outlook stays usable. Stop = interrupt."), uiInfo, True

    StartSend = True
    CampaignSendNext
    Exit Function
ErrH:
    gSendActive = False
    ExcelStatus ctx, ""
    modDebugLog.DError MOD_NAME, "StartSend " & Err.Number & " - " & Err.Description
    modUi.Toast T("Envoi : erreur ", "Send: error ") & Err.Number, uiError, True
End Function


' Cible modDefer "CampaignNext" : envoie la ligne courante, ecrit son statut,
' arme la suivante. Silencieux (contexte differe).
' v12.1 : (F5-01) la ligne est retrouvee par sa cle, une ligne dont le
' Statut n'est plus vide est sautee ; (F5-02) "Envoye (en cours) le" est
' pose et relu AVANT .Send : Excel qui refuse = meme ligne retentee 10 s
' plus tard, envoi arrete apres 6 refus consecutifs.
Public Sub CampaignSendNext()
    On Error Resume Next
    If Not gSendActive Then Exit Sub
    If gSendPos < 1 Or gSendPos > gSendTotal Then
        FinishSend
        Exit Sub
    End If
    If gSendCtx.Ws Is Nothing Then
        modDebugLog.DError MOD_NAME, "CampaignSendNext : feuille perdue - arret"
        FinishSend
        Exit Sub
    End If
    ' La feuille repond-elle encore ? (classeur ferme pendant l'envoi)
    Dim probe As String
    Err.Clear
    probe = CStr(gSendCtx.Ws.Name)
    If Err.Number <> 0 Then
        modDebugLog.DError MOD_NAME, "CampaignSendNext : classeur ferme (" & Err.Number & ") - arret"
        Err.Clear
        FinishSend
        Exit Sub
    End If

    ' F5-01 : ligne visee retrouvee par sa cle ; 0 = ne pas envoyer.
    Dim rw As Long, why As String
    rw = LocateSendRow(gSendPos, why)
    If rw = 0 Then
        gSendSkip = gSendSkip + 1
        modDebugLog.DInfo MOD_NAME, "Envoi : position " & gSendPos & " (ligne prevue " & gSendRows(gSendPos) & ") sautee - " & why
        StepToNextRow True
        Exit Sub
    End If

    ' F5-02 : statut pose et relu AVANT le depart du mail. Un statut "en
    ' cours" deja pose par un essai precedent de la meme position est garde.
    If gSendKind = KIND_MAIN Then
        If Len(gSendClaim) = 0 Or Trim$(Cell(gSendCtx, rw, R_STATUS)) <> gSendClaim Then
            gSendClaim = ST_SENDING & Format$(Now, "dd/mm/yyyy hh:nn:ss")
            gSendClaimRow = rw
            If Not PutCellChecked(gSendCtx, rw, R_STATUS, gSendClaim) Then
                gSendBusyTries = gSendBusyTries + 1
                If gSendBusyTries >= SEND_BUSY_MAX Then
                    gSendAbortReason = T("Excel occup~e (statut impossible ~a ~ecrire) : envoi interrompu.", _
                                         "Excel busy (status could not be written): send interrupted.")
                    modDebugLog.DError MOD_NAME, "Envoi interrompu : statut refuse " & gSendBusyTries & " fois de suite (ligne " & rw & ")"
                    FinishSend
                Else
                    modDebugLog.DInfo MOD_NAME, "Statut non ecrit (Excel occupe ?) ligne " & rw & " : nouvel essai dans " & SEND_BUSY_RETRY_SEC & " s"
                    ExcelStatus gSendCtx, T("Campagne : Excel occup~e, nouvel essai dans ", "Campaign: Excel busy, retrying in ") & SEND_BUSY_RETRY_SEC & " s"
                    modDefer.ArmUnique DEFER_SEND_KEY, SEND_BUSY_RETRY_SEC
                End If
                Exit Sub
            End If
        End If
    End If
    gSendBusyTries = 0

    Err.Clear
    If SendRowNow(rw) Then
        If gSendKind = KIND_MAIN Then
            ' Un echec ici laisse "Envoye (en cours) le" : la ligne reste protegee
            ' et l'arrivee dans Elements envoyes (ou l'actualisation) la corrige.
            If Not PutCellChecked(gSendCtx, rw, R_STATUS, ST_SENT & Format$(Now, "dd/mm/yyyy hh:nn:ss")) Then
                modDebugLog.DError MOD_NAME, "Ligne " & rw & " envoyee, statut final non ecrit (reste 'en cours', protegee)"
            End If
        End If
        gSendOk = gSendOk + 1
        ' F5-09 : copie envoyee dans les Elements envoyes du compte secondaire.
        If gSendSecondary Then modDefer.ArmUnique DEFER_SENT_KEY, SENT_FLUSH_DELAY_SEC
    Else
        If gSendKind = KIND_MAIN Then
            PutCell gSendCtx, rw, R_STATUS, ST_FAIL & Format$(Now, "dd/mm hh:nn") & IIf(Len(gLastFailure) > 0, " - " & gLastFailure, "")
        Else
            PutCell gSendCtx, rw, R_CHECK, T("relance : ", "follow-up: ") & gLastFailure
        End If
        gSendKo = gSendKo + 1
    End If
    If Err.Number <> 0 Then
        modDebugLog.DError MOD_NAME, "CampaignSendNext ligne " & rw & " : err " & Err.Number & " - " & Err.Description
        Err.Clear
    End If

    StepToNextRow False
End Sub


' v12.1 : position suivante (fin de l'envoi, pause de paquet ou cadence).
' quick=True : la position courante a ete sautee, rien n'est parti ->
' suivante dans 1 s. F5-02 : classeur enregistre toutes les SAVE_EVERY_ROWS
' positions et a chaque pause de paquet.
Private Sub StepToNextRow(ByVal quick As Boolean)
    On Error Resume Next
    gSendClaim = ""
    gSendClaimRow = 0
    gSendBusyTries = 0
    gSendPos = gSendPos + 1
    If gSendPos > gSendTotal Then
        FinishSend
        Exit Sub
    End If

    Dim waitSec As Long, done As Long
    done = gSendPos - 1
    If (done Mod gSendBatchSize) = 0 Then
        waitSec = gSendPauseSec
        modDebugLog.DInfo MOD_NAME, "Paquet de " & gSendBatchSize & " atteint, pause " & gSendPauseSec & "s."
        SaveSendProgress
    Else
        waitSec = gSendStepSec
        If (done Mod SAVE_EVERY_ROWS) = 0 Then SaveSendProgress
    End If
    If quick Then waitSec = 1
    If waitSec < 1 Then waitSec = 1

    ExcelStatus gSendCtx, T("Campagne : ", "Campaign: ") & done & "/" & gSendTotal & _
                          T(" trait~es, prochain dans ", " processed, next in ") & waitSec & " s"
    modDefer.ArmUnique DEFER_SEND_KEY, waitSec
End Sub


' v12.1 (F5-01) : ligne a envoyer pour la position pos. La ligne prevue au
' lancement est verifiee par sa cle (destinataire) ; si elle ne correspond
' plus (tri, ligne inseree ou supprimee), la ligne est cherchee par sa cle
' (destinataire + objet). Envoi principal : une ligne dont le Statut n'est
' plus vide (deja envoyee, passee en brouillon entre-temps) est sautee,
' sauf le statut "en cours" pose par un essai precedent de cette position.
' 0 = ne pas envoyer (why = raison, pour le journal).
Private Function LocateSendRow(ByVal pos As Long, ByRef why As String) As Long
    On Error Resume Next
    LocateSendRow = 0
    why = ""
    Dim rw As Long, key As String, cur As String
    rw = gSendRows(pos)
    key = gSendKeys(pos)
    cur = LCase$(Trim$(CleanAddressList(Cell(gSendCtx, rw, R_TO))))
    If Len(key) = 0 Then
        why = "cle vide"
        Exit Function
    End If
    If cur <> key Then
        Dim found As Long
        found = FindRowByKey(gSendCtx, 0, key, gSendSubj(pos))
        If found = 0 Then
            why = "ligne deplacee ou modifiee : introuvable par sa cle"
            Exit Function
        End If
        modDebugLog.DInfo MOD_NAME, "Envoi : ligne " & rw & " retrouvee en " & found & " (tri ou insertion pendant l'envoi)"
        rw = found
    End If
    If gSendKind = KIND_MAIN Then
        Dim st As String
        st = Trim$(Cell(gSendCtx, rw, R_STATUS))
        If Len(st) > 0 Then
            If Len(gSendClaim) = 0 Or st <> gSendClaim Then
                why = "statut deja renseigne"
                Exit Function
            End If
        End If
    End If
    LocateSendRow = rw
End Function


' v12.1 (F5-02) : enregistrement intermediaire du classeur pendant un envoi
' long (un plantage d'Excel ou une fermeture sans enregistrer ne doit pas
' effacer les statuts deja poses). Jamais d'erreur visible.
Private Sub SaveSendProgress()
    On Error Resume Next
    If gSendCtx.Wb Is Nothing Then Exit Sub
    Dim ro As Boolean, pth As String
    ro = True
    ro = CBool(gSendCtx.Wb.ReadOnly)
    pth = ""
    pth = CStr(gSendCtx.Wb.Path)
    If ro Or Len(pth) = 0 Then Exit Sub
    modExcelBridge.SaveWorkbookVisible gSendCtx.XlApp, gSendCtx.Wb
    modDebugLog.DDebug MOD_NAME, "Envoi : classeur enregistre (" & gSendOk & " envoye(s))"
End Sub


' Construit et envoie la ligne (mail ou relance). True si .Send effectue.
Private Function SendRowNow(ByVal rw As Long) As Boolean
    On Error GoTo ErrH
    SendRowNow = False
    Dim mail As Object
    If gSendKind = KIND_FU Then
        Set mail = BuildFollowUp(gSendCtx, rw)
    Else
        Set mail = BuildMail(gSendCtx, rw)
    End If
    If mail Is Nothing Then Exit Function

    Dim ko As String
    ko = UnresolvedAddresses(mail)
    If Len(ko) > 0 Then
        gLastFailure = T("adresse non r~esolue (", "unresolved address (") & ko & ")"
        mail.Close 1
        Exit Function
    End If
    mail.Send
    modDebugLog.DInfo MOD_NAME, "Ligne " & rw & " : .Send effectue (" & gSendKind & ")"
    SendRowNow = True
    Exit Function
ErrH:
    gLastFailure = "err " & Err.Number
    modDebugLog.DError MOD_NAME, "SendRowNow ligne " & rw & " : " & Err.Number & " - " & Err.Description
    SendRowNow = False
End Function


' Fin de campagne (normale ou arret) : etat remis a zero, recapitulatif,
' rappel du classeur (ou enregistrement + fermeture s'il etait ferme).
Private Sub FinishSend()
    On Error Resume Next
    Dim ok As Long, ko As Long, total As Long, done As Long
    ok = gSendOk
    ko = gSendKo
    total = gSendTotal
    done = gSendPos - 1
    If done < 0 Then done = 0
    Dim skipped As Long, reason As String
    skipped = gSendSkip
    reason = gSendAbortReason
    gSendActive = False
    gSendTotal = 0
    gSendPos = 0
    gSendSkip = 0
    gSendAbortReason = ""
    gSendBusyTries = 0

    ' v12.1 (F5-02) : statut "en cours" pose pour une ligne finalement non
    ' envoyee (arret sur Excel occupe) : retire si Excel le permet, la ligne
    ' redevient envoyable.
    If Len(gSendClaim) > 0 And gSendClaimRow > 0 And gSendKind = KIND_MAIN Then
        If Trim$(Cell(gSendCtx, gSendClaimRow, R_STATUS)) = gSendClaim Then PutCell gSendCtx, gSendClaimRow, R_STATUS, ""
    End If
    gSendClaim = ""
    gSendClaimRow = 0

    ExcelStatus gSendCtx, ""
    modDebugLog.DInfo MOD_NAME, "Envoi termine : ok=" & ok & " ko=" & ko & " sautees=" & skipped & " (" & done & "/" & total & ")"

    modUi.Toast T("Envoi termin~e : ", "Send complete: ") & ok & T(" mail(s) envoy~e(s).", " mail(s) sent.") & _
                 IIf(ko > 0, vbLf & ko & T(" ~echec(s) (voir Statut / Contr~ole).", " failure(s) (see Status / Check)."), "") & _
                 IIf(skipped > 0, vbLf & skipped & T(" ligne(s) saut~ee(s) : d~ej~a trait~ee(s) ou d~eplac~ee(s) pendant l'envoi.", _
                                                     " row(s) skipped: already processed or moved during the send."), "") & _
                 IIf(done < total, vbLf & (total - done) & T(" non envoy~e(s) (arr~xt).", " not sent (stopped)."), "") & _
                 IIf(Len(reason) > 0, vbLf & reason, ""), _
                 IIf(Len(reason) > 0, uiError, IIf(ko > 0 Or skipped > 0, uiWarn, uiSuccess)), True

    If gSendCtx.Loaded Then
        Dim ctx As CampaignCtx
        ctx = gSendCtx
        ReleaseContext ctx, gSendWasOpen, gSendCreatedByUs, True
    End If
    ClearSendContext
    ' v12.1 (F5-09) : dernieres copies envoyees du compte secondaire.
    If gSendSecondary Then modDefer.ArmUnique DEFER_SENT_KEY, SWEEP_AFTER_SEND_SEC
    gSendSecondary = False
End Sub


Private Sub ClearSendContext()
    On Error Resume Next
    Dim blank As CampaignCtx
    gSendCtx = blank
End Sub

End Function

End Function

End Function

End Function

' Relit le fil de chaque ligne envoyee : Reponses, Derniere reponse, Suivi,
' _ReplyID. Lignes envoyees avant v12 (sans identifiant) : mail retrouve
' dans Elements envoyes par objet + destinataire.
' v12.1 : (F5-09) envois non encore enregistres rattrapes d'abord ; (F5-23)
' identifiant perime (mail range, archive, dossiers fusionnes) : mail
' retrouve et identifiants reecrits ; (F5-08) une reponse l'emporte sur un
' rapport de non-remise ; (F5-06) "Relance en brouillon" conserve tant
' qu'aucune reponse ni relance envoyee n'est vue.
Private Sub RefreshReplies(ByRef ctx As CampaignCtx, ByVal interactive As Boolean)
    On Error GoTo ErrH
    If ctx.Legacy Then
        If interactive Then
            modUi.Inform T("Feuille ancienne (sans colonnes de suivi) : utilise le classeur des campagnes (", _
                           "Legacy sheet (no tracking columns): use the campaigns workbook (") & _
                         modConfig.CampaignWorkbookFilename() & ").", T("Campagnes", "Campaigns"), vbExclamation
        End If
        Exit Sub
    End If

    ' F5-09 : brouillons envoyes a la main depuis un compte secondaire (ou
    ' pendant qu'Outlook ne surveillait pas les envois) : enregistres avant
    ' la lecture des statuts.
    SweepForContext ctx

    Dim lastRow As Long
    lastRow = LastDataRow(ctx)
    If lastRow <= ctx.HeaderRow Then Exit Sub
    Dim first As Long, nb As Long
    first = ctx.HeaderRow + 1
    nb = lastRow - ctx.HeaderRow

    Dim stB As Variant, cvB As Variant, idB As Variant, soB As Variant, fuB As Variant, trB As Variant
    stB = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_STATUS), lastRow, ctx.Map(R_STATUS))
    cvB = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_CONV), lastRow, ctx.Map(R_CONV))
    idB = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_ENTRY), lastRow, ctx.Map(R_ENTRY))
    soB = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_SENTON), lastRow, ctx.Map(R_SENTON))
    fuB = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_FOLLOWUPS), lastRow, ctx.Map(R_FOLLOWUPS))
    trB = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_TRACK), lastRow, ctx.Map(R_TRACK))

    Dim repB() As Variant, lastB() As Variant, ridB() As Variant
    ReDim repB(1 To nb, 1 To 1)
    ReDim lastB(1 To nb, 1 To 1)
    ReDim ridB(1 To nb, 1 To 1)
    Dim repOld As Variant, lastOld As Variant, ridOld As Variant
    repOld = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_REPLIES), lastRow, ctx.Map(R_REPLIES))
    lastOld = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_LASTREPLY), lastRow, ctx.Map(R_LASTREPLY))
    ridOld = modExcelBridge.ReadBlock(ctx.Ws, first, ctx.Map(R_REPLYID), lastRow, ctx.Map(R_REPLYID))

    Dim ns As Object
    Set ns = Application.GetNamespace("MAPI")

    ' F5-23 : dossiers ou chercher un mail deplace (resolus une fois).
    Dim filingFld As Object, nFiling As Long, subjFolders As Object
    Set filingFld = ResolveFilingFolder(ParamValue(ctx, P_FOLDER), nFiling)
    Set subjFolders = CreateObject("Scripting.Dictionary")

    Dim i As Long, rw As Long, st As String, eid As String, convID As String, sentOn As Date, fuCount As Long
    Dim replies As Long, lastDate As Date, lastEID As String, ndr As Boolean
    Dim nReplied As Long, nNone As Long, nNdr As Long, nLost As Long, nScanned As Long, nRelinked As Long
    Dim okScan As Boolean, oldEid As String, curTrack As String

    For i = 1 To nb
        rw = ctx.HeaderRow + i
        repB(i, 1) = repOld(i, 1)
        lastB(i, 1) = lastOld(i, 1)
        ridB(i, 1) = ridOld(i, 1)
        st = SafeText(stB(i, 1))
        If Left$(st, 6) = "Envoye" Then
            nScanned = nScanned + 1
            If (nScanned Mod 5) = 0 Then
                ExcelStatus ctx, T("Campagne : lecture des r~eponses ", "Campaign: reading replies ") & nScanned & "..."
                DoEvents
            End If
            eid = SafeText(idB(i, 1))
            convID = SafeText(cvB(i, 1))
            curTrack = SafeText(trB(i, 1))
            sentOn = 0
            fuCount = 0
            On Error Resume Next
            sentOn = CDate(soB(i, 1))
            fuCount = CLng(Val(CStr(fuB(i, 1))))
            On Error GoTo ErrH

            okScan = False
            If Len(eid) > 0 Then
                okScan = ScanThread(ns, eid, convID, sentOn, replies, lastDate, lastEID, ndr)
            End If
            If Not okScan Then
                ' Ligne envoyee avant v12 (sans identifiant) ou, v12.1 (F5-23),
                ' identifiant perime : le mail a ete range, archive ou ses
                ' dossiers fusionnes apres l'envoi. Recherche par objet +
                ' destinataire + date d'envoi, puis identifiants reecrits.
                oldEid = eid
                eid = ""
                If RecoverSentMail(ctx, rw, st, eid, convID, sentOn, filingFld, subjFolders) Then
                    If eid <> oldEid Then
                        PutCell ctx, rw, R_ENTRY, eid
                        PutCell ctx, rw, R_CONV, convID
                        PutCell ctx, rw, R_SENTON, sentOn
                        If Len(oldEid) > 0 Then nRelinked = nRelinked + 1
                        okScan = ScanThread(ns, eid, convID, sentOn, replies, lastDate, lastEID, ndr)
                    End If
                End If
            End If

            If okScan Then
                ' F5-02 : mail retrouve -> un statut "en cours" devient "Envoye le".
                If Left$(st, Len(ST_SENDING)) = ST_SENDING And sentOn > 0 Then
                    PutCell ctx, rw, R_STATUS, ST_SENT & Format$(sentOn, "dd/mm/yyyy hh:nn:ss")
                End If
                repB(i, 1) = replies
                If lastDate > 0 Then lastB(i, 1) = lastDate Else lastB(i, 1) = ""
                ridB(i, 1) = lastEID
                ' F5-08 : une reponse prouve la remise (avant "Non remis").
                If replies > 0 Then
                    trB(i, 1) = TrkReplied()
                    nReplied = nReplied + 1
                ElseIf ndr Then
                    trB(i, 1) = TrkNdr()
                    nNdr = nNdr + 1
                ElseIf fuCount > 0 Then
                    trB(i, 1) = TrkFollowed()
                    nNone = nNone + 1
                ElseIf curTrack = TrkFuDraft() Then
                    ' F5-06 : relance encore en brouillon : statut conserve (sinon
                    ' un nouveau "Relancer" recreerait un second brouillon).
                    nNone = nNone + 1
                Else
                    trB(i, 1) = TrkNoReply()
                    nNone = nNone + 1
                End If
            Else
                trB(i, 1) = TrkLost()
                nLost = nLost + 1
            End If
        End If
    Next i

    ctx.Ws.Range(ctx.Ws.Cells(first, ctx.Map(R_REPLIES)), ctx.Ws.Cells(lastRow, ctx.Map(R_REPLIES))).Value = repB
    ctx.Ws.Range(ctx.Ws.Cells(first, ctx.Map(R_LASTREPLY)), ctx.Ws.Cells(lastRow, ctx.Map(R_LASTREPLY))).Value = lastB
    ctx.Ws.Range(ctx.Ws.Cells(first, ctx.Map(R_LASTREPLY)), ctx.Ws.Cells(lastRow, ctx.Map(R_LASTREPLY))).NumberFormat = "dd/mm/yyyy hh:mm"
    ctx.Ws.Range(ctx.Ws.Cells(first, ctx.Map(R_REPLYID)), ctx.Ws.Cells(lastRow, ctx.Map(R_REPLYID))).Value = ridB
    ctx.Ws.Range(ctx.Ws.Cells(first, ctx.Map(R_TRACK)), ctx.Ws.Cells(lastRow, ctx.Map(R_TRACK))).Value = trB
    ExcelStatus ctx, ""

    Dim msg As String
    msg = nScanned & T(" envoy~e(s) : ", " sent: ") & nReplied & T(" r~epondu, ", " replied, ") & nNone & T(" sans r~eponse", " unanswered") & _
          IIf(nNdr > 0, ", " & nNdr & T(" non remis", " undeliverable"), "") & _
          IIf(nLost > 0, ", " & nLost & T(" introuvable(s)", " not found"), "") & _
          IIf(nRelinked > 0, vbLf & nRelinked & T(" mail(s) d~eplac~e(s) retrouv~e(s).", " moved mail(s) found again."), "")
    modDebugLog.DInfo MOD_NAME, "Reponses " & ctx.SheetName & " : " & Replace(msg, vbLf, " ")
    If interactive Then
        modUi.Inform msg, T("Campagnes - r~eponses", "Campaigns - replies")
    Else
        modUi.Toast msg, uiInfo, True
    End If
    Exit Sub
ErrH:
    ExcelStatus ctx, ""
    modDebugLog.DError MOD_NAME, "RefreshReplies " & Err.Number & " - " & Err.Description
    modUi.Toast T("Actualisation des r~eponses : erreur ", "Replies refresh: error ") & Err.Number, uiError, True
End Sub


' Mail envoye d'une ligne sans identifiant valable : meme objet, destinataire
' present, date la plus proche. True si trouve (eid / convID / sentOn
' renseignes). v12.1 (F5-23) : cherche aussi hors des Elements envoyes
' (mail range ou archive apres l'envoi) : dossier de classement de la
' campagne (filingFld), puis dossier du code trouve dans l'objet ; date de
' reference = _SentOn quand il est connu (ecart < 1 jour), sinon la date du
' statut (ecart < 2 jours, regle des lignes d'avant v12). Hors des Elements
' envoyes, seul un mail parti de mes adresses est retenu.
' subjFolders : cache objet -> dossier du code (partage par l'appelant).
Private Function RecoverSentMail(ByRef ctx As CampaignCtx, ByVal rw As Long, ByVal st As String, _
                                 ByRef eid As String, ByRef convID As String, ByRef sentOn As Date, _
                                 ByVal filingFld As Object, ByVal subjFolders As Object) As Boolean
    On Error Resume Next
    RecoverSentMail = False
    Dim subj As String, adr As String
    subj = EffectiveSubject(ctx, rw)
    adr = LCase$(Trim$(Split(CleanAddressList(Cell(ctx, rw, R_TO)) & ";", ";")(0)))
    If Len(subj) = 0 Or Len(adr) = 0 Then Exit Function

    Dim refDate As Date, maxGap As Double
    If sentOn > 0 Then
        refDate = sentOn
        maxGap = 1
    Else
        refDate = 0
        Dim p As Long
        p = InStr(st, " le ")
        If p > 0 Then refDate = CDate(Mid$(st, p + 4))
        If refDate = 0 Then refDate = Now
        maxGap = 2
    End If

    ' Dossiers fouilles, dans l'ordre (chacun une seule fois).
    Dim flds As Collection, seenF As Object, f As Object
    Set flds = New Collection
    Set seenF = CreateObject("Scripting.Dictionary")
    Set f = Nothing
    If Not ctx.Store Is Nothing Then Set f = modStoreContext.SysFolder(ctx.Store, OL_FOLDER_SENT)
    If f Is Nothing Then Set f = Application.GetNamespace("MAPI").GetDefaultFolder(OL_FOLDER_SENT)
    AddFolderOnce flds, seenF, f
    AddFolderOnce flds, seenF, filingFld
    If Not subjFolders Is Nothing Then
        If Not subjFolders.Exists(subj) Then subjFolders.Add subj, FolderFromSubjectCode(subj)
        Set f = Nothing
        Set f = subjFolders(subj)
        AddFolderOnce flds, seenF, f
    End If
    If flds.Count = 0 Then Exit Function

    Dim filtered As Object, itm As Object, best As Object, gap As Double, bestGap As Double
    Dim i As Long, n As Long, k As Long, cls As Long, mine As Boolean
    bestGap = 1E+30
    For k = 1 To flds.Count
        Set filtered = Nothing
        Set filtered = flds.Item(k).Items.Restrict("[Subject] = '" & Replace(subj, "'", "''") & "'")
        n = 0
        If Not filtered Is Nothing Then n = filtered.Count
        For i = 1 To n
            Set itm = Nothing
            Set itm = filtered.Item(i)
            cls = 0
            If Not itm Is Nothing Then cls = itm.Class
            If cls = OL_MAIL_CLASS Then
                If InStr(1, LCase$(CStr(itm.To)), adr, vbTextCompare) > 0 Or RecipientsContain(itm, adr) Then
                    mine = True
                    If k > 1 Then mine = modSmtp.IsFromMe(itm)
                    If mine Then
                        gap = 1E+30
                        gap = Abs(CDbl(itm.SentOn) - CDbl(refDate))
                        If gap < bestGap And gap < maxGap Then
                            bestGap = gap
                            Set best = itm
                        End If
                    End If
                End If
            End If
        Next i
        If Not best Is Nothing Then Exit For
    Next k
    If best Is Nothing Then Exit Function
    eid = CStr(best.EntryID)
    convID = CStr(best.ConversationID)
    sentOn = best.SentOn
    RecoverSentMail = (Len(eid) > 0)
    modDebugLog.DInfo MOD_NAME, "Ligne " & rw & " : mail envoye retrouve (" & Format$(sentOn, "dd/mm hh:nn") & _
                                IIf(k > 1, ", hors Elements envoyes", "") & ")"
End Function


' Ajoute fld a la liste s'il n'y est pas deja (meme EntryID).
Private Sub AddFolderOnce(ByVal flds As Collection, ByVal seen As Object, ByVal fld As Object)
    On Error Resume Next
    If fld Is Nothing Then Exit Sub
    Dim fid As String
    fid = ""
    fid = CStr(fld.EntryID)
    If Len(fid) = 0 Then Exit Sub
    If seen.Exists(fid) Then Exit Sub
    seen.Add fid, True
    flds.Add fld
End Sub


' v12.1 (F5-23) : dossier unique du code trouve dans un objet, par la meme
' detection que le classement automatique des envoyes (ExtractPrefillSilent
' sur un mail tampon jamais enregistre ni affiche). Nothing si aucun code
' ou plusieurs dossiers.
Private Function FolderFromSubjectCode(ByVal subj As String) As Object
    On Error Resume Next
    Set FolderFromSubjectCode = Nothing
    If Len(Trim$(subj)) = 0 Then Exit Function
    Dim probe As Object, code As String
    Set probe = Nothing
    Set probe = Application.CreateItem(0)      ' olMailItem
    If probe Is Nothing Then Exit Function
    probe.Subject = subj
    code = ""
    code = modQuickMove.ExtractPrefillSilent(probe)
    probe.Close 1                              ' olDiscard
    Set probe = Nothing
    If Len(code) = 0 Then Exit Function
    If Not modQuickMove.IsIndexLoaded() Then modQuickMove.SmartLoad
    Dim idx() As Long, n As Long
    n = CodeMatches(code, idx)
    If n = 1 Then Set FolderFromSubjectCode = modQuickMove.IndexFolderObject(idx(1))
End Function


' v12.1 : dossier(s) d'un code DETECTE automatiquement dans un objet. Meme
' regle que le classement automatique des envoyes (modQuickMove v12.2,
' F3-12) : un code numerique ne designe pas un dossier ou il est colle a
' d'autres chiffres ("1500" ne range plus dans "... 01500 ...") ; repli sur
' le chemin entier pour un code non numerique seulement. Le "Dossier de
' classement" saisi dans le bloc reste une recherche libre
' (ResolveFilingFolder).
Private Function CodeMatches(ByVal code As String, ByRef idx() As Long) As Long
    CodeMatches = modQuickMove.FindCodeMatches(code, idx)
    If CodeMatches = 0 And (code Like "*[!0-9]*") Then CodeMatches = modQuickMove.FindMatches(code, idx)
End Function


Private Function RecipientsContain(ByVal itm As Object, ByVal adrLower As String) As Boolean
    On Error Resume Next
    RecipientsContain = False
    Dim recips As Object, r As Object, i As Long, found As String
    Set recips = itm.Recipients
    For i = 1 To recips.Count
        Set r = recips.Item(i)
        found = LCase$(CStr(r.Address))
        If found = adrLower Then RecipientsContain = True: Exit Function
        found = ""
        found = LCase$(CStr(r.AddressEntry.GetExchangeUser.PrimarySmtpAddress))
        If found = adrLower Then RecipientsContain = True: Exit Function
    Next i
End Function


' Lit le fil du mail envoye (index de conversations Exchange). Reponses =
' mails recus apres l'envoi, pas de moi, hors reponses automatiques
' d'absence ; un rapport de non-remise -> ndr. False si le mail d'ancrage
' est introuvable (identifiant perime) ou sans index de conversations.
' v12.1 : (F5-08) nature du message par modSmtp.MessageClassKind : un
' accuse de remise ou de lecture (REPORT.* sans .NDR) est ignore, seul un
' rapport de NON-remise donne ndr ; (F5-22) reponses automatiques reconnues
' par modSmtp.IsAutoReplyItem (classe, en-tetes RFC 3834, objet, prefixe
' "[EXT]") ; "de moi" par modSmtp.IsFromMe (adresses et alias du profil).
Private Function ScanThread(ByVal ns As Object, ByVal anchorEID As String, ByVal convID As String, _
                            ByVal sentOn As Date, ByRef replies As Long, _
                            ByRef lastDate As Date, ByRef lastEID As String, ByRef ndr As Boolean) As Boolean
    On Error GoTo Dead
    ScanThread = False
    replies = 0
    lastDate = 0
    lastEID = ""
    ndr = False

    Dim anchor As Object
    Set anchor = ns.GetItemFromID(anchorEID)
    If anchor Is Nothing Then Exit Function
    If anchor.Class <> OL_MAIL_CLASS Then Exit Function
    If sentOn = 0 Then sentOn = anchor.SentOn

    Dim conv As Object
    Set conv = anchor.GetConversation
    If conv Is Nothing Then Exit Function

    Dim storeID As String
    storeID = CStr(anchor.Parent.StoreID)
    Dim tbl As Object
    Set tbl = conv.GetTable

    Dim guard As Long, rw As Object, eid As String, mc As String, itm As Object
    Dim rcv As Date, cls As Long, mcItem As String
    On Error Resume Next
    Do Until tbl.EndOfTable
        guard = guard + 1
        If guard > 300 Then Exit Do
        Set rw = Nothing
        Set rw = tbl.GetNextRow
        If rw Is Nothing Then Exit Do
        eid = ""
        eid = CStr(rw.Item("EntryID"))
        mc = ""
        mc = UCase$(CStr(rw.Item("MessageClass")))
        If Len(eid) > 0 And eid <> anchorEID Then
            Select Case modSmtp.MessageClassKind(mc)
                Case modSmtp.MSG_KIND_NDR
                    ndr = True
                Case modSmtp.MSG_KIND_NORMAL
                    Set itm = Nothing
                    Set itm = ns.GetItemFromID(eid, storeID)
                    If Not itm Is Nothing Then
                        cls = 0
                        cls = itm.Class
                        If cls = OL_REPORT_CLASS Then
                            ' Rapport ouvert : seul un rapport de NON-remise compte.
                            mcItem = ""
                            mcItem = CStr(itm.MessageClass)
                            If modSmtp.MessageClassKind(mcItem) = modSmtp.MSG_KIND_NDR Then ndr = True
                        ElseIf cls = OL_MAIL_CLASS Then
                            If itm.Sent Then
                                If Not modSmtp.IsAutoReplyItem(itm) Then
                                    If Not modSmtp.IsFromMe(itm) Then
                                        rcv = 0
                                        rcv = itm.ReceivedTime
                                        If rcv > sentOn Then
                                            replies = replies + 1
                                            If rcv > lastDate Then
                                                lastDate = rcv
                                                lastEID = eid
                                            End If
                                        End If
                                    End If
                                End If
                            End If
                        End If
                    End If
                Case Else
                    ' reponse automatique Exchange, accuse de remise ou de
                    ' lecture, reunion : ignore
            End Select
        End If
    Loop
    ScanThread = True
    Exit Function
Dead:
    modDebugLog.DDebug MOD_NAME, "ScanThread : ancrage mort ou fil indisponible (" & Err.Number & ")"
    ScanThread = False
End Function


'=============================================================================
' RELANCES
'=============================================================================

' mode "DRAFTS" : relances en brouillons (Suivi "Relance en brouillon").
' L'envoi direct passe par StartSend(KIND_FU). rowsArg : lignes cochees
' ("12;15") ou vide = toutes les lignes "Sans reponse" (le suivi est
' actualise d'abord si aucune ligne envoyee ne porte de Suivi).
Private Sub FollowUp(ByRef ctx As CampaignCtx, ByVal mode As String, ByVal rowsArg As String, ByVal interactive As Boolean)
    On Error GoTo ErrH
    If ctx.Legacy Then
        modUi.Toast T("Feuille ancienne : relances non disponibles.", "Legacy sheet: follow-ups not available."), uiWarn
        Exit Sub
    End If
    If Len(Trim$(rowsArg)) = 0 Then
        If Not TrackingPresent(ctx) Then RefreshReplies ctx, False
    End If

    Dim rows() As Long, n As Long, ok As Long, pbs As Long
    n = CandidateRows(ctx, KIND_FU, rowsArg, rows)
    If n = 0 Then
        modUi.Toast T("Aucune ligne ~a relancer (Suivi 'Sans r~eponse' ou lignes coch~ees).", "No row to follow up (Tracking 'Unanswered' or checked rows)."), uiInfo, True
        Exit Sub
    End If
    If interactive Then
        If modUi.Ask(n & T(" relance(s) en brouillon ?", " follow-up draft(s)?"), T("Campagnes - relances", "Campaigns - follow-ups"), vbQuestion + vbYesNo, False) <> vbYes Then Exit Sub
    End If

    ExcelStatus ctx, T("Campagne : v~erification des relances...", "Campaign: checking follow-ups...")
    pbs = Preflight(ctx, rows, n, KIND_FU, ok)
    If pbs > 0 Then
        ExcelStatus ctx, ""
        modUi.Toast pbs & T(" ligne(s) ~a corriger (colonne Contr~ole). Aucune relance cr~e~ee.", " row(s) to fix (Check column). No follow-up created."), uiWarn, True
        Exit Sub
    End If

    ' v12.1 (F5-06) : brouillon de relance deja present pour ces lignes
    ' (relance refaite) : supprime, un seul brouillon par ligne.
    Dim purged As Long
    purged = PurgeCampaignDrafts(ctx.WbName, ctx.SheetName, RowKeys(ctx, rows, n), KIND_FU)
    If purged > 0 Then modDebugLog.DInfo MOD_NAME, "Relances : " & purged & " ancien(s) brouillon(s) de relance supprime(s)"
    If IsSecondarySendStore(ctx) Then RegisterSweepStore ctx.Store   ' F5-09

    Dim drafts As Object
    Set drafts = TargetDraftsFolder(ctx)
    Dim i As Long, rw As Long, done As Long, failed As Long, fu As Object
    For i = 1 To n
        rw = rows(i)
        ExcelStatus ctx, T("Campagne : relance ligne ", "Campaign: follow-up row ") & rw & " (" & (i) & "/" & n & ")"
        Set fu = BuildFollowUp(ctx, rw)
        If fu Is Nothing Then
            PutCell ctx, rw, R_CHECK, T("relance : ", "follow-up: ") & gLastFailure
            failed = failed + 1
        Else
            SaveToDrafts fu, drafts
            PutCell ctx, rw, R_TRACK, TrkFuDraft()
            done = done + 1
        End If
        If (i Mod 10) = 0 Then DoEvents
    Next i
    ExcelStatus ctx, ""
    modDebugLog.DInfo MOD_NAME, "Relances " & ctx.SheetName & " : brouillons=" & done & " ko=" & failed
    modUi.Toast done & T(" relance(s) en brouillon (", " follow-up draft(s) (") & ctx.AccountLabel & ")" & _
                 IIf(failed > 0, vbLf & failed & T(" ~echec(s) : voir Contr~ole.", " failure(s): see Check."), ""), _
                 IIf(failed > 0, uiWarn, uiSuccess), True
    Exit Sub
ErrH:
    ExcelStatus ctx, ""
    modDebugLog.DError MOD_NAME, "FollowUp " & Err.Number & " - " & Err.Description
    modUi.Toast T("Relances : erreur ", "Follow-ups: error ") & Err.Number, uiError, True
End Sub

Public Function HandleSentItem(ByVal itm As Object) As Boolean

' Mail de campagne arrive dans des Elements envoyes : classement (jamais de
' fenetre), puis ligne mise en file pour l'ecriture Excel (gPendingSent).
' False si le payload est d'un ancien format (pas d'ecriture Excel).
Private Function QueueSentItem(ByVal itm As Object, ByVal payload As String) As Boolean
    On Error Resume Next
    QueueSentItem = False
    Set itm = FileSentItem(itm, payload)
    If gSweptIDs Is Nothing Then Set gSweptIDs = CreateObject("Scripting.Dictionary")
    gSweptIDs(CStr(itm.EntryID)) = True        ' F5-09 : jamais releve deux fois

    Dim f() As String
    f = Split(payload, vbTab)
    If UBound(f) < 4 Then
        modDebugLog.DTrace MOD_NAME, "Mail de campagne : payload ancien format, statut Excel non ecrit"
        Exit Function
    End If
    If gPendingSent Is Nothing Then Set gPendingSent = New Collection
    gPendingSent.Add Array(payload, CStr(itm.ConversationID), CStr(itm.EntryID), CDbl(itm.SentOn))
    QueueSentItem = True
End Function


' Range le mail de campagne envoye (dossier de classement, code de l'objet,
' dossier du mail d'origine pour une relance). Renvoie l'element APRES
' deplacement (l'identifiant change sur Exchange), sinon l'element recu.
Private Function FileSentItem(ByVal itm As Object, ByVal payload As String) As Object
    On Error Resume Next
    Set FileSentItem = itm
    Dim f() As String
    f = Split(payload, vbTab)
    Dim kind As String, folderText As String, origEID As String
    kind = KIND_MAIN
    If UBound(f) >= 5 Then kind = f(5)
    If UBound(f) >= 6 Then folderText = f(6)
    If UBound(f) >= 7 Then origEID = f(7)

    Dim target As Object
    Set target = FolderTargetFor(folderText, itm, kind, origEID)
    If target Is Nothing Then Exit Function
    Dim curID As String, tgtID As String
    curID = ""
    curID = CStr(itm.Parent.EntryID)
    tgtID = ""
    tgtID = CStr(target.EntryID)
    If Len(tgtID) = 0 Or curID = tgtID Then Exit Function
    Dim moved As Object
    Set moved = Nothing
    Set moved = itm.Move(target)
    If Not moved Is Nothing Then
        Set FileSentItem = moved
        modDebugLog.DInfo MOD_NAME, "Mail de campagne range : " & target.Name
    Else
        modDebugLog.DError MOD_NAME, "Mail de campagne : Move refuse vers " & target.Name
    End If
End Function


' Dossier cible d'un mail de campagne envoye :
'   1. parametre "Dossier de classement" (index QuickMove : feuille unique,
'      sinon chemin unique) ;
'   2. code trouve dans l'objet / le corps (meme detection que le classement
'      automatique des envoyes) : dossier unique ;
'   3. relance : dossier du mail d'origine ;
'   4. sinon Nothing (le mail reste dans Envoyes).
Private Function FolderTargetFor(ByVal folderText As String, ByVal itm As Object, ByVal kind As String, _
                                 ByVal origEID As String) As Object
    On Error Resume Next
    Set FolderTargetFor = Nothing
    If Not modQuickMove.IsIndexLoaded() Then modQuickMove.SmartLoad

    Dim idx() As Long, n As Long, txt As String
    txt = Trim$(folderText)
    If Len(txt) > 0 Then
        Set FolderTargetFor = ResolveFilingFolder(txt, n)
        If Not FolderTargetFor Is Nothing Then Exit Function
        If n <> 1 Then modDebugLog.DInfo MOD_NAME, "Dossier de classement '" & txt & "' : " & n & " correspondance(s), ignore"
    End If

    Dim prefill As String
    prefill = modQuickMove.ExtractPrefillSilent(itm)
    If Len(prefill) > 0 Then
        n = CodeMatches(prefill, idx)
        If n = 1 Then
            Set FolderTargetFor = modQuickMove.IndexFolderObject(idx(1))
            If Not FolderTargetFor Is Nothing Then Exit Function
        End If
    End If

    If kind = KIND_FU And Len(origEID) > 0 Then
        Dim orig As Object
        Set orig = Application.Session.GetItemFromID(origEID)
        If Not orig Is Nothing Then Set FolderTargetFor = orig.Parent
    End If
End Function


' v12.1 (F5-13) : dossier designe par le parametre "Dossier de classement"
' (index QuickMove : nom de dossier unique, sinon chemin unique). Meme
' resolution au pre-vol (CampaignLevelProblems), au classement des envoyes
' (FolderTargetFor) et pour retrouver un mail deplace (F5-23).
' n = nombre de dossiers correspondants (1 attendu). Nothing si 0, plusieurs,
' ou dossier inaccessible.
Private Function ResolveFilingFolder(ByVal folderText As String, ByRef n As Long) As Object
    On Error Resume Next
    Set ResolveFilingFolder = Nothing
    n = 0
    Dim txt As String
    txt = Trim$(folderText)
    If Len(txt) = 0 Then Exit Function
    If Not modQuickMove.IsIndexLoaded() Then modQuickMove.SmartLoad
    Dim idx() As Long
    n = modQuickMove.FindMatchesLeaf(txt, idx)
    If n <> 1 Then n = modQuickMove.FindMatches(txt, idx)
    If n = 1 Then Set ResolveFilingFolder = modQuickMove.IndexFolderObject(idx(1))
End Function


' Cible modDefer "CampaignSent" : ecrit dans le classeur les envois detectes.
' v12.1 : (F5-09) releve d'abord les Elements envoyes des comptes
' secondaires de la session ; (F5-10) un seul contexte (donc un seul
' enregistrement et un seul rappel du classeur) par feuille, quel que soit
' le nombre de mails du lot.
Public Sub CampaignSentFlush()
    On Error Resume Next
    SweepRegisteredStores
    If gPendingSent Is Nothing Then Exit Sub
    If gPendingSent.Count = 0 Then Exit Sub
    Dim batch As Collection
    Set batch = gPendingSent
    Set gPendingSent = New Collection

    Dim groups As Object, grp As Collection, i As Long, it As Variant, f() As String, gk As String
    Set groups = CreateObject("Scripting.Dictionary")
    For i = 1 To batch.Count
        it = batch.Item(i)
        f = Split(CStr(it(0)), vbTab)
        If UBound(f) >= 4 Then
            gk = LCase$(LeafName(f(0))) & vbTab & LCase$(f(1))
            If Not groups.Exists(gk) Then
                Set grp = New Collection
                groups.Add gk, grp
            End If
            groups(gk).Add it
        End If
    Next i
    Dim k As Variant
    For Each k In groups.Keys
        Set grp = groups(k)
        WriteSentGroup grp
    Next k
End Sub


' v12.1 (F5-10) : ecrit les envois d'une meme feuille avec UN seul contexte.
' Envoi automatique en cours sur cette feuille : ecriture directe dans le
' contexte de l'envoi (le classeur est peut-etre ouvert en silence dans une
' instance Excel a nous : le rouvrir ailleurs le passerait en lecture
' seule). Sinon classeur retrouve par nom/chemin, ou rouvert en silence
' (enregistre puis referme une seule fois).
Private Sub WriteSentGroup(ByVal items As Collection)
    On Error GoTo ErrH
    If items Is Nothing Then Exit Sub
    If items.Count = 0 Then Exit Sub
    Dim it As Variant, f() As String, wbRef As String, sheet As String
    it = items.Item(1)
    f = Split(CStr(it(0)), vbTab)
    wbRef = f(0)
    sheet = f(1)

    Dim ctx As CampaignCtx
    Dim wasOpen As Boolean, createdByUs As Boolean, viaSend As Boolean, acquired As Boolean
    viaSend = False
    If gSendActive And gSendCtx.Loaded Then
        If PayloadTargets(gSendCtx, wbRef, sheet) Then
            ctx = gSendCtx
            viaSend = True
        End If
    End If
    If Not viaSend Then
        If Not AcquireContext(wbRef, sheet, ctx, wasOpen, createdByUs) Then
            modDebugLog.DError MOD_NAME, "Envois a enregistrer : classeur/feuille inaccessible - " & sheet & _
                                         " (" & items.Count & " mail(s))"
            Exit Sub
        End If
        acquired = True
    End If

    Dim i As Long, nOk As Long
    For i = 1 To items.Count
        it = items.Item(i)
        If WriteSentRow(ctx, CStr(it(0)), CStr(it(1)), CStr(it(2)), CDate(it(3))) Then nOk = nOk + 1
    Next i
    If items.Count > 1 Then modDebugLog.DInfo MOD_NAME, "Envois enregistres en un lot : " & nOk & "/" & items.Count & " (" & sheet & ")"
    If acquired Then ReleaseContext ctx, wasOpen, createdByUs, (nOk > 0)
    Exit Sub
ErrH:
    modDebugLog.DError MOD_NAME, "WriteSentGroup " & Err.Number & " - " & Err.Description
    On Error Resume Next
    If acquired Then ReleaseContext ctx, wasOpen, createdByUs, True
End Sub


' Le payload (classeur, feuille) vise-t-il la feuille de ctx ?
Private Function PayloadTargets(ByRef ctx As CampaignCtx, ByVal wbRef As String, ByVal sheet As String) As Boolean
    PayloadTargets = False
    If StrComp(ctx.SheetName, sheet, vbTextCompare) <> 0 Then Exit Function
    If StrComp(ctx.WbPath, wbRef, vbTextCompare) = 0 Then
        PayloadTargets = True
        Exit Function
    End If
    PayloadTargets = (StrComp(ctx.WbName, LeafName(wbRef), vbTextCompare) = 0)
End Function


' Ecrit identifiants, date d'envoi, statut (envoi manuel) ou compteur de
' relances dans la ligne referencee par le payload, dans un contexte DEJA
' charge (v12.1, F5-10 : l'acquisition et la liberation sont faites une
' fois par feuille par l'appelant). True si la ligne a ete trouvee.
'   - mail principal : statut vide ou "Brouillon..." -> "Envoye manuellement
'     le" ; v12.1 (F5-02) statut "Envoye (en cours) le" -> "Envoye le" ;
'   - relance : compteur + date ; v12.1 : une relance deja comptee (date de
'     derniere relance >= date d'envoi) ne l'est pas deux fois (releve des
'     Elements envoyes, F5-09).
Private Function WriteSentRow(ByRef ctx As CampaignCtx, ByVal payload As String, ByVal convID As String, _
                              ByVal eid As String, ByVal sentOn As Date) As Boolean
    On Error GoTo ErrH
    WriteSentRow = False
    Dim f() As String
    f = Split(payload, vbTab)
    If UBound(f) < 4 Then Exit Function
    Dim rowHint As Long, toKey As String, subjKey As String, kind As String
    rowHint = CLng(Val(f(2))): toKey = f(3): subjKey = f(4)
    kind = KIND_MAIN
    If UBound(f) >= 5 Then kind = f(5)

    Dim rw As Long
    rw = FindRowByKey(ctx, rowHint, toKey, subjKey)
    If rw = 0 Then
        modDebugLog.DError MOD_NAME, "Envoi a enregistrer : ligne introuvable (destinataire ou objet modifie ?) - " & ctx.SheetName
        Exit Function
    End If

    If kind = KIND_FU Then
        Dim prevFu As Variant, prevD As Date
        prevD = 0
        If HasRole(ctx, R_LASTFU) Then
            prevFu = ctx.Ws.Cells(rw, ctx.Map(R_LASTFU)).Value
            If IsDate(prevFu) Then prevD = CDate(prevFu)
        End If
        If prevD > 0 And sentOn <= prevD + TimeSerial(0, 0, 1) Then
            modDebugLog.DDebug MOD_NAME, "Relance deja enregistree : ligne " & rw
        Else
            Dim nFu As Long
            nFu = CLng(Val(Cell(ctx, rw, R_FOLLOWUPS))) + 1
            PutCell ctx, rw, R_FOLLOWUPS, nFu
            PutCell ctx, rw, R_LASTFU, sentOn
            If HasRole(ctx, R_LASTFU) Then ctx.Ws.Cells(rw, ctx.Map(R_LASTFU)).NumberFormat = "dd/mm/yyyy hh:mm"
            PutCell ctx, rw, R_TRACK, TrkFollowed()
            modDebugLog.DInfo MOD_NAME, "Relance enregistree : ligne " & rw & " (" & nFu & ")"
        End If
    Else
        Dim st As String
        st = Trim$(Cell(ctx, rw, R_STATUS))
        If Len(st) = 0 Or LCase$(Left$(st, 9)) = "brouillon" Then
            PutCell ctx, rw, R_STATUS, ST_SENT_MANUAL & Format$(sentOn, "dd/mm/yyyy hh:nn")
        ElseIf Left$(st, Len(ST_SENDING)) = ST_SENDING Then
            PutCell ctx, rw, R_STATUS, ST_SENT & Format$(sentOn, "dd/mm/yyyy hh:nn:ss")
        End If
        PutCell ctx, rw, R_CONV, convID
        PutCell ctx, rw, R_ENTRY, eid
        PutCell ctx, rw, R_SENTON, sentOn
        If HasRole(ctx, R_SENTON) Then ctx.Ws.Cells(rw, ctx.Map(R_SENTON)).NumberFormat = "dd/mm/yyyy hh:mm"
        If Len(Trim$(Cell(ctx, rw, R_TRACK))) = 0 Then PutCell ctx, rw, R_TRACK, TrkNoReply()
        modDebugLog.DInfo MOD_NAME, "Envoi enregistre : ligne " & rw
    End If
    WriteSentRow = True
    Exit Function
ErrH:
    modDebugLog.DError MOD_NAME, "WriteSentRow " & Err.Number & " - " & Err.Description
    WriteSentRow = False
End Function


' v12.1 (F5-09) : ecrit dans ctx (deja charge) les envois en file qui visent
' sa feuille ; les autres restent en file (differe "CampaignSent" arme).
' Renvoie le nombre de lignes mises a jour.
Private Function WritePendingFor(ByRef ctx As CampaignCtx) As Long
    On Error Resume Next
    WritePendingFor = 0
    If gPendingSent Is Nothing Then Exit Function
    If gPendingSent.Count = 0 Then Exit Function
    Dim rest As Collection, i As Long, it As Variant, f() As String, hit As Boolean
    Set rest = New Collection
    For i = 1 To gPendingSent.Count
        it = gPendingSent.Item(i)
        f = Split(CStr(it(0)), vbTab)
        hit = False
        If UBound(f) >= 4 Then hit = PayloadTargets(ctx, f(0), f(1))
        If hit Then
            If WriteSentRow(ctx, CStr(it(0)), CStr(it(1)), CStr(it(2)), CDate(it(3))) Then WritePendingFor = WritePendingFor + 1
        Else
            rest.Add it
        End If
    Next i
    Set gPendingSent = rest
    If rest.Count > 0 Then modDefer.ArmUnique DEFER_SENT_KEY, SENT_FLUSH_DELAY_SEC
End Function


' v12.1 (F5-09) : Elements envoyes des comptes secondaires notes pendant la
' session (envoi, brouillons, envoi manuel vu par CampaignOnItemSend) :
' mails de campagne des dernieres 24 h pas encore traites, mis en file.
Private Sub SweepRegisteredStores()
    On Error Resume Next
    If gSweepStores Is Nothing Then Exit Sub
    If gSweepStores.Count = 0 Then Exit Sub
    Dim k As Variant, st As Object, n As Long
    For Each k In gSweepStores.Keys
        Set st = Nothing
        Set st = gSweepStores(k)
        If Not st Is Nothing Then
            n = n + SweepSentFolder(modStoreContext.SysFolder(st, OL_FOLDER_SENT), Now - 1, Nothing, "", "")
        End If
    Next k
    If n > 0 Then modDebugLog.DInfo MOD_NAME, "Comptes secondaires : " & n & " mail(s) de campagne releve(s) dans les Elements envoyes"
End Sub


' v12.1 (F5-09 / F5-23) : avant "Actualiser les reponses", rattrape les
' envois que l'evenement d'Elements envoyes n'a pas vus : brouillon envoye
' a la main depuis un compte secondaire, ou pendant que les liaisons
' d'evenements etaient perdues. Ne tourne que si la feuille a des lignes
' "Brouillon cree le" ou "Relance en brouillon" ; periode = depuis le plus
' ancien de ces brouillons (90 jours au plus). Elements envoyes de la boite
' par defaut et du compte d'envoi de la campagne. Les mails dont
' l'identifiant figure deja dans _EntryID sont ignores.
Private Sub SweepForContext(ByRef ctx As CampaignCtx)
    On Error Resume Next
    If ctx.Legacy Then Exit Sub
    Dim lastRow As Long
    lastRow = LastDataRow(ctx)
    If lastRow <= ctx.HeaderRow Then Exit Sub
    Dim stB As Variant, trB As Variant, soB As Variant, idB As Variant
    stB = Empty: trB = Empty: soB = Empty: idB = Empty
    stB = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_STATUS), lastRow, ctx.Map(R_STATUS))
    trB = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_TRACK), lastRow, ctx.Map(R_TRACK))
    soB = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_SENTON), lastRow, ctx.Map(R_SENTON))
    idB = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_ENTRY), lastRow, ctx.Map(R_ENTRY))
    If Not IsArray(stB) Or Not IsArray(trB) Or Not IsArray(soB) Or Not IsArray(idB) Then Exit Sub

    Dim known As Object, i As Long, st As String, eid As String, d As Date, p As Long, since As Date
    Set known = CreateObject("Scripting.Dictionary")
    since = 0
    For i = 1 To UBound(stB, 1)
        st = SafeText(stB(i, 1))
        eid = SafeText(idB(i, 1))
        If Len(eid) > 0 Then known(eid) = True
        d = 0
        If LCase$(Left$(st, 9)) = "brouillon" Then
            p = InStr(st, " le ")
            If p > 0 Then d = CDate(Mid$(st, p + 4))
            If d = 0 Then d = Now - 30
        ElseIf SafeText(trB(i, 1)) = TrkFuDraft() Then
            If IsDate(soB(i, 1)) Then d = CDate(soB(i, 1))
            If d = 0 Then d = Now - 30
        End If
        If d > 0 Then
            If since = 0 Or d < since Then since = d
        End If
    Next i
    If since = 0 Then Exit Sub
    If since < Now - 90 Then since = Now - 90
    since = since - TimeSerial(0, 5, 0)

    If IsSecondarySendStore(ctx) Then RegisterSweepStore ctx.Store
    Dim n As Long, stDef As Object, isDef As Boolean
    Set stDef = modStoreContext.DefaultStore()
    n = SweepSentFolder(modStoreContext.SysFolder(stDef, OL_FOLDER_SENT), since, known, ctx.WbName, ctx.SheetName)
    If Not ctx.Store Is Nothing Then
        isDef = True
        isDef = modStoreContext.IsDefaultStore(ctx.Store)
        If Not isDef Then
            n = n + SweepSentFolder(modStoreContext.SysFolder(ctx.Store, OL_FOLDER_SENT), since, known, ctx.WbName, ctx.SheetName)
        End If
    End If
    If n = 0 Then Exit Sub
    Dim w As Long
    w = WritePendingFor(ctx)
    modDebugLog.DInfo MOD_NAME, "Releve des envoyes (" & ctx.SheetName & ") : " & n & " mail(s) de campagne rattrape(s), " & _
                                w & " ligne(s) mise(s) a jour"
End Sub


' v12.1 (F5-09) : mails de campagne des Elements envoyes fld envoyes depuis
' sinceDate et pas encore traites (identifiant absent de gSweptIDs et de
' skipIDs). Filtre serveur sur la propriete EmaileurCampagne (les mails
' ordinaires ne sont pas ouverts) ; repli : parcours de la periode, borne
' a SWEEP_SCAN_CAP elements. onlySheet non vide : seuls les mails de cette
' feuille (onlyWb = nom du classeur). Chaque mail retenu est range comme a
' l'arrivee puis mis en file d'ecriture. Renvoie le nombre mis en file.
Private Function SweepSentFolder(ByVal fld As Object, ByVal sinceDate As Date, ByVal skipIDs As Object, _
                                 ByVal onlyWb As String, ByVal onlySheet As String) As Long
    On Error Resume Next
    SweepSentFolder = 0
    If fld Is Nothing Then Exit Function
    If gSweptIDs Is Nothing Then Set gSweptIDs = CreateObject("Scripting.Dictionary")

    Dim byDate As Object, flt As Object
    Set byDate = Nothing
    Set byDate = fld.Items.Restrict("[SentOn] >= '" & modCommon.FormatDaslDate(sinceDate) & "'")
    If byDate Is Nothing Then Exit Function
    Dim i As Long, n As Long, e As Long
    Set flt = Nothing
    Set flt = byDate.Restrict("@SQL=NOT(""" & DASL_CAMPAIGN_PROP & """ IS NULL)")
    n = -1
    If Not flt Is Nothing Then
        Err.Clear
        n = flt.Count
        e = Err.Number
        Err.Clear
        If e <> 0 Then n = -1
    End If
    If n < 0 Then
        Set flt = byDate
        n = 0
        n = flt.Count
        modDebugLog.DDebug MOD_NAME, "Releve des envoyes : filtre sur la propriete refuse, parcours de la periode"
    End If
    flt.Sort "[SentOn]", False
    Err.Clear

    ' 1. Reperage (aucun deplacement pendant le parcours : les indices
    '    de la collection filtree bougeraient).
    Dim found As Collection, pls As Collection
    Set found = New Collection
    Set pls = New Collection
    Dim itm As Object, up As Object, cls As Long
    Dim eid As String, pl As String, f() As String, skip As Boolean
    If n > SWEEP_SCAN_CAP Then
        modDebugLog.DInfo MOD_NAME, "Releve des envoyes : " & n & " elements, seuls les " & SWEEP_SCAN_CAP & " plus anciens sont lus"
        n = SWEEP_SCAN_CAP
    End If
    For i = 1 To n
        Set itm = Nothing
        Set itm = flt.Item(i)
        cls = 0
        If Not itm Is Nothing Then cls = itm.Class
        If cls = OL_MAIL_CLASS Then
            Set up = Nothing
            Set up = itm.UserProperties.Find(UPROP_CAMPAIGN)
            pl = ""
            If Not up Is Nothing Then pl = CStr(up.Value)
            eid = ""
            If Len(pl) > 0 Then eid = CStr(itm.EntryID)
            If Len(eid) > 0 Then
                skip = gSweptIDs.Exists(eid)
                If Not skip Then
                    If Not skipIDs Is Nothing Then skip = skipIDs.Exists(eid)
                End If
                If Not skip And Len(onlySheet) > 0 Then
                    skip = True
                    f = Split(pl, vbTab)
                    If UBound(f) >= 1 Then
                        skip = Not (StrComp(LeafName(f(0)), onlyWb, vbTextCompare) = 0 And _
                                    StrComp(f(1), onlySheet, vbTextCompare) = 0)
                    End If
                End If
                If Not skip Then
                    found.Add itm
                    pls.Add pl
                End If
            End If
        End If
    Next i

    ' 2. Classement + mise en file.
    For i = 1 To found.Count
        Set itm = found.Item(i)
        If QueueSentItem(itm, CStr(pls.Item(i))) Then SweepSentFolder = SweepSentFolder + 1
    Next i
End Function


' Ligne du payload : indice d'abord, sinon balayage par cle (destinataire +
' objet effectif, insensible a la casse). v12.1 : la colonne destinataire
' est lue en UN bloc ; l'objet n'est compare que pour les lignes dont le
' destinataire correspond (une recherche par ligne pendant un envoi cadence
' apres un tri, F5-01).
Private Function FindRowByKey(ByRef ctx As CampaignCtx, ByVal rowHint As Long, ByVal toKey As String, ByVal subjKey As String) As Long
    On Error Resume Next
    FindRowByKey = 0
    If rowHint > ctx.HeaderRow Then
        If RowMatches(ctx, rowHint, toKey, subjKey) Then FindRowByKey = rowHint: Exit Function
    End If
    Dim lastRow As Long, r As Long, toB As Variant, hitTo As Boolean
    lastRow = LastDataRow(ctx)
    If lastRow <= ctx.HeaderRow Then Exit Function
    toB = Empty
    toB = modExcelBridge.ReadBlock(ctx.Ws, ctx.HeaderRow + 1, ctx.Map(R_TO), lastRow, ctx.Map(R_TO))
    If Not IsArray(toB) Then Exit Function
    For r = 1 To UBound(toB, 1)
        hitTo = (LCase$(Trim$(CleanAddressList(SafeText(toB(r, 1))))) = toKey)
        If hitTo Then
            If StrComp(EffectiveSubject(ctx, ctx.HeaderRow + r), subjKey, vbTextCompare) = 0 Then
                FindRowByKey = ctx.HeaderRow + r
                Exit Function
            End If
        End If
    Next r
End Function


Private Function RowMatches(ByRef ctx As CampaignCtx, ByVal rw As Long, ByVal toKey As String, ByVal subjKey As String) As Boolean
    On Error Resume Next
    RowMatches = False
    Dim vTo As String
    vTo = LCase$(Trim$(CleanAddressList(Cell(ctx, rw, R_TO))))
    If vTo <> toKey Then Exit Function
    RowMatches = (StrComp(EffectiveSubject(ctx, rw), subjKey, vbTextCompare) = 0)
End Function


'=============================================================================
' BROUILLONS DE CAMPAGNE (comptage / purge, toutes boites)
' wbName / sheet non vides : seuls les brouillons de CETTE feuille.
' v12.1 (F5-05 / F5-06) : keys (adresses en minuscules, 4e champ du
' marqueur) : seuls les brouillons de ces lignes ; kindFilter (M / F, 6e
' champ, M pour un ancien brouillon a 5 champs) : seuls ceux de ce genre.
'=============================================================================

Private Function CountCampaignDrafts(ByVal wbName As String, ByVal sheet As String) As Long
    CountCampaignDrafts = ScanCampaignDrafts(wbName, sheet, False)
End Function


Private Function PurgeCampaignDrafts(ByVal wbName As String, ByVal sheet As String, _
                                     Optional ByVal keys As Object = Nothing, _
                                     Optional ByVal kindFilter As String = "") As Long
    PurgeCampaignDrafts = ScanCampaignDrafts(wbName, sheet, True, keys, kindFilter)
End Function


Private Function ScanCampaignDrafts(ByVal wbName As String, ByVal sheet As String, ByVal doDelete As Boolean, _
                                    Optional ByVal keys As Object = Nothing, _
                                    Optional ByVal kindFilter As String = "") As Long
    On Error Resume Next
    Dim st As Object, fld As Object, itm As Object, up As Object, items As Object
    Dim c As Long, i As Long, n As Long, keep As Boolean, f() As String, pl As String, k As String
    For Each st In Application.Session.Stores
        Set fld = Nothing
        Set fld = st.GetDefaultFolder(OL_FOLDER_DRAFTS)
        If Not fld Is Nothing Then
            Set items = fld.Items
            n = 0
            n = items.Count
            For i = n To 1 Step -1
                Set itm = Nothing
                Set itm = items.Item(i)
                If Not itm Is Nothing Then
                    Set up = Nothing
                    Set up = itm.UserProperties.Find(UPROP_CAMPAIGN)
                    If Not up Is Nothing Then
                        ' Payload relu a chaque element (jamais celui du precedent).
                        pl = ""
                        pl = CStr(up.Value)
                        f = Split(pl, vbTab)
                        keep = True
                        If Len(wbName) > 0 Then
                            keep = False
                            If UBound(f) >= 1 Then
                                If StrComp(LeafName(f(0)), wbName, vbTextCompare) = 0 And StrComp(f(1), sheet, vbTextCompare) = 0 Then keep = True
                            End If
                        End If
                        If keep And Not keys Is Nothing Then
                            keep = False
                            If UBound(f) >= 3 Then keep = keys.Exists(LCase$(Trim$(f(3))))
                        End If
                        If keep And Len(kindFilter) > 0 Then
                            k = KIND_MAIN
                            If UBound(f) >= 5 Then k = f(5)
                            keep = (k = kindFilter)
                        End If
                        If keep Then
                            c = c + 1
                            If doDelete Then itm.Delete
                        End If
                    End If
                End If
            Next i
        End If
    Next st
    ScanCampaignDrafts = c
End Function


' v12.1 (F5-05 / F5-06) : cles des lignes rows(1..n) (adresse destinataire
' nettoyee, en minuscules : meme forme que le 4e champ du marqueur).
Private Function RowKeys(ByRef ctx As CampaignCtx, ByRef rows() As Long, ByVal n As Long) As Object
    Dim d As Object, i As Long, k As String
    Set d = CreateObject("Scripting.Dictionary")
    d.CompareMode = 1   ' TextCompare
    Set RowKeys = d
    On Error Resume Next
    For i = 1 To n
        k = LCase$(Trim$(CleanAddressList(Cell(ctx, rows(i), R_TO))))
        If Len(k) > 0 Then d(k) = rows(i)
    Next i
End Function

Private Function ChooseTargetWorkbook(ByVal xlApp As Object, ByRef wb As Object) As Long
    Dim nb As Long
    nb = xlApp.Workbooks.Count
    If nb = 0 Then ChooseTargetWorkbook = 1: Exit Function

    Dim listTxt As String, i As Long
    listTxt = T("O~u cr~eer la feuille ? Tape un num~ero :", "Where should the sheet be created? Type a number:") & vbCrLf & vbCrLf & _
              T("1 - Nouveau classeur", "1 - New workbook")
    For i = 1 To nb
        listTxt = listTxt & vbCrLf & (i + 1) & " - " & xlApp.Workbooks(i).Name
    Next i
    Dim s As String
    s = Trim$(modUi.AskText(listTxt, T("Campagnes - cr~eer une feuille", "Campaigns - create a sheet")))
    If Len(s) = 0 Or Not IsNumeric(s) Then ChooseTargetWorkbook = 0: Exit Function
    Dim c As Long
    c = CLng(Val(s))
    If c = 1 Then
        ChooseTargetWorkbook = 1
    ElseIf c >= 2 And c <= nb + 1 Then
        Set wb = xlApp.Workbooks(c - 1)
        ChooseTargetWorkbook = 2
    Else
        ChooseTargetWorkbook = 0
    End If
End Function


Private Function UniqueSheetName(ByVal wb As Object, ByVal base As String) As String
    On Error Resume Next
    Dim nm As String, k As Long, j As Long, found As Boolean
    nm = base
    k = 1
    Do
        found = False
        For j = 1 To wb.Worksheets.Count
            If LCase$(wb.Worksheets(j).Name) = LCase$(nm) Then found = True: Exit For
        Next j
        If Not found Then Exit Do
        k = k + 1
        nm = base & " " & k
    Loop
    UniqueSheetName = nm
End Function

Private Function CleanAddressList(ByVal listText As String) As String
    Dim parts() As String, i As Long, one As String, out As String
    parts = Split(Replace(Replace(listText, ",", ";"), vbLf, ";"), ";")
    For i = LBound(parts) To UBound(parts)
        one = Trim$(Replace(parts(i), vbCr, ""))
        If Len(one) > 0 Then
            If Len(out) > 0 Then out = out & "; "
            out = out & one
        End If
    Next i
    CleanAddressList = out
End Function


Private Function CleanPath(ByVal s As String) As String
    Dim x As String
    x = Trim$(s)
    If Len(x) >= 2 Then
        If Left$(x, 1) = """" Then x = Mid$(x, 2)
        If Right$(x, 1) = """" Then x = Left$(x, Len(x) - 1)
    End If
    CleanPath = Trim$(x)
End Function


' Chemin relatif -> prefixe par le dossier du classeur. Absolu (lettre de
' lecteur, UNC, URL) : inchange. Classeur non enregistre ou SharePoint :
' inchange (le pre-vol signale alors "introuvable" avec le chemin en clair).
Private Function ResolveRelPath(ByRef ctx As CampaignCtx, ByVal p As String) As String
    On Error Resume Next
    ResolveRelPath = p
    If Len(p) = 0 Then Exit Function
    If InStr(p, ":") > 0 Then Exit Function
    If Left$(p, 2) = "\\" Then Exit Function
    Dim base As String
    base = CStr(ctx.Wb.Path)
    If Len(base) = 0 Then Exit Function
    If LCase$(Left$(base, 4)) = "http" Then Exit Function
    If Right$(base, 1) <> "\" Then base = base & "\"
    ResolveRelPath = base & p
End Function


' Retours a la ligne dans la cellule acceptes comme separateurs, en plus de ";".
Private Function NormalizeAttachSep(ByVal s As String) As String
    Dim res As String
    res = Replace(s, vbCrLf, ";")
    res = Replace(res, vbCr, ";")
    res = Replace(res, vbLf, ";")
    NormalizeAttachSep = res
End Function
